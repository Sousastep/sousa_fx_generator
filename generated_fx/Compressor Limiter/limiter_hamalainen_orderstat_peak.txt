// ============================================================================
// limiter_hamalainen_orderstat_peak.txt
// Stereo Lookahead Order-Statistic (Running-Max) + Boxcar Peak Limiter
// Based on Perttu Hamalainen (DAFx-2002, "Smoothing of the Control Signal
// without Clipped Output in Digital Peak Limiters", Section 3, Eqs. 6-21, Fig. 4)
// Standalone Stereo RNBO codebox~ Effect (in1, in2 -> out1, out2)
// ============================================================================

@param({ min: 0.0, max: 30.0 }) drive_db = 9.0;
@param({ min: -18.0, max: -0.05 }) ceiling_db = -0.3;
@param({ min: 4.0, max: 128.0 }) lookahead_samp = 48.0;
@param({ min: 10.0, max: 1000.0 }) release_ms = 120.0;
@param({ min: 1.0, max: 64.0 }) boxcar_smooth = 24.0;
@param({ min: 0.0, max: 1.0 }) pump_character = 0.35;
@param({ min: 0.0, max: 1.0 }) mix = 1.0;

// Audio lookahead delay lines z^{-N} (Fig. 4)
@state dly_L = new delay(256);
@state dly_R = new delay(256);

// Sidechain delay line for recursive M-point moving average (Eq. 21)
@state emax_dly = new delay(128);

// Running-max 4-stage cascaded peak-hold buckets covering N lookahead samples
@state hold_val_0 = 0.0;
@state hold_val_1 = 0.0;
@state hold_val_2 = 0.0;
@state hold_val_3 = 0.0;
@state hold_cnt = 0;

// Envelope states e_max(n-1) and recursive moving average e_ave(n-1) (Eqs. 7, 19, 21)
@state e_max_state = 0.0;
@state e_ave_state = 0.0;
@state pump_LP = 0.0;

let sr_safe = max(samplerate(), 8000.0);
let dryL = in1;
let dryR = (abs(in2) > 1e-18) ? in2 : in1;

// Input drive & ceiling threshold
let drv = dbtoa(clamp(drive_db, 0.0, 30.0));
let xL = dryL * drv;
let xR = dryR * drv;

let thresh = dbtoa(clamp(ceiling_db, -18.0, -0.05));
let N = floor(clamp(lookahead_samp, 4.0, 128.0));
let M = floor(clamp(boxcar_smooth, 1.0, 64.0));

// Delay audio signals by N samples: y(n) = x(n - N) * g(n) (Eq. 5)
dly_L.write(xL);
dly_R.write(xR);
let xL_delayed = dly_L.read(N, { interp: "none" });
let xR_delayed = dly_R.read(N, { interp: "none" });

// Instantaneous stereo peak |x(n)|
let abs_x = max(abs(xL), abs(xR));

// Hamalainen Eq. 16: Choose attack coefficient a for N lookahead samples
// with max overshoot alpha_max = 1.02 so e(n+N) >= |x(n)| without clipping
let a_att = 1.0 - pow(0.0196, 1.0 / max(N + 1.0, 2.0));
let beta = pow(1.0 - a_att, N + 1.0);

// Clipping control block c(n) (Hamalainen Eq. 14 & Eq. 19):
// Aims slightly higher than |x(n)| so the 1-pole + boxcar envelope is guaranteed
// to reach |x(n)| within N samples without overshooting into clipping
let denom = max(1.0 - beta, 0.001);
let c_target = (abs_x - beta * e_ave_state) / denom;
let c_n = max(abs_x, c_target);

// Order-Statistics Running Max Filter x_max(n) = max[c(n-N), ..., c(n)] (Eq. 6)
// Implemented via 4 rotating sub-window max buckets of length ceil(N/4)
let sub_len = max(1.0, floor(N * 0.25));
hold_cnt = hold_cnt + 1;
if (hold_cnt >= sub_len) {
    hold_cnt = 0;
    hold_val_3 = hold_val_2;
    hold_val_2 = hold_val_1;
    hold_val_1 = hold_val_0;
    hold_val_0 = c_n;
} else {
    hold_val_0 = max(hold_val_0, c_n);
}
let x_max = max(max(hold_val_0, hold_val_1), max(hold_val_2, hold_val_3));

// Level Detection 1-pole filter (Eq. 7): attack uses a_att, release uses release_ms
let tau_rel = clamp(release_ms, 10.0, 1000.0) * 0.001;
let a_rel = 1.0 - exp(-1.0 / (tau_rel * sr_safe));
let a_coeff = (x_max > e_max_state) ? a_att : a_rel;
e_max_state = a_coeff * x_max + (1.0 - a_coeff) * e_max_state;

// M-point Averaging Filter after level detection (Eq. 17 & recursive form Eq. 21)
emax_dly.write(e_max_state);
let emax_old = emax_dly.read(M, { interp: "none" });
// Leaky recursive boxcar integration (prevents float drift while smoothing corners)
let box_step = (e_max_state - emax_old) / max(M, 1.0);
e_ave_state = 0.9995 * (e_ave_state + box_step) + 0.0005 * e_max_state;

// Ensure fail-safe non-clipping envelope bound: e_env >= max(e_ave, |x(n-N)|)
let e_env = max(e_ave_state, max(abs(xL_delayed), abs(xR_delayed)));

// Optional musical "pump_character" resonant envelope under-swing on release
let p_amt = clamp(pump_character, 0.0, 1.0);
let g_pump = 1.0 - exp(-1.0 / (0.045 * sr_safe));
pump_LP = pump_LP + g_pump * (e_env - pump_LP);
let e_shaped = max(e_env + 0.45 * p_amt * (pump_LP - e_env), max(abs(xL_delayed), abs(xR_delayed)));

// Gain processor g(n) = min(1, threshold / e(n)) (Hamalainen Eq. 3)
let g_n = (e_shaped > thresh) ? (thresh / max(e_shaped, 1e-9)) : 1.0;

let wetL = xL_delayed * g_n;
let wetR = xR_delayed * g_n;

let mx = clamp(mix, 0.0, 1.0);
out1 = (1.0 - mx) * dryL + mx * wetL;
out2 = (1.0 - mx) * dryR + mx * wetR;
