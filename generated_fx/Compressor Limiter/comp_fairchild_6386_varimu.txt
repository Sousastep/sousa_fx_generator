// ============================================================================
// comp_fairchild_6386_varimu.txt
// Stereo Fairchild 670 / 6386 Remote-Cutoff Triode Variable-Mu Feedback Limiter
// Based on Raffensperger (DAFx-2012, "Toward a Wave Digital Filter Model of
// the Fairchild 670 Limiter", Eqs. 1, 10-14, Fig. 8, Table 3)
// Standalone Stereo RNBO codebox~ Effect (in1, in2 -> out1, out2)
// ============================================================================

@param({ min: 0.0, max: 24.0 }) drive_db = 8.0;
@param({ min: 0.0, max: 1.0 }) ac_thresh = 0.55;
@param({ min: 0.0, max: 1.0 }) dc_knee = 0.25;
@param({ min: 1.0, max: 6.0 }) tc_mode = 5.0;
@param({ min: 0.0, max: 1.0 }) tube_harmonics = 0.5;
@param({ min: 0.0, max: 1.0 }) lat_vert_mode = 0.35;
@param({ min: 0.0, max: 1.0 }) mix = 1.0;

// Feedback output state (unit delay breaking feedback loop, Eq. 13)
@state y_prev_L = 0.0;
@state y_prev_R = 0.0;

// 3-Capacitor sidechain network states (CT fast, CU medium, CV slow memory)
// Channel 1 (Left or Mid/Lateral)
@state v_ct_1 = 0.0;
@state v_cu_1 = 0.0;
@state v_cv_1 = 0.0;

// Channel 2 (Right or Side/Vertical)
@state v_ct_2 = 0.0;
@state v_cu_2 = 0.0;
@state v_cv_2 = 0.0;

// Transformer DC-blocking & high-frequency inductive roll-off states
@state dc_x_L = 0.0;
@state dc_y_L = 0.0;
@state dc_x_R = 0.0;
@state dc_y_R = 0.0;
@state hf_lp_L = 0.0;
@state hf_lp_R = 0.0;

// Helper: Raffensperger Eq. 10 Class-B sidechain crossover nonlinearity
function fairchild_stage1(vpot: number, phi_dc_prime: number): number {
    let a = clamp(vpot - phi_dc_prime, -40.0, 40.0);
    let b = clamp(-vpot - phi_dc_prime, -40.0, 40.0);
    let num = log(1.0 + exp(a));
    let den = log(1.0 + exp(b));
    return num - den;
}

// Helper: Push-Pull 6386 remote-cutoff variable-mu triode amplifier
// Models balanced odd-order harmonic saturation that increases with V_sc
function varimu_6386_amp(x: number, v_sc: number, harm_amt: number): number {
    // Control grid bias shifts more negative as sidechain voltage v_sc grows
    let gr_gain = 1.0 / (1.0 + 0.38 * v_sc + 0.045 * v_sc * v_sc);
    // Raffensperger Sec. 6 Fig. 13: 3rd-harmonic distortion rises with gain reduction
    let gr_depth = 1.0 - gr_gain;
    let k3 = harm_amt * (0.14 + 0.55 * gr_depth);
    let u = x * gr_gain;
    let u_sat = u + k3 * (u * u * u);
    return u_sat / (1.0 + 0.35 * k3 + 0.18 * u_sat * u_sat);
}

let sr_safe = max(samplerate(), 8000.0);
let dt = 1.0 / sr_safe;

let dryL = in1;
let dryR = (abs(in2) > 1e-18) ? in2 : in1;

// Input transformer step gain
let in_gain = dbtoa(clamp(drive_db, 0.0, 24.0));
let xL = dryL * in_gain;
let xR = dryR * in_gain;

// Continuous blend between Left/Right (0) and Lateral/Vertical Mid-Side (1)
let lv = clamp(lat_vert_mode, 0.0, 1.0);
let inv_sqrt2 = 0.7071067811865475;
let mid_in = (xL + xR) * inv_sqrt2;
let side_in = (xL - xR) * inv_sqrt2;

let ch1_in = (1.0 - lv) * xL + lv * mid_in;
let ch2_in = (1.0 - lv) * xR + lv * side_in;

// Feedback signals from previous sample (in L/R or M/S space)
let mid_prev = (y_prev_L + y_prev_R) * inv_sqrt2;
let side_prev = (y_prev_L - y_prev_R) * inv_sqrt2;
let fb1 = (1.0 - lv) * y_prev_L + lv * mid_prev;
let fb2 = (1.0 - lv) * y_prev_R + lv * side_prev;

// Sidechain AC and DC threshold controls (Raffensperger Eq. 10)
let phi_ac = clamp(ac_thresh, 0.0, 1.0);
let phi_dc = clamp(dc_knee, 0.0, 1.0);
let phi_dc_prime = 12.2 * (phi_dc + 0.08);
let sc_scale = 18.0 * (0.08 + 0.92 * phi_ac);

// Stage 1-3 sidechain voltage and diode-bridge current for Channel 1 & Channel 2
let v_st1_1 = fairchild_stage1(fb1 * sc_scale, phi_dc_prime);
let v_st3_1 = clamp(8.4 * v_st1_1, -100.0, 100.0);
let v_diff_1 = max(0.0, abs(v_st3_1) - v_ct_1);
let i_sc_1 = 0.45 * log(1.0 + exp(clamp(2.5 * v_diff_1 - 1.2, -20.0, 25.0)));

let v_st1_2 = fairchild_stage1(fb2 * sc_scale, phi_dc_prime);
let v_st3_2 = clamp(8.4 * v_st1_2, -100.0, 100.0);
let v_diff_2 = max(0.0, abs(v_st3_2) - v_ct_2);
let i_sc_2 = 0.45 * log(1.0 + exp(clamp(2.5 * v_diff_2 - 1.2, -20.0, 25.0)));

// Fairchild 670 6-position Time Constant network (Table 3)
// Positions 1-4: single RC release (0.3s, 0.8s, 2.0s, 5.0s)
// Position 5: dual RC program-dependent release (CT + CU)
// Position 6: triple RC dual-program memory release (CT + CU + CV)
let sel = clamp(tc_mode, 1.0, 6.0);
let tau_att = 0.00025 + 0.00015 * clamp(sel - 1.0, 0.0, 3.0);
let tau_rel_fast = 0.12;
if (sel < 1.5) {
    tau_rel_fast = 0.10;
} else if (sel < 2.5) {
    tau_rel_fast = 0.30;
} else if (sel < 3.5) {
    tau_rel_fast = 0.88;
} else if (sel < 4.5) {
    tau_rel_fast = 1.76;
} else if (sel < 5.5) {
    tau_rel_fast = 0.40;
} else {
    tau_rel_fast = 0.20;
}

let g_cu = (sel >= 4.5) ? (1.0 / 0.80) : 0.0;
let g_cv = (sel >= 5.5) ? (1.0 / 2.50) : 0.0;

// Update 3-capacitor network for Channel 1
let charge_1 = (i_sc_1 > 0.02) ? (i_sc_1 / tau_att) : 0.0;
let d_ct_1 = charge_1 - (v_ct_1 / tau_rel_fast) - g_cu * (v_ct_1 - v_cu_1) - g_cv * (v_ct_1 - v_cv_1);
v_ct_1 = clamp(v_ct_1 + dt * d_ct_1, 0.0, 60.0);
v_cu_1 = clamp(v_cu_1 + dt * (g_cu * 0.25 * (v_ct_1 - v_cu_1) - v_cu_1 / 2.2), 0.0, 60.0);
v_cv_1 = clamp(v_cv_1 + dt * (g_cv * 0.10 * (v_ct_1 - v_cv_1) - v_cv_1 / 6.5), 0.0, 60.0);

// Update 3-capacitor network for Channel 2
let charge_2 = (i_sc_2 > 0.02) ? (i_sc_2 / tau_att) : 0.0;
let d_ct_2 = charge_2 - (v_ct_2 / tau_rel_fast) - g_cu * (v_ct_2 - v_cu_2) - g_cv * (v_ct_2 - v_cv_2);
v_ct_2 = clamp(v_ct_2 + dt * d_ct_2, 0.0, 60.0);
v_cu_2 = clamp(v_cu_2 + dt * (g_cu * 0.25 * (v_ct_2 - v_cu_2) - v_cu_2 / 2.2), 0.0, 60.0);
v_cv_2 = clamp(v_cv_2 + dt * (g_cv * 0.10 * (v_ct_2 - v_cv_2) - v_cv_2 / 6.5), 0.0, 60.0);

// Apply 6386 Push-Pull Variable-Mu Amplifier to Ch1 & Ch2
let harm = clamp(tube_harmonics, 0.0, 1.0);
let y_ch1 = varimu_6386_amp(ch1_in, v_ct_1, harm);
let y_ch2 = varimu_6386_amp(ch2_in, v_ct_2, harm);

// Decode back from Lateral/Vertical (M/S) if engaged
let ms_out_L = (y_ch1 + y_ch2) * inv_sqrt2;
let ms_out_R = (y_ch1 - y_ch2) * inv_sqrt2;
let wetL = (1.0 - lv) * y_ch1 + lv * ms_out_L;
let wetR = (1.0 - lv) * y_ch2 + lv * ms_out_R;

// Store feedback output state
y_prev_L = wetL;
y_prev_R = wetR;

// Output transformer gentle high-frequency silk & DC block
let g_hf = 1.0 - exp(-2.0 * PI * 15500.0 / sr_safe);
hf_lp_L = hf_lp_L + g_hf * (wetL - hf_lp_L);
hf_lp_R = hf_lp_R + g_hf * (wetR - hf_lp_R);

let out_comp = 1.0 / (1.0 + 0.18 * (in_gain - 1.0) * (1.0 - 0.55 * phi_ac));
let procL = hf_lp_L * out_comp;
let procR = hf_lp_R * out_comp;

dc_y_L = procL - dc_x_L + 0.9992 * dc_y_L;
dc_x_L = procL;
dc_y_R = procR - dc_x_R + 0.9992 * dc_y_R;
dc_x_R = procR;

let m = clamp(mix, 0.0, 1.0);
out1 = (1.0 - m) * dryL + m * dc_y_L;
out2 = (1.0 - m) * dryR + m * dc_y_R;
