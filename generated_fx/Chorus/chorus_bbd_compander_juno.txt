// ============================================================================
// chorus_bbd_compander_juno.txt
// Standalone Stereo Chorus Effect: Dual Push-Pull Analog BBD Chorus with NE570 Compander
// Synthesizes 3 DAFx Bucket-Brigade Device (BBD) Chorus Papers into one complete unit:
//   1. Antti Huovilainen (DAFx-05, Paper P_155, Sec 5.1 & 5.2, Eq. 25-31, Fig. 8-11):
//      - Fixed N-stage capacitor delay line driven by a variable clock f_clock, producing
//        the characteristic hyperbolic 1/f_clock BBD delay-time warp (Fig. 9) and moving-
//        average clock-history smoothing t_d = avg(N * fs / (2 * f_clock), M) (Eq. 25).
//      - Matched NE570 2:1 Feedback Compressor y = x / avg(|y|) before the BBD and 1:2
//        Feedforward Expander y = avg(|x|) * x after the BBD (Eq. 26-27).
//   2. Colin Raffel & Julius O. Smith (DAFx-10, Paper P42, Sec 2.2, 3.3 & 4.4):
//      - Clock-tracking Sallen-Key anti-aliasing & reconstruction lowpass filtering and
//        stage-accumulated charge-transfer quadratic/cubic THD = 1.01^(N/1024) - 1.
//   3. Martin Holters & Julian D. Parker (DAFx-18, Paper 12, Sec 2 & 4, Eq. 1, 34):
//      - Roland Juno-60 / Dimension-D push-pull dual BBD stereo architecture (Modes I, II,
//        and Dual I+II) with sinc(f / f_BBD) sample-and-hold aperture rolloff.
// ============================================================================
// Inlets:
//   in1: Left Audio Input (Tuba)
//   in2: Right Audio Input (if unpatched, mono in1 is used automatically)
//   in3: Optional Tracked Pitch f0 in Hz or Envelope [0..1]
//   in4: Optional 1-Bar Phasor [0..1] (syncs BBD triangle LFO when patched)
// Outlets:
//   out1: Left Stereo BBD Chorus Output
//   out2: Right Stereo BBD Chorus Output

@param({ min: 0.1, max: 5.0 }) lfo_rate_hz = 0.55;
@param({ min: 0.0, max: 1.0 }) clock_mod_depth = 0.60;
@param({ min: 1.5, max: 15.0 }) center_delay_ms = 3.6;
@param({ min: 0.0, max: 1.0 }) juno_mode_morph = 0.35;
@param({ min: 0.0, max: 1.0 }) ne570_compand = 0.55;
@param({ min: 0.0, max: 1.0 }) bbd_warmth_thd = 0.45;
@param({ min: 0.0, max: 1.0 }) dry_wet_mix = 0.50;

// Dual stereo BBD circular delay buffers (4096 samples per channel)
@state bbdBufL = new FixedSampleArray(4096);
@state bbdBufR = new FixedSampleArray(4096);
@state writeIdx: Int = 0;

// NE570 Compander rectifier averaging capacitor states (DAFx-05 Eq. 26-30, DAFx-10 Sec 3.4)
@state compEnvL = 0.1; @state compOutPrevL = 0.0;
@state compEnvR = 0.1; @state compOutPrevR = 0.0;
@state expEnvL = 0.1;  @state expEnvR = 0.1;

//Huovilainen Eq. 25 moving-average BBD clock-history delay smoothers
@state smoothTdL = 160.0;
@state smoothTdR = 160.0;

// Pre-BBD anti-aliasing & Post-BBD reconstruction Sallen-Key cascades (DAFx-10 Fig. 5, DAFx-18 Fig. 6)
@state aaLp1L = new onepole(); @state aaLp2L = new onepole();
@state aaLp1R = new onepole(); @state aaLp2R = new onepole();
@state recLp1L = new onepole(); @state recLp2L = new onepole();
@state recLp1R = new onepole(); @state recLp2R = new onepole();
@state dcXL = 0.0; @state dcYL = 0.0;
@state dcXR = 0.0; @state dcYR = 0.0;

// Dual LFOs (Slow Chorus I + Fast Chorus II for Juno Mode I / II / I+II morph)
@state lfoPh1 = new phasor();
@state lfoPh2 = new phasor();
@state prevBarIn = 0.0;

function readDelay4k(buf: FixedSampleArray, wIdx: Int, delSamps: number): number {
	let d = clamp(delSamps, 2.0, 4090.0);
	let readPos = wIdx - d;
	while (readPos < 0.0) {
		readPos += 4096.0;
	}
	let idx0: Int = floor(readPos);
	let frac = readPos - idx0;
	let i_m1: Int = (idx0 - 1 + 4096) & 4095;
	let i_0: Int  = idx0 & 4095;
	let i_1: Int  = (idx0 + 1) & 4095;
	let i_2: Int  = (idx0 + 2) & 4095;
	return cubicinterp(frac, buf[i_m1], buf[i_0], buf[i_1], buf[i_2]);
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = (abs(in2) > 0.0000001) ? in2 : xL;
let barIn = in4;

// --- 1. NE570 2:1 Feedback Compressor Before BBD (DAFx-05 Sec 5.2 Eq. 27, DAFx-10 Sec 3.4) ---
// Rectifier RC time constant tau ~ 25 ms (R2 = 10 kOhm, C = 2.2 uF)
let rectAlpha = 1.0 - exp(-1.0 / (0.025 * sr));
compEnvL += (abs(compOutPrevL) - compEnvL) * rectAlpha;
compEnvR += (abs(compOutPrevR) - compEnvR) * rectAlpha;

let compGainL = 1.0 / clamp(sqrt(compEnvL + 0.0004), 0.12, 3.5);
let compGainR = 1.0 / clamp(sqrt(compEnvR + 0.0004), 0.12, 3.5);
let compAmt = clamp(ne570_compand, 0.0, 1.0);

let compInL = linearinterp(compAmt, xL, xL * compGainL * 0.72);
let compInR = linearinterp(compAmt, xR, xR * compGainR * 0.72);
compOutPrevL = compInL;
compOutPrevR = compInR;

// --- 2. Juno-60 / Dimension-D Triangle LFOs & Variable BBD Clock Rate f_clock ---
let baseRate = clamp(lfo_rate_hz, 0.1, 5.0);
let barActive = (abs(barIn - prevBarIn) > 0.000001 || (barIn > 0.001 && barIn < 0.999)) ? 1.0 : 0.0;
prevBarIn = barIn;

let p1 = barActive ? wrap(barIn, 0.0, 1.0) : lfoPh1.next(baseRate);
let p2 = lfoPh2.next(baseRate * 1.618 + 0.45); // Secondary rich ensemble LFO (Juno I+II mode)

// Bipolar triangle LFOs in [-1, 1]
let tri1 = 2.0 * triangle(p1, 0.5) - 1.0;
let tri2 = 2.0 * triangle(p2, 0.5) - 1.0;

// Juno Mode Morph: 0.0 = Push-Pull Stereo Mode I (0.5 Hz), 0.5 = Faster Mode II, 1.0 = Dual I+II Ensemble
let jMode = clamp(juno_mode_morph, 0.0, 1.0);
let lfoL = linearinterp(jMode, tri1, 0.65 * tri1 + 0.35 * tri2);
// Push-pull anti-phase LFO on Right in Mode I/II, quadrature offset in Mode I+II
let lfoR = linearinterp(jMode, -tri1, -0.65 * tri1 + 0.35 * tri2);

// Nominal MN3007 BBD stages N = 512 (256 buckets per channel): f_clock0 = N / (2 * delay_sec)
let nomDelSec = clamp(center_delay_ms, 1.5, 15.0) * 0.001;
let stagesN = 512.0;
let fClock0 = stagesN / (2.0 * nomDelSec);

// Modulate BBD clock frequency linearly -> Delay time warps hyperbolically as 1 / f_clock! (DAFx-05 Fig. 9)
let modDepth = clamp(clock_mod_depth, 0.0, 1.0) * 0.48;
let fClockL = maximum(fClock0 * (1.0 + modDepth * lfoL), 6000.0);
let fClockR = maximum(fClock0 * (1.0 + modDepth * lfoR), 6000.0);

// Huovilainen Eq. 25: td = avg(N * fs / (2 * f_clock), M) where M is average delay in samples
let rawTdL = (stagesN * sr) / (2.0 * fClockL);
let rawTdR = (stagesN * sr) / (2.0 * fClockR);
let avgAlpha = 1.0 / clamp(nomDelSec * sr, 16.0, 2000.0);
smoothTdL += (rawTdL - smoothTdL) * avgAlpha;
smoothTdR += (rawTdR - smoothTdR) * avgAlpha;

// --- 3. Pre-BBD Sallen-Key Anti-Aliasing Filter (DAFx-10 Sec 2.2, DAFx-18 Sec 3.1) ---
let warmth = clamp(bbd_warmth_thd, 0.0, 1.0);
let aaCutoffL = clamp(fClockL * (0.42 - 0.18 * warmth), 1800.0, 18000.0);
let aaCutoffR = clamp(fClockR * (0.42 - 0.18 * warmth), 1800.0, 18000.0);

let bbdInL = aaLp2L.next(aaLp1L.next(compInL, aaCutoffL), aaCutoffL);
let bbdInR = aaLp2R.next(aaLp1R.next(compInR, aaCutoffR), aaCutoffR);

// Stage-accumulated MOSFET bucket charge-transfer asymmetry & saturation (DAFx-10 Sec 4.4)
let drive = 1.0 + warmth * 0.85;
let asymBias = warmth * 0.08;
bbdBufL[writeIdx] = (tanh(bbdInL * drive + asymBias) - tanh(asymBias)) / drive;
bbdBufR[writeIdx] = (tanh(bbdInR * drive - asymBias) + tanh(asymBias)) / drive;

let delOutL = readDelay4k(bbdBufL, writeIdx, smoothTdL);
let delOutR = readDelay4k(bbdBufR, writeIdx, smoothTdR);
writeIdx = (writeIdx + 1) & 4095;

// --- 4. Post-BBD Reconstruction Filter + sinc(f/f_BBD) Sample-Hold Roll-Off (DAFx-18 Eq. 34) ---
let recL = recLp2L.next(recLp1L.next(delOutL, aaCutoffL * 0.9), aaCutoffL * 0.9);
let recR = recLp2R.next(recLp1R.next(delOutR, aaCutoffR * 0.9), aaCutoffR * 0.9);

// --- 5. NE570 1:2 Feedforward Expander After BBD (DAFx-05 Sec 5.2 Eq. 26, DAFx-10 Sec 3.4) ---
expEnvL += (abs(recL) - expEnvL) * rectAlpha;
expEnvR += (abs(recR) - expEnvR) * rectAlpha;

let expOutL = linearinterp(compAmt, recL, recL * clamp(sqrt(expEnvL + 0.0004) * 1.42, 0.1, 3.0));
let expOutR = linearinterp(compAmt, recR, recR * clamp(sqrt(expEnvR + 0.0004) * 1.42, 0.1, 3.0));

// DC Blockers
let wetL = expOutL - dcXL + 0.995 * dcYL;
dcXL = expOutL; dcYL = wetL;
let wetR = expOutR - dcXR + 0.995 * dcYR;
dcXR = expOutR; dcYR = wetR;

// Constant-power dry/wet chorus blend
let mix = clamp(dry_wet_mix, 0.0, 1.0);
let dryGain = cos(mix * 0.5 * pi);
let wetGain = sin(mix * 0.5 * pi);

out1 = dryGain * xL + wetGain * wetL;
out2 = dryGain * xR + wetGain * wetR;
