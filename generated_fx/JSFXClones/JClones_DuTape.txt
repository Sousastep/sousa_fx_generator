 * in1/in2: stereo in; out1/out2: stereo out.
 * A tape-saturation emulation: per-channel tape hiss (a 24-bit shift-register
 * noise source through two low-passes and a high-pass, level and EQ set per
 * machine), an input-gain stage, the machine's play/record tone EQ as cascaded
 * one-poles, a 256-point measured tape transfer curve per machine, an optional
 * "random gain" dither on negative noise half-cycles, and a 3 Hz DC remover on
 * the asymmetric machines. Original slider units, ranges and defaults kept.
 *
 * Table layout (s_tab): 6 machines x 512 points; base = machine * 512, and the
 * negative half of the curve lives at base + 256 (it is NOT always a mirror of
 * the positive half — the transfer curves are measured, so keep both).
 */
// Input (dB)
@param({ min: -12, max: 6 }) input_db = 0;
// Output (dB)
@param({ min: -12, max: 6 }) output_db = 0;
// Machine
@param({ enum: ["Ideal Tape", "80s 2 Tk.", "70s 24Tk.", "60s 2 Tk.", "60s Tube", "1/2 M.Tk."] }) machine = 0;
// Speed
@param({ enum: ["30 ips", "15 ips", "7 1/2 ips"] }) speed = 0;
// Noise Reduction
@param({ enum: ["Off", "Type 1", "Type 2", "Noiseless"] }) noise_reduction = 3;
// Dithering
@param({ enum: ["Off", "On"] }) dither = 1;

@state s_tab = new FixedFloat64Array(3072);
@state s_tabReady = 0;
@state s_rand = new FixedInt32Array(2);
@state s_st = new FixedFloat64Array(14);
@state s_inGain = 1;
@state s_outGain = 1;
@state s_model = 0;
@state s_nLp1 = 0;
@state s_nLp2 = 0;
@state s_nHp3 = 0;
@state s_nGain = 0;
@state s_randGain = 1;
@state s_sLp = 0;
@state s_sHp = 0;
@state s_sLs = 0;
@state s_sLsG = 0;
@state s_dcK = 0;
@state s_dither = 1;
@state frame: Int = 0;

function fnDbToK(x) {
    return pow(10.0, x / 20.0);
}

function fnSetIPS(ips) {
    let ss = 44100.0 / maximum(samplerate(), 1.0);
    if (ips < 0.5) {
        s_sLp = 0.86928473 * ss;
        s_sHp = 0.00317097 * ss;
        s_sLs = 0.01162779 * ss;
        s_sLsG = 0.15116276;
    } else if (ips < 1.5) {
        s_sLp = 0.67441853 * ss;
        s_sHp = 0.00706673 * ss;
        s_sLs = 0.01321328 * ss;
        s_sLsG = 0.2176744;
    } else {
        s_sLp = 0.45348828 * ss;
        s_sHp = 0.01413346 * ss;
        s_sLs = 0.02325571 * ss;
        s_sLsG = 0.2790697;
    }
}

function fnSetModel(model, nr) {
    let ss = 44100.0 / maximum(samplerate(), 1.0);
    let m = trunc(model);
    s_model = model;
    if (m == 0) {
        s_nLp1 = 0.93840837 * ss;
        s_nLp2 = 0.55813951 * ss;
        s_nHp3 = 0.0;
        s_nGain = 0.00072396;
        s_randGain = 1.0;
    } else if (m == 1) {
        s_nLp1 = 0.88372086 * ss;
        s_nLp2 = 0.17441859 * ss;
        s_nHp3 = 0.02325571 * ss;
        s_nGain = 0.00347507;
        s_randGain = 1.0 - 0.00176024;
    } else if (m == 2) {
        s_nLp1 = 0.63953491 * ss;
        s_nLp2 = 0.40697675 * ss;
        s_nHp3 = 0.02325571 * ss;
        s_nGain = 0.00225973;
        s_randGain = 1.0 - 0.00227785;
    } else if (m == 3) {
        s_nLp1 = 0.45348828 * ss;
        s_nLp2 = 0.26744178 * ss;
        s_nHp3 = 0.00679493 * ss;
        s_nGain = 0.00394523;
        s_randGain = 1.0 - 0.00306559;
    } else if (m == 4) {
        s_nLp1 = 0.39534883 * ss;
        s_nLp2 = 0.16279068 * ss;
        s_nHp3 = 0.00163078 * ss;
        s_nGain = 0.00910485;
        s_randGain = 1.0 - 0.00382733;
    } else {
        s_nLp1 = 0.67441853 * ss;
        s_nLp2 = 0.49999994 * ss;
        s_nHp3 = 0.08139528 * ss;
        s_nGain = 0.00226498;
        s_randGain = 1.0 - 0.00715435;
    }
    let rel = 1.0;
    if (nr < 0.5) {
        rel = 1.0;
    } else if (nr < 1.5) {
        rel = 0.4671902;
    } else if (nr < 2.5) {
        rel = 0.21542778;
    } else {
        rel = 0.0;
    }
    s_nGain = s_nGain * rel;
    s_dcK = 1.0 - exp(0.0 - 2.0 * pi * 3.0 / maximum(samplerate(), 1.0));
}

// 24-bit shift-register noise, one generator per channel (seeds 1 and 2).
function fnNextRand(c) {
    let p: Int = s_rand[c];
    let neg = (p & 8388608) != 0;
    let n: Int = p << 1;
    if (neg) n = n ^ 1935147;
    n = n & 16777215;
    let sv: Int = 0;
    if ((n & 8388608) == 0) sv = n;
    else sv = 0 - (((n ^ 16777215) & 16777215) + 1);
    s_rand[c] = n;
    return sv / 8388607.0;
}

function fnShape(x) {
    let base: Int = s_model * 512.0;
    if (x <= 0.0) base = base + 256;
    let fe = abs(x) * 255.0;
    let y = 0.0;
    if (fe > 255.0) {
        y = s_tab[base + 255];
    } else {
        let t0: Int = floor(fe);
        let t1: Int = t0 + 1;
        if (t1 > 255) t1 = 255;
        let k = fe - t0;
        y = s_tab[base + t1] * k + s_tab[base + t0] * (1.0 - k);
    }
    if (x <= 0.0) y = 0.0 - y;
    return y;
}

function fnProcess(x, c) {
    let k: Int = c * 7;
    let y = x * s_inGain;
    let ny = fnNextRand(c);

    // tape hiss: two low-passes and a high-pass, then its level
    s_st[k] = s_st[k] + (ny - s_st[k]) * s_nLp1;
    ny = s_st[k];
    s_st[k + 1] = s_st[k + 1] + (ny - s_st[k + 1]) * s_nLp2;
    ny = s_st[k + 1];
    s_st[k + 2] = s_st[k + 2] + (ny - s_st[k + 2]) * s_nHp3;
    ny = ny - s_st[k + 2];

    y = y + ny * s_nGain;

    if (s_dither >= 0.5 && ny < 0.0) y = y * s_randGain;

    // play/record tone EQ
    s_st[k + 3] = s_st[k + 3] + (y - s_st[k + 3]) * s_sLp;
    y = s_st[k + 3];
    s_st[k + 4] = s_st[k + 4] + (y - s_st[k + 4]) * s_sHp;
    y = y - s_st[k + 4];
    s_st[k + 5] = s_st[k + 5] + (y - s_st[k + 5]) * s_sLs;
    y = y + s_st[k + 5] * s_sLsG;

    y = fnShape(y);

    // DC remover on the asymmetric machines
    if (s_model >= 2.0) {
        s_st[k + 6] = s_st[k + 6] + (y - s_st[k + 6]) * s_dcK;
        y = y - s_st[k + 6];
    }
    return y * s_outGain;
}
