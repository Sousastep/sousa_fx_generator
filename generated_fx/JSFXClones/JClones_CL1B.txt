function fnDbToK(x) {
    return pow(10.0, x / 20.0);
}

function fnClamp(x) {
    return clamp(x, -0.99999988, 0.99999988);
}

// Returns the mantissa in [0.5, 1) and stores the exponent in s_frexpExp.
function fnFrexp(x) {
    let sign = 0.0;
    if (x < 0.0) {
        x = -x;
        sign = 1.0;
    }
    let ex = ceil(log(maximum(x, 1.0e-300)) / 0.6931471805599453);
    let a1 = x / pow(2.0, ex);
    if (a1 == 1.0) {
        a1 = 0.5;
        ex = ex + 1.0;
    }
    if (sign >= 0.5) a1 = -a1;
    s_frexpExp = ex;
    return a1;
}

function fnInterpLin(x, off, size) {
    x = fnClamp(x);
    if (x < 0.0) x = 0.0;
    x *= size - 1.0;
    let i: Int = floor(x);
    if (i == size - 1) return s_linTab[off + size - 1];
    let frac = x - i;
    return s_linTab[off + i] * (1.0 - frac) + s_linTab[off + i + 1] * frac;
}

function fnInterpExp(x, off, isNeg) {
    let o: Int = off;
    if (isNeg >= 0.5) o = off + 23;
    x = fnClamp(x);
    if (x * 1000000.0 == 0.0) return s_expTab[o + 23];

    let mant = fnFrexp(x);
    let ex: Int = s_frexpExp;
    let idx: Int = 1 - ex;
    if (idx < 0) ex = 0;
    let frac = 0.0;
    if (idx > 22) {
        frac = mant * pow(2.0, 22.0 + ex);
        idx = 23;
    } else {
        if (mant <= 0.0) frac = (mant + 0.5) * 2.0;
        else frac = (mant - 0.5) * 2.0;
    }
    if (x < 0.0 && isNeg >= 0.5) {
        idx = -idx;
        frac = frac + 1.0;
    }
    return frac * (s_expTab[o + idx] - s_expTab[o + idx + 1]) + s_expTab[o + idx + 1];
}

function fnSetParameters(ratioK, thrDb, mode, atk, rel, outDb) {
    let sr = maximum(samplerate(), 1.0);
    s_t3 = fnInterpExp(ratioK, 0, 0.0);
    s_t10 = fnInterpExp(ratioK, 50, 0.0);
    s_t8 = fnInterpLin(atk, 48, 46) * (44100.0 / sr);
    s_t9 = fnInterpLin(rel, 94, 24) * (44100.0 / sr);
    s_t4 = fnInterpLin(s_t3, 0, 24);
    s_mode = mode;
    s_t7 = fnDbToK(-40.0 - thrDb);
    s_t11 = fnDbToK(-30.0 + outDb);
}

function fnProcess(inL, inR) {
    let invGr = s_l1s * 0.2998201 + s_l2s * 0.079904087;
    let gr = 0.0029900903 / fnClamp(invGr + 0.0029900903);

    let t5 = fnInterpExp(invGr, 25, 0.0);
    let t6 = fnInterpLin(t5, 24, 24);

    let m1 = 0.01193628 * t6 + 0.9323384 * (1.0 - t6);
    let m2 = 0.4595526 * t6 + 1.0 * (1.0 - t6);
    let mult = m1 * (1.0 - s_t4) + m2 * s_t4;

    let det1 = (inL * 0.08098298) * mult * s_t7;
    let det2 = (inR * 0.08098298) * mult * s_t7;

    let t12a = fnInterpExp(det1, 75, 1.0);
    let t12b = fnInterpExp(det2, 75, 1.0);
    let lvl = maximum(abs(t12a), abs(t12b));

    if (s_mode != 0.0) {
        if (lvl >= s_lvlS) s_lvlS = minimum(s_lvlS + s_t8, lvl);
        else s_lvlS = maximum(s_lvlS * s_relK - s_t9, lvl);
    }

    let combined = lvl;
    if (s_mode == 2.0) combined = s_lvlS;
    else if (s_mode == 1.0) combined = maximum(lvl, s_lvlS);

    let t13 = fnInterpLin(combined, 118, 252);

    let k1 = s_l1R;
    if (t13 > s_l1s) k1 = s_l1A;
    s_l1s = t13 + (s_l1s - t13) * k1;

    let k2 = s_l2R;
    if (t13 > s_l2s) k2 = s_l2A;
    s_l2s = t13 + (s_l2s - t13) * k2;

    let y1 = inL * s_t10 * s_t11 * gr * 33.768673;
    let y2 = inR * s_t10 * s_t11 * gr * 33.768673;

    s_peq1 += (y1 - s_peq1) * s_peqK;
    s_peq2 += (y2 - s_peq2) * s_peqK;
    s_y1 = s_peq1;
    s_y2 = s_peq2;
}

if (frame == 0) {
    let sr = maximum(samplerate(), 1.0);
    if (s_tablesReady == 0) {
        fnInitTables();
        s_tablesReady = 1;
    }
    if (sr != s_sr) {
        s_l1A = exp(-1.0 / (sr * 0.001324200));
        s_l1R = exp(-1.0 / (sr * 0.001782562));
        s_l2A = exp(-1.0 / (sr * 0.028011420));
        s_l2R = exp(-1.0 / (sr * 0.026260180));
        s_relK = exp(-1.0 / (sr * 5.898));
        s_peqK = 1.0 - exp(-2.0 * pi * (20000.0 / sr));
        s_sr = sr;
    }
    fnSetParameters((ratio - 2.0) / 8.0, threshold, ar_select, attack * 0.1, release * 0.1, output_gain);
}

out2 = s_y2;
