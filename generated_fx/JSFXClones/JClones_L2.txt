// Threshold (dB)
@param({ min: -30, max: 0 }) threshold = 0;
// Ceiling (dB)
@param({ min: -30, max: 0 }) ceiling = 0;
// Auto Release
@param({ enum: ["off", "on"] }) auto_release = 1;
// Release (ms)
@param({ enum: ["0.01", "0.1", "1.0", "10.0", "100", "1000"] }) release = 2;
// Quantize
@param({ enum: ["16", "18", "20", "22", "24"] }) quantize = 4;
// Dither
@param({ enum: ["none", "type1", "type2"] }) dither = 1;
// Shaping
@param({ enum: ["none", "moderate", "normal", "ultra"] }) shaping = 2;

@state s_inGain = 1;
@state s_outGain = 1;
@state s_auto = 1;
@state s_fastG = 0;
@state s_fastS = 1;
@state s_slowS = 1;
@state s_dynS = 1;
@state s_dynRelG = 0;
@state s_slowAtkG = 0;
@state s_slowRelG = 0;
@state s_dynG1 = 0;
@state s_dynG2 = 0;
@state s_dynG3 = 0;
@state s_laSize: Int = 64;
@state s_maSize: Int = 33;
@state s_grMinBuf = new FixedFloat64Array(2048);
@state s_grMinHead: Int = 0;
@state s_ma1 = new FixedFloat64Array(2048);
@state s_ma2 = new FixedFloat64Array(2048);
@state s_ma1Head: Int = 0;
@state s_ma2Head: Int = 0;
@state s_ma1Sum = 0;
@state s_ma2Sum = 0;
@state s_dlyL = new delay(0.02);
@state s_dlyR = new delay(0.02);
@state s_table = new FixedFloat64Array(144);
@state s_histL = new FixedFloat64Array(9);
@state s_histR = new FixedFloat64Array(9);
@state s_histIdx: Int = 0;
@state s_t1State = 0;
@state s_t2a = 0;
@state s_t2b = 0.25;
@state s_scaling = 1;
@state s_srIdx: Int = 0;
@state s_shaping: Int = 2;
@state s_dithType: Int = 1;
@state s_tableReady = 0;
@state s_histReady = 0;
@state s_sr = 0;
@state frame: Int = 0;

function fnDbToK(x) {
    return pow(10.0, x / 20.0);
}

function fnStepG(t) {
    return exp(-1.0 / (t * (maximum(samplerate(), 1.0) / 2.302585093)));
}

// EEL's filter1_G(s, G, x): x + G*(s - x)
function fnOnePole(s, g, x) {
    return x + g * (s - x);
}

function fnRandType1() {
    let st = 10368889.0 * s_t1State + 1.0;
    st = st - floor(st / 4294967296.0) * 4294967296.0;
    if (st > 2147483648.0) st = st - 4294967296.0;
    s_t1State = st;
    return st;
}

function fnDitherSample(input, r1, r2, ch) {
    let tbl: Int = 9 * (s_shaping + 4 * s_srIdx);
    let filtered = 0.0;
    let k: Int = 0;
    let i: Int = s_histIdx;
    for (let n: Int = 0; n < 9; n++) {
        let h = s_histL[i];
        if (ch >= 0.5) h = s_histR[i];
        filtered += h * s_table[tbl + k];
        k += 1;
        i += 1;
        if (i >= 9) i -= 9;
    }

    input = clamp(input, -1.0, 1.0);
    let x1 = input * 8388608.0;
    let x2 = x1 + r2 * s_scaling;
    let x3 = x2 + (r1 + 0.5) * s_scaling + filtered;
    let outFp = floor(x3);
    if (outFp > 8388607.0) outFp = 8388607.0;
    if (outFp < -8388608.0) outFp = -8388608.0;
    if (s_scaling != 1.0) outFp = floor(outFp / s_scaling) * s_scaling;

    let err = x2 - outFp;
    if (ch < 0.5) s_histL[s_histIdx] = err;
    else s_histR[s_histIdx] = err;
    return outFp / 8388608.0;
}

if (frame == 0) {
    let sr = maximum(samplerate(), 1.0);
    if (sr != s_sr) {
        let la = floor((0.0014 * sr) / 64.0 + 0.6) * 64.0;
        if (la < 64.0) la = 64.0;
        if (la > 2040.0) la = 2040.0;
        s_laSize = la;
        s_maSize = floor(la / 2.0 + 1.0);
        for (let i: Int = 0; i < 2048; i++) {
            s_grMinBuf[i] = 1.0;
            s_ma1[i] = 1.0;
            s_ma2[i] = 1.0;
        }
        s_grMinHead = 0;
        s_ma1Head = 0;
        s_ma2Head = 0;
        s_ma1Sum = s_maSize;
        s_ma2Sum = s_maSize;
        s_slowAtkG = fnStepG(0.250);
        s_slowRelG = fnStepG(0.500);
        s_dynG1 = fnStepG(0.281);
        s_dynG2 = fnStepG(0.057);
        s_dynG3 = fnStepG(0.00054);
        s_dynRelG = fnStepG(0.001);
        if (sr <= 44100.0) s_srIdx = 0;
        else if (sr <= 48000.0) s_srIdx = 1;
        else if (sr <= 88200.0) s_srIdx = 2;
        else s_srIdx = 3;
        s_sr = sr;
    }

    if (s_tableReady == 0) {
        fnInitTable();
        s_tableReady = 1;
    }
    if (s_histReady == 0) {
        for (let i: Int = 0; i < 9; i++) {
            s_histL[i] = 0.0;
            s_histR[i] = 0.0;
        }
        s_t1State = 0;
        s_t2a = 0.0;
        s_t2b = 0.25;
        s_histIdx = 0;
        s_histReady = 1;
    }

    s_inGain = 1.0 / fnDbToK(threshold);
    s_outGain = fnDbToK(ceiling);
    s_auto = auto_release;
    s_fastG = fnStepG(pow(10.0, release - 5.0));
    s_dithType = dither;
    s_shaping = shaping;
    s_scaling = pow(2.0, 24.0 - (16.0 + quantize * 2.0));
}

let lL = in1 * s_inGain;
let lR = in2 * s_inGain;

let level = maximum(abs(lL), abs(lR));
let gr = 1.0;
if (level >= 1.0) gr = 1.0 / level;

// push and take the running minimum over the look-ahead window
s_grMinBuf[s_grMinHead] = gr;
s_grMinHead += 1;
if (s_grMinHead >= s_laSize) s_grMinHead = 0;
let minGr = s_grMinBuf[0];
for (let i: Int = 1; i < s_laSize; i++) minGr = minimum(minGr, s_grMinBuf[i]);

let outGr = 1.0;
if (s_auto < 0.5) {
    if (minGr < s_fastS) s_fastS = minGr;
    else s_fastS = fnOnePole(s_fastS, s_fastG, minGr);
    outGr = s_fastS;
} else {
    if (minGr < s_slowS) s_slowS = fnOnePole(s_slowS, s_slowAtkG, minGr);
    else s_slowS = fnOnePole(s_slowS, s_slowRelG, minGr);

    let combined = minimum(minGr, s_slowS);
    if (combined < s_dynS) s_dynS = combined;
    else s_dynS = fnOnePole(s_dynS, s_dynRelG, combined);
    outGr = s_dynS;

    let dynRelease = pow(1.0 - outGr, 4.0);
    let dynRelG1 = maximum(fnStepG(dynRelease), 0.36787944117);

    let dynFactor = minimum(s_slowS * 2.0 - outGr, 1.0);
    let dynRelG2 = 1.0;
    if (dynFactor < 0.25) dynRelG2 = 1.0;
    else if (dynFactor < 0.5) dynRelG2 = (dynFactor - 0.25) / 0.25 * (s_dynG1 - 1.0) + 1.0;
    else if (dynFactor < 0.71) dynRelG2 = (dynFactor - 0.5) / 0.21 * (s_dynG2 - s_dynG1) + s_dynG1;
    else dynRelG2 = (dynFactor - 0.71) / 0.29 * (s_dynG3 - s_dynG2) + s_dynG2;

    s_dynRelG = dynRelG1 * dynRelG2;
}

// two cascaded box averages
s_ma1Sum -= s_ma1[s_ma1Head];
s_ma1[s_ma1Head] = outGr;
s_ma1Head += 1;
if (s_ma1Head >= s_maSize) s_ma1Head = 0;
s_ma1Sum += outGr;
outGr = s_ma1Sum / s_maSize;

s_ma2Sum -= s_ma2[s_ma2Head];
s_ma2[s_ma2Head] = outGr;
s_ma2Head += 1;
if (s_ma2Head >= s_maSize) s_ma2Head = 0;
s_ma2Sum += outGr;
outGr = s_ma2Sum / s_maSize;

let outL = s_dlyL.next(lL, s_laSize) * outGr * s_outGain;
let outR = s_dlyR.next(lR, s_laSize) * outGr * s_outGain;

// dither
let r1L = 0.0;
let r1R = 0.0;
let r2L = 0.0;
let r2R = 0.0;
if (s_dithType == 1) {
    let a = fnRandType1();
    let b = fnRandType1();
    r1L = (a + b) / 4294967296.0;
    r1R = (a - b) / 4294967296.0;
} else if (s_dithType == 2) {
    s_t2a -= 0.16666667;
    if (s_t2a < -0.16666667) s_t2a = 0.16666667;
    s_t2b = -s_t2b;
    r2L = s_t2a + s_t2b;
    r2R = s_t2a - s_t2b;
}

out1 = fnDitherSample(outL, r1L, r2L, 0.0);
out2 = fnDitherSample(outR, r1R, r2R, 1.0);

s_histIdx += 1;
if (s_histIdx >= 9) s_histIdx -= 9;

frame++;
if (frame >= vectorsize()) frame = 0;
