 * with the source's drive/bias auto-compensation and a 5 Hz DC remover.
 * Metering and GUI omitted.

// Input Gain (dB)
@param({ min: -12, max: 12 }) input_gain = 0;
// Drive
@param({ min: 0, max: 10 }) drive = 2;
// Bias
@param({ min: 0, max: 10 }) bias = 5;
// Boost
@param({ enum: ["off", "on"] }) boost = 0;
// Output Gain (dB)
@param({ min: -12, max: 12 }) output_gain = 0;
@state s_dcL = 0;
@state s_dcR = 0;
@state s_dcK = 0;
@state s_sr = 0;
@state s_inGain = 1;
@state s_outGain = 1;
@state s_drive = 0;
@state s_bias = 0;
@state s_comp = 0;
@state frame: Int = 0;

// EEL tanh() by definition; the clamp only stops exp() overflow and is
// numerically identical to the source everywhere tanh is not exactly +-1.
function fnTanh(x) {
    let a = clamp(x, -20.0, 20.0);
    let e2x = exp(2.0 * a);
    return (e2x - 1.0) / (e2x + 1.0);
}

function fnSaturate(x) {
    let tx = fnTanh(x * s_drive);
    return tx / (tx + s_bias) * s_comp;
}

if (frame == 0) {
    let sr = maximum(samplerate(), 1.0);
    if (sr != s_sr) {
        s_dcK = 5.0 / sr * 2.0 * pi;
        s_sr = sr;
    }
    s_inGain = pow(10.0, input_gain / 20.0);
    s_outGain = pow(10.0, output_gain / 20.0);
    let pDrive = drive / 10.0;
    let pBias = bias / 10.0;
    let drv = pDrive * 5.0 * (boost * 4.0 + 1.0) + 0.5;
    let bia = pBias * pBias * 30.0 + 2.0;
    let xx = 0.1;
    let td = fnTanh(xx * drv);
    s_drive = drv;
    s_bias = bia;
    s_comp = xx * (td + bia) / td;
}

let xl = in1 * s_inGain;
let yl = fnSaturate(xl - s_dcL);
s_dcL += s_dcK * yl;

let xr = in2 * s_inGain;
let yr = fnSaturate(xr - s_dcR);
s_dcR += s_dcK * yr;

out1 = yl * s_outGain;
out2 = yr * s_outGain;

frame++;
if (frame >= vectorsize()) frame = 0;
