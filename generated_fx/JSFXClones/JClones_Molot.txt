 * sidechain = Ext SC — this overrides the usual in3 = pitch Hz convention);
 * out1/out2: stereo out.
 * A valve-flavoured feed-forward compressor: phon-80-shaped sidechain, a
 * diode AR detector, a 9th-order polynomial p(x) gain-clipper with a soft knee,
 * an output saturator and six stereo topologies. Original slider units, ranges
 * and defaults kept. The JClones port omits Molot's limiter, EQ and
 * oversampling stages and so does this one.
 *
 * Table layout (s_k, row-major 10 per row): row i = ratio index 0..98
 * (ratio x10 - 11); cols 0..8 = x^ratio polynomial, col 9 = clamp ceiling.
 * Biquad slots (5 coef + 4 state each):
 *   0 c1 sc2  1 c1 ls1  2 c1 ls2  3 c1 fabs  4 c1 diode
 *   5 c2 sc2  6 c2 ls1  7 c2 ls2  8 c2 fabs  9 c2 diode
 */

@state s_st = new FixedFloat64Array(40);

@state s_tabReady = 0;

@state s_ax1 = 0;
@state s_ax2 = 0;

function fnDbToK(x) {
    return pow(10.0, x / 20.0);
}

function fnSecToK(sec) {
    let sr = maximum(samplerate(), 1.0);
    let f = sr / (2.0 * pi);
    if (sec > 0.0) f = 1.0 / sec;
    return f / sr;
}

function fnBqSet(k, a0, a1, a2, b1, b2) {
    let cb: Int = k * 5;
    s_co[cb] = a0;
    s_co[cb + 1] = a1;
    s_co[cb + 2] = a2;
    s_co[cb + 3] = b1;
    s_co[cb + 4] = b2;
}

function fnBq(x, k) {
    let cb: Int = k * 5;
    let sb: Int = k * 4;
    let y = x * s_co[cb] + s_st[sb] * s_co[cb + 1] + s_st[sb + 1] * s_co[cb + 2] + s_st[sb + 2] * s_co[cb + 3] + s_st[sb + 3] * s_co[cb + 4];
    s_st[sb + 1] = s_st[sb];
    s_st[sb] = x;
    s_st[sb + 3] = s_st[sb + 2];
    s_st[sb + 2] = y;
    return y;
}

function fnBqReset(k, v) {
    let sb: Int = k * 4;
    s_st[sb] = v;
    s_st[sb + 1] = v;
    s_st[sb + 2] = v;
    s_st[sb + 3] = v;
}

function fnLp2(freqK, q, k) {
    let w0 = freqK * 2.0 * pi;
    let kk = 1.0 / tan(w0 / 2.0);
    let kk2 = kk * kk;
    let b0 = kk2 + kk / q + 1.0;
    fnBqSet(k, 1.0 / b0, 2.0 / b0, 1.0 / b0,
        0.0 - 2.0 * (1.0 - kk2) / b0, 0.0 - (kk2 - kk / q + 1.0) / b0);
}

function fnHp2(freqK, q, k) {
    let w0 = freqK * 2.0 * pi;
    let cw = cos(w0);
    let al = sin(w0) / (2.0 * q);
    let d = 1.0 + al;
    fnBqSet(k, ((1.0 + cw) / 2.0) / d, 0.0 - (1.0 + cw) / d, ((1.0 + cw) / 2.0) / d,
        0.0 - (0.0 - 2.0 * cw) / d, 0.0 - (1.0 - al) / d);
}

function fnLs2(freqK, q, aLin, k) {
    let a = sqrt(aLin);
    let w0 = freqK * 2.0 * pi;
    let cw = cos(w0);
    let al = sin(w0) / (2.0 * q);
    let sa = sqrt(a);
    let a0 = a * ((a + 1.0) - (a - 1.0) * cw + 2.0 * sa * al);
    let a1 = 2.0 * a * ((a - 1.0) - (a + 1.0) * cw);
    let a2 = a * ((a + 1.0) - (a - 1.0) * cw - 2.0 * sa * al);
    let b0 = (a + 1.0) + (a - 1.0) * cw + 2.0 * sa * al;
    let b1 = 0.0 - 2.0 * ((a - 1.0) + (a + 1.0) * cw);
    let b2 = (a + 1.0) + (a - 1.0) * cw - 2.0 * sa * al;
    fnBqSet(k, a0 / b0, a1 / b0, a2 / b0, 0.0 - b1 / b0, 0.0 - b2 / b0);
}

// LowShelfPhon80: two cascaded 35.15 Hz high-passes whose Q tracks the dial.
function fnPhonQ(x) {
    let f = maximum(x, 23.756);
    let ff = f / 35.15;
    let ff2 = ff * ff;
    return 1.0 / sqrt((sqrt(2.0) - 1.0) * ff2 + 2.0 - 1.0 / ff2);
}

function fnPhonSet(freqHz, c) {
    let sr = maximum(samplerate(), 1.0);
    let q = fnPhonQ(freqHz);
    fnHp2(35.15 / sr, q, c * 5.0 + 1.0);
    fnHp2(35.15 / sr, q, c * 5.0 + 2.0);
}

function fnPhonProc(x, c) {
    return fnBq(fnBq(x, c * 5.0 + 1.0), c * 5.0 + 2.0);
}

function fnDiode(x) {
    if (x < 0.0) return 0.0;
    if (x > 0.025) return x;
    let x2 = x * x;
    let x4 = x2 * x2;
    let x8 = x4 * x4;
    let x12 = x8 * x4;
    return x8 * x * s_ax0 + x8 * x2 * s_ax1 + x8 * x2 * x * s_ax2 +
        x12 * s_ax3 + x12 * x * s_ax4 + x12 * x2 * s_ax5 +
        x12 * x2 * x * s_ax6 + x12 * x4 * s_ax7;
}

// ARFilter: diode charge/discharge detector, then a smoothing low-pass.
function fnArSet(c, attackK, releaseK, signalK, sharp) {
    let fl = signalK;
    if (sharp < 0.5) fl = maximum(attackK, releaseK);
    fnLp2(fl, 0.7071067811865476, c * 5.0 + 4.0);
    s_arFa[c] = 1.0 / (1.0 / (attackK * 2.0 * pi) + 1.0);
    s_arFr[c] = 1.0 / (1.0 / (releaseK * 2.0 * pi) + 1.0);
}

function fnArProc(x, c) {
    let s = s_arS[c];
    let d = s + s_arFa[c] * fnDiode(x - s) + s_arFr[c] * (x - s);
    s_arS[c] = d;
    return fnBq(d, c * 5.0 + 4.0);
}

function fnEnvSet(c, attackK, releaseK, signalK, sharp) {
    fnLp2(signalK, 0.7071067811865476, c * 5.0 + 3.0);
    fnArSet(c, attackK, releaseK, signalK, sharp);
}

function fnEnvAc(x, c) {
    return fnArProc(fnBq(abs(x), c * 5.0 + 3.0), c);
}

function fnEnvDc(x, c) {
    return fnArProc(x, c);
}

// EnvelopeClip: soft-knee x^ratio gain clipper.
function fnEcH2(c, x) {
    let k = c * 7.0;
    if (s_ec[k + 1] <= 0.0) return s_ec[k + 2];
    let ri: Int = s_ec[k];
    let top = s_k[ri * 10 + 9];
    if (x > top) x = top;
    let x2 = x * x;
    let x4 = x2 * x2;
    let p = s_k[ri * 10] + x * s_k[ri * 10 + 1] + x2 * s_k[ri * 10 + 2] + x2 * x * s_k[ri * 10 + 3] +
        x4 * s_k[ri * 10 + 4] + x4 * x * s_k[ri * 10 + 5] + x4 * x2 * s_k[ri * 10 + 6] +
        x4 * x2 * x * s_k[ri * 10 + 7] + x4 * x4 * s_k[ri * 10 + 8];
    return p * s_ec[k + 4];
}

function fnEcDh2(c, x) {
    let k = c * 7.0;
    if (s_ec[k + 1] <= 0.0) return 0.0;
    if (x <= 0.0) return 0.0;
    return fnEcH2(c, x) * s_ec[k + 1] / x;
}

function fnEcH1(c, x) {
    let k = c * 7.0;
    let x2 = x * x;
    return x + x2 * x * s_ec[k + 5] + x2 * x2 * s_ec[k + 6];
}

function fnEcH(c, x) {
    let k = c * 7.0;
    if (x < s_ec[k + 3]) return fnEcH1(c, x);
    return fnEcH2(c, x);
}

function fnEcGr(c, x) {
    if (x < 0.000001) return 1.0;
    return fnEcH(c, x) / x;
}

function fnEcProc(c, x) {
    let k = c * 7.0;
    if (s_ec[k + 1] == 1.0) return 1.0;
    return fnEcGr(c, x);
}

function fnEcSet(c, thresh, knee, ratioIn) {
    let k = c * 7.0;
    let r = ratioIn;
    let rt = 0.0;
    let ridx = 0.0;
    if (r <= 10.9) {
        r = floor(r * 10.0) / 10.0;
        ridx = r * 10.0 - 11.0;
        rt = 1.0 / r;
        if (ridx < 0.0) ridx = 0.0;
        if (ridx >= 99.0) ridx = 98.0;
    }
    s_ec[k + 2] = thresh;
    s_ec[k + 0] = ridx;
    s_ec[k + 1] = rt;
    if (rt > 0.0) s_ec[k + 4] = pow(1.0 / thresh, rt) * thresh;
    let x2 = thresh + (1.0 - knee * 0.95) * thresh;
    s_ec[k + 3] = x2;
    let y2 = fnEcH2(c, x2);
    let dy2 = fnEcDh2(c, x2);
    s_ec[k + 5] = 0.0 - (3.0 * x2 - 4.0 * y2 + x2 * dy2) / (x2 * x2 * x2);
    s_ec[k + 6] = (2.0 * x2 - 3.0 * y2 + x2 * dy2) / (x2 * x2 * x2 * x2);
}

function fnSat(x) {
    let t = s_satThresh;
    let u = x / t;
    if (u < 0.0 - 1.927384823613229) u = 0.0 - 1.927384823613229;
    if (u > 1.925547073273503) u = 1.925547073273503;
    let x2 = u * u;
    let x4 = x2 * x2;
    let y = 1.043653138515045 * u + 0.051012188210767 * x2 + 0.0 - 0.048182108389193 * x2 * u +
        0.0 - 0.076851680685586 * x4 + 0.0 - 0.076851680685586 * x4 * u + 0.019456557521838 * x4 * x2 +
        0.011322488095808 * x4 * x2 * u + 0.0 - 0.001614452827801 * x4 * x4;
    return y * t;
}

function fnCompPre(x, c) {
    let sc = x;
    if (s_useSc[c] >= 0.5) sc = fnPhonProc(x, c);
    else sc = fnBq(x, c * 5.0);
    return fnEnvAc(sc * s_inGain[c], c) * s_detFix[c];
}

function fnCompPost(xDry, env, c) {
    let gr = fnEcProc(c, env);
    let ar = 2.0 - fnEnvDc(2.0 - gr, c);
    let xw = xDry * ar * s_inGain[c];
    if (s_useSat[c] >= 0.5) xw = fnSat(xw);
    return xw * s_gain[c] * (1.0 - s_dryK[c]) + xDry * s_dryK[c];
}

function fnCompProc(x, sc, c) {
    return fnCompPost(x, fnCompPre(sc, c), c);
}

function fnSetEnvelope(attackS, releaseS, scFreq, sharp) {
    let ea = 0.001;
    let er = 0.050;
    let fix = 0.547;
    if (sharp < 0.5) {
        if (attackS > 0.0245) {
            ea = 0.025;
            fix = 2.237;
        } else if (attackS > 0.0095) {
            ea = 0.010;
            fix = 1.536;
        }
    } else {
        ea = 0.00005;
        er = 0.001;
        fix = 0.878;
    }
    let dk = fnDbToK(fix);
    s_detFix[0] = dk;
    s_detFix[1] = dk;
    let sigK = 21500.0 / maximum(samplerate(), 1.0);
    let c = 0.0;
    while (c < 2.0) {
        fnEnvSet(c, fnSecToK(ea), fnSecToK(er), sigK, 0.0);
        c += 1.0;
    }
    if (scFreq > 0.0) {
        fnPhonSet(scFreq, 0.0);
        fnPhonSet(scFreq, 1.0);
        s_useSc[0] = 1.0;
        s_useSc[1] = 1.0;
    } else {
        let sr = maximum(samplerate(), 1.0);
        fnLs2(296.0 / sr, 0.7, fnDbToK(0.0 - 3.02), 0.0);
        fnLs2(296.0 / sr, 0.7, fnDbToK(0.0 - 3.02), 5.0);
        s_useSc[0] = 0.0;
        s_useSc[1] = 0.0;
    }
    c = 0.0;
    while (c < 2.0) {
        fnArSet(c, fnSecToK(attackS), fnSecToK(releaseS), sigK, sharp);
        c += 1.0;
    }
}

function fnSetMode(m) {
    s_stereoMode = m;
    let us = 0.0;
    if (m != 3.0 && m != 4.0) us = s_useSatOn;
    s_useSat[0] = us;
    s_useSat[1] = us;
}

function fnSetSatDrive(driveDb) {
    s_satThresh = fnDbToK(12.0 - driveDb);
}

function fnSetGain(gain, dm, ig) {
    s_gain[0] = gain;
    s_gain[1] = gain;
    s_dryK[0] = dm;
    s_dryK[1] = dm;
    s_inGain[0] = ig;
    s_inGain[1] = ig;
}

function fnProcess(x1, x2, sc1, sc2) {
    let m = s_stereoMode;
    if (m == 0.0) {
        s_y1 = fnCompProc(x1, sc1, 0.0);
        s_y2 = s_y1;
    } else if (m == 1.0) {
        s_y1 = fnCompProc(x1, sc1, 0.0);
        s_y2 = fnCompProc(x2, sc2, 1.0);
    } else if (m == 2.0) {
        let xm = (x1 + x2) / 2.0;
        let xs = (x1 - x2) / 2.0;
        let ym = fnCompProc(xm, (sc1 + sc2) / 2.0, 0.0);
        s_y1 = ym + xs;
        s_y2 = ym - xs;
    } else if (m == 3.0) {
        let xm = (x1 + x2) / 2.0;
        let xs = (x1 - x2) / 2.0;
        let ys = fnCompProc(xs, (sc1 - sc2) / 2.0, 0.0);
        s_y1 = xm + ys;
        s_y2 = xm - ys;
    } else if (m == 4.0) {
        let e0 = fnCompPre((sc1 + sc2) / 2.0, 0.0);
        let e1 = fnCompPre((sc1 - sc2) / 2.0, 1.0);
        let env = maximum(e0, e1);
        let xm = (x1 + x2) / 2.0;
        let xs = (x1 - x2) / 2.0;
        let ym = fnCompPost(xm, env, 0.0);
        let ys = fnCompPost(xs, env, 1.0);
        s_y1 = fnSat(ym + ys);
        s_y2 = fnSat(ym - ys);
    } else {
        let e0 = fnCompPre(sc1, 0.0);
        let e1 = fnCompPre(sc2, 1.0);
        let env = (e0 + e1) / 2.0;
        s_y1 = fnCompPost(x1, env, 0.0);
        s_y2 = fnCompPost(x2, env, 1.0);
    }
}
