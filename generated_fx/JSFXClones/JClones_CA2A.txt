@param({ min: 0, max: 100 }) r37 = 50;

@param({ enum: ["classic", "fast"] }) opto_cell = 1;

@state s_tA1 = 0;
@state s_tA2 = 0;
@state s_tB1 = 0;
@state s_tB2 = 0;
@state s_b1a0 = 1; @state s_b1a1 = 0; @state s_b1a2 = 0; @state s_b1b1 = 0; @state s_b1b2 = 0;

@state s_b4a0 = 1; @state s_b4a1 = 0; @state s_b4a2 = 0; @state s_b4b1 = 0; @state s_b4b2 = 0;

@state s_b8a0 = 1; @state s_b8a1 = 0; @state s_b8a2 = 0; @state s_b8b1 = 0; @state s_b8b2 = 0;
@state s_b1x1 = 0; @state s_b1x2 = 0; @state s_b1y1 = 0; @state s_b1y2 = 0;
@state s_b2x1 = 0; @state s_b2x2 = 0; @state s_b2y1 = 0; @state s_b2y2 = 0;
@state s_b3x1 = 0; @state s_b3x2 = 0; @state s_b3y1 = 0; @state s_b3y2 = 0;
@state s_b4x1 = 0; @state s_b4x2 = 0; @state s_b4y1 = 0; @state s_b4y2 = 0;
@state s_b5x1 = 0; @state s_b5x2 = 0; @state s_b5y1 = 0; @state s_b5y2 = 0;
@state s_b6x1 = 0; @state s_b6x2 = 0; @state s_b6y1 = 0; @state s_b6y2 = 0;
@state s_b7x1 = 0; @state s_b7x2 = 0; @state s_b7y1 = 0; @state s_b7y2 = 0;
@state s_b8x1 = 0; @state s_b8x2 = 0; @state s_b8y1 = 0; @state s_b8y2 = 0;
@state s_rmsK = 0;
@state s_atkK = 1;

@state s_lvlAtk = 1;

@state s_l1A = 0;
@state s_l1R = 0;
@state s_l2A = 0;
@state s_l2R = 0;
@state s_l3A = 0;
@state s_l3R = 0;
@state s_l4R = 0;
@state s_dynA = 0;

@state s_prK = 0.5;

@state s_logRatio = 0.079181246;
@state s_r47 = 0.5;
@state s_outGain = 1;
@state s_classic = 1;
@state s_rmsL = 0;
@state s_rmsR = 0;
@state s_grFilterS = 0;
@state s_scLevelS = 0;

function fnDbToK(x) {
    return pow(10.0, x / 20.0);
}

function fnToDb(x) {
    return 20.0 * log10(maximum(x, 1.0e-30));
}

function fnF1K(msec) {
    // 1 - 0.05^(1/(sr*ms/1000)); the 0 ms case is the mathematical limit 1.0
    if (msec <= 0.0) return 1.0;
    return 1.0 - pow(0.05, 1.0 / (maximum(samplerate(), 1.0) * (msec / 1000.0)));
}

function fnGenLS(freqK, gainDb, q) {
    let w0 = freqK * 2.0 * pi;
    let sw = sin(w0);
    let cw = cos(w0);
    let g1 = fnDbToK(gainDb);
    let g0 = g1 - 1.0;
    let g2 = (g1 * g1 + 1.0) / q - (g1 - 1.0) * (g1 - 1.0);
    let ss = sqrt(maximum(g2, 0.0)) * sw;
    let x1 = g1 + 1.0 + g0 * cw;
    let x2 = g1 + 1.0 - g0 * cw;
    let x3 = g0 - (g1 + 1.0) * cw;
    let b1 = ((g1 + 1.0) * cw + g0) * -2.0 / (ss + x1);
    let b2 = (x1 - ss) / (ss + x1);
    let a0 = (x2 + ss) * g1 / (ss + x1);
    let a1 = x3 * (g1 * 2.0) / (ss + x1);
    let a2 = (x2 - ss) * g1 / (ss + x1);
    s_tA0 = a0;
    s_tA1 = a1;
    s_tA2 = a2;
    s_tB1 = -b1;
    s_tB2 = -b2;
}

function fnGenHS(freqK, gainDb, q) {
    let w0 = freqK * 2.0 * pi;
    let sw = sin(w0);
    let cw = cos(w0);
    let g1 = fnDbToK(gainDb);
    let g0 = g1 - 1.0;
    let g2 = (g1 * g1 + 1.0) / q - (g1 - 1.0) * (g1 - 1.0);
    let ss = sqrt(maximum(g2, 0.0)) * sw;
    let x1 = g1 + 1.0 - g0 * cw;
    let x2 = g1 + 1.0 + g0 * cw;
    let x3 = g0 + (g1 + 1.0) * cw;
    let b1 = (g0 - (g1 + 1.0) * cw) * 2.0 / (ss + x1);
    let b2 = (x1 - ss) / (ss + x1);
    let a0 = (x2 + ss) * g1 / (ss + x1);
    let a1 = x3 * (g1 * -2.0) / (ss + x1);
    let a2 = (x2 - ss) * g1 / (ss + x1);
    s_tA0 = a0;
    s_tA1 = a1;
    s_tA2 = a2;
    s_tB1 = -b1;
    s_tB2 = -b2;
}

function fnGenPK(freqK, gainDb) {
    let w0 = freqK * 2.0 * pi;
    let sw = sin(w0);
    let cw = cos(w0);
    let g1 = fnDbToK(gainDb);
    let x1 = ((freqK * 2.0 + 1.0) * 0.187 / (1.0 - freqK * 2.0) + 0.81) * sw / 0.42;
    let x2 = x1 / g1;
    let x3 = x1 * g1;
    let b1 = cw * -2.0 / (x2 + 1.0);
    let b2 = (1.0 - x2) / (x2 + 1.0);
    let a0 = (x3 + 1.0) / (x2 + 1.0);
    let a1 = cw * -2.0 / (x2 + 1.0);
    let a2 = (1.0 - x3) / (x2 + 1.0);
    s_tA0 = a0;
    s_tA1 = a1;
    s_tA2 = a2;
    s_tB1 = -b1;
    s_tB2 = -b2;
}

function fnBq1(x0) {
    let y0 = x0 * s_b1a0 + s_b1x1 * s_b1a1 + s_b1x2 * s_b1a2 + s_b1y1 * s_b1b1 + s_b1y2 * s_b1b2;
    s_b1x2 = s_b1x1; s_b1x1 = x0; s_b1y2 = s_b1y1; s_b1y1 = y0;
    return y0;
}
function fnBq2(x0) {
    let y0 = x0 * s_b2a0 + s_b2x1 * s_b2a1 + s_b2x2 * s_b2a2 + s_b2y1 * s_b2b1 + s_b2y2 * s_b2b2;
    s_b2x2 = s_b2x1; s_b2x1 = x0; s_b2y2 = s_b2y1; s_b2y1 = y0;
    return y0;
}
function fnBq3(x0) {
    let y0 = x0 * s_b3a0 + s_b3x1 * s_b3a1 + s_b3x2 * s_b3a2 + s_b3y1 * s_b3b1 + s_b3y2 * s_b3b2;
    s_b3x2 = s_b3x1; s_b3x1 = x0; s_b3y2 = s_b3y1; s_b3y1 = y0;
    return y0;
}
function fnBq4(x0) {
    let y0 = x0 * s_b4a0 + s_b4x1 * s_b4a1 + s_b4x2 * s_b4a2 + s_b4y1 * s_b4b1 + s_b4y2 * s_b4b2;
    s_b4x2 = s_b4x1; s_b4x1 = x0; s_b4y2 = s_b4y1; s_b4y1 = y0;
    return y0;
}
function fnBq5(x0) {
    let y0 = x0 * s_b5a0 + s_b5x1 * s_b5a1 + s_b5x2 * s_b5a2 + s_b5y1 * s_b5b1 + s_b5y2 * s_b5b2;
    s_b5x2 = s_b5x1; s_b5x1 = x0; s_b5y2 = s_b5y1; s_b5y1 = y0;
    return y0;
}
function fnBq6(x0) {
    let y0 = x0 * s_b6a0 + s_b6x1 * s_b6a1 + s_b6x2 * s_b6a2 + s_b6y1 * s_b6b1 + s_b6y2 * s_b6b2;
    s_b6x2 = s_b6x1; s_b6x1 = x0; s_b6y2 = s_b6y1; s_b6y1 = y0;
    return y0;
}
function fnBq7(x0) {
    let y0 = x0 * s_b7a0 + s_b7x1 * s_b7a1 + s_b7x2 * s_b7a2 + s_b7y1 * s_b7b1 + s_b7y2 * s_b7b2;
    s_b7x2 = s_b7x1; s_b7x1 = x0; s_b7y2 = s_b7y1; s_b7y1 = y0;
    return y0;
}
function fnBq8(x0) {
    let y0 = x0 * s_b8a0 + s_b8x1 * s_b8a1 + s_b8x2 * s_b8a2 + s_b8y1 * s_b8b1 + s_b8y2 * s_b8b2;
    s_b8x2 = s_b8x1; s_b8x1 = x0; s_b8y2 = s_b8y1; s_b8y1 = y0;
    return y0;
}

function fnRefreshFilters() {
    let sr = maximum(samplerate(), 1.0);
    fnGenLS(10.0 / sr, -9.0, 0.5);
    s_b1a0 = s_tA0; s_b1a1 = s_tA1; s_b1a2 = s_tA2; s_b1b1 = s_tB1; s_b1b2 = s_tB2;
    s_b2a0 = s_tA0; s_b2a1 = s_tA1; s_b2a2 = s_tA2; s_b2b1 = s_tB1; s_b2b2 = s_tB2;

    fnGenHS(17000.0 / sr, -6.0, 0.3);
    s_b3a0 = s_tA0; s_b3a1 = s_tA1; s_b3a2 = s_tA2; s_b3b1 = s_tB1; s_b3b2 = s_tB2;
    s_b4a0 = s_tA0; s_b4a1 = s_tA1; s_b4a2 = s_tA2; s_b4b1 = s_tB1; s_b4b2 = s_tB2;

    fnGenPK(220.0 / sr, s_r47 * -13.0 * 0.5);
    s_b5a0 = s_tA0; s_b5a1 = s_tA1; s_b5a2 = s_tA2; s_b5b1 = s_tB1; s_b5b2 = s_tB2;
    s_b6a0 = s_tA0; s_b6a1 = s_tA1; s_b6a2 = s_tA2; s_b6b1 = s_tB1; s_b6b2 = s_tB2;

    fnGenLS(43.0 / sr, 30.0, 0.6);
    s_b7a0 = s_tA0; s_b7a1 = s_tA1; s_b7a2 = s_tA2; s_b7b1 = s_tB1; s_b7b2 = s_tB2;
    s_b8a0 = s_tA0; s_b8a1 = s_tA1; s_b8a2 = s_tA2; s_b8b1 = s_tB1; s_b8b2 = s_tB2;
}

function fnGetGainReduction(input) {
    if (input < s_gr1) s_gr1 += (input - s_gr1) * s_l1A;
    else s_gr1 += (input - s_gr1) * s_l1R;
    s_gr1 = maximum(s_gr1, 0.7);

    if (input < s_gr2) s_gr2 += (input - s_gr2) * s_l2A;
    else s_gr2 += (minimum(input, s_gr1) - s_gr2) * s_l2R;
    s_gr2 = maximum(s_gr2, 0.4);

    if (input < s_gr4) s_gr3 += (input - s_gr3) * s_l3A;
    else s_gr3 += (input - s_gr3) * s_l3R;

    let sv = (s_gr3 - 0.5) * 2.0;
    let maxGr = input * 12.0;
    if (s_classic < 0.5) {
        sv /= 5.5;
        maxGr = input * 6.0;
        s_dynS = 0.9;
    }

    let attackMsec = 70.0 * maximum(0.3, sv);
    let dynAtk = fnF1K(attackMsec);

    if (input < s_gr4) {
        if (input + 0.0001 < s_gr4) s_dynS += (1.0 - s_dynS) * s_dynA;
        s_gr4 += (input - s_gr4) * dynAtk * s_dynS;
        if (s_gr4 >= maxGr) s_gr4 = maxGr;
    } else {
        s_dynS += (0.0 - s_dynS) * s_dynR;
        s_gr4 += (minimum(input, s_gr2) - s_gr4) * s_l4R;
    }
    return s_gr4;
}

function fnSaturate(x) {
    if (x >= 0.0) return x / (pow(x * 0.05, 1.3) + 1.0);
    return x / (1.0 - x * 0.005);
}

function fnProcess(inL, inR) {
    let scL = inL;
    let scR = inR;
    scL = fnBq5(fnBq3(fnBq1(scL)));
    scR = fnBq6(fnBq4(fnBq2(scR)));

    s_rmsL += (scL * scL - s_rmsL) * s_rmsK;
    s_rmsR += (scR * scR - s_rmsR) * s_rmsK;

    let lvL = sqrt(s_rmsL) * 0.25 + abs(scL) * 0.75;
    let lvR = sqrt(s_rmsR) * 0.25 + abs(scR) * 0.75;
    let lvMax = maximum(maximum(lvL, lvR), 1.0e-10);

    let lvDb = fnToDb(lvMax) - s_grA;
    let valueDb = log10(pow(s_grB, lvDb) + 1.0) / s_grC + s_grA;
    let value = fnDbToK(valueDb);
    let grLevel = (value - s_grD) / lvMax;

    let k = s_relK;
    if (grLevel > s_grFilterS) k = s_atkK;
    s_grFilterS += (grLevel - s_grFilterS) * k;

    scL *= s_grFilterS;
    scR *= s_grFilterS;

    let bL = fnBq7(scL);
    let bR = fnBq8(scR);

    let lvSc = maximum(maximum(abs(scL), abs(scR)), 1.0e-10);
    let lvB = maximum(maximum(abs(bL), abs(bR)), 1.0e-10);

    k = s_lvlRel;
    if (lvSc > s_scLevelS) k = s_lvlAtk;
    s_scLevelS += (lvSc - s_scLevelS) * k;

    k = s_lvlRel;
    if (lvB > s_bLevelS) k = s_lvlAtk;
    s_bLevelS += (lvB - s_bLevelS) * k;

    let v1 = fnToDb(s_bLevelS);
    let v2 = log10(pow(s_ratio, v1 - s_thrDb) + 1.0);
    let v3 = fnDbToK(v2 / s_logRatio + s_thrDb);
    let v4 = s_thrLin / v3 * s_scLevelS;
    let expo = 0.79;
    if (s_isLimit >= 0.5) expo = 0.94;
    let v5 = pow(s_grD / (s_grD + v4), expo);

    let gr = fnGetGainReduction(v5);

    let satX1 = fnSaturate(inL);
    let dX1 = satX1 - inL;
    let satX2 = fnSaturate(inR);
    let dX2 = satX2 - inR;

    let gainScale = s_prK * s_prK * 14.0 + 1.0;
    let y1 = inL * gr * gainScale;
    let y2 = inR * gr * gainScale;

    let satY1 = fnSaturate(y1);
    let dY1 = satY1 - y1;
    let satY2 = fnSaturate(y2);
    let dY2 = satY2 - y2;

    let dGr = maximum((1.0 - pow(gr * 0.98, 0.1)) * 18.0, 0.0);

    s_y1 = ((dY1 * 0.5 + dX1 * 0.5) * dGr + inL) * gr * s_outGain;
    s_y2 = ((dY2 * 0.5 + dX2 * 0.5) * dGr + inR) * gr * s_outGain;
}

if (frame == 0) {
    let sr = maximum(samplerate(), 1.0);
    if (sr != s_sr) {
        s_rmsK = fnF1K(36.0);
        s_atkK = fnF1K(0.0);
        s_relK = fnF1K(20.0);
        s_lvlAtk = fnF1K(0.0);
        s_lvlRel = fnF1K(20.0);
        s_l1A = fnF1K(35000.0);
        s_l1R = fnF1K(9000.0);
        s_l2A = fnF1K(1400.0);
        s_l2R = fnF1K(900.0);
        s_l3A = fnF1K(1100.0);
        s_l3R = fnF1K(12000.0);
        s_l4R = fnF1K(70.0);
        s_dynA = fnF1K(15.0);
        s_dynR = fnF1K(2.0);
        s_sr = sr;
    }

    let g = gain / 100.0;
    s_outGain = pow((pow(24.0, g) - 1.0) / 23.0, 1.35) * 24.0;

    let pr = peak_reduction / 100.0;
    s_grA = (pr * 1.03 - 0.03) * -49.0;
    s_thrDb = s_grA + 47.0;
    s_thrLin = fnDbToK(s_thrDb);
    let a = pow(2.0, 1.0 - pr);
    s_grB = a + 1.2;
    s_grC = log10(s_grB);
    s_grD = fnDbToK(s_grA);
    s_prK = pr;

    s_isLimit = mode;
    s_ratio = 1.2;
    if (s_isLimit >= 0.5) s_ratio = 1.3;
    s_logRatio = log10(s_ratio);

    s_r47 = 1.0 - r37 / 100.0;
    s_classic = 1.0;
    if (opto_cell >= 0.5) s_classic = 0.0;

    fnRefreshFilters();
}

fnProcess(in1, in2);
out1 = s_y1;
out2 = s_y2;

frame++;
if (frame >= vectorsize()) frame = 0;
