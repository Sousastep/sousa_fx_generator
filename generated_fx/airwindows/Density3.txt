// Density: VST A = (density + 1) / 5, as the VST displays it (0 = clean; VST default -1)
@param({ min: -1, max: 4 }) density = 0.0;
// Highpass: VST B, one-pole coefficient B^3 at 44.1 kHz; 0 = off
@param({ min: 0, max: 4000, exponent: 3 }) highpass_hz = 0.0;
// Output: VST C = dbtoa(output_db)
@param({ min: -36, max: 0 }) output_db = 0.0;
// Dry/Wet: VST D = mix_pct / 100
@param({ min: 0, max: 100 }) mix_pct = 100.0;

@state iirSampleL = 0;
@state iirSampleR = 0;
@state fpdL = 305419896.0;
@state fpdR = 2271560481.0;
@state overallscale = 0;
@state densityAmt = 0;
@state iirAmount = 0;
@state output = 0;
@state wet = 0;
@state frame: Int = 0;
// Xorshift32 on RNBO's int32 bit operators (logical >> via mask), returned
// as an unsigned 0..2^32-1 value; bit-identical to the airwindows uint32_t.
function unsigned32(x) {
    return x < 0 ? x + 4294967296.0 : x;
}
function advanceRandom(x) {
    let a: Int = x;
    a = a ^ (a << 13);
    a = a ^ ((a >> 17) & 32767);
    a = a ^ (a << 5);
    let r = a;
    return unsigned32(r);
}

function init() {

	iirSampleL = 0.0;
	iirSampleR = 0.0;

}
function dspsetup() { frame = 0; }

if (frame == 0) {
overallscale = 1.0;
	overallscale /= 44100.0;
	overallscale *= samplerate();
	densityAmt = clamp(density, -1.0, 4.0) + 1.0;
	let hpHz = clamp(highpass_hz, 0.0, 4000.0);
	iirAmount = 0.0;
	if (hpHz > 0.0) iirAmount = 1.0 - exp(-2.0 * PI * minimum(hpHz, samplerate() * 0.45) / samplerate());
	if (iirAmount == 0.0) {iirSampleL = 0.0; iirSampleR = 0.0;}
	output = dbtoa(clamp(output_db, -36.0, 0.0));
	wet = clamp(mix_pct, 0.0, 100.0) * 0.01;

}
let sampleFrames = vectorsize() - frame - 1;
let inFramesToProcess = vectorsize();

		let inputSampleL = in1;
		let inputSampleR = in2;
		if (abs(inputSampleL)<1.18e-23) inputSampleL = fpdL * 1.18e-17;
		if (abs(inputSampleR)<1.18e-23) inputSampleR = fpdR * 1.18e-17;
		let drySampleL = inputSampleL;
		let drySampleR = inputSampleR;
		
		iirSampleL = (iirSampleL * (1.0 - iirAmount)) + (inputSampleL * iirAmount);
		inputSampleL -= iirSampleL;
		iirSampleR = (iirSampleR * (1.0 - iirAmount)) + (inputSampleR * iirAmount);
		inputSampleR -= iirSampleR;
		
		let altered = inputSampleL;
		if (densityAmt > 1.0) {
			altered = maximum(minimum(inputSampleL*densityAmt*(PI*0.5),(PI*0.5)),-(PI*0.5));
			let X = altered*altered;
			let temp = altered*X;
			altered -= (temp / 6.0); temp *= X;
			altered += (temp / 120.0); temp *= X;
			altered -= (temp / 5040.0); temp *= X;
			altered += (temp / 362880.0); temp *= X;
			altered -= (temp / 39916800.0);
		}
		if (densityAmt < 1.0) {
			altered = maximum(minimum(inputSampleL,1.0),-1.0);
			let polarity = altered;
			let X = inputSampleL * altered;
			let temp = X;
			altered = (temp / 2.0); temp *= X;
			altered -= (temp / 24.0); temp *= X;
			altered += (temp / 720.0); temp *= X;
			altered -= (temp / 40320.0); temp *= X;
			altered += (temp / 3628800.0);
			altered *= ((polarity<0.0)?-1.0:1.0);
		}
		if (densityAmt > 2.0) inputSampleL = altered;
		else inputSampleL = (inputSampleL*(1.0-abs(densityAmt-1.0)))+(altered*abs(densityAmt-1.0));
		
		altered = inputSampleR;
		if (densityAmt > 1.0) {
			altered = maximum(minimum(inputSampleR*densityAmt*(PI*0.5),(PI*0.5)),-(PI*0.5));
			let X = altered*altered;
			let temp = altered*X;
			altered -= (temp / 6.0); temp *= X;
			altered += (temp / 120.0); temp *= X;
			altered -= (temp / 5040.0); temp *= X;
			altered += (temp / 362880.0); temp *= X;
			altered -= (temp / 39916800.0);
		}
		if (densityAmt < 1.0) {
			altered = maximum(minimum(inputSampleR,1.0),-1.0);
			let polarity = altered;
			let X = inputSampleR * altered;
			let temp = X;
			altered = (temp / 2.0); temp *= X;
			altered -= (temp / 24.0); temp *= X;
			altered += (temp / 720.0); temp *= X;
			altered -= (temp / 40320.0); temp *= X;
			altered += (temp / 3628800.0);
			altered *= ((polarity<0.0)?-1.0:1.0);
		}
		if (densityAmt > 2.0) inputSampleR = altered;
		else inputSampleR = (inputSampleR*(1.0-abs(densityAmt-1.0)))+(altered*abs(densityAmt-1.0));
		
		inputSampleL = (drySampleL*(1.0-wet))+(inputSampleL*output*wet);
		inputSampleR = (drySampleR*(1.0-wet))+(inputSampleR*output*wet);

		fpdL = advanceRandom(fpdL);

		fpdR = advanceRandom(fpdR);
		if (unsigned32(fpdL-fpdR) < 1073741824 || unsigned32(fpdR-fpdL) < 1073741824) {
			fpdR = advanceRandom(fpdR);}

out1 = inputSampleL;
out2 = inputSampleR;
frame++;
if (frame >= vectorsize()) frame = 0;
