// Compress: VST A = compress_db / 14 (input drive and compression depth)
@param({ min: 0, max: 14 }) compress_db = 0.0;
// Output: VST B = dbtoa(output_db) / 2
@param({ min: -24, max: 6 }) output_db = 0.0;
// Dry/Wet: VST C = mix_pct / 100
@param({ min: 0, max: 100 }) mix_pct = 100.0;

@state controlAposL = 0;
@state controlAnegL = 0;
@state controlBposL = 0;
@state controlBnegL = 0;
@state targetposL = 0;
@state targetnegL = 0;
@state lastOutputL = 0;
@state controlAposR = 0;
@state controlAnegR = 0;
@state controlBposR = 0;
@state controlBnegR = 0;
@state targetposR = 0;
@state targetnegR = 0;
@state lastOutputR = 0;
@state flip = 0;
@state overallscale = 0;
@state inputgain = 0;
@state compfactor = 0;
@state compscaled = 0;
@state output = 0;
@state wet = 0;
@state outputgain = 0;
@state frame: Int = 0;

function init() {

	controlAposL = 1.0;
	controlAnegL = 1.0;
	controlBposL = 1.0;
	controlBnegL = 1.0;
	targetposL = 1.0;
	targetnegL = 1.0;
	lastOutputL = 0.0;

	controlAposR = 1.0;
	controlAnegR = 1.0;
	controlBposR = 1.0;
	controlBnegR = 1.0;
	targetposR = 1.0;
	targetnegR = 1.0;
	lastOutputR = 0.0;

	flip = 0;

}
function dspsetup() { frame = 0; }

if (frame == 0) {
overallscale = 1.0;
	overallscale /= 44100.0;
	overallscale *= samplerate();
	
	let compA = clamp(compress_db, 0.0, 14.0) / 14.0;
	inputgain = dbtoa(compA * 14.0);
	compfactor = 0.012 * (compA / 135.0);
	compscaled = compfactor / overallscale;
	output = dbtoa(clamp(output_db, -24.0, 6.0));
	wet = clamp(mix_pct, 0.0, 100.0) * 0.01;
	
	outputgain = inputgain;
	outputgain -= 1.0;
	outputgain /= 1.5;
	outputgain += 1.0;

}
let sampleFrames = vectorsize() - frame - 1;
let inFramesToProcess = vectorsize();

		let inputSampleL = in1;
		let inputSampleR = in2;

		// Legacy ~1e-16 residue noise omitted; compression is unchanged.
let drySampleL = inputSampleL;
		let drySampleR = inputSampleR;
		
		inputSampleL *= inputgain;
		inputSampleR *= inputgain;
		
		let divisor = compscaled / (1.0+abs(lastOutputL));
		let remainder = divisor;
		divisor = 1.0 - divisor;

		let inputposL = inputSampleL + 1.0;
		if (inputposL < 0.0) inputposL = 0.0;
		let outputposL = inputposL / 2.0;
		if (outputposL > 1.0) outputposL = 1.0;		
		inputposL *= inputposL;
		targetposL *= divisor;
		targetposL += (inputposL * remainder);
		
		let inputnegL = (-inputSampleL) + 1.0;
		if (inputnegL < 0.0) inputnegL = 0.0;
		let outputnegL = inputnegL / 2.0;
		if (outputnegL > 1.0) outputnegL = 1.0;		
		inputnegL *= inputnegL;
		targetnegL *= divisor;
		targetnegL += (inputnegL * remainder);

		if (inputSampleL > 0)
		{ 
			let invposL = 1.0/targetposL;
			let calcposL = invposL*invposL;
			if (flip)
			{
				controlAposL *= divisor;
				controlAposL += (calcposL*remainder);
				
			}
			else
			{
				controlBposL *= divisor;
				controlBposL += (calcposL*remainder);
			}
		}
		else
		{ 
			let invnegL = 1.0/targetnegL;
			let calcnegL = invnegL*invnegL;
			if (flip)
			{
				controlAnegL *= divisor;
				controlAnegL += (calcnegL*remainder);
			}
			else
			{
				controlBnegL *= divisor;
				controlBnegL += (calcnegL*remainder);
			}
		}

		divisor = compscaled / (1.0+abs(lastOutputR));
		remainder = divisor;
		divisor = 1.0 - divisor;

		let inputposR = inputSampleR + 1.0;
		if (inputposR < 0.0) inputposR = 0.0;
		let outputposR = inputposR / 2.0;
		if (outputposR > 1.0) outputposR = 1.0;		
		inputposR *= inputposR;
		targetposR *= divisor;
		targetposR += (inputposR * remainder);
		
		let inputnegR = (-inputSampleR) + 1.0;
		if (inputnegR < 0.0) inputnegR = 0.0;
		let outputnegR = inputnegR / 2.0;
		if (outputnegR > 1.0) outputnegR = 1.0;		
		inputnegR *= inputnegR;
		targetnegR *= divisor;
		targetnegR += (inputnegR * remainder);

		if (inputSampleR > 0)
		{ 
			let invposR = 1.0/targetposR;
			let calcposR = invposR*invposR;
			if (flip)
			{
				controlAposR *= divisor;
				controlAposR += (calcposR*remainder);
				
			}
			else
			{
				controlBposR *= divisor;
				controlBposR += (calcposR*remainder);
			}
		}
		else
		{ 
			let invnegR = 1.0/targetnegR;
			let calcnegR = invnegR*invnegR;
			if (flip)
			{
				controlAnegR *= divisor;
				controlAnegR += (calcnegR*remainder);
			}
			else
			{
				controlBnegR *= divisor;
				controlBnegR += (calcnegR*remainder);
			}
		}

		let totalmultiplierL;
		let totalmultiplierR;
		if (flip)
		{
			totalmultiplierL = (controlAposL * outputposL) + (controlAnegL * outputnegL);
			totalmultiplierR = (controlAposR * outputposR) + (controlAnegR * outputnegR);
		}
		else
		{
			totalmultiplierL = (controlBposL * outputposL) + (controlBnegL * outputnegL);
			totalmultiplierR = (controlBposR * outputposR) + (controlBnegR * outputnegR);
		}

		inputSampleL *= totalmultiplierL;
		inputSampleL /= outputgain;
		
		inputSampleR *= totalmultiplierR;
		inputSampleR /= outputgain;
		
		if (output != 1.0) {
			inputSampleL *= output;
			inputSampleR *= output;
		}
		
		if (wet !=1.0) {
			inputSampleL = (inputSampleL * wet) + (drySampleL * (1.0-wet));
			inputSampleR = (inputSampleR * wet) + (drySampleR * (1.0-wet));
		}
		
		lastOutputL = inputSampleL;
		lastOutputR = inputSampleR;

		flip = !flip;

out1 = inputSampleL;
out2 = inputSampleR;
frame++;
if (frame >= vectorsize()) frame = 0;
