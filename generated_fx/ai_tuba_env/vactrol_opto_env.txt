// ============================================================================
// vactrol_opto_env.txt — vactrol LED + CdS photoresistor envelope
//
// From: J. Najnudel, R. Müller, T. Hélie & D. Roze, "Power-Balanced Dynamic
//       Modeling of Vactrols: Application to a VTL5C3/2", DAFx-23, paper 50,
//       §1 (LDR turn-on is fast, turn-off is hundreds of ms and slows after
//       sustained light), §3 Eqs. 3-4 (photocarrier storage and recombination),
//       §5 (identified VTL5C3/2 resistance law).
//
// This is a reduced signal-level model of that cell, not the full port-
// Hamiltonian circuit: rectified audio drives the LED, the LDR conductance
// follows light^gamma, and the release time is stretched by a short exposure
// memory so a sustained note leaves the cell "charged" and slower to fall.
// The attack shortens as the LED gets brighter, which is the vactrol's
// characteristic program dependence. Output is normalized photoconductance:
// 0 = dark (R_dark), 1 = full light (R_light).
//
// Inlets:
//   in1: tuba audio (mono)
// Outlets:
//   out1: normalized LDR conductance, 0..1 (audio-rate), scaled by out_db
// ============================================================================

@param({ min: -24.0, max: 24.0 }) sensitivity_db = 0.0; // LED drive trim
@param({ min: 20.0, max: 2000.0 }) release_ms = 300.0;  // dark decay
@param({ min: 0.0, max: 1.0 }) program = 0.6;           // exposure memory depth
@param({ min: 0.5, max: 1.5 }) curve = 0.8;             // LDR gamma
@param({ min: -24.0, max: 24.0 }) out_db = 0.0;         // output level

@state ldrCond = 0.0;
@state exposure = 0.0;

let sr = maximum(samplerate(), 1.0);
let drive = minimum(abs(in1) * dbtoa(clamp(sensitivity_db, -24.0, 24.0)), 1.0);
let target = pow(drive, clamp(curve, 0.5, 1.5));

// 0.8 s exposure memory: how much light the cell has recently seen
exposure = drive + exp(0.0 - 1.0 / (0.8 * sr)) * (exposure - drive);
let prog = clamp(program, 0.0, 1.0) * clamp(exposure * 2.0, 0.0, 1.0);

// Attack shortens with brightness; release stretches after sustained light
let attMs = 8.0 / (1.0 + 6.0 * drive);
let relMs = maximum(release_ms, 20.0) * (1.0 + 4.0 * prog);
let tauMs = (target > ldrCond) ? attMs : relMs;
ldrCond = target + exp(0.0 - 1.0 / (tauMs * 0.001 * sr)) * (ldrCond - target);

out1 = ldrCond * dbtoa(clamp(out_db, -24.0, 24.0));
