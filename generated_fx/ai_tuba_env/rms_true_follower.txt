// ============================================================================
// rms_true_follower.txt — true-RMS energy envelope
//
// From: F. Floru, "Attack and Release Time Constants in RMS-Based Feedback
//       Compressors", AES 104th Convention 1998, preprint 4703, §1-2
//       (true RMS detector, linear- vs log-domain output).
//       P. Dutilleux, DAFx-98, §5.1-5.2, §5.4 (squarer detector + averager;
//       RMS detector + AR-averager as the preferred amplitude follower).
//
// Square the input, average the power with a symmetric RMS window, take the
// square root, then let an attack/release smoother shape how fast the reading
// is allowed to move. The symmetric window is what makes it true RMS: an
// attack/release average of the square is actually a peak-biased detector at
// low frequencies (a 100 Hz sine with 10/500 ms reads about +2.7 dB high).
// The AR stage after the root gives the musical ballistics without that bias.
//
// Inlets:
//   in1: tuba audio (mono)
// Outlets:
//   out1: RMS envelope control signal, 0..1 linear (audio-rate), scaled by out_db
// ============================================================================

@param({ min: 5.0, max: 500.0 }) rms_ms = 50.0;       // RMS measurement window
@param({ min: 0.5, max: 200.0 }) attack_ms = 5.0;    // reading attack
@param({ min: 10.0, max: 4000.0 }) release_ms = 500.0; // reading release
@param({ min: -24.0, max: 24.0 }) out_db = 0.0;      // output level

@state powerAvg = 0.0;
@state rmsSm = 0.0;

let sr = maximum(samplerate(), 1.0);
let tRms = maximum(rms_ms, 5.0) * 0.001;
let cRms = exp(0.0 - 1.0 / (tRms * sr));

let x = in1;
let p = x * x;
powerAvg = p + cRms * (powerAvg - p);
let rmsRaw = sqrt(maximum(powerAvg, 0.0));

let tAtt = maximum(attack_ms, 0.05) * 0.001;
let tRel = maximum(release_ms, 0.05) * 0.001;
let cAtt = exp(0.0 - 1.0 / (tAtt * sr));
let cRel = exp(0.0 - 1.0 / (tRel * sr));
let coef = (rmsRaw > rmsSm) ? cAtt : cRel;
rmsSm = rmsRaw + coef * (rmsSm - rmsRaw);

out1 = rmsSm * dbtoa(clamp(out_db, -24.0, 24.0));
