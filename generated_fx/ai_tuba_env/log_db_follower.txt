// ============================================================================
// log_db_follower.txt — envelope followed in the decibel domain
//
// From: F. Floru, "Attack and Release Time Constants in RMS-Based Feedback
//       Compressors", AES 104th Convention 1998, preprint 4703, §2.2.1-2.2.2
//       (logarithmic-output RMS detector and dB-linear integrator) and §4.3
//       (the log filter's diode gives a slower attack than a linear RC filter).
//       P. Dutilleux, DAFx-98, §5.2 (one-pole averager time constant).
//
// Rectify, convert to dB, run the attack/release one-pole on the dB value, then
// convert back. Because the averaging happens in dB, the release falls at a
// constant number of dB per second instead of a constant fraction of the
// amplitude: a long tuba note fades like a fader or an opto cell, not like an
// RC discharge. The floor_db clamp keeps log(0) out of the math and subtracts
// the floor so silence is exactly zero on out1.
//
// Inlets:
//   in1: tuba audio (mono)
// Outlets:
//   out1: envelope control signal, 0..1 linear (audio-rate), scaled by out_db
// ============================================================================

@param({ min: 0.5, max: 500.0 }) attack_ms = 5.0;     // dB attack time
@param({ min: 10.0, max: 5000.0 }) release_ms = 500.0; // dB release time
@param({ min: -90.0, max: -20.0 }) floor_db = -60.0;   // silence floor
@param({ min: -24.0, max: 24.0 }) out_db = 0.0;        // output level

@state envDb = -120.0;

let sr = maximum(samplerate(), 1.0);
let tAtt = maximum(attack_ms, 0.05) * 0.001;
let tRel = maximum(release_ms, 0.05) * 0.001;
let cAtt = exp(0.0 - 1.0 / (tAtt * sr));
let cRel = exp(0.0 - 1.0 / (tRel * sr));

let x = maximum(abs(in1), 0.000001);
let xDb = log(x) * 8.685889638;   // 20 * log10(x)
let flrDb = clamp(floor_db, -90.0, -20.0);
let target = maximum(xDb, flrDb);

let coef = (target > envDb) ? cAtt : cRel;
envDb = target + coef * (envDb - target);

let flrAmp = dbtoa(flrDb);
let amp = maximum(0.0, dbtoa(envDb) - flrAmp);
out1 = amp * dbtoa(clamp(out_db, -24.0, 24.0));
