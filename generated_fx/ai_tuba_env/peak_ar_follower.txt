// ============================================================================
// peak_ar_follower.txt — full-wave peak envelope with independent attack/release
//
// From: F. Floru, "Attack and Release Time Constants in RMS-Based Feedback
//       Compressors", AES 104th Convention 1998, preprint 4703, §1-2
//       (peak vs average vs true-RMS detectors).
//       P. Dutilleux, "Filters, Delays, Modulations and Demodulations:
//       A Tutorial", DAFx-98, §5.1-5.2 (half/full-wave detectors and the
//       AR-averager with McNally's two fixed coefficients).
//
// The detector is a full-wave rectifier followed by one exponential averager
// whose coefficient is chosen per sample from the attack/release pair. Output
// is the linear peak envelope, the fast and literal reading of the tuba's
// loudness; the release is the only memory. No look-ahead, no limiter.
//
// Inlets:
//   in1: tuba audio (mono)
// Outlets:
//   out1: envelope control signal, 0..1 linear (audio-rate), scaled by out_db
// ============================================================================

@param({ min: 0.1, max: 100.0 }) attack_ms = 5.0;    // attack time
@param({ min: 5.0, max: 2000.0 }) release_ms = 250.0; // release time
@param({ min: -24.0, max: 24.0 }) out_db = 0.0;      // output level

@state envPeak = 0.0;

let sr = maximum(samplerate(), 1.0);
let tAtt = maximum(attack_ms, 0.05) * 0.001;
let tRel = maximum(release_ms, 0.05) * 0.001;
let cAtt = exp(0.0 - 1.0 / (tAtt * sr));
let cRel = exp(0.0 - 1.0 / (tRel * sr));

let x = abs(in1);
let coef = (x > envPeak) ? cAtt : cRel;
envPeak = x + coef * (envPeak - x);

out1 = envPeak * dbtoa(clamp(out_db, -24.0, 24.0));
