// ============================================================================
// cascade_env_follower.txt — cascaded exponential-smoother envelope
//
// From: D. Sanfilippo, "Envelope Following via Cascaded Exponential Smoothers
//       for Low-Distortion Peak Limiting and Maximisation", SMC-22, §2.2
//       Eqs. 2-4 (M cascaded one-poles with per-stage attack/release) and
//       §2.4 (overshoot / THD of the cascaded attack-release smoother).
//
// One attack/release one-pole has a corner in its slope at every attack-to-
// release transition. Cascading M identical stages makes the envelope
// continuous in its derivatives up to order M-1, so when this control signal
// multiplies a gain, a filter cutoff or a pitch it produces far less sideband
// splatter than a single-stage follower. The paper normalizes the stages by
// their cutoff (C_M); here the stages are normalized by step time instead, so
// attack_ms/release_ms are the *composite* 63% times a musician expects:
// q_M = 1.0, 2.15, 3.26, 4.35, 5.43, 6.51, 7.57, 8.64 for M = 1..8.
//
// Inlets:
//   in1: tuba audio (mono)
// Outlets:
//   out1: envelope control signal, 0..1 linear (audio-rate), scaled by out_db
// ============================================================================

@param({ min: 0.1, max: 100.0 }) attack_ms = 5.0;    // composite attack time
@param({ min: 5.0, max: 2000.0 }) release_ms = 300.0; // composite release time
@param({ min: 2, max: 8, steps: 7 }) stages = 4;      // cascade order
@param({ min: -24.0, max: 24.0 }) out_db = 0.0;      // output level

@state casc = new FixedSampleArray(8);
@state qTab = new FixedFloat32Array([1.0, 2.1457, 3.2577, 4.3513, 5.4333, 6.5072, 7.5748, 8.6377]);

let sr = maximum(samplerate(), 1.0);
let tAtt = maximum(attack_ms, 0.05) * 0.001;
let tRel = maximum(release_ms, 0.05) * 0.001;

let nSt: Int = clamp(floor(stages + 0.5), 2, 8);
let qM = qTab[nSt - 1];
let aAtt = exp(0.0 - qM / (tAtt * sr));
let aRel = exp(0.0 - qM / (tRel * sr));

let x = abs(in1);
for (let i: Int = 0; i < nSt; i++) {
    let s = casc[i];
    let a = (x > s) ? aAtt : aRel;
    s = x + a * (s - x);
    casc[i] = s;
    x = s;
}

out1 = x * dbtoa(clamp(out_db, -24.0, 24.0));
