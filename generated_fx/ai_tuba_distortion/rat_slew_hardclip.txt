// ============================================================================
// rat_slew_hardclip.txt — RAT-style distortion: a slow op amp whose bandwidth
//                         shrinks as its gain rises and whose output
//                         slew-limits, into hard silicon diodes to ground,
//                         then a passive lowpass "filter" knob
//
// From: D. T. Yeh, J. S. Abel, J. O. Smith, "Simplified, Physically-Informed
//       Models of Distortion and Overdrive Guitar Effects Pedals", DAFx-07
//       p189: §3.3 (op amp gain stage: 1 + Zf/Zi with RC legs to ground, the
//       closed-loop bandwidth falls as GBW / gain), §3.4 (diode clipper to
//       ground after a series resistor), §3.5 (tone stage lowpass);
//       D. T. Yeh, "Simulation of the Diode Limiter in Guitar Distortion
//       Circuits by Numerical Solution of ODEs", DAFx-07 p197 §1.3 (the
//       diode clipper equation: near-hard limiting at ~0.6 V);
//       K. J. Werner, E. Azelborn, "Antialiasing Piecewise Polynomial
//       Waveshapers", DAFx-23 paper 61 §3 (ADAA of piecewise-linear
//       shapers: the hard clipper, F(u) = u^2/2 inside, |u| - 1/2 outside).
//
// Signal (volts, 1.0 digital = 1 V):
//   v    = x + G * hp60(x)                     (G = dbtoa(drive_db) - 1)
//   v    = lp(v, 1 MHz / (1 + G))              (gain-bandwidth limit)
//   v    = 4.5 * tanh(v / 4.5)                 (+-4.5 V rails, 9 V battery)
//   v    = slew(v, slew_v_us)                  (slew rate; 0.3 = LM308 spec)
//   d    = 0.6 * hardclip_adaa(v / 0.6)        (1N914 pair to ground)
//   wet  = lp(d, filter_hz) / 0.6 * 0.5        (filter knob, -6 dBFS peak)
//
// Inlets:
//   in1 / in2: stereo tuba
// Outlets:
//   out1 / out2: stereo
// No safety limiter, no metering.
// ============================================================================

// Distortion (dB of op-amp gain)
@param({ min: 0, max: 60 }) drive_db = 36;
// Op-amp slew rate (V/us)
@param({ min: 0.001, max: 0.3, exponent: 3 }) slew_v_us = 0.005;
// Filter lowpass (Hz)
@param({ min: 400, max: 12000, exponent: 2 }) filter_hz = 3000;
// Dry/wet
@param({ min: 0, max: 1 }) mix = 1;
// Output gain (dB)
@param({ min: -36, max: 12 }) output_db = 0;

// TPT one-pole states: 0/1 hp60, 2/3 gbw lp, 4/5 filter lp, 6/7 output coupling cap
@state zs = new FixedSampleArray(8);
@state prevU = new FixedSampleArray(2);
@state slewY = new FixedSampleArray(2);

// TPT one-pole lowpass on state slot idx; gg = g / (1 + g)
function lp1(idx: Int, x: number, gg: number): number {
	let v = (x - zs[idx]) * gg;
	let y = v + zs[idx];
	zs[idx] = y + v;
	return y;
}

// Antiderivative of the unit hard clipper
function clipF(u: number): number {
	let a = abs(u);
	return (a <= 1.0) ? 0.5 * u * u : a - 0.5;
}

// First-order ADAA hard clip on channel ch
function clipAdaa(ch: Int, u: number): number {
	let u1 = prevU[ch];
	prevU[ch] = u;
	let du = u - u1;
	let y = 0.0;
	if (abs(du) < 0.00001) {
		y = clamp(0.5 * (u + u1), -1.0, 1.0);
	} else {
		y = (clipF(u) - clipF(u1)) / du;
	}
	return y;
}

// Slew limiter on channel ch: step at most maxStep per sample
function slew(ch: Int, v: number, maxStep: number): number {
	let y = slewY[ch] + clamp(v - slewY[ch], 0.0 - maxStep, maxStep);
	slewY[ch] = y;
	return y;
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = in2;

let G = maximum(dbtoa(clamp(drive_db, 0.0, 60.0)) - 1.0, 0.0);
let gHp = tan(pi * 60.0 / sr);
let ggHp = gHp / (1.0 + gHp);
let fGbw = minimum(1000000.0 / (1.0 + G), sr * 0.45);
let gGbw = tan(pi * fGbw / sr);
let ggGbw = gGbw / (1.0 + gGbw);
let gFil = tan(pi * minimum(clamp(filter_hz, 400.0, 12000.0), sr * 0.45) / sr);
let ggFil = gFil / (1.0 + gFil);
let maxStep = clamp(slew_v_us, 0.001, 0.3) * 1000000.0 / sr;

// --- Op amp: gain branch above 60 Hz, GBW lowpass, rails, slew ---------------
let vL = xL + G * (xL - lp1(0, xL, ggHp));
let vR = xR + G * (xR - lp1(1, xR, ggHp));
vL = lp1(2, vL, ggGbw);
vR = lp1(3, vR, ggGbw);
vL = slew(0, 4.5 * tanh(vL / 4.5), maxStep);
vR = slew(1, 4.5 * tanh(vR / 4.5), maxStep);

// --- Diodes to ground --------------------------------------------------------
let dL = 0.6 * clipAdaa(0, vL / 0.6);
let dR = 0.6 * clipAdaa(1, vR / 0.6);

// --- Filter knob, scale 0.6 V to -6 dBFS -------------------------------------
let wetL = lp1(4, dL, ggFil) * 0.8333333333;
let wetR = lp1(5, dR, ggFil) * 0.8333333333;

// --- Output coupling cap (10 Hz): the clipper makes DC on asymmetric notes --
let gDc = tan(pi * 10.0 / sr);
let ggDc = gDc / (1.0 + gDc);
wetL = wetL - lp1(6, wetL, ggDc);
wetR = wetR - lp1(7, wetR, ggDc);

let w = clamp(mix, 0.0, 1.0);
let outGain = dbtoa(clamp(output_db, -36.0, 12.0));
out1 = linearinterp(w, xL, wetL) * outGain;
out2 = linearinterp(w, xR, wetR) * outGain;
