// ============================================================================
// transformer_flux_sat.txt — output-transformer core saturation with
//                            hysteresis: the core flux is the integral of the
//                            voltage, so low notes saturate long before high
//                            ones (frequency-dependent breakup with no
//                            crossover filter anywhere)
//
// From: J. Macak, J. Schimmel, "Simulation of a Vacuum-Tube Push-Pull Guitar
//       Power Amplifier", DAFx-11 (2011_05_e): §2.2 (output transformer
//       model: magnetizing inductance whose current rises steeply as the
//       core saturates), §3.2 (circuit with the nonlinear transformer: the
//       saturating magnetizing current loads the source, so the output
//       collapses on the low-frequency peaks);
//       J. Chowdhury, "Real-time Physical Modelling for Analog Tape
//       Machines", DAFx-19 paper 3: §2.2 (magnetisation M follows the field
//       H through a hysteresis loop, Jiles–Atherton), §3 (discretizing the
//       stiff magnetic ODE: an implicit step keeps it stable).
//
// Model, per channel (flux phi; phis = 0.0004 core_size;
//                     a = 2 pi primary_hz / (1 + 3 hyst)):
//   dphi/dt = v - a * i(phi),     out = dphi/dt,     v = x * G
//   i(phi)  = phis * atanh(phi / phis)          (anhysteretic: linear when
//                                                small, unbounded at phis)
//           + hc * (phi - p) / w                 (hysteresis: play operator
//             p = clamp(p, phi - w, phi + w)      of width w = 0.1 phis,
//                                                 hc = 0.3 phis * hyst)
// Linear region: a one-pole high-pass at primary_hz (a is divided by the
// play operator's in-band stiffness 1 + 3 hyst, so hysteresis does not move
// the corner). Flux amplitude = V / (2 pi f): halving the frequency doubles
// how hard the core is hit. A fixed 8 kHz lowpass (leakage inductance and
// winding capacitance) rounds the steps as the core leaves saturation.
// Discretized with implicit Euler (4 Newton iterations, flux = phis tanh u;
// the hysteresis current is taken from the previous sample).
// wet = out / G (unity below saturation).
//
// Inlets:
//   in1 / in2: stereo tuba
// Outlets:
//   out1 / out2: stereo
// No safety limiter, no metering.
// ============================================================================

// Drive into the transformer (dB)
@param({ min: 0, max: 36 }) drive_db = 12;
// Core size (saturation flux, relative)
@param({ min: 0.1, max: 2, exponent: 2 }) core_size = 1;
// Hysteresis
@param({ min: 0, max: 100 }) hysteresis_pct = 40;
// Primary inductance corner (Hz)
@param({ min: 5, max: 200, exponent: 2 }) primary_hz = 30;
// Dry/wet
@param({ min: 0, max: 1 }) mix = 1;
// Output gain (dB)
@param({ min: -36, max: 12 }) output_db = 0;

// core state u: flux = phis * tanh(u), so the flux can never pass phis
@state ucore = new FixedSampleArray(2);
@state play = new FixedSampleArray(2);
// TPT one-pole states: 0/1 leakage lowpass
@state zs = new FixedSampleArray(2);

// TPT one-pole lowpass on state slot idx; gg = g / (1 + g)
function lp1(idx: Int, x: number, gg: number): number {
	let v = (x - zs[idx]) * gg;
	let y = v + zs[idx];
	zs[idx] = y + v;
	return y;
}

// One implicit Euler step of the core on channel ch; returns dphi/dt.
// Solves phis (tanh u' + c u') = flux + dt (v - a ih) by Newton in u
// (the left side is strictly increasing, slope >= phis c).
function core(ch: Int, v: number, a: number, phis: number, hc: number, w: number, dt: number): number {
	let u0 = ucore[ch];
	let ph = phis * tanh(u0);
	let p = clamp(play[ch], ph - w, ph + w);
	let ih = hc * (ph - p) / w;
	let c = dt * a;
	let rhs = ph + dt * (v - a * ih);
	let u = u0;
	for (let it: Int = 0; it < 4; it++) {
		let t = tanh(u);
		let gu = phis * (t + c * u) - rhs;
		let du = gu / (phis * (1.0 - t * t + c));
		u = u - clamp(du, -2.0, 2.0);
	}
	ucore[ch] = u;
	let phNew = phis * tanh(u);
	play[ch] = clamp(p, phNew - w, phNew + w);
	return (phNew - ph) / dt;
}

let sr = maximum(samplerate(), 1.0);
let dt = 1.0 / sr;
let xL = in1;
let xR = in2;

let G = dbtoa(clamp(drive_db, 0.0, 36.0));
let hy = clamp(hysteresis_pct, 0.0, 100.0) * 0.01;
let a = 2.0 * pi * clamp(primary_hz, 5.0, 200.0) / (1.0 + 3.0 * hy);
let phis = 0.0004 * clamp(core_size, 0.1, 2.0);
let hc = 0.3 * phis * hy;
let gLk = tan(pi * minimum(8000.0, sr * 0.45) / sr);
let ggLk = gLk / (1.0 + gLk);
let w = 0.1 * phis;

let wetL = lp1(0, core(0, G * xL, a, phis, hc, w, dt), ggLk) / G;
let wetR = lp1(1, core(1, G * xR, a, phis, hc, w, dt), ggLk) / G;

let mx = clamp(mix, 0.0, 1.0);
let outGain = dbtoa(clamp(output_db, -36.0, 12.0));
out1 = linearinterp(mx, xL, wetL) * outGain;
out2 = linearinterp(mx, xR, wetR) * outGain;
