// ============================================================================
// 01_harmonic_isolator.txt
// Unix-Style Effect 1: Pitch-Tracking Harmonic Isolator & Sine-ifier
// Isolates the fundamental or a chosen harmonic of the stereo tuba signal
// using a cascaded resonant bell boost + SVF bandpass, and outputs a clean,
// normalized stereo sine-approximated carrier/modulator on out3 & out4.
// ============================================================================
// Inlets:
//   in1: Left Audio Input
//   in2: Right Audio Input
//   in3: Tracked Fundamental Pitch in Hz (from fluid.pitch~ YinFFT; <=0 uses internal tracker)
// Outlets:
//   out1: Left Audio Output (Filtered / Boosted Harmonic)
//   out2: Right Audio Output (Filtered / Boosted Harmonic)
//   out3: Left Normalized Sine-ified Carrier/Modulator [-1, 1]
//   out4: Right Normalized Sine-ified Carrier/Modulator [-1, 1]

@param({ min: 1.0, max: 8.0 }) harmonic = 1.0;
@param({ min: 0.0, max: 48.0 }) boost_db = 24.0;
@param({ min: 0.0, max: 1.0 }) focus = 0.65;
@param({ min: 0.0, max: 1.0 }) norm_amount = 0.75;

// Biquad peak EQ states (Left and Right)
@state bqL = new biquad();
@state bqR = new biquad();

// Cascaded 2-stage SVF bandpass states for steep harmonic isolation
@state svfL1 = new svf();
@state svfR1 = new svf();
@state svfL2 = new svf();
@state svfR2 = new svf();

// Internal fallback zero-crossing pitch tracker & conditioning state
@state lpTrack = new onepole();
@state prevTrackSig = 0.0;
@state zcCounter = 0.0;
@state zcFreq = 55.0;
@state smoothF0 = 55.0;

// Envelope followers for dynamic decoupling / amplitude normalization
@state envFastL = 0.0;
@state envFastR = 0.0;
@state envSlowL = 0.0;
@state envSlowR = 0.0;

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = in2;
let extPitch = in3;

// --- 1. Pitch Conditioning (External YinFFT or Internal Zero-Crossing Fallback) ---
let monoLow = lpTrack.next(0.5 * (xL + xR), 180.0);
zcCounter += 1.0;
if (prevTrackSig <= 0.0 && monoLow > 0.0) {
	if (zcCounter > sr / 500.0 && zcCounter < sr / 22.0) {
		zcFreq = sr / zcCounter;
	}
	zcCounter = 0.0;
}
prevTrackSig = monoLow;

let rawF0 = (extPitch > 20.0 && extPitch < 1000.0) ? extPitch : zcFreq;
// Slew-limit pitch in log/Hz domain to suppress tracker glitches
let pitchSlew = 1.0 - exp(-1.0 / (0.008 * sr));
smoothF0 += (rawF0 - smoothF0) * pitchSlew;
let targetHz = clamp(smoothF0 * harmonic, 24.0, sr * 0.42);

// --- 2. Musical Tapers ---
// focus (0..1) maps exponentially to Q from 1.0 (gentle) to 28.0 (surgical harmonic lock)
let qVal = 1.0 * pow(28.0, pow(clamp(focus, 0.0, 1.0), 1.4));
// boost_db (0..48 dB) maps to RBJ peaking filter linear gain A = 10^(dB/40)
let A = pow(10.0, clamp(boost_db, 0.0, 48.0) / 40.0);

// --- 3. RBJ Peaking EQ Coefficients at targetHz ---
let w0 = 2.0 * pi * targetHz / sr;
let cosw = cos(w0);
let sinw = sin(w0);
let alpha = sinw / (2.0 * maximum(qVal * 0.5, 0.5));

let b0 = 1.0 + alpha * A;
let b1 = -2.0 * cosw;
let b2 = 1.0 - alpha * A;
let a0_inv = 1.0 / (1.0 + alpha / A);
let a1 = (-2.0 * cosw) * a0_inv;
let a2 = (1.0 - alpha / A) * a0_inv;

let c_b0 = b0 * a0_inv;
let c_b1 = b1 * a0_inv;
let c_b2 = b2 * a0_inv;

// Compensate peak gain so +48 dB doesn't clip wildly before normalization
let compGain = 1.0 / pow(A, 1.35);
let peakedL = bqL.next(xL, c_b0, c_b1, c_b2, a1, a2) * compGain;
let peakedR = bqR.next(xR, c_b0, c_b1, c_b2, a1, a2) * compGain;

// --- 4. Cascaded SVF Bandpass Isolation ---
// svf.next(x, freq, Q) returns [lp, hp, bp, notch]
let svfQ = clamp(0.5 + focus * 0.45, 0.5, 0.96);
let bpL1 = svfL1.next(peakedL, targetHz, svfQ)[2];
let bpR1 = svfR1.next(peakedR, targetHz, svfQ)[2];
let bpL2 = svfL2.next(bpL1, targetHz, svfQ)[2];
let bpR2 = svfR2.next(bpR1, targetHz, svfQ)[2];

// Crossfade between bell-boosted tuba and pure isolated harmonic bandpass with focus
let isoL = linearinterp(focus, peakedL, bpL2 * 2.2);
let isoR = linearinterp(focus, peakedR, bpR2 * 2.2);

// --- 5. Instantaneous Envelope & Dynamic Decoupling ---
let absL = abs(isoL);
let absR = abs(isoR);
let fastAtk = 1.0 - exp(-1.0 / (0.0015 * sr));
let fastRel = 1.0 - exp(-1.0 / (0.025 * sr));
let slowCoeff = 1.0 - exp(-1.0 / (0.060 * sr));

envFastL = (absL > envFastL) ? (envFastL + (absL - envFastL) * fastAtk) : (envFastL + (absL - envFastL) * fastRel);
envFastR = (absR > envFastR) ? (envFastR + (absR - envFastR) * fastAtk) : (envFastR + (absR - envFastR) * fastRel);
envSlowL += (absL - envSlowL) * slowCoeff;
envSlowR += (absR - envSlowR) * slowCoeff;

// Gate threshold to prevent noise floor amplification during silence
let gateL = clamp((envSlowL - 0.0008) * 250.0, 0.0, 1.0);
let gateR = clamp((envSlowR - 0.0008) * 250.0, 0.0, 1.0);

let unitSinL = (isoL / (envFastL * 1.25 + 0.0005)) * gateL;
let unitSinR = (isoR / (envFastR * 1.25 + 0.0005)) * gateR;
unitSinL = tanh(unitSinL);
unitSinR = tanh(unitSinR);

let nAmt = pow(clamp(norm_amount, 0.0, 1.0), 1.2);
out1 = linearinterp(nAmt, isoL, unitSinL);
out2 = linearinterp(nAmt, isoR, unitSinR);
out3 = unitSinL;
out4 = unitSinR;
