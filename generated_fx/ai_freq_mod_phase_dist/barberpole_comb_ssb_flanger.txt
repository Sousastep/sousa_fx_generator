// ============================================================================
// barberpole_comb_ssb_flanger.txt
// Unix-Style Stereo Effect: Endless Barberpole Dual-Comb & SSB Spectral-Delay Flanger
// Implements Fabian Esqueda, Vesa Valimaki & Julian Parker (DAFx-15, Paper 67):
// "Barberpole Phasing and Flanging Illusions" (Sections 3, 4 & 4.1, Eq. 9-14, Fig. 7-12).
// Combines BOTH time-domain/hybrid barberpole comb engines from the paper:
//   1. Synchronized Dual Flanger (Sec 3, Fig. 7-8): Two cubic-interpolated comb
//      delay lines modulated by 90-degree-offset sawtooth ramps (Eq. 9) and
//      seamlessly crossfaded by synchronized triangular window gains so comb
//      notches sweep infinitely upward or downward without reset clicks!
//   2. Single-Sideband (SSB) Spectral-Delay Barberpole (Sec 4 & 4.1, Fig. 10):
//      Hilbert SSB frequency shifting mixed with a dispersive 6-stage allpass
//      spectral delay chain to warp the moving comb notches logarithmically!
// ============================================================================
// Inlets:
//   in1: Left Audio Input (Tuba)
//   in2: Right Audio Input (if unpatched, mono in1 is used automatically)
//   in3: Optional Tracked Pitch f0 in Hz (if > 20 Hz, tunes comb sweep range around f0)
//   in4: Optional 1-Bar Phasor [0..1] (syncs barberpole sweep cycle to host tempo)
// Outlets:
//   out1: Left Endless Barberpole Comb/SSB Output
//   out2: Right Endless Barberpole Comb/SSB Output

@param({ min: -16.0, max: 16.0 }) sweep_rate_hz = 3;
@param({ min: 0.4, max: 12.0 }) center_delay_ms = 2.2;
@param({ min: -0.92, max: 0.92 }) feedback = 0.72;
@param({ min: 0.0, max: 1.0 }) dual_to_ssb_mode = 0.4;
@param({ min: -0.8, max: 0.8 }) notch_warp_a1 = -0.5;
@param({ min: 0.0, max: 1.0 }) stereo_counter_dir = 0.5;

// Circular delay buffers for Dual Barberpole Flanger (2048 samples per channel)
@state flgBufL = new FixedSampleArray(2048);
@state flgBufR = new FixedSampleArray(2048);
@state writeIdx: Int = 0;

// 4-stage Quadrature IIR Hilbert states for SSB Barberpole (Paper Sec 4, Fig. 10)
@state lReX = new FixedSampleArray(4); @state lReY = new FixedSampleArray(4);
@state lImX = new FixedSampleArray(4); @state lImY = new FixedSampleArray(4);
@state lDelRe = 0.0;

@state rReX = new FixedSampleArray(4); @state rReY = new FixedSampleArray(4);
@state rImX = new FixedSampleArray(4); @state rImY = new FixedSampleArray(4);
@state rDelRe = 0.0;

// 6-stage First-Order Allpass Spectral Delay Chain for Log-Warped Notches (Sec 4.1)
@state apXL = new FixedSampleArray(6); @state apYL = new FixedSampleArray(6);
@state apXR = new FixedSampleArray(6); @state apYR = new FixedSampleArray(6);

// Feedback states & LFO phasor
@state fbStateL = 0.0;
@state fbStateR = 0.0;
@state prevBarIn = 0.0;
@state smoothCenterMs = 2.2;
@state sweepPhase = 0.0;
@state smoothRate = 0.0;
@state prevBarActive = 0.0;

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = (abs(in2) > 0.0000001) ? in2 : xL;
let pitchIn = in3;
let barIn = in4;

let slew = 1.0 - exp(-1.0 / (0.006 * sr));
smoothCenterMs += (clamp(center_delay_ms, 0.4, 12.0) - smoothCenterMs) * slew;

// If pitch tracker is connected on in3, bias center delay toward 1/(2*f0)
let baseDelaySamps = mstosamps(smoothCenterMs);
if (pitchIn > 25.0 && pitchIn < 1000.0) {
	baseDelaySamps = linearinterp(0.5, baseDelaySamps, clamp(sr / (2.0 * pitchIn), 15.0, 1200.0));
}

// Per DAFx-15 Sec 3 (p. 5): D_min should be > 0.5 * D_max to hide resets cleanly
let dMin = clamp(baseDelaySamps * 0.68, 4.0, 1200.0);
let dMax = clamp(baseDelaySamps * 1.32, 8.0, 1800.0);

// --- 1. Barberpole Sweep Phase (signed integration: continuous through 0 Hz) ---
let rateTarget = clamp(sweep_rate_hz, -16.0, 16.0);
let rateSlew = 1.0 - exp(-1.0 / (0.02 * sr));
smoothRate += (rateTarget - smoothRate) * rateSlew;

let barActive = (abs(barIn - prevBarIn) > 0.000001 || (barIn > 0.001 && barIn < 0.999)) ? 1.0 : 0.0;

if (barActive > 0.5) {
	// Tempo-synced: advance by the signed change in the bar phasor (cycles per bar = rounded |rate|)
	if (prevBarActive > 0.5) {
		let d = barIn - prevBarIn;
		d = d - floor(d + 0.5); // wrap delta to [-0.5, 0.5)
		let cyc = maximum(floor(abs(smoothRate) + 0.5), 1.0);
		let sgn = (smoothRate >= 0.0) ? 1.0 : -1.0;
		sweepPhase = wrap(sweepPhase + d * cyc * sgn, 0.0, 1.0);
	}
} else {
	// Free-running: signed rate, so the phase simply slows, stops and reverses
	sweepPhase = wrap(sweepPhase + smoothRate / sr, 0.0, 1.0);
}
prevBarIn = barIn;
prevBarActive = barActive;

let phL = sweepPhase;
// Stereo counter-direction option: Right channel sweeps opposite direction when stereo_counter_dir > 0.5
let counter = clamp(stereo_counter_dir, 0.0, 1.0);
let phR = (counter > 0.5) ? (1.0 - phL) : wrap(phL + 0.25 * counter, 0.0, 1.0);

let fb = clamp(feedback, -0.92, 0.92);
let uL = tanh(xL + fb * fbStateL);
let uR = tanh(xR + fb * fbStateR);

flgBufL[writeIdx] = uL;
flgBufR[writeIdx] = uR;

// --- 2. Engine A: Synchronized Dual Flanger (Paper Sec 3, Eq. 9, Fig. 7-8) ---
// Two sawtooth LFOs offset by 0.5 cycle (so reset of one falls at midpoint of the other)
// with complementary raised-triangle/cosine crossfade gains g1 + g2 = 1.0
let ph1L = phL;
let ph2L = wrap(phL + 0.5, 0.0, 1.0);
let del1L = dMin + (dMax - dMin) * ph1L;
let del2L = dMin + (dMax - dMin) * ph2L;
let g1L = triangle(ph1L, 0.5);
let g2L = 1.0 - g1L;
let rpC1L = writeIdx - clamp(del1L, 2.0, 2042.0);
while (rpC1L < 0.0) { rpC1L += 2048.0; }
let iC1L: Int = floor(rpC1L);
let c1L = cubicinterp(rpC1L - iC1L, flgBufL[(iC1L - 1 + 2048) & 2047], flgBufL[iC1L & 2047], flgBufL[(iC1L + 1) & 2047], flgBufL[(iC1L + 2) & 2047]);
let rpC2L = writeIdx - clamp(del2L, 2.0, 2042.0);
while (rpC2L < 0.0) { rpC2L += 2048.0; }
let iC2L: Int = floor(rpC2L);
let c2L = cubicinterp(rpC2L - iC2L, flgBufL[(iC2L - 1 + 2048) & 2047], flgBufL[iC2L & 2047], flgBufL[(iC2L + 1) & 2047], flgBufL[(iC2L + 2) & 2047]);
let dualCombL = 0.5 * (uL + g1L * c1L + g2L * c2L);

let ph1R = phR;
let ph2R = wrap(phR + 0.5, 0.0, 1.0);
let del1R = dMin + (dMax - dMin) * ph1R;
let del2R = dMin + (dMax - dMin) * ph2R;
let g1R = triangle(ph1R, 0.5);
let g2R = 1.0 - g1R;
let rpC1R = writeIdx - clamp(del1R, 2.0, 2042.0);
while (rpC1R < 0.0) { rpC1R += 2048.0; }
let iC1R: Int = floor(rpC1R);
let c1R = cubicinterp(rpC1R - iC1R, flgBufR[(iC1R - 1 + 2048) & 2047], flgBufR[iC1R & 2047], flgBufR[(iC1R + 1) & 2047], flgBufR[(iC1R + 2) & 2047]);
let rpC2R = writeIdx - clamp(del2R, 2.0, 2042.0);
while (rpC2R < 0.0) { rpC2R += 2048.0; }
let iC2R: Int = floor(rpC2R);
let c2R = cubicinterp(rpC2R - iC2R, flgBufR[(iC2R - 1 + 2048) & 2047], flgBufR[iC2R & 2047], flgBufR[(iC2R + 1) & 2047], flgBufR[(iC2R + 2) & 2047]);
let dualCombR = 0.5 * (uR + g1R * c1R + g2R * c2R);

// --- 3. Engine B: SSB + Spectral-Delay Warped Barberpole (Paper Sec 4 & 4.1, Fig. 10) ---
// Quadrature allpass: y[n] = c*(x[n] + y[n-1]) - x[n-1]; this cR/cI order is the
// negative Hilbert, so the imaginary branch is negated before the SSB shift below.
let cR0 = 0.4794008656; let cR1 = 0.8762184935; let cR2 = 0.9765975895; let cR3 = 0.9974992559;
let cI0 = 0.1617584983; let cI1 = 0.7330289323; let cI2 = 0.9453497003; let cI3 = 0.9905991567;

let reL = uL;
let t = cR0 * (reL + lReY[0]) - lReX[0]; lReX[0] = reL; lReY[0] = t; reL = t;
t = cR1 * (reL + lReY[1]) - lReX[1]; lReX[1] = reL; lReY[1] = t; reL = t;
t = cR2 * (reL + lReY[2]) - lReX[2]; lReX[2] = reL; lReY[2] = t; reL = t;
t = cR3 * (reL + lReY[3]) - lReX[3]; lReX[3] = reL; lReY[3] = t; reL = t;
let sReL = lDelRe; lDelRe = reL;

let imL = uL;
t = cI0 * (imL + lImY[0]) - lImX[0]; lImX[0] = imL; lImY[0] = t; imL = t;
t = cI1 * (imL + lImY[1]) - lImX[1]; lImX[1] = imL; lImY[1] = t; imL = t;
t = cI2 * (imL + lImY[2]) - lImX[2]; lImX[2] = imL; lImY[2] = t; imL = t;
t = cI3 * (imL + lImY[3]) - lImX[3]; lImX[3] = imL; lImY[3] = t; imL = t;
let sImL = 0.0 - imL;

let reR = uR;
t = cR0 * (reR + rReY[0]) - rReX[0]; rReX[0] = reR; rReY[0] = t; reR = t;
t = cR1 * (reR + rReY[1]) - rReX[1]; rReX[1] = reR; rReY[1] = t; reR = t;
t = cR2 * (reR + rReY[2]) - rReX[2]; rReX[2] = reR; rReY[2] = t; reR = t;
t = cR3 * (reR + rReY[3]) - rReX[3]; rReX[3] = reR; rReY[3] = t; reR = t;
let sReR = rDelRe; rDelRe = reR;

let imR = uR;
t = cI0 * (imR + rImY[0]) - rImX[0]; rImX[0] = imR; rImY[0] = t; imR = t;
t = cI1 * (imR + rImY[1]) - rImX[1]; rImX[1] = imR; rImY[1] = t; imR = t;
t = cI2 * (imR + rImY[2]) - rImX[2]; rImX[2] = imR; rImY[2] = t; imR = t;
t = cI3 * (imR + rImY[3]) - rImX[3]; rImX[3] = imR; rImY[3] = t; imR = t;
let sImR = 0.0 - imR;

// Quadrature SSB shift by phL / phR
let ssbL = sReL * cos(2.0 * pi * phL) - sImL * sin(2.0 * pi * phL);
let ssbR = sReR * cos(2.0 * pi * phR) - sImR * sin(2.0 * pi * phR);

// Pass delayed SSB signal through 6-stage allpass spectral delay to warp notch spacing (Sec 4.1)
let a1 = clamp(notch_warp_a1, -0.8, 0.8);
let rpDspL = writeIdx - clamp(baseDelaySamps * 0.5, 2.0, 2042.0);
while (rpDspL < 0.0) { rpDspL += 2048.0; }
let iDspL: Int = floor(rpDspL);
let dispL = cubicinterp(rpDspL - iDspL, flgBufL[(iDspL - 1 + 2048) & 2047], flgBufL[iDspL & 2047], flgBufL[(iDspL + 1) & 2047], flgBufL[(iDspL + 2) & 2047]);

let rpDspR = writeIdx - clamp(baseDelaySamps * 0.5, 2.0, 2042.0);
while (rpDspR < 0.0) { rpDspR += 2048.0; }
let iDspR: Int = floor(rpDspR);
let dispR = cubicinterp(rpDspR - iDspR, flgBufR[(iDspR - 1 + 2048) & 2047], flgBufR[iDspR & 2047], flgBufR[(iDspR + 1) & 2047], flgBufR[(iDspR + 2) & 2047]);

for (let k: Int = 0; k < 6; k++) {
	let ynL = a1 * dispL + apXL[k] - a1 * apYL[k];
	apXL[k] = dispL; apYL[k] = ynL; dispL = ynL;

	let ynR = a1 * dispR + apXR[k] - a1 * apYR[k];
	apXR[k] = dispR; apYR[k] = ynR; dispR = ynR;
}

let ssbCombL = 0.5 * (ssbL + dispL);
let ssbCombR = 0.5 * (ssbR + dispR);

// --- 4. Blend Dual-Comb Flanger & SSB Spectral-Delay Barberpole ---
let mode = clamp(dual_to_ssb_mode, 0.0, 1.0);
let outL = linearinterp(mode, dualCombL, ssbCombL);
let outR = linearinterp(mode, dualCombR, ssbCombR);

fbStateL = outL;
fbStateR = outR;
writeIdx = (writeIdx + 1) & 2047;

out1 = outL;
out2 = outR;
