// ============================================================================
// 08_suboctave_phase_ringmod.txt
// Unix-Style Effect 8: Phase-Divided Sub-Octave Ring Modulator & Feedback AM
// Implements Approach 3b & 5 from "Novel Distortion Approaches for Live Tuba"
// and Kleimola et al. (2011) Feedback Amplitude Modulation (y[n] = x[n]*(1 + k*y[n-1])).
// Isolates the fundamental (2-stage bandpass at f0), takes its analytic phase and
// locks a monotonic PLL phasor to it; counting the PLL's wraps (mod 6) builds a
// zero-drift phase-divided sub-octave modulator cos(theta_unwrapped / 2) and
// sub-twelfth cos(theta_unwrapped / 3) that never beats against the tuba. (The raw
// atan2 phase of the lowpassed audio wrapped twice per period whenever the 2nd
// harmonic was the stronger one, which is normal for tuba, and the "sub-octave"
// then sat on f0.)
// ============================================================================
// Inlets:
//   in1: Left Audio Input
//   in2: Right Audio Input
//   in3: Tracked Pitch in Hz (tunes the fundamental bandpass and the PLL; outside
//        20..1000 Hz an internal zero-crossing tracker is used)
// Outlets:
//   out1: Left Sub-Octave Ring-Mod / Feedback AM Output
//   out2: Right Sub-Octave Ring-Mod / Feedback AM Output

@param({ min: 0.0, max: 1.0 }) sub_harmonic = 0.0;
@param({ min: 0.0, max: 1.0 }) rm_depth = 0.85;
@param({ min: 0.0, max: 0.92 }) fb_am = 0.35;
@param({ min: 0.0, max: 1.0 }) mix = 0.8;

// Hilbert quadrature pair for instantaneous phase extraction
@state hReX = new FixedSampleArray(4); @state hReY = new FixedSampleArray(4);
@state hImX = new FixedSampleArray(4); @state hImY = new FixedSampleArray(4);
@state hDelRe = 0.0;

// PLL phasor + wrap counter for exact phase division (theta / 2 and theta / 3)
@state pllPhase = 0.0;
@state wrapCount: Int = 0;

// Feedback AM state (Kleimola et al. 2011)
@state prevYL = 0.0;
@state prevYR = 0.0;

// Pre-filter for clean fundamental phase tracking, fallback tracker + DC blockers
@state lpTrack = new onepole();
@state bpIc1 = new FixedSampleArray(2);
@state bpIc2 = new FixedSampleArray(2);
@state prevSig = 0.0;
@state zcCount = 0.0;
@state zcHz = 55.0;
@state smoothF0 = 55.0;
@state in2Hold = 0.0;
@state dcX1L = 0.0; @state dcY1L = 0.0;
@state dcX1R = 0.0; @state dcY1R = 0.0;

// Unity-peak TPT state-variable bandpass (Zavalishin / Simper), one stage
function svfBp(idx: Int, x, g, kd) {
	let a1 = 1.0 / (1.0 + g * (g + kd));
	let a2 = g * a1;
	let a3 = g * a2;
	let v3 = x - bpIc2[idx];
	let v1 = a1 * bpIc1[idx] + a2 * v3;
	let v2 = bpIc2[idx] + a2 * bpIc1[idx] + a3 * v3;
	bpIc1[idx] = 2.0 * v1 - bpIc1[idx];
	bpIc2[idx] = 2.0 * v2 - bpIc2[idx];
	return kd * v1;
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let pitchIn = in3;

// Mono fallback: in2 counts as patched for 250 ms after its last non-zero sample
if (abs(in2) > 0.0000001) {
	in2Hold = 0.25 * sr;
} else if (in2Hold > 0.0) {
	in2Hold -= 1.0;
}
let xR = (in2Hold > 0.0) ? in2 : xL;
let mono = 0.5 * (xL + xR);

// --- 1. Pitch (in3, else zero-crossing fallback) and Fundamental Isolation ---
let monoLow = lpTrack.next(mono, 200.0);
zcCount += 1.0;
if (prevSig <= 0.0 && monoLow > 0.0) {
	if (zcCount > sr / 500.0 && zcCount < sr / 22.0) {
		zcHz = sr / zcCount;
	}
	zcCount = 0.0;
}
prevSig = monoLow;
let rawF0 = (pitchIn > 20.0 && pitchIn < 1000.0) ? pitchIn : zcHz;
smoothF0 += (rawF0 - smoothF0) * (1.0 - exp(-1.0 / (0.005 * sr)));
let f0 = clamp(smoothF0, 24.0, 1000.0);

// 2-stage bandpass (Q 2 per stage): the 2nd harmonic ends up about 20 dB down
let gBp = tan(pi * f0 / sr);
let monoFund = svfBp(1, svfBp(0, mono, gBp, 0.5), gBp, 0.5);

// --- 2. Analytic Phase Extraction ---
let cRe0 = 0.4794008656; let cRe1 = 0.8762184935; let cRe2 = 0.9765975895; let cRe3 = 0.9974992559;
let cIm0 = 0.1617584983; let cIm1 = 0.7330289323; let cIm2 = 0.9453497003; let cIm3 = 0.9905991567;

let sRe = monoFund;
let yRe0 = cRe0 * (sRe + hReY[0]) - hReX[0]; hReX[0] = sRe; hReY[0] = yRe0; sRe = yRe0;
let yRe1 = cRe1 * (sRe + hReY[1]) - hReX[1]; hReX[1] = sRe; hReY[1] = yRe1; sRe = yRe1;
let yRe2 = cRe2 * (sRe + hReY[2]) - hReX[2]; hReX[2] = sRe; hReY[2] = yRe2; sRe = yRe2;
let yRe3 = cRe3 * (sRe + hReY[3]) - hReX[3]; hReX[3] = sRe; hReY[3] = yRe3;
let outRe = hDelRe; hDelRe = yRe3;

let sIm = monoFund;
let yIm0 = cIm0 * (sIm + hImY[0]) - hImX[0]; hImX[0] = sIm; hImY[0] = yIm0; sIm = yIm0;
let yIm1 = cIm1 * (sIm + hImY[1]) - hImX[1]; hImX[1] = sIm; hImY[1] = yIm1; sIm = yIm1;
let yIm2 = cIm2 * (sIm + hImY[2]) - hImX[2]; hImX[2] = sIm; hImY[2] = yIm2; sIm = yIm2;
let yIm3 = cIm3 * (sIm + hImY[3]) - hImX[3]; hImX[3] = sIm; hImY[3] = yIm3;
let outIm = 0.0 - yIm3; // negate: this cR/cI order is the negative Hilbert (+f analytic phase)

let pol = cartopol(outRe, outIm);
let instAmp = pol[0];
let fundPh = wrap((pol[1] + pi) / (2.0 * pi), 0.0, 1.0);

// --- 3. PLL (~8 Hz loop, strictly monotonic) + Wrap Counter (Modulo 6 = LCM of 2, 3) ---
let dtF0 = f0 / sr;
let phErr = wrap(fundPh - pllPhase + 0.5, 0.0, 1.0) - 0.5;
let pllGain = (instAmp > 0.0003) ? (2.0 * pi * 8.0 / sr) : 0.0;
let pllNext = pllPhase + maximum(dtF0 + pllGain * phErr, 0.25 * dtF0);
if (pllNext >= 1.0) {
	pllNext -= 1.0;
	wrapCount = (wrapCount + 1) % 6;
}
pllPhase = pllNext;

let theta = 2.0 * pi * pllPhase;
let unwrappedTheta = theta + 2.0 * pi * wrapCount;

// Locked Sub-Octave (theta / 2), Sub-Twelfth (theta / 3), and Octave-Up (2 * theta)
let subOctL = cos(0.5 * unwrappedTheta);
let subOctR = cos(0.5 * unwrappedTheta + 0.25 * pi); // Stereo quadrature spread
let subFifthL = cos(unwrappedTheta / 3.0);
let subFifthR = cos(unwrappedTheta / 3.0 + 0.25 * pi);
let octUp = cos(2.0 * theta);

// Morph modulator across Sub-Octave (0.0) -> Sub-Twelfth (0.5) -> Sub+Octave Growl (1.0)
let sh = clamp(sub_harmonic, 0.0, 1.0);
let modWaveL = (sh < 0.5) ? linearinterp(sh * 2.0, subOctL, subFifthL) : linearinterp((sh - 0.5) * 2.0, subFifthL, 0.7 * subOctL + 0.5 * octUp);
let modWaveR = (sh < 0.5) ? linearinterp(sh * 2.0, subOctR, subFifthR) : linearinterp((sh - 0.5) * 2.0, subFifthR, 0.7 * subOctR - 0.5 * octUp);

// Silence gate so phase noise doesn't chatter when not playing
let gate = clamp((instAmp - 0.0008) * 250.0, 0.0, 1.0);
modWaveL *= gate;
modWaveR *= gate;

// --- 4. AM-to-RM Continuum + Feedback AM (Kleimola et al. 2011) ---
// y = x * ( (1 - d) + d * modWave + k * y[n-1] )
let d = pow(clamp(rm_depth, 0.0, 1.0), 1.2);
let kFb = pow(clamp(fb_am / 0.92, 0.0, 1.0), 1.4) * 0.92;

let amModL = (1.0 - d) + d * modWaveL;
let amModR = (1.0 - d) + d * modWaveR;

let rawYL = xL * (amModL + kFb * prevYL);
let rawYR = xR * (amModR + kFb * prevYR);

// Soft-saturate feedback loop to prevent runaway while keeping sawtooth bite
prevYL = tanh(rawYL);
prevYR = tanh(rawYR);

// --- 5. DC Blocking Filter (Highpass at ~12 Hz) ---
let dcR = 1.0 - (2.0 * pi * 12.0 / sr);
let cleanL = prevYL - dcX1L + dcR * dcY1L;
dcX1L = prevYL; dcY1L = cleanL;

let cleanR = prevYR - dcX1R + dcR * dcY1R;
dcX1R = prevYR; dcY1R = cleanR;

out1 = linearinterp(clamp(mix, 0.0, 1.0), xL, cleanL);
out2 = linearinterp(clamp(mix, 0.0, 1.0), xR, cleanR);
