// ============================================================================
// quadrature_chebyshev_shaper.txt
// Unix-Style Stereo Effect: Phase-Quadrature Chebyshev T_k & U_k Waveshaper
// Implements Timoney, Lazzarini, Gibney & Pekonen (DAFx-10, Paper P65):
// "Digital Emulation of Distortion Effects by Wave and Phase Shaping Methods"
// (Section 2.1, Eq. 5-15).
// Classical Chebyshev waveshaping T_k(cos(w*t)) = cos(k*w*t) only controls
// harmonic magnitudes with fixed 0-degree phase. By extracting the quadrature
// signal pair (x_I = cos(w*t), x_Q = sin(w*t)) via a 4-stage Hilbert network and
// combining Chebyshev polynomials of the FIRST kind T_k(x_I) with Chebyshev
// polynomials of the SECOND kind x_Q * U_{k-1}(x_I) = sin(k*w*t) (Eq. 11-15):
//   y(t) = sum_k a_k * [ cos(phi_k) * T_k(x_I) + sin(phi_k) * x_Q * U_{k-1}(x_I) ]
// you gain independent control over BOTH harmonic amplitudes AND harmonic phase
// angles phi_k (even continuously rotating harmonic phases at audio/LFO rates)!
// ============================================================================
// Inlets:
//   in1: Left Audio Input (prefer 01_harmonic_isolator or raw tuba)
//   in2: Right Audio Input (if unpatched, mono in1 is used automatically)
//   in3: Optional Mod Envelope [0..1] (dynamically rotates harmonic phase angles phi_k)
//   in4: Optional 1-Bar Phasor [0..1] (drives continuous per-harmonic Shepard/Barberpole
//        phase rotation when phase_spin > 0)
// Outlets:
//   out1: Left Quadrature Chebyshev Output
//   out2: Right Quadrature Chebyshev Output

@param({ min: 0.0, max: 18.0 }) drive_db = 6.0;
@param({ min: 0.0, max: 1.0 }) even_harmonics = 0.55;
@param({ min: 0.0, max: 1.0 }) odd_harmonics = 0.75;
@param({ min: 0.0, max: 1.0 }) high_order_tilt = 0.5;
@param({ min: -3.14159, max: 3.14159 }) quad_phase_angle = 0.785;
@param({ min: 0.0, max: 1.0 }) phase_spin = 0.35;
@param({ min: 0.0, max: 1.0 }) stereo_quad_spread = 0.5;

// Stereo 4-stage IIR Hilbert transform states for Quadrature (x_I, x_Q) extraction
@state lReX = new FixedSampleArray(4); @state lReY = new FixedSampleArray(4);
@state lImX = new FixedSampleArray(4); @state lImY = new FixedSampleArray(4);
@state lDelRe = 0.0;

@state rReX = new FixedSampleArray(4); @state rReY = new FixedSampleArray(4);
@state rImX = new FixedSampleArray(4); @state rImY = new FixedSampleArray(4);
@state rDelRe = 0.0;

// Warm lowpass pre-conditioning so Chebyshev polynomials shape the fundamental + low brass body cleanly
@state preLpL = new onepole();
@state preLpR = new onepole();

// Glitch-free bar phasor hold & smoothers
@state barFallback = new phasor();
@state prevBarIn = 0.0;
@state barHold = 0.0;
@state smoothEnvMod = 0.0;
@state dcXL = 0.0; @state dcYL = 0.0;
@state dcXR = 0.0; @state dcYR = 0.0;

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = (abs(in2) > 0.0000001) ? in2 : xL;
let envMod = (in3 > -2.0 && in3 < 2.0) ? in3 : 0.0;
let barIn = in4;

// Smooth in3 modulation to prevent clicks
let modSlew = 1.0 - exp(-1.0 / (0.005 * sr));
smoothEnvMod += (envMod - smoothEnvMod) * modSlew;

// --- 1. Stereo 4-Stage Hilbert Quadrature Extraction (x_I = cos, x_Q = sin) ---
// Pre-condition high brass fizz slightly so 6th-order Chebyshev recurrences don't intermodulate harshly
let cleanL = 0.75 * preLpL.next(xL, 450.0) + 0.25 * xL;
let cleanR = 0.75 * preLpR.next(xR, 450.0) + 0.25 * xR;

let cR0 = 0.4794008656; let cR1 = 0.8762184935; let cR2 = 0.9765975895; let cR3 = 0.9974992559;
let cI0 = 0.1617584983; let cI1 = 0.7330289323; let cI2 = 0.9453497003; let cI3 = 0.9905991567;

// Quadrature allpass: y[n] = c * (x[n] + y[n-1]) - x[n-1] (H = (c - z^-1)/(1 - c z^-1))
// This cR/cI order is the negative Hilbert; the imaginary branch is negated below.
let reL = cleanL;
let t = cR0 * (reL + lReY[0]) - lReX[0]; lReX[0] = reL; lReY[0] = t; reL = t;
t = cR1 * (reL + lReY[1]) - lReX[1]; lReX[1] = reL; lReY[1] = t; reL = t;
t = cR2 * (reL + lReY[2]) - lReX[2]; lReX[2] = reL; lReY[2] = t; reL = t;
t = cR3 * (reL + lReY[3]) - lReX[3]; lReX[3] = reL; lReY[3] = t; reL = t;
let rawIL = lDelRe; lDelRe = reL;

let imL = cleanL;
t = cI0 * (imL + lImY[0]) - lImX[0]; lImX[0] = imL; lImY[0] = t; imL = t;
t = cI1 * (imL + lImY[1]) - lImX[1]; lImX[1] = imL; lImY[1] = t; imL = t;
t = cI2 * (imL + lImY[2]) - lImX[2]; lImX[2] = imL; lImY[2] = t; imL = t;
t = cI3 * (imL + lImY[3]) - lImX[3]; lImX[3] = imL; lImY[3] = t; imL = t;
let rawQL = 0.0 - imL; // negate: this order is the negative Hilbert

let reR = cleanR;
t = cR0 * (reR + rReY[0]) - rReX[0]; rReX[0] = reR; rReY[0] = t; reR = t;
t = cR1 * (reR + rReY[1]) - rReX[1]; rReX[1] = reR; rReY[1] = t; reR = t;
t = cR2 * (reR + rReY[2]) - rReX[2]; rReX[2] = reR; rReY[2] = t; reR = t;
t = cR3 * (reR + rReY[3]) - rReX[3]; rReX[3] = reR; rReY[3] = t; reR = t;
let rawIR = rDelRe; rDelRe = reR;

let imR = cleanR;
t = cI0 * (imR + rImY[0]) - rImX[0]; rImX[0] = imR; rImY[0] = t; imR = t;
t = cI1 * (imR + rImY[1]) - rImX[1]; rImX[1] = imR; rImY[1] = t; imR = t;
t = cI2 * (imR + rImY[2]) - rImX[2]; rImX[2] = imR; rImY[2] = t; imR = t;
t = cI3 * (imR + rImY[3]) - rImX[3]; rImX[3] = imR; rImY[3] = t; imR = t;
let rawQR = 0.0 - imR; // negate: this order is the negative Hilbert

// Radial unit-disk saturation: preserves exact quadrature angle (cos(theta), sin(theta))
// while bounding radius r_sat in [0, 1) so Chebyshev T_k and U_k never blow up or clip!
let drive = dbtoa(clamp(drive_db, 0.0, 18.0));
let magL = sqrt(rawIL * rawIL + rawQL * rawQL + 0.00000001);
let satMagL = tanh(magL * drive);
let scaleL = satMagL / magL;
let xIL = rawIL * scaleL;
let xQL = rawQL * scaleL;

let magR = sqrt(rawIR * rawIR + rawQR * rawQR + 0.00000001);
let satMagR = tanh(magR * drive);
let scaleR = satMagR / magR;
let xIR = rawIR * scaleR;
let xQR = rawQR * scaleR;

// Dynamic envelope window (r_sat^2) for even-order T_2, T_4, T_6 so their constant terms
// (-1, +1, -1) vanish smoothly at zero signal and never cause gate/breath thumps!
let envWinL = satMagL * satMagL;
let envWinR = satMagR * satMagR;

// --- 2. Compute Chebyshev Polynomials of 1st Kind (T_1..T_6) & 2nd Kind (U_0..U_5) ---
// Recurrences: T_k(x) = 2*x*T_{k-1}(x) - T_{k-2}(x)
//              U_k(x) = 2*x*U_{k-1}(x) - U_{k-2}(x), where sin(k*w*t) = sin(w*t) * U_{k-1}(cos(w*t))
// Left Channel:
let T1L = xIL;
let T2L = (2.0 * xIL * T1L - 1.0) * envWinL;
let T3L = 2.0 * xIL * (2.0 * xIL * T1L - 1.0) - T1L;
let T4L = (2.0 * xIL * T3L - (2.0 * xIL * T1L - 1.0)) * envWinL;
let T5L = 2.0 * xIL * (2.0 * xIL * T3L - (2.0 * xIL * T1L - 1.0)) - T3L;
let T6L = (2.0 * xIL * T5L - (2.0 * xIL * T3L - (2.0 * xIL * T1L - 1.0))) * envWinL;

let U0L = 1.0;
let U1L = 2.0 * xIL;
let U2L = 2.0 * xIL * U1L - U0L;
let U3L = 2.0 * xIL * U2L - U1L;
let U4L = 2.0 * xIL * U3L - U2L;
let U5L = 2.0 * xIL * U4L - U3L;

// Right Channel:
let T1R = xIR;
let T2R = (2.0 * xIR * T1R - 1.0) * envWinR;
let T3R = 2.0 * xIR * (2.0 * xIR * T1R - 1.0) - T1R;
let T4R = (2.0 * xIR * T3R - (2.0 * xIR * T1R - 1.0)) * envWinR;
let T5R = 2.0 * xIR * (2.0 * xIR * T3R - (2.0 * xIR * T1R - 1.0)) - T3R;
let T6R = (2.0 * xIR * T5R - (2.0 * xIR * T3R - (2.0 * xIR * T1R - 1.0))) * envWinR;

let U0R = 1.0;
let U1R = 2.0 * xIR;
let U2R = 2.0 * xIR * U1R - U0R;
let U3R = 2.0 * xIR * U2R - U1R;
let U4R = 2.0 * xIR * U3R - U2R;
let U5R = 2.0 * xIR * U4R - U3R;

// --- 3. Harmonic Phase Angle Rotation phi_k (Paper Eq. 10-15) ---
// Glitch-free bar phasor detection (100 ms hold prevents wrap-around dropout clicks every 2 seconds!)
if (abs(barIn - prevBarIn) > 0.0000001) {
	barHold = 0.1 * sr;
} else if (barHold > 0.0) {
	barHold -= 1.0;
}
prevBarIn = barIn;
let fbPh = barFallback.next(0.5);
let spinPh = (barHold > 0.0) ? wrap(barIn * 2.0, 0.0, 1.0) : fbPh;
let spinAngle = 2.0 * pi * spinPh * clamp(phase_spin, 0.0, 1.0);

let baseAngle = clamp(quad_phase_angle, -3.14159, 3.14159) + smoothEnvMod * pi;
let stOffset = clamp(stereo_quad_spread, 0.0, 1.0) * 0.5 * pi;

// Progressive harmonic phase dispersion: phi_k = k * baseAngle + spinAngle
let evAmt = pow(clamp(even_harmonics, 0.0, 1.0), 1.2);
let odAmt = pow(clamp(odd_harmonics, 0.0, 1.0), 1.2);
let hiTilt = clamp(high_order_tilt, 0.0, 1.0);

let a2 = evAmt * (0.75 - 0.25 * hiTilt);
let a3 = odAmt * (0.75 - 0.25 * hiTilt);
let a4 = evAmt * (0.25 + 0.55 * hiTilt);
let a5 = odAmt * (0.25 + 0.55 * hiTilt);
let a6 = evAmt * (0.15 + 0.50 * hiTilt);

let p2L = 2.0 * baseAngle + spinAngle;       let p2R = 2.0 * (baseAngle + stOffset) - spinAngle;
let p3L = 3.0 * baseAngle + spinAngle * 2.0; let p3R = 3.0 * (baseAngle + stOffset) - spinAngle * 2.0;
let p4L = 4.0 * baseAngle + spinAngle * 3.0; let p4R = 4.0 * (baseAngle + stOffset) - spinAngle * 3.0;
let p5L = 5.0 * baseAngle + spinAngle * 4.0; let p5R = 5.0 * (baseAngle + stOffset) - spinAngle * 4.0;
let p6L = 6.0 * baseAngle + spinAngle * 5.0; let p6R = 6.0 * (baseAngle + stOffset) - spinAngle * 5.0;

// Paper Eq. 15: y(t) = sum_k a_k * [ cos(phi_k)*T_k(x_I) + sin(phi_k)*x_Q*U_{k-1}(x_I) ]
let sumL = T1L
	+ a2 * (cos(p2L) * T2L + sin(p2L) * xQL * U1L * envWinL)
	+ a3 * (cos(p3L) * T3L + sin(p3L) * xQL * U2L)
	+ a4 * (cos(p4L) * T4L + sin(p4L) * xQL * U3L * envWinL)
	+ a5 * (cos(p5L) * T5L + sin(p5L) * xQL * U4L)
	+ a6 * (cos(p6L) * T6L + sin(p6L) * xQL * U5L * envWinL);

let sumR = T1R
	+ a2 * (cos(p2R) * T2R + sin(p2R) * xQR * U1R * envWinR)
	+ a3 * (cos(p3R) * T3R + sin(p3R) * xQR * U2R)
	+ a4 * (cos(p4R) * T4R + sin(p4R) * xQR * U3R * envWinR)
	+ a5 * (cos(p5R) * T5R + sin(p5R) * xQR * U4R)
	+ a6 * (cos(p6R) * T6R + sin(p6R) * xQR * U5R * envWinR);

let norm = 1.0 / (1.0 + 0.50 * (a2 + a3 + a4 + a5 + a6));
let yL = sumL * norm;
let yR = sumR * norm;

// DC blocker
let outL = yL - dcXL + 0.995 * dcYL;
dcXL = yL; dcYL = outL;
let outR = yR - dcXR + 0.995 * dcYR;
dcXR = yR; dcYR = outR;

out1 = tanh(outL * 1.15);
out2 = tanh(outR * 1.15);
