// ============================================================================
// vector_phaseshaper_lissajous.txt
// Unix-Style Stereo Effect: 2D Vector Phaseshaping (VPS) & Lissajous Formant Engine
// Implements Kleimola, Lazzarini, Timoney & Valimaki (DAFx-11, Paper 55_e):
// "Vector Phaseshaping Synthesis" (Sec 3, 3.1, 3.2, 4.3, 5.2, Eq. 5-12, 20).
// Takes the phase-locked fundamental phasor phBase(t) in [0, 1) of the tuba,
// warps it through a 2D breakpoint vector p = (d, v) with incomplete-period
// alias suppression (Eq. 11-12), and modulates (d, v) along LFO or audio-rate
// Lissajous trajectories (Eq. 20) to synthesize vocal formants centered at
// f_f = (2*v - 1) * f0.
// ============================================================================
// Inlets:
//   in1: Left Audio Input (Tuba or 01_harmonic_isolator)
//   in2: Right Audio Input (if unpatched, mono in1 is used automatically)
//   in3: Tracked Fundamental Pitch f0 in Hz (from tuba_freq_tracker_v2; fallback ZC if <=0)
//   in4: Optional 1-Bar Phasor [0..1] (drives Lissajous LFO when audio_rate_mod < 1.0)
//   in5: Optional Phase-Locked Tuba Phasor [0..1) (tuba_freq_tracker_v2 out3).
//        While it moves the voice is locked to it; static -> internal Hilbert PLL,
//        continued from the last phase.
//   in6: Optional Smoothed Tuba Envelope [0..1] (e.g. cascade_env_follower).
//        Static for 250 ms -> internal 4-stage cascaded follower.
// Outlets:
//   out1: Left Vector-Phaseshaped Formant Output
//   out2: Right Vector-Phaseshaped Formant Output

// Breakpoint x (d)
@param({ min: 0.05, max: 0.95 }) vec_d = 0.5;
// Breakpoint y (v): formant height
@param({ min: 0.5, max: 6.0 }) vec_v_formant = 2.2;
// Lissajous orbit depth
@param({ min: 0.0, max: 1.0 }) liss_depth = 0.35;
// Lissajous v:d ratio
@param({ min: 1.0, max: 4.0, steps: 4 }) liss_ratio = 2.0;
// LFO -> audio-rate orbit
@param({ min: 0.0, max: 1.0 }) audio_rate_mod = 0.0;
// Incomplete-period alias suppression
@param({ min: 0.0, max: 1.0 }) alias_suppress = 0.85;
// Gate sustain -> tuba envelope
@param({ min: 0.0, max: 1.0 }) env_follow = 0.85;

// Fallback phase lock: linear TPT bandpass + 4-stage Quadrature IIR Hilbert network
@state bpIc1 = 0.0; @state bpIc2 = 0.0;
@state hReX = new FixedSampleArray(4); @state hReY = new FixedSampleArray(4);
@state hImX = new FixedSampleArray(4); @state hImY = new FixedSampleArray(4);
@state hDelRe = 0.0;
@state pllPhase = 0.0;
@state prevPhIn = 0.0;
@state phInHold = 0.0;

// Lissajous modulators, DC blockers & fallback trackers
@state lfoPhase = 0.0;
@state prevBarIn = 0.0;
@state barHold = 0.0;
@state lfoDL = 0.0; @state lfoVL = 1.0;
@state lfoDR = 1.0; @state lfoVR = 0.0;
@state lpTrack = new onepole();
@state prevSig = 0.0;
@state zcCount = 0.0;
@state zcHz = 58.27;
@state smoothF0 = 58.27;
@state envCasc = new FixedSampleArray(4);
@state prevEnvIn = 0.0;
@state envInHold = 0.0;
@state gateSm = 0.0;
@state ampSm = 0.0;
@state smoothD = 0.5;
@state smoothV = 2.2;
@state in2Hold = 0.0;
@state dcXL = 0.0; @state dcYL = 0.0;
@state dcXR = 0.0; @state dcYR = 0.0;

// VPS 2D piecewise phase distortion + DAFx-11 Sec 3.2 Alias Suppression (Eq. 6, 11, 12)
function evalVPS(xIn: number, dIn: number, vIn: number, suppressAmt: number): number {
	let x = wrap(xIn, 0.0, 1.0);
	let d = clamp(dIn, 0.04, 0.96);
	let v = maximum(vIn, 0.25);

	// Eq. 6: 2D Vector Phaseshaping function phi_vps(x)
	let p = (x < d) ? ((v * x) / d) : ((1.0 - v) * ((x - d) / (1.0 - d)) + v);
	let rawCos = 0.0 - cos(2.0 * pi * p);

	// Eq. 11 & 12: Incomplete period alias suppression when p > floor(v).
	// Only for v >= 1: below that the shaper is one complete (classic PD) period
	// and there is no incomplete period to fold.
	let vFloor = floor(v);
	let b = v - vFloor;
	let sa = rawCos;

	if (vFloor >= 1.0 && p > vFloor && b > 0.001 && b < 0.999) {
		let pMod = p - vFloor;
		let c = cos(2.0 * pi * b);
		// 0 < b <= 0.5 branch: half a sinusoid over the incomplete period
		let phLo = clamp(pMod / (2.0 * b), 0.0, 1.0);
		let saLo = 0.5 * ((1.0 - c) * (0.0 - cos(2.0 * pi * phLo)) - 1.0 - c);
		// 0.5 < b < 1 branch: a full sinusoid, the second half scaled/offset.
		// The first half is s(n) itself (not the trivial cosine), so the
		// segment is continuous at phHi = 0.5.
		let phHi = clamp(pMod / b, 0.0, 1.0);
		let sHi = 0.0 - cos(2.0 * pi * phHi);
		let saHi = (phHi > 0.5) ? (0.5 * ((1.0 + c) * sHi + 1.0 - c)) : sHi;
		// The two branches give different shapes at b = 0.5; crossfading them
		// over b = 0.35..0.65 keeps the waveform continuous while v sweeps.
		let wHi = clamp((b - 0.35) / 0.3, 0.0, 1.0);
		wHi = wHi * wHi * (3.0 - 2.0 * wHi);
		sa = saLo + (saHi - saLo) * wHi;
	}
	return linearinterp(clamp(suppressAmt, 0.0, 1.0), rawCos, sa);
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
// Mono fallback: in2 counts as patched for 250 ms after its last non-zero sample
if (abs(in2) > 0.0000001) {
	in2Hold = 0.25 * sr;
} else if (in2Hold > 0.0) {
	in2Hold -= 1.0;
}
let xR = (in2Hold > 0.0) ? in2 : xL;
let pitchIn = in3;
let barIn = in4;
let phIn = in5;
let envIn = in6;

// --- 1. Pitch Conditioning ---
let mono = 0.5 * (xL + xR);
let monoLow = lpTrack.next(mono, 190.0);
zcCount += 1.0;
if (prevSig <= 0.0 && monoLow > 0.0) {
	if (zcCount > sr / 500.0 && zcCount < sr / 22.0) {
		zcHz = sr / zcCount;
	}
	zcCount = 0.0;
}
prevSig = monoLow;

let rawF0 = (pitchIn > 20.0 && pitchIn < 1000.0) ? pitchIn : zcHz;
let slew = 1.0 - exp(-1.0 / (0.006 * sr));
smoothF0 += (rawF0 - smoothF0) * slew;
let f0 = clamp(smoothF0, 24.0, 1000.0);

// --- 2. Envelope: in6 while it moves, else internal 4-stage cascade (5 ms / 120 ms) ---
// A single fast peak follower ripples ~12% at f0 on low notes (audible AM buzz);
// the cascade keeps it under 2% (Sanfilippo, SMC-22, see cascade_env_follower).
if (abs(envIn - prevEnvIn) > 0.0000001) {
	envInHold = 0.25 * sr;
} else if (envInHold > 0.0) {
	envInHold -= 1.0;
}
prevEnvIn = envIn;
let fullAmp = maximum(abs(xL), abs(xR));
let envAtk = exp(-4.3513 / (0.005 * sr));
let envRel = exp(-4.3513 / (0.12 * sr));
let ev = fullAmp;
for (let i: Int = 0; i < 4; i++) {
	let s = envCasc[i];
	let a = (ev > s) ? envAtk : envRel;
	s = ev + a * (s - ev);
	envCasc[i] = s;
	ev = s;
}
let env = (envInHold > 0.0 && envIn >= 0.0 && envIn <= 2.0) ? envIn : ev;

// --- 3. Phase: in5 while it moves, else Hilbert PLL on a linear bandpass ---
let kd = 0.25; // Q 4
let g = tan(pi * clamp(f0, 20.0, sr * 0.45) / sr);
let a1 = 1.0 / (1.0 + g * (g + kd));
let a2 = g * a1;
let a3 = g * a2;
let v3 = mono - bpIc2;
let v1 = a1 * bpIc1 + a2 * v3;
let v2 = bpIc2 + a2 * bpIc1 + a3 * v3;
bpIc1 = 2.0 * v1 - bpIc1;
bpIc2 = 2.0 * v2 - bpIc2;
let isoFund = kd * v1;

let cR0 = 0.4794008656; let cR1 = 0.8762184935; let cR2 = 0.9765975895; let cR3 = 0.9974992559;
let cI0 = 0.1617584983; let cI1 = 0.7330289323; let cI2 = 0.9453497003; let cI3 = 0.9905991567;

// Quadrature allpass: y[n] = c * (x[n] + y[n-1]) - x[n-1] (H = (c - z^-1)/(1 - c z^-1))
// This cR/cI order is the negative Hilbert; the imaginary branch is negated below.
let re = isoFund;
let t = cR0 * (re + hReY[0]) - hReX[0]; hReX[0] = re; hReY[0] = t; re = t;
t = cR1 * (re + hReY[1]) - hReX[1]; hReX[1] = re; hReY[1] = t; re = t;
t = cR2 * (re + hReY[2]) - hReX[2]; hReX[2] = re; hReY[2] = t; re = t;
t = cR3 * (re + hReY[3]) - hReX[3]; hReX[3] = re; hReY[3] = t; re = t;
let sigRe = hDelRe;
hDelRe = re;

let im = isoFund;
t = cI0 * (im + hImY[0]) - hImX[0]; hImX[0] = im; hImY[0] = t; im = t;
t = cI1 * (im + hImY[1]) - hImX[1]; hImX[1] = im; hImY[1] = t; im = t;
t = cI2 * (im + hImY[2]) - hImX[2]; hImX[2] = im; hImY[2] = t; im = t;
t = cI3 * (im + hImY[3]) - hImX[3]; hImX[3] = im; hImY[3] = t; im = t;
let sigIm = 0.0 - im; // negate: this order is the negative Hilbert (+f analytic phase)

// in5 counts as live for 100 ms after it last moved, while it stays in 0..1
let phInStep = wrap(phIn - prevPhIn + 0.5, 0.0, 1.0) - 0.5;
if (abs(phIn - prevPhIn) > 0.0000001) {
	phInHold = 0.1 * sr;
} else if (phInHold > 0.0) {
	phInHold -= 1.0;
}
prevPhIn = phIn;
let phInLive = phInHold > 0.0 && phIn >= 0.0 && phIn <= 1.0;

// Phase-Locked Loop: a monotonic [0, 1) ramp. With in5 it advances by in5's own
// step and converges onto in5 exactly; without it, it locks gently to the Hilbert
// phase. The correction is capped at +/-25% of the step, so a re-lock bends the
// pitch a little instead of chirping (an uncapped gain swept it by ~800 Hz).
// NOTE: Named phBase (not phi, which RNBO expands to the golden ratio 1.618!)
let hilbPh = wrap((atan2(sigIm, sigRe) + pi) / (2.0 * pi), 0.0, 1.0);
let phStep = f0 / sr;
let phRef = hilbPh;
let lockGain = (env > 0.002) ? 0.0025 : 0.0;
if (phInLive) {
	phStep = clamp(phInStep, 0.0, 0.05);
	phRef = phIn;
	lockGain = 0.02;
}
let phErr = wrap(phRef - pllPhase + 0.5, 0.0, 1.0) - 0.5;
let phCorr = clamp(lockGain * phErr, -0.25 * phStep, 0.25 * phStep);
pllPhase = wrap(pllPhase + phStep + phCorr, 0.0, 1.0);
let phBase = pllPhase;

// --- 4. 2D Lissajous Vector Modulation (Paper Sec 4.3 & 5.2, Eq. 20) ---
// The bar counts as running for 100 ms after it last moved; otherwise the LFO
// phase continues at 0.75 Hz from where the bar left it.
if (abs(barIn - prevBarIn) > 0.0000001) {
	barHold = 0.1 * sr;
} else if (barHold > 0.0) {
	barHold -= 1.0;
}
prevBarIn = barIn;
lfoPhase = (barHold > 0.0) ? wrap(barIn * 2.0, 0.0, 1.0) : wrap(lfoPhase + 0.75 / sr, 0.0, 1.0);
// Audio-rate Lissajous modulation is phase-locked directly to phBase (Paper Sec 5.2)
let audPh = phBase;

// Blend between bar-synced LFO Lissajous and pitch-synchronous audio-rate Lissajous (Sec 5.2)
let arBlend = clamp(audio_rate_mod, 0.0, 1.0);
let ratioN = floor(clamp(liss_ratio, 1.0, 4.0) + 0.5);

// LFO figure is smoothed (5 ms) so a bar start/stop or ratio step glides
let lfoSm = 1.0 - exp(-1.0 / (0.005 * sr));
lfoDL += (cos(2.0 * pi * lfoPhase + 0.5 * pi) - lfoDL) * lfoSm;
lfoVL += (cos(2.0 * pi * ratioN * lfoPhase) - lfoVL) * lfoSm;
lfoDR += (sin(2.0 * pi * lfoPhase + 0.5 * pi) - lfoDR) * lfoSm;
lfoVR += (sin(2.0 * pi * ratioN * lfoPhase) - lfoVR) * lfoSm;

let lissDL = linearinterp(arBlend, lfoDL, cos(2.0 * pi * audPh + 0.5 * pi));
let lissVL = linearinterp(arBlend, lfoVL, cos(2.0 * pi * ratioN * audPh));

// Right channel has quadrature stereo offset on the Lissajous figure
let lissDR = linearinterp(arBlend, lfoDR, sin(2.0 * pi * audPh + 0.5 * pi));
let lissVR = linearinterp(arBlend, lfoVR, sin(2.0 * pi * ratioN * audPh));

smoothD += (clamp(vec_d, 0.05, 0.95) - smoothD) * slew;
smoothV += (clamp(vec_v_formant, 0.5, 6.0) - smoothV) * slew;

let depth = pow(clamp(liss_depth, 0.0, 1.0), 1.25);
let vModL = clamp(smoothV + depth * 1.8 * lissVL, 0.35, 7.5);
let vModR = clamp(smoothV + depth * 1.8 * lissVR, 0.35, 7.5);
// Anti-alias guard: segment 1 sweeps v cycles in d, segment 2 sweeps |1 - v| in
// 1 - d, so their instantaneous frequencies are v/d*f0 and |1-v|/(1-d)*f0. Keep
// both under 0.2*sr; past that a tiny d with a high v (or a high f0) turns each
// period into an aliased click. Inactive at normal settings (v 2.2, f0 58 Hz:
// d >= 0.013).
let fRate = phStep * sr / (0.2 * sr);
let dLoL = clamp(vModL * fRate, 0.04, 0.45);
let dHiL = 1.0 - clamp(abs(1.0 - vModL) * fRate, 0.04, 0.45);
let dLoR = clamp(vModR * fRate, 0.04, 0.45);
let dHiR = 1.0 - clamp(abs(1.0 - vModR) * fRate, 0.04, 0.45);
let dModL = clamp(smoothD + depth * 0.38 * lissDL, dLoL, dHiL);
let dModR = clamp(smoothD + depth * 0.38 * lissDR, dLoR, dHiR);

// --- 5. Evaluate Stereo VPS with Alias Suppression & DC Blocker ---
let rawYL = evalVPS(phBase, dModL, vModL, alias_suppress);
let rawYR = evalVPS(wrap(phBase + 0.015, 0.0, 1.0), dModR, vModR, alias_suppress);

// DC blocker (DAFx-11 Section 3.2 notes that incomplete-period folding introduces a DC offset)
let dcPole = 1.0 - (2.0 * pi * 10.0 / sr); // 10 Hz (a fixed 0.995 was 38 Hz at 48 kHz: -1.6 dB at 58 Hz, -4 dB at 29 Hz)
let yL = rawYL - dcXL + dcPole * dcYL;
dcXL = rawYL; dcYL = yL;
let yR = rawYR - dcXR + dcPole * dcYR;
dcXR = rawYR; dcYR = yR;

// Apply natural tuba envelope or gated unit synthesis level.
// The gate is a smoothstep over env 0.002..0.007 followed by a 4 ms / 40 ms
// one-pole, so it can neither snap on nor chatter on a decaying note.
let gTarget = clamp((env - 0.002) * 200.0, 0.0, 1.0);
gTarget = gTarget * gTarget * (3.0 - 2.0 * gTarget);
let gCoef = (gTarget > gateSm) ? (1.0 - exp(-1.0 / (0.004 * sr))) : (1.0 - exp(-1.0 / (0.04 * sr)));
gateSm += (gTarget - gateSm) * gCoef;
let ampTarget = linearinterp(clamp(env_follow, 0.0, 1.0), gateSm * 0.75, clamp(env * 0.9, 0.0, 1.3));
ampSm += (ampTarget - ampSm) * (1.0 - exp(-1.0 / (0.001 * sr)));

out1 = tanh(yL * ampSm);
out2 = tanh(yR * ampSm);
