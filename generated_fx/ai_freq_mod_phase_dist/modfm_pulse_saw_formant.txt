// ============================================================================
// modfm_pulse_saw_formant.txt
// Unix-Style Stereo Effect: Modified FM (ModFM) Bandlimited Pulse, Saw & Formant Engine
// Implements Timoney, Lazzarini & Lysaght (DAFx-08, Paper 05):
// "A Modified FM Synthesis Approach to Bandlimited Signal Generation"
// (Sections 2, 3 & 4, Eq. 1-4, Fig. 1-7) combined with Phase-Synchronous ModFM
// Formant Crossfading (Lazzarini & Timoney JAES 2010 / DAFx-11 Ref. 11 & 18).
// Modulates the tuba's carrier (or resynthesizes from its phase-locked fundamental)
// using strictly real exponential cosine modulation:
//   y_pulse(t) = exp(k * (cos(w_0 * t) - 1)) * cos(w_c * t)   (Eq. 1)
// Because the sidebands are governed by scaled Modified Bessel functions
// exp(-k) * I_n(k) (Eq. 2, Fig. 4), harmonic magnitudes decay monotonically
// without classic FM Bessel zero-crossings and never exceed 0 dB for any k >= 0!
// Includes mean removal + leaky integration + DC blocking to transform the ModFM
// pulse train into a fat, bandlimited ModFM Sawtooth (Sec 3, Eq. 3-4, Fig. 5-6).
// ============================================================================
// Inlets:
//   in1: Left Audio Input (Tuba)
//   in2: Right Audio Input (if unpatched, mono in1 is used automatically)
//   in3: Tracked Fundamental Pitch f0 in Hz (from 00_tuba_freq_tracker; fallback ZC if <=0)
//   in4: Optional 1-Bar Phasor [0..1] (sweeps ModFM bandwidth index k when k_wobble > 0)
// Outlets:
//   out1: Left ModFM Pulse / Saw / Formant Output
//   out2: Right ModFM Pulse / Saw / Formant Output

@param({ min: 0.1, max: 60.0 }) modfm_index_k = 12.0;
@param({ min: 1.0, max: 12.0 }) formant_harmonic = 1.0;
@param({ min: 0.0, max: 1.0 }) pulse_to_saw = 0.5;
@param({ min: 0.0, max: 1.0 }) tuba_carrier_mix = 0.4;
@param({ min: 0.0, max: 1.0 }) k_wobble = 0.35;
@param({ min: 0.0, max: 1.0 }) stereo_width = 0.25;

// Fundamental isolation SVF + 4-stage Quadrature IIR Hilbert network for PLL phase lock
@state fundSvf = new svf();
@state hReX = new FixedSampleArray(4); @state hReY = new FixedSampleArray(4);
@state hImX = new FixedSampleArray(4); @state hImY = new FixedSampleArray(4);
@state hDelRe = 0.0;
@state pllPhase = 0.0;

// Leaky Integrator + DC Blocker states for ModFM Sawtooth (Paper Sec 3, Fig. 5, Eq. 3)
@state runningMeanL = 0.0; @state runningMeanR = 0.0;
@state integStateL = 0.0;  @state integStateR = 0.0;
@state dcXL = 0.0; @state dcYL = 0.0;
@state dcXR = 0.0; @state dcYR = 0.0;

// Fallback trackers & smoothing
@state barFallback = new phasor();
@state prevBarIn = 0.0;
@state barHold = 0.0;
@state lpTrack = new onepole();
@state prevSig = 0.0;
@state zcCount = 0.0;
@state zcHz = 58.27;
@state smoothF0 = 58.27;
@state smoothEnv = 0.0;
@state smoothK = 12.0;
@state smoothHarm = 1.0;
@state in2Hold = 0.0;
@state outXL = 0.0; @state outYL = 0.0;
@state outXR = 0.0; @state outYR = 0.0;

let sr = maximum(samplerate(), 1.0);
let xL = in1;
// Mono fallback: in2 counts as patched for 250 ms after its last non-zero sample
if (abs(in2) > 0.0000001) {
	in2Hold = 0.25 * sr;
} else if (in2Hold > 0.0) {
	in2Hold -= 1.0;
}
let xR = (in2Hold > 0.0) ? in2 : xL;
let pitchIn = in3;
let barIn = in4;

// --- 1. Pitch Conditioning ---
let mono = 0.5 * (xL + xR);
let monoLow = lpTrack.next(mono, 190.0);
zcCount += 1.0;
if (prevSig <= 0.0 && monoLow > 0.0) {
	if (zcCount > sr / 500.0 && zcCount < sr / 22.0) {
		zcHz = sr / zcCount;
	}
	zcCount = 0.0;
}
prevSig = monoLow;

let rawF0 = (pitchIn > 20.0 && pitchIn < 1000.0) ? pitchIn : zcHz;
let slew = 1.0 - exp(-1.0 / (0.006 * sr));
smoothF0 += (rawF0 - smoothF0) * slew;
let f0 = clamp(smoothF0, 24.0, 1000.0);

// --- 2. Extract Phase-Locked Fundamental Phasor phBase & Hilbert Envelope ---
// Isolate the fundamental before Hilbert phase extraction so overtones don't cause intra-period phase reversals
let isoFund = fundSvf.next(mono, f0, 0.80)[2];

let cR0 = 0.4794008656; let cR1 = 0.8762184935; let cR2 = 0.9765975895; let cR3 = 0.9974992559;
let cI0 = 0.1617584983; let cI1 = 0.7330289323; let cI2 = 0.9453497003; let cI3 = 0.9905991567;

// Quadrature allpass: y[n] = c * (x[n] + y[n-1]) - x[n-1] (H = (c - z^-1)/(1 - c z^-1))
// This cR/cI order is the negative Hilbert; the imaginary branch is negated below.
let re = isoFund;
let t = cR0 * (re + hReY[0]) - hReX[0]; hReX[0] = re; hReY[0] = t; re = t;
t = cR1 * (re + hReY[1]) - hReX[1]; hReX[1] = re; hReY[1] = t; re = t;
t = cR2 * (re + hReY[2]) - hReX[2]; hReX[2] = re; hReY[2] = t; re = t;
t = cR3 * (re + hReY[3]) - hReX[3]; hReX[3] = re; hReY[3] = t; re = t;
let sigRe = hDelRe;
hDelRe = re;

let im = isoFund;
t = cI0 * (im + hImY[0]) - hImX[0]; hImX[0] = im; hImY[0] = t; im = t;
t = cI1 * (im + hImY[1]) - hImX[1]; hImX[1] = im; hImY[1] = t; im = t;
t = cI2 * (im + hImY[2]) - hImX[2]; hImX[2] = im; hImY[2] = t; im = t;
t = cI3 * (im + hImY[3]) - hImX[3]; hImX[3] = im; hImY[3] = t; im = t;
let sigIm = 0.0 - im; // negate: this order is the negative Hilbert (+f analytic phase)

// Full-band RMS/peak envelope of the live tuba
let fullAmp = maximum(abs(xL), abs(xR));
let envAtk = 1.0 - exp(-1.0 / (0.0025 * sr));
let envRel = 1.0 - exp(-1.0 / (0.025 * sr));
smoothEnv = (fullAmp > smoothEnv) ? (smoothEnv + (fullAmp - smoothEnv) * envAtk) : (smoothEnv + (fullAmp - smoothEnv) * envRel);

// Phase-Locked Loop (PLL) locks a clean monotonic [0, 1) ramp to the tuba's fundamental analytic phase
let hilbPh = wrap((atan2(sigIm, sigRe) + pi) / (2.0 * pi), 0.0, 1.0);
let phErr = wrap(hilbPh - pllPhase + 0.5, 0.0, 1.0) - 0.5;
let lockGain = (smoothEnv > 0.002) ? 0.035 : 0.0;
pllPhase = wrap(pllPhase + (f0 / sr) + lockGain * phErr, 0.0, 1.0);
let phBase = pllPhase;

// --- 3. ModFM Index k Calculation & Anti-Aliasing Ceiling (Paper Fig. 7) ---
// Glitch-free bar phasor activity detector (100 ms hold prevents wrap-around dropout clicks)
if (abs(barIn - prevBarIn) > 0.0000001) {
	barHold = 0.1 * sr;
} else if (barHold > 0.0) {
	barHold -= 1.0;
}
prevBarIn = barIn;
let fbPh = barFallback.next(0.5);
let barPh = (barHold > 0.0) ? wrap(barIn * 2.0, 0.0, 1.0) : fbPh;
let wobMod = 0.5 * (1.0 - cos(2.0 * pi * barPh));

// Smooth ModFM index k (0.1 to 60.0) and formant harmonic ratio (1.0 to 12.0)
let kParam = clamp(modfm_index_k, 0.1, 60.0);
smoothK += (kParam - smoothK) * slew;
smoothHarm += (clamp(formant_harmonic, 1.0, 12.0) - smoothHarm) * slew;

// Automatic pitch-dependent k clamp so upper harmonics never alias past Nyquist (Paper Fig. 7)
let maxSafeK = clamp((sr * 0.42) / maximum(f0 * smoothHarm, 25.0), 1.0, 80.0);
let wobMultiplier = 1.0 + clamp(k_wobble, 0.0, 1.0) * 2.5 * wobMod;
let width = clamp(stereo_width, 0.0, 1.0);

let kL = clamp(smoothK * wobMultiplier * (1.0 - 0.25 * width), 0.1, maxSafeK);
let kR = clamp(smoothK * wobMultiplier * (1.0 + 0.25 * width), 0.1, maxSafeK);

// --- 4. ModFM Bandlimited Pulse Train Generation (Paper Sec 2, Eq. 1) ---
// y_pulse(t) = exp(k * (cos(w0*t) - 1)) * carrier(t)
let phR = wrap(phBase + 0.06 * width, 0.0, 1.0);
let modExpL = exp(kL * (cos(2.0 * pi * phBase) - 1.0));
let modExpR = exp(kR * (cos(2.0 * pi * phR) - 1.0));

// Dual-harmonic crossfaded formant carrier for click-free formant sweeps (Lazzarini & Timoney JAES 2010)
let hInt = floor(smoothHarm);
let hFrac = smoothHarm - hInt;
let synthCarL = (1.0 - hFrac) * cos(2.0 * pi * hInt * phBase) + hFrac * cos(2.0 * pi * (hInt + 1.0) * phBase);
let synthCarR = (1.0 - hFrac) * cos(2.0 * pi * hInt * phR) + hFrac * cos(2.0 * pi * (hInt + 1.0) * phR);

// Compensate narrow pulse duty-cycle energy drop at high k:
// Since integral(exp(2k*(cos(theta)-1))) ~ 1 / sqrt(4*pi*k), RMS drops as k^(-0.25).
// Multiplying by (1 + 0.45 * k)^0.28 keeps perceived loudness rock-solid across k = 0.1..60!
let pulseCompL = pow(1.0 + 0.45 * kL, 0.28);
let pulseCompR = pow(1.0 + 0.45 * kR, 0.28);

let envGain = clamp(smoothEnv * 1.4, 0.0, 1.2);
let carMix = clamp(tuba_carrier_mix, 0.0, 1.0);
let carSigL = linearinterp(carMix, synthCarL * envGain, xL * (1.0 + 0.6 * modExpL));
let carSigR = linearinterp(carMix, synthCarR * envGain, xR * (1.0 + 0.6 * modExpR));

let pulseL = modExpL * carSigL * pulseCompL;
let pulseR = modExpR * carSigR * pulseCompR;

// --- 5. Bandlimited Sawtooth Integration Stage (Paper Sec 3, Eq. 3-4, Fig. 5-6) ---
// When formant_harmonic == 1, ModFM pulse has positive DC mean exp(-k)*I_0(k) ~ 1/sqrt(2*pi*k).
// Subtract running mean at cutoff 0.35 * f0 so the leaky integrator integrates zero-mean pulses into a clean ramp!
let meanCoeff = 1.0 - exp(-2.0 * pi * (f0 * 0.35) / sr);
runningMeanL += (modExpL * carSigL - runningMeanL) * meanCoeff;
runningMeanR += (modExpR * carSigR - runningMeanR) * meanCoeff;

let zeroMeanPulseL = modExpL * carSigL - runningMeanL;
let zeroMeanPulseR = modExpR * carSigR - runningMeanR;

// Pitch-tracked leaky integrator pole g_leak = exp(-2*pi * 0.15 * f0 / sr)
// and period-normalized integration gain so sawtooth amplitude matches pulse amplitude at any f0 and k!
let leakPole = clamp(exp(-2.0 * pi * (f0 * 0.15) / sr), 0.96, 0.9995);
let normSawL = (f0 / sr) * 3.4 * sqrt(1.0 + kL * 0.85 * smoothHarm);
let normSawR = (f0 / sr) * 3.4 * sqrt(1.0 + kR * 0.85 * smoothHarm);

integStateL = leakPole * integStateL + zeroMeanPulseL * normSawL;
integStateR = leakPole * integStateR + zeroMeanPulseR * normSawR;

// High-pass DC Blocker after integration (Paper Fig. 5)
let dcPole = 1.0 - (2.0 * pi * 10.0 / sr); // 10 Hz (a fixed 0.995 was 38 Hz at 48 kHz: -1.6 dB at 58 Hz, -4 dB at 29 Hz)
let sawL = integStateL - dcXL + dcPole * dcYL;
dcXL = integStateL; dcYL = sawL;

let sawR = integStateR - dcXR + dcPole * dcYR;
dcXR = integStateR; dcYR = sawR;

// 15 Hz DC blocker on the blend: the ModFM pulse is unipolar (mean exp(-k)*I_1(k)
// at formant 1), so without it the pulse side thumped with the breath envelope
let sawBlend = clamp(pulse_to_saw, 0.0, 1.0);
let blendL = linearinterp(sawBlend, pulseL, sawL);
let blendR = linearinterp(sawBlend, pulseR, sawR);
let outDc = 1.0 - (2.0 * pi * 15.0 / sr);
let cleanL = blendL - outXL + outDc * outYL; outXL = blendL; outYL = cleanL;
let cleanR = blendR - outXR + outDc * outYR; outXR = blendR; outYR = cleanR;
out1 = tanh(cleanL);
out2 = tanh(cleanR);
