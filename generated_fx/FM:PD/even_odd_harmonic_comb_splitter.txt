// ============================================================================
// even_odd_harmonic_comb_splitter.txt
// Unix-Style Stereo Utility/Effect: Universal Comb Even/Odd Harmonic Separator
// Implements DAFx-21 (Paper 2):
// "Separation of Harmonic Signals" (Sections 2.3 & 4.1-4.5, Eq. 1-18, Fig. 4-6).
// Uses a canonical single-delay-line Universal Comb Filter (UCF) tuned to the
// second harmonic f2 = 2 * f0 (delay M = fs / (2 * f0) = half fundamental period):
//   xh[n] = x[n] + a_M * xh[n - M]
//   y_even[n] = b_0 * xh[n] + b_M * xh[n - M]     (peaks at 2*f0, 4*f0, 6*f0...)
//   y_odd[n]  = b_0 * xh[n] - b_M * xh[n - M]     (peaks shifted by pi/M -> f0, 3*f0, 5*f0...)
// With exact unity-gain Q normalization (Eq. 10 & 14):
//   beta = tan(pi / (2*Q)),  a_M = (1 - beta)/(1 + beta),  b_0 = b_M = beta/(1 + beta)
// Separates the live stereo tuba into EVEN harmonics and ODD harmonics so they
// can be routed to separate distortion/FM chains (out1..out4) OR differentially
// driven inside the module without intermodulation mud!
// ============================================================================
// Inlets:
//   in1: Left Audio Input (Tuba)
//   in2: Right Audio Input (if unpatched, mono in1 is used automatically)
//   in3: Tracked Fundamental Pitch f0 in Hz (from 00_tuba_freq_tracker; fallback ZC if <=0)
//   in4: Optional 1-Bar Phasor [0..1] (modulates odd/even tilt when tilt_wobble > 0)
// Outlets:
//   out1: Left Main Output (Even/Odd Morphed & Differentially Driven Stereo Mix)
//   out2: Right Main Output (Even/Odd Morphed & Differentially Driven Stereo Mix)
//   out3: Isolated Odd Harmonics Output (f0, 3*f0, 5*f0... for external Unix chaining)
//   out4: Isolated Even Harmonics Output (2*f0, 4*f0, 6*f0... for external Unix chaining)

@param({ min: 2.0, max: 35.0 }) comb_selectivity_q = 8.0;
@param({ min: -1.0, max: 1.0 }) odd_even_balance = 0.0;
@param({ min: 0.0, max: 24.0 }) odd_drive_db = 6.0;
@param({ min: 0.0, max: 24.0 }) even_fold_db = 9.0;
@param({ min: 0.0, max: 1.0 }) stereo_odd_even_split = 0.35;
@param({ min: 0.0, max: 1.0 }) tilt_wobble = 0.0;

// Canonical Universal Comb delay buffers xh[n] (4096 samples per channel)
@state xhBufL = new FixedSampleArray(4096);
@state xhBufR = new FixedSampleArray(4096);
@state writeIdx: Int = 0;

// Fallback trackers & smoothing
@state barFallback = new phasor();
@state prevBarIn = 0.0;
@state lpTrack = new onepole();
@state prevSig = 0.0;
@state zcCount = 0.0;
@state zcHz = 58.27;
@state smoothF0 = 58.27;
@state smoothQ = 8.0;

function readDelay4k(buf: FixedSampleArray, wIdx: Int, delSamps: number): number {
	let d = clamp(delSamps, 2.0, 4090.0);
	let readPos = wIdx - d;
	while (readPos < 0.0) {
		readPos += 4096.0;
	}
	let idx0: Int = floor(readPos);
	let frac = readPos - idx0;
	let i_m1: Int = (idx0 - 1 + 4096) & 4095;
	let i_0: Int  = idx0 & 4095;
	let i_1: Int  = (idx0 + 1) & 4095;
	let i_2: Int  = (idx0 + 2) & 4095;
	return cubicinterp(frac, buf[i_m1], buf[i_0], buf[i_1], buf[i_2]);
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = (abs(in2) > 0.0000001) ? in2 : xL;
let pitchIn = in3;
let barIn = in4;

// --- 1. Pitch Conditioning ---
let monoLow = lpTrack.next(0.5 * (xL + xR), 190.0);
zcCount += 1.0;
if (prevSig <= 0.0 && monoLow > 0.0) {
	if (zcCount > sr / 500.0 && zcCount < sr / 22.0) {
		zcHz = sr / zcCount;
	}
	zcCount = 0.0;
}
prevSig = monoLow;

let rawF0 = (pitchIn > 20.0 && pitchIn < 1000.0) ? pitchIn : zcHz;
let slew = 1.0 - exp(-1.0 / (0.006 * sr));
smoothF0 += (rawF0 - smoothF0) * slew;
let f0 = clamp(smoothF0, 24.0, 1000.0);

// --- 2. Universal Comb Filter Coefficients (DAFx-21 Sec 4.4, Eq. 9-14, 18) ---
// Delay M = fs / f2 = fs / (2 * f0) so comb teeth sit at multiples of 2*f0
let delayM = clamp(sr / (2.0 * f0), 4.0, 4088.0);

smoothQ += (clamp(comb_selectivity_q, 2.0, 35.0) - smoothQ) * slew;
// Eq. 14: beta = tan(pi / (2 * Q))
let beta = tan(pi / (2.0 * maximum(smoothQ, 2.0)));
let onePlusBetaInv = 1.0 / (1.0 + beta);
// Eq. 10: Unity-gain normalized IIR and FIR coefficients
let aM = clamp((1.0 - beta) * onePlusBetaInv, 0.0, 0.96);
let b0 = clamp(beta * onePlusBetaInv, 0.02, 0.5);

// --- 3. Canonical Single-Delay-Line Universal Comb Separation (Eq. 8) ---
let xhDelL = readDelay4k(xhBufL, writeIdx, delayM);
let xhDelR = readDelay4k(xhBufR, writeIdx, delayM);

let xhL = xL + aM * xhDelL;
let xhR = xR + aM * xhDelR;

xhBufL[writeIdx] = xhL;
xhBufR[writeIdx] = xhR;
writeIdx = (writeIdx + 1) & 4095;

// Positive feedforward extracts EVEN harmonics (2*f0, 4*f0, 6*f0...)
// Negative feedforward shifts comb peaks by pi/M, extracting ODD harmonics (f0, 3*f0, 5*f0...)
let evenL = b0 * (xhL + xhDelL);
let oddL  = b0 * (xhL - xhDelL);
let evenR = b0 * (xhR + xhDelR);
let oddR  = b0 * (xhR - xhDelR);

// --- 4. Independent Harmonic-Group Distortion Without Intermodulation (Sec 2.3 & 6) ---
let oddGain = dbtoa(clamp(odd_drive_db, 0.0, 24.0));
let evenGain = dbtoa(clamp(even_fold_db, 0.0, 24.0));

// Odd harmonics get warm symmetric tanh saturation (preserving odd symmetry + fundamental weight)
let distOddL = tanh(oddL * oddGain) / tanh(maximum(oddGain * 0.6, 1.0));
let distOddR = tanh(oddR * oddGain) / tanh(maximum(oddGain * 0.6, 1.0));

// Even harmonics get sine-wavefolded octave bloom
let uEvenL = evenL * evenGain;
let uEvenR = evenR * evenGain;
let distEvenL = (sin(uEvenL) + 0.5 * tanh(uEvenL)) * 0.65;
let distEvenR = (sin(uEvenR) + 0.5 * tanh(uEvenR)) * 0.65;

// --- 5. Odd/Even Balance, Bar Wobble & Stereo Harmonic Splitting ---
let barActive = (abs(barIn - prevBarIn) > 0.000001 || (barIn > 0.001 && barIn < 0.999)) ? 1.0 : 0.0;
prevBarIn = barIn;
let barPh = barActive ? wrap(barIn * 4.0, 0.0, 1.0) : barFallback.next(1.0);
let wobLfo = sin(2.0 * pi * barPh) * clamp(tilt_wobble, 0.0, 1.0);

let bal = clamp(odd_even_balance + wobLfo * 0.7, -1.0, 1.0);
let stSplit = clamp(stereo_odd_even_split, 0.0, 1.0) * 0.45;

let balL = clamp(0.5 * (bal + 1.0) - stSplit, 0.0, 1.0);
let balR = clamp(0.5 * (bal + 1.0) + stSplit, 0.0, 1.0);

out1 = linearinterp(balL, distOddL, distEvenL);
out2 = linearinterp(balR, distOddR, distEvenR);
// Dedicated clean/driven Odd & Even stems on out3 & out4 for Unix chaining!
out3 = 0.5 * (oddL + oddR);
out4 = 0.5 * (evenL + evenR);
