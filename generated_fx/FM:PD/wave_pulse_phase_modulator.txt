// ============================================================================
// wave_pulse_phase_modulator.txt
// Unix-Style Stereo Effect: Wave Pulse Phase Modulation (WPPM) Hybrid PD-PM
// Implements Matthew Smart (DAFx-25, Paper 48):
// "Wave Pulse Phase Modulation: Hybridising Phase Modulation and Phase Distortion"
// (Sections 2.2, 3.1, 3.2 & 4, Eq. 5-10, Fig. 4-5).
// Segments each period of the tuba's extracted analytic phase phi[n] at breakpoint d
// into two normalized [0, 1] sub-pulses, applies half-sine windowed Phase Modulation
// g1(t) and g2(t) with independent harmonic ratios R1, R2 and bipolar indices I1, I2,
// and applies DAFx-25 Eq. 10 duration-dependent antialiasing windowing + PolyBLEP.
// Produces acid TB-303-like resonant sweeps and dual-formant dubstep bass growls.
// ============================================================================
// Inlets:
//   in1: Left Audio Input (Tuba or 01_harmonic_isolator)
//   in2: Right Audio Input (if unpatched, mono in1 is used automatically)
//   in3: Tracked Fundamental Pitch f0 in Hz (from 00_tuba_freq_tracker; fallback ZC if <=0)
//   in4: Optional 1-Bar Phasor [0..1] (sweeps duration breakpoint d when d_sweep > 0)
// Outlets:
//   out1: Left WPPM Hybrid PD-PM Output
//   out2: Right WPPM Hybrid PD-PM Output

@param({ min: 0.04, max: 0.96 }) pulse_split_d = 0.35;
@param({ min: 0.5, max: 8.0 }) ratio_r1 = 1.0;
@param({ min: -2.0, max: 2.0 }) index_i1 = 0.85;
@param({ min: 0.5, max: 8.0 }) ratio_r2 = 2.0;
@param({ min: -2.0, max: 2.0 }) index_i2 = -0.75;
@param({ min: 0.0, max: 1.0 }) d_sweep = 0.3;
@param({ min: 0.0, max: 1.0 }) env_follow = 0.85;

// 4-stage Quadrature IIR Hilbert network for analytic phase & envelope extraction
@state hReX = new FixedSampleArray(4); @state hReY = new FixedSampleArray(4);
@state hImX = new FixedSampleArray(4); @state hImY = new FixedSampleArray(4);
@state hDelRe = 0.0;

// Fallback trackers & smoothing
@state barFallback = new phasor();
@state prevBarIn = 0.0;
@state lpTrack = new onepole();
@state prevSig = 0.0;
@state zcCount = 0.0;
@state zcHz = 58.27;
@state smoothF0 = 58.27;
@state smoothEnv = 0.0;
@state smoothD = 0.35;
@state smoothI1 = 0.85;
@state smoothI2 = -0.75;

// Evaluate WPPM with Eq. 5, 8, 9, 10 antialiasing windowing
function evalWPPM(phiIn: number, dVal: number, r1: number, i1: number, r2: number, i2: number, pOff: number): number {
	let phi = wrap(phiIn, 0.0, 1.0);
	let d = clamp(dVal, 0.03, 0.97);
	let f5 = 0.0;

	if (phi < d) {
		// Segment 1: normalized sub-phase t1 in [0, 1)
		let t1 = phi / d;
		// Window w(t1) = sin(pi * t1) ensures 0 amplitude at both segment boundaries (Eq. 5)
		let wSeg1 = sin(pi * t1);
		// Antialiasing duration window w(d / 2) attenuates g1 as d -> 0 (Eq. 10)
		let wAnti1 = sin(pi * 0.5 * d);
		let g1 = wAnti1 * wSeg1 * i1 * sin(2.0 * pi * (r1 * t1 + pOff));
		f5 = (phi / (2.0 * d)) + g1;
	} else {
		// Segment 2: normalized sub-phase t2 in [0, 1)
		let t2 = (phi - d) / (1.0 - d);
		let wSeg2 = sin(pi * t2);
		// Antialiasing duration window w(1 - d / 2) attenuates g2 as d -> 1 (Eq. 10)
		let wAnti2 = sin(pi * (1.0 - 0.5 * d));
		let g2 = wAnti2 * wSeg2 * i2 * sin(2.0 * pi * (r2 * t2 - pOff));
		f5 = 0.5 + ((phi - d) / (2.0 * (1.0 - d))) + g2;
	}

	return -cos(2.0 * pi * f5);
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = (abs(in2) > 0.0000001) ? in2 : xL;
let pitchIn = in3;
let barIn = in4;

// --- 1. Pitch Conditioning ---
let mono = 0.5 * (xL + xR);
let monoLow = lpTrack.next(mono, 190.0);
zcCount += 1.0;
if (prevSig <= 0.0 && monoLow > 0.0) {
	if (zcCount > sr / 500.0 && zcCount < sr / 22.0) {
		zcHz = sr / zcCount;
	}
	zcCount = 0.0;
}
prevSig = monoLow;

let rawF0 = (pitchIn > 20.0 && pitchIn < 1000.0) ? pitchIn : zcHz;
let slew = 1.0 - exp(-1.0 / (0.006 * sr));
smoothF0 += (rawF0 - smoothF0) * slew;

// --- 2. Analytic Phase & Amplitude Extraction ---
let cR0 = 0.4794008656; let cR1 = 0.8762184935; let cR2 = 0.9765975895; let cR3 = 0.9974992559;
let cI0 = 0.1617584983; let cI1 = 0.7330289323; let cI2 = 0.9453497003; let cI3 = 0.9905991567;

let re = mono;
let t = cR0 * (re + hReY[0]) - hReX[0]; hReX[0] = re; hReY[0] = t; re = t;
t = cR1 * (re + hReY[1]) - hReX[1]; hReX[1] = re; hReY[1] = t; re = t;
t = cR2 * (re + hReY[2]) - hReX[2]; hReX[2] = re; hReY[2] = t; re = t;
t = cR3 * (re + hReY[3]) - hReX[3]; hReX[3] = re; hReY[3] = t; re = t;
let sigRe = hDelRe;
hDelRe = re;

let im = mono;
t = cI0 * (im + hImY[0]) - hImX[0]; hImX[0] = im; hImY[0] = t; im = t;
t = cI1 * (im + hImY[1]) - hImX[1]; hImX[1] = im; hImY[1] = t; im = t;
t = cI2 * (im + hImY[2]) - hImX[2]; hImX[2] = im; hImY[2] = t; im = t;
t = cI3 * (im + hImY[3]) - hImX[3]; hImX[3] = im; hImY[3] = t; im = t;
let sigIm = im;

let instAmp = sqrt(sigRe * sigRe + sigIm * sigIm);
smoothEnv += (instAmp - smoothEnv) * (1.0 - exp(-1.0 / (0.004 * sr)));
let phi = wrap((atan2(sigIm, sigRe) + pi) / (2.0 * pi), 0.0, 1.0);

// --- 3. Bar-Synced Duration Ratio Sweep & Parameter Smoothing ---
let barActive = (abs(barIn - prevBarIn) > 0.000001 || (barIn > 0.001 && barIn < 0.999)) ? 1.0 : 0.0;
prevBarIn = barIn;
let barPh = barActive ? wrap(barIn * 4.0, 0.0, 1.0) : barFallback.next(1.0);
let sweepTri = 2.0 * triangle(barPh, 0.5) - 1.0;

smoothD += (clamp(pulse_split_d, 0.04, 0.96) - smoothD) * slew;
smoothI1 += (clamp(index_i1, -2.0, 2.0) - smoothI1) * slew;
smoothI2 += (clamp(index_i2, -2.0, 2.0) - smoothI2) * slew;

let swAmt = clamp(d_sweep, 0.0, 1.0) * 0.36;
let dL = clamp(smoothD + swAmt * sweepTri - 0.03, 0.04, 0.96);
let dR = clamp(smoothD - swAmt * sweepTri + 0.03, 0.04, 0.96);

let r1 = clamp(ratio_r1, 0.5, 8.0);
let r2 = clamp(ratio_r2, 0.5, 8.0);

// --- 4. Evaluate Stereo WPPM ---
let yL = evalWPPM(phi, dL, r1, smoothI1, r2, smoothI2, 0.0);
let yR = evalWPPM(phi, dR, r1, smoothI1, r2, smoothI2, 0.12);

let gate = clamp((smoothEnv - 0.002) * 200.0, 0.0, 1.0);
let ampScale = linearinterp(clamp(env_follow, 0.0, 1.0), gate * 0.7, clamp(smoothEnv * 1.6, 0.0, 1.5));

out1 = yL * ampScale;
out2 = yR * ampScale;
