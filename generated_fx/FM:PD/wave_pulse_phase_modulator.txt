// ============================================================================
// wave_pulse_phase_modulator.txt
// Unix-Style Stereo Effect: Wave Pulse Phase Modulation (WPPM) Hybrid PD-PM
// Implements Matthew Smart (DAFx-25, Paper 48):
// "Wave Pulse Phase Modulation: Hybridising Phase Modulation and Phase Distortion"
// (Sections 2.2, 3.1, 3.2 & 4, Eq. 5-10, Fig. 4-5).
// Segments each period of the tuba's phase-locked fundamental phasor phBase[n] at
// breakpoint d into two normalized [0, 1] sub-pulses, applies half-sine windowed
// Phase Modulation g1(t) and g2(t) with independent harmonic ratios R1, R2 and
// bipolar indices I1, I2, and applies DAFx-25 Section 4 Tukey duration-dependent
// antialiasing windowing.
// Produces acid TB-303-like resonant sweeps and dual-formant dubstep bass growls.
// ============================================================================
// Inlets:
//   in1: Left Audio Input (Tuba or 01_harmonic_isolator)
//   in2: Right Audio Input (if unpatched, mono in1 is used automatically)
//   in3: Tracked Fundamental Pitch f0 in Hz (from 00_tuba_freq_tracker; fallback ZC if <=0)
//   in4: Optional 1-Bar Phasor [0..1] (sweeps duration breakpoint d when d_sweep > 0)
// Outlets:
//   out1: Left WPPM Hybrid PD-PM Output
//   out2: Right WPPM Hybrid PD-PM Output

@param({ min: 0.04, max: 0.96 }) pulse_split_d = 0.35;
@param({ min: 0.5, max: 8.0 }) ratio_r1 = 1.0;
@param({ min: -2.0, max: 2.0 }) index_i1 = 0.85;
@param({ min: 0.5, max: 8.0 }) ratio_r2 = 2.0;
@param({ min: -2.0, max: 2.0 }) index_i2 = -0.75;
@param({ min: 0.0, max: 1.0 }) d_sweep = 0.3;
@param({ min: 0.0, max: 1.0 }) env_follow = 0.85;

// Fundamental isolation SVF + 4-stage Quadrature IIR Hilbert network for PLL phase lock
@state fundSvf = new svf();
@state hReX = new FixedSampleArray(4); @state hReY = new FixedSampleArray(4);
@state hImX = new FixedSampleArray(4); @state hImY = new FixedSampleArray(4);
@state hDelRe = 0.0;
@state pllPhase = 0.0;

// Fallback trackers, DC blockers & smoothing
@state barFallback = new phasor();
@state prevBarIn = 0.0;
@state barHold = 0.0;
@state lpTrack = new onepole();
@state prevSig = 0.0;
@state zcCount = 0.0;
@state zcHz = 58.27;
@state smoothF0 = 58.27;
@state smoothEnv = 0.0;
@state smoothD = 0.35;
@state smoothI1 = 0.85;
@state smoothI2 = -0.75;
@state dcXL = 0.0; @state dcYL = 0.0;
@state dcXR = 0.0; @state dcYR = 0.0;

// Evaluate WPPM with Eq. 5, 8, 9, 10 + Section 4 Tukey duration antialiasing window
// NOTE: Uses phVal (never `phi`, which RNBO compiles as the constant golden ratio 1.618!)
function evalWPPM(phIn: number, dVal: number, r1: number, i1: number, r2: number, i2: number, pOff: number, f0Hz: number, srHz: number): number {
	let phVal = wrap(phIn, 0.0, 1.0);
	let d = clamp(dVal, 0.04, 0.96);
	let f5 = 0.0;

	if (phVal < d) {
		// Segment 1: normalized sub-phase t1 in [0, 1)
		let t1 = phVal / d;
		// Window w(t1) = sin(pi * t1) ensures 0 amplitude at both segment boundaries (Eq. 5)
		let wSeg1 = sin(pi * t1);
		// Section 4 Tukey-style duration antialiasing window + Nyquist rolloff as d -> 0
		let wTukey1 = sin(0.5 * pi * clamp(d / 0.25, 0.0, 1.0));
		let effFreq1 = (f0Hz * r1) / d;
		let nyqAtten1 = clamp((srHz * 0.42 - effFreq1) / (srHz * 0.25), 0.0, 1.0);
		let g1 = wTukey1 * nyqAtten1 * wSeg1 * i1 * sin(2.0 * pi * (r1 * t1 + pOff));
		f5 = (0.5 * t1) + g1;
	} else {
		// Segment 2: normalized sub-phase t2 in [0, 1)
		let oneMinusD = 1.0 - d;
		let t2 = (phVal - d) / oneMinusD;
		let wSeg2 = sin(pi * t2);
		// Section 4 Tukey-style duration antialiasing window + Nyquist rolloff as d -> 1
		let wTukey2 = sin(0.5 * pi * clamp(oneMinusD / 0.25, 0.0, 1.0));
		let effFreq2 = (f0Hz * r2) / oneMinusD;
		let nyqAtten2 = clamp((srHz * 0.42 - effFreq2) / (srHz * 0.25), 0.0, 1.0);
		let g2 = wTukey2 * nyqAtten2 * wSeg2 * i2 * sin(2.0 * pi * (r2 * t2 - pOff));
		f5 = 0.5 + (0.5 * t2) + g2;
	}

	return 0.0 - cos(2.0 * pi * f5);
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = (abs(in2) > 0.0000001) ? in2 : xL;
let pitchIn = in3;
let barIn = in4;

// --- 1. Pitch Conditioning ---
let mono = 0.5 * (xL + xR);
let monoLow = lpTrack.next(mono, 190.0);
zcCount += 1.0;
if (prevSig <= 0.0 && monoLow > 0.0) {
	if (zcCount > sr / 500.0 && zcCount < sr / 22.0) {
		zcHz = sr / zcCount;
	}
	zcCount = 0.0;
}
prevSig = monoLow;

let rawF0 = (pitchIn > 20.0 && pitchIn < 1000.0) ? pitchIn : zcHz;
let slew = 1.0 - exp(-1.0 / (0.006 * sr));
smoothF0 += (rawF0 - smoothF0) * slew;
let f0 = clamp(smoothF0, 24.0, 1000.0);

// --- 2. Analytic Phase & Amplitude Extraction via Fundamental SVF + 4-Stage Hilbert Network ---
let isoFund = fundSvf.next(mono, f0, 0.80)[2];

let cR0 = 0.4794008656; let cR1 = 0.8762184935; let cR2 = 0.9765975895; let cR3 = 0.9974992559;
let cI0 = 0.1617584983; let cI1 = 0.7330289323; let cI2 = 0.9453497003; let cI3 = 0.9905991567;

// Correct negative-feedback allpass Hilbert network: y[n] = c * (x[n] - y[n-1]) + x[n-1]
let re = isoFund;
let t = cR0 * (re - hReY[0]) + hReX[0]; hReX[0] = re; hReY[0] = t; re = t;
t = cR1 * (re - hReY[1]) + hReX[1]; hReX[1] = re; hReY[1] = t; re = t;
t = cR2 * (re - hReY[2]) + hReX[2]; hReX[2] = re; hReY[2] = t; re = t;
t = cR3 * (re - hReY[3]) + hReX[3]; hReX[3] = re; hReY[3] = t; re = t;
let sigRe = hDelRe;
hDelRe = re;

let im = isoFund;
t = cI0 * (im - hImY[0]) + hImX[0]; hImX[0] = im; hImY[0] = t; im = t;
t = cI1 * (im - hImY[1]) + hImX[1]; hImX[1] = im; hImY[1] = t; im = t;
t = cI2 * (im - hImY[2]) + hImX[2]; hImX[2] = im; hImY[2] = t; im = t;
t = cI3 * (im - hImY[3]) + hImX[3]; hImX[3] = im; hImY[3] = t; im = t;
let sigIm = im;

let fullAmp = maximum(abs(xL), abs(xR));
let envAtk = 1.0 - exp(-1.0 / (0.0025 * sr));
let envRel = 1.0 - exp(-1.0 / (0.025 * sr));
smoothEnv = (fullAmp > smoothEnv) ? (smoothEnv + (fullAmp - smoothEnv) * envAtk) : (smoothEnv + (fullAmp - smoothEnv) * envRel);

// Phase-Locked Loop (PLL) locks a clean monotonic [0, 1) ramp to the tuba's fundamental
// NOTE: Named phBase (never `phi`, which RNBO compiles as the constant golden ratio 1.618!)
let hilbPh = wrap((atan2(sigIm, sigRe) + pi) / (2.0 * pi), 0.0, 1.0);
let phErr = wrap(hilbPh - pllPhase + 0.5, 0.0, 1.0) - 0.5;
let lockGain = (smoothEnv > 0.002) ? 0.035 : 0.0;
pllPhase = wrap(pllPhase + (f0 / sr) + lockGain * phErr, 0.0, 1.0);
let phBase = pllPhase;

// --- 3. Bar-Synced Duration Ratio Sweep & Parameter Smoothing ---
if (abs(barIn - prevBarIn) > 0.0000001) {
	barHold = 0.1 * sr;
} else if (barHold > 0.0) {
	barHold -= 1.0;
}
prevBarIn = barIn;
let fbPh = barFallback.next(0.5);
let barPh = (barHold > 0.0) ? wrap(barIn * 2.0, 0.0, 1.0) : fbPh;
let sweepTri = 2.0 * triangle(barPh, 0.5) - 1.0;

smoothD += (clamp(pulse_split_d, 0.04, 0.96) - smoothD) * slew;
smoothI1 += (clamp(index_i1, -2.0, 2.0) - smoothI1) * slew;
smoothI2 += (clamp(index_i2, -2.0, 2.0) - smoothI2) * slew;

let swAmt = clamp(d_sweep, 0.0, 1.0) * 0.34;
let dL = clamp(smoothD + swAmt * sweepTri - 0.025, 0.04, 0.96);
let dR = clamp(smoothD - swAmt * sweepTri + 0.025, 0.04, 0.96);

let r1 = clamp(ratio_r1, 0.5, 8.0);
let r2 = clamp(ratio_r2, 0.5, 8.0);

// --- 4. Evaluate Stereo WPPM + DC Block ---
let rawYL = evalWPPM(phBase, dL, r1, smoothI1, r2, smoothI2, 0.0, f0, sr);
let rawYR = evalWPPM(phBase, dR, r1, smoothI1, r2, smoothI2, 0.08, f0, sr);

let yL = rawYL - dcXL + 0.995 * dcYL;
dcXL = rawYL; dcYL = yL;
let yR = rawYR - dcXR + 0.995 * dcYR;
dcXR = rawYR; dcYR = yR;

let gate = clamp((smoothEnv - 0.002) * 200.0, 0.0, 1.0);
let ampScale = linearinterp(clamp(env_follow, 0.0, 1.0), gate * 0.75, clamp(smoothEnv * 1.5, 0.0, 1.3));

out1 = tanh(yL * ampScale);
out2 = tanh(yR * ampScale);
