// ============================================================================
// fbam_delay_comb_operator.txt
// Unix-Style Stereo Effect: Variable-Delay Feedback AM (FBAM) Variations I-V
// Implements Lazzarini, Timoney, Kleimola & Valimaki (DAFx-09, Paper 31):
// "Five Variations on a Feedback Theme" (Sections 2 & 3.1-3.6, Eq. 6, 8, 10, 11, 12).
// Realizes the complete Variable-Delay Feedback Amplitude Modulation operator
// generalized for arbitrary instrument inputs with an internal feedback damping
// one-pole lowpass filter (Section 3.6 "Coda"):
//   - Var 0 (Theme / Var V, Eq. 6 & 12): y(n) = x(n) * [1 + beta * LP(y(n - D))]
//   - Var 0.5 (Var IV Waveshaped FBAM, Eq. 11): y(n) = x(n) * [1 + cos(beta * LP(y(n - D)))]
//   - Var 1.0 (Var III Secondary Heterodyne FBAM, Eq. 10):
//              y(n) = cos(N * w0 * n) * x(n) * [1 + beta * LP(y(n - D))]
// Sweeping the feedback delay D around fs / f0 creates intense "phase difference
// modulation" (Section 4.2) and sharp pulse-comb harmonic spectra!
// ============================================================================
// Inlets:
//   in1: Left Audio Input (Tuba)
//   in2: Right Audio Input (if unpatched, mono in1 is used automatically)
//   in3: Tracked Fundamental Pitch f0 in Hz (from 00_tuba_freq_tracker; fallback ZC if <=0)
//   in4: Optional 1-Bar Phasor [0..1] (sweeps delay D for "phase difference modulation")
// Outlets:
//   out1: Left Feedback-AM Comb Operator Output
//   out2: Right Feedback-AM Comb Operator Output

@param({ min: 0.0, max: 1.8 }) fbam_beta = 0.95;
@param({ min: 0.25, max: 4.0 }) delay_harmonic_d = 1.0;
@param({ min: 0.0, max: 1.0 }) variation_morph = 0.25;
@param({ min: 1.0, max: 8.0 }) het_ratio_n = 2.0;
@param({ min: 200.0, max: 10000.0 }) loop_lp_hz = 2800.0;
@param({ min: 0.0, max: 1.0 }) phase_diff_sweep = 0.3;

// Feedback delay buffers y(n - D) (4096 samples per channel, Paper Sec 3.5, Fig. 10)
@state fbBufL = new FixedSampleArray(4096);
@state fbBufR = new FixedSampleArray(4096);
@state writeIdx: Int = 0;

// Section 3.6 "Coda": 1st-order IIR lowpass filter in feedback loop for stability with beta > 1
@state fbLpL = new onepole();
@state fbLpR = new onepole();
@state dcXL = 0.0; @state dcYL = 0.0;
@state dcXR = 0.0; @state dcYR = 0.0;

// Auxiliary heterodyne oscillator for Variation III (Eq. 10) & fallback trackers
@state hetPhasor = new phasor();
@state barFallback = new phasor();
@state prevBarIn = 0.0;
@state lpTrack = new onepole();
@state prevSig = 0.0;
@state zcCount = 0.0;
@state zcHz = 58.27;
@state smoothF0 = 58.27;
@state smoothBeta = 0.95;

function readDelay4k(buf: FixedSampleArray, wIdx: Int, delSamps: number): number {
	let d = clamp(delSamps, 1.0, 4090.0);
	let readPos = wIdx - d;
	while (readPos < 0.0) {
		readPos += 4096.0;
	}
	let idx0: Int = floor(readPos);
	let frac = readPos - idx0;
	let i_m1: Int = (idx0 - 1 + 4096) & 4095;
	let i_0: Int  = idx0 & 4095;
	let i_1: Int  = (idx0 + 1) & 4095;
	let i_2: Int  = (idx0 + 2) & 4095;
	return cubicinterp(frac, buf[i_m1], buf[i_0], buf[i_1], buf[i_2]);
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = (abs(in2) > 0.0000001) ? in2 : xL;
let pitchIn = in3;
let barIn = in4;

// --- 1. Pitch Conditioning ---
let monoLow = lpTrack.next(0.5 * (xL + xR), 190.0);
zcCount += 1.0;
if (prevSig <= 0.0 && monoLow > 0.0) {
	if (zcCount > sr / 500.0 && zcCount < sr / 22.0) {
		zcHz = sr / zcCount;
	}
	zcCount = 0.0;
}
prevSig = monoLow;

let rawF0 = (pitchIn > 20.0 && pitchIn < 1000.0) ? pitchIn : zcHz;
let slew = 1.0 - exp(-1.0 / (0.006 * sr));
smoothF0 += (rawF0 - smoothF0) * slew;
let f0 = clamp(smoothF0, 24.0, 1000.0);
smoothBeta += (clamp(fbam_beta, 0.0, 1.8) - smoothBeta) * slew;

// --- 2. Feedback Delay D & "Phase Difference Modulation" (Paper Sec 3.5 & 4.2) ---
let barActive = (abs(barIn - prevBarIn) > 0.000001 || (barIn > 0.001 && barIn < 0.999)) ? 1.0 : 0.0;
prevBarIn = barIn;
let barPh = barActive ? wrap(barIn * 4.0, 0.0, 1.0) : barFallback.next(0.75);
let sweepLfo = sin(2.0 * pi * barPh);

// When delay_harmonic_d = 1.0, D = fs / f0 aligns all cosine product phases (Eq. 12-14, Fig. 9)
let baseD = (sr / f0) / clamp(delay_harmonic_d, 0.25, 4.0);
let swDepth = clamp(phase_diff_sweep, 0.0, 1.0) * 0.45;
let delL = clamp(baseD * pow(2.0, swDepth * sweepLfo - 0.02), 1.0, 4088.0);
let delR = clamp(baseD * pow(2.0, -swDepth * sweepLfo + 0.02), 1.0, 4088.0);

// Read delayed feedback y(n - D) and pass through Section 3.6 Coda lowpass filter
let rawYD_L = readDelay4k(fbBufL, writeIdx, delL);
let rawYD_R = readDelay4k(fbBufR, writeIdx, delR);

let lpCut = clamp(loop_lp_hz, 200.0, 10000.0);
let yDL = fbLpL.next(rawYD_L, lpCut);
let yDR = fbLpR.next(rawYD_R, lpCut);

// --- 3. Evaluate Variations V, IV, and III (Paper Eq. 10, 11, 12) ---
// Variation V (Eq. 12): y(n) = x(n) * [1 + beta * y(n - D)]
let modLinL = 1.0 + smoothBeta * yDL;
let modLinR = 1.0 + smoothBeta * yDR;

// Variation IV (Eq. 11): y(n) = x(n) * [1 + cos(beta * y(n - D))] (odd-harmonic partial feedback FM)
let modCosL = 1.0 + cos(smoothBeta * pi * yDL) - 0.5;
let modCosR = 1.0 + cos(smoothBeta * pi * yDR) - 0.5;

// Variation III (Eq. 10): auxiliary heterodyne carrier cos(N * w0 * n)
let hetFreq = clamp(f0 * clamp(het_ratio_n, 1.0, 8.0), 20.0, sr * 0.42);
let hetPh = hetPhasor.next(hetFreq);
let hetCarL = cos(2.0 * pi * hetPh);
let hetCarR = cos(2.0 * pi * (hetPh + 0.08));

let vMorph = clamp(variation_morph, 0.0, 1.0);
let yL = 0.0;
let yR = 0.0;

if (vMorph <= 0.5) {
	let m = vMorph * 2.0;
	yL = xL * linearinterp(m, modLinL, modCosL);
	yR = xR * linearinterp(m, modLinR, modCosR);
} else {
	let m = (vMorph - 0.5) * 2.0;
	let var4L = xL * modCosL;
	let var4R = xR * modCosR;
	let var3L = hetCarL * (xL * modLinL);
	let var3R = hetCarR * (xR * modLinR);
	yL = linearinterp(m, var4L, var3L);
	yR = linearinterp(m, var4R, var3R);
}

// Soft bound and write to feedback buffer
let boundedL = tanh(yL * 0.85);
let boundedR = tanh(yR * 0.85);
fbBufL[writeIdx] = boundedL;
fbBufR[writeIdx] = boundedR;
writeIdx = (writeIdx + 1) & 4095;

// DC blocker (FBAM generates an inherent x^2 rectified DC component at even powers)
let outL = boundedL - dcXL + 0.995 * dcYL;
dcXL = boundedL; dcYL = outL;
let outR = boundedR - dcXR + 0.995 * dcYR;
dcXR = boundedR; dcYR = outR;

out1 = outL;
out2 = outR;
