// ============================================================================
// flanger_through_zero_jet.txt
// Through-zero flanger: two taps swept symmetrically through the zero point.
// From: Antti Huovilainen, "Enhanced Digital Models for Analog Modulation
//       Effects", DAFx-05, 2005_P_155, §1.1 (fractional-delay flanger as a
//       time-varying comb, feedback turns the FIR comb into an IIR comb) and
//       §4 (delay-time warping / loop behaviour); structure after the flanger
//       block of Esqueda, Välimäki & Parker, "Barberpole Phasing and Flanging
//       Illusions", DAFx-15, 2015_DAFx-15_submission_67, Fig. 6;
//       fractional-delay choice after Nam & Välimäki, DAFx-09 paper 72, §2.1
//       (cubicinterp is 3rd-order Lagrange: FIR, no Thiran allpass
//       dispersion; slight HF droop at half-sample positions).
//
// Two taps sit at c -/+ s around a centre delay c = 0.15 ms + base, so their
// relative delay is 2s. In a true through-zero flanger the moving tap passes
// the reference, so the relative delay crosses zero with full slope and |2s|
// makes a sharp V-shaped bounce there - the notches slam up to Nyquist and
// straight back, the "jet". Because the two taps are summed equally, swapping
// them is inaudible, so s = base * sin(pi * p) (>= 0, continuous across the
// phase wrap, a V-kink at p = 0) is exactly that crossing, once per cycle.
// (A raised sine 0.5 + 0.5 sin only *touches* zero with zero slope - an
// ordinary flanger reaching 0 ms, not through-zero.) Neither tap is ever
// instantaneous (the 0.15 ms floor keeps the output off the dry path), so this
// file is 100 % wet with no dry term at all.
// ============================================================================
// Inlets:
//   in1 / in2: stereo audio
//   in3: bound only - this sweep is time-based, not pitch-locked
//        (use flanger_enharmonic_diesis / phaser_harmonic_track for f0 lock)
//   in4: 0..1 one-bar phasor (exception to the Hz contract) - while it moves
//        the sweep runs one full cycle per bar (zero crossing on the downbeat)
//        and sweep_rate_hz is ignored; held constant (or unconnected) the
//        sweep free-runs at sweep_rate_hz
// Outlets:
//   out1 / out2: stereo, wet only
// No dry/wet control, no limiter, no metering.
// ============================================================================

@param({ min: 0.05, max: 12 }) sweep_ms = 2.2;      // sweep span
@param({ min: 0.02, max: 2 }) sweep_rate_hz = 0.14; // sweep rate
@param({ min: -0.95, max: 0.95 }) feedback = -0.55; // feedback
@param({ min: 0, max: 1 }) stereo_offset = 0.25;    // stereo offset
@param({ min: 800, max: 14000 }) tone_hz = 7000;    // loop damping
@param({ min: -24, max: 12 }) out_gain_db = 0;      // output gain

@state bufL = new FixedSampleArray(4096);
@state bufR = new FixedSampleArray(4096);
@state wi: Int = 0;

@state dampL = 0.0;
@state dampR = 0.0;

@state phF = 0.0;
@state phOff = 0.0;
@state prevBarIn = 0.0;

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = in2;
let unused_in3 = in3;  // keeps the in3 port present; this sweep is not pitch-locked
let barIn = in4;

// --- Sweep phase -----------------------------------------------------------
// The bar phase is carried as an offset against a free-running phase, so a
// moving phasor takes over exactly and a stopped one hands back with no jump.
let rate = clamp(sweep_rate_hz, 0.02, 2.0);
phF = wrap(phF + rate / sr, 0.0, 1.0);
if (abs(barIn - prevBarIn) > 0.0000001) {
	phOff = wrap(barIn - phF, 0.0, 1.0);
}
prevBarIn = barIn;

let offR = clamp(stereo_offset, 0.0, 1.0) * 0.5;
let pL = wrap(phF + phOff, 0.0, 1.0);
let pR = wrap(pL + offR, 0.0, 1.0);

// --- Delay geometry --------------------------------------------------------
// centre c = dmin + base, half-spread s = base * sin(pi * p) in 0 .. base
let base = clamp(mstosamps(clamp(sweep_ms, 0.05, 12.0)), 2.0, 1600.0);
let dminS = maximum(mstosamps(0.15), 1.0);
let centre = dminS + base;

let spreadL = base * sin(pi * pL);
let dAL = centre - spreadL;
let dBL = centre + spreadL;

let spreadR = base * sin(pi * pR);
let dAR = centre - spreadR;
let dBR = centre + spreadR;

let fbAmt = clamp(feedback, -0.95, 0.95);
let dampF = clamp(tone_hz, 200.0, sr * 0.45);
let dampC = 1.0 - exp(0.0 - 2.0 * pi * dampF / sr);

// --- Left channel: write input + damped feedback, read the two swept taps ----
bufL[wi] = xL + fbAmt * dampL;

let rAL = wi - dAL;
while (rAL < 0.0) { rAL += 4096.0; }
let qAL: Int = floor(rAL);
let tapAL = cubicinterp(rAL - qAL,
	bufL[(qAL - 1 + 4096) & 4095],
	bufL[qAL & 4095],
	bufL[(qAL + 1) & 4095],
	bufL[(qAL + 2) & 4095]);

let rBL = wi - dBL;
while (rBL < 0.0) { rBL += 4096.0; }
let qBL: Int = floor(rBL);
let tapBL = cubicinterp(rBL - qBL,
	bufL[(qBL - 1 + 4096) & 4095],
	bufL[qBL & 4095],
	bufL[(qBL + 1) & 4095],
	bufL[(qBL + 2) & 4095]);

let yL = 0.5 * (tapAL + tapBL);
dampL = dampL + dampC * (yL - dampL);

// --- Right channel ---------------------------------------------------------
bufR[wi] = xR + fbAmt * dampR;

let rAR = wi - dAR;
while (rAR < 0.0) { rAR += 4096.0; }
let qAR: Int = floor(rAR);
let tapAR = cubicinterp(rAR - qAR,
	bufR[(qAR - 1 + 4096) & 4095],
	bufR[qAR & 4095],
	bufR[(qAR + 1) & 4095],
	bufR[(qAR + 2) & 4095]);

let rBR = wi - dBR;
while (rBR < 0.0) { rBR += 4096.0; }
let qBR: Int = floor(rBR);
let tapBR = cubicinterp(rBR - qBR,
	bufR[(qBR - 1 + 4096) & 4095],
	bufR[qBR & 4095],
	bufR[(qBR + 1) & 4095],
	bufR[(qBR + 2) & 4095]);

let yR = 0.5 * (tapAR + tapBR);
dampR = dampR + dampC * (yR - dampR);

wi = (wi + 1) & 4095;

let gain = dbtoa(clamp(out_gain_db, -24.0, 12.0));
out1 = yL * gain;
out2 = yR * gain;
