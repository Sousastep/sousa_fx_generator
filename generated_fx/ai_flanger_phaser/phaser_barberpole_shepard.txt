// ============================================================================
// phaser_barberpole_shepard.txt
// Barberpole phaser: indefinitely rising (or falling) spectral notches, built
// the way Shepard built the endless scale - a ladder of notches one octave
// apart under a raised-cosine depth envelope, so the notch leaving the top is
// replaced by the one entering at the bottom.
// From: Esqueda, Välimäki & Parker, "Barberpole Phasing and Flanging Illusions",
//       DAFx-15, 2015_DAFx-15_submission_67:
//         §2, Eq. 1-2 (M cascaded notches, centre frequencies one octave apart),
//         Eq. 3-4 (raised-cosine depth envelope over the whole ladder),
//         Eq. 5-8 (constant-Q parametric notch, G2B = (1+G^2)/2),
//         Fig. 2-5.
//
// The paper steps a table of centre frequencies and resets every filter one
// octave down at the end of each cycle, carrying the biquad state across the
// jump. Here each stage instead climbs the *whole* ladder on its own (position
// (m + u) / stages, with u counting cycles), so a stage only wraps when it
// reaches the edge of the ladder - where the raised-cosine depth is 0 dB and
// the stage is an exact identity. Its state is cleared at that moment, so the
// wrap is inaudible: no table, no global reset, no transient. Notches are also
// faded out over the octave below 0.45*sr so stages above Nyquist never stack.
//
// Each notch is a TPT state-variable bandpass subtracted from the input,
// y = x - (1 - G) * bp / Q, which gives exactly G at fc for any Q.
// ============================================================================
// Inlets:
//   in1 / in2: stereo audio
//   in3: bound only - the ladder is anchored to ladder_base_hz, not to f0
//   in4: 0..1 one-bar phasor (exception to the Hz contract) - while it moves
//        the ladder climbs one octave per bar and notch_rate_hz is ignored;
//        held constant (or unconnected) it free-runs at notch_rate_hz
// Outlets:
//   out1 / out2: stereo, wet only
// No dry/wet control, no limiter, no metering.
// ============================================================================

@param({ min: 0.01, max: 0.3 }) notch_rate_hz = 0.06; // notch rate
@param({ min: 20, max: 400 }) ladder_base_hz = 63;    // ladder base
@param({ min: 4, max: 10 }) stages = 8;               // notch stages
@param({ min: 2, max: 40 }) notch_q = 12;             // notch width
@param({ min: 3, max: 36 }) notch_depth_db = 18;      // deepest notch
@param({ enum: ["falling", "rising"] }) direction = 1; // notch direction
@param({ min: -24, max: 12 }) out_gain_db = 0;        // output gain

@state ic1L = new FixedSampleArray(10);
@state ic2L = new FixedSampleArray(10);
@state ic1R = new FixedSampleArray(10);
@state ic2R = new FixedSampleArray(10);
@state posPrevL = new FixedSampleArray(10);
@state posPrevR = new FixedSampleArray(10);

@state phF = 0.0;
@state phOff = 0.0;
@state prevBarIn = 0.0;
@state prevBase = 0.0;
@state cycles: Int = 0;

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = in2;
let unused_in3 = in3;  // keeps the in3 port present
let barIn = in4;

// --- Ladder phase ----------------------------------------------------------
// The bar phase is carried as an offset against a free-running phase, so a
// moving phasor takes over exactly and a stopped one hands back with no jump.
let rate = clamp(notch_rate_hz, 0.01, 0.3);
phF = wrap(phF + rate / sr, 0.0, 1.0);
if (abs(barIn - prevBarIn) > 0.0000001) {
	phOff = wrap(barIn - phF, 0.0, 1.0);
}
prevBarIn = barIn;
let base = wrap(phF + phOff, 0.0, 1.0);

// Count whole cycles (one octave each) so every stage can travel the full
// ladder. 2520 = lcm(4..10), so the count wraps seamlessly for any stage count.
if (base < prevBase - 0.5) { cycles = (cycles + 1) % 2520; }
else if (base > prevBase + 0.5) { cycles = (cycles + 2519) % 2520; }
prevBase = base;
let u = cycles + base;

let rising = direction > 0.5;
let uL = rising ? u : 2520.0 - u;
// A quarter-stage offset puts the right ladder a quarter-octave above the left
let uR = uL + 0.25;

let nStages: Int = clamp(floor(stages + 0.5), 4, 10);
let qv = clamp(notch_q, 2.0, 40.0);
let kq = 1.0 / qv;
let fBase = clamp(ladder_base_hz, 20.0, 400.0);

// Lmin = 0 dB at the ends of the ladder (a stage there is an identity, so it
// can wrap), Lmax = -notch_depth_db in the middle
let lmax = 0.0 - clamp(notch_depth_db, 3.0, 36.0);
let fTop = sr * 0.45;

let sigL = xL;
let sigR = xR;

for (let m: Int = 0; m < nStages; m++) {
	let posL = wrap((m + uL) / nStages, 0.0, 1.0);
	let posR = wrap((m + uR) / nStages, 0.0, 1.0);

	// a stage that just wrapped is at the 0 dB edge: clear its state there
	if (abs(posL - posPrevL[m]) > 0.5) { ic1L[m] = 0.0; ic2L[m] = 0.0; }
	if (abs(posR - posPrevR[m]) > 0.5) { ic1R[m] = 0.0; ic2R[m] = 0.0; }
	posPrevL[m] = posL;
	posPrevR[m] = posR;

	// Eq. 2: centre frequencies one octave apart
	let fRawL = fBase * exp2(posL * nStages);
	let fRawR = fBase * exp2(posR * nStages);
	let fcL = clamp(fRawL, 20.0, fTop);
	let fcR = clamp(fRawR, 20.0, fTop);

	// Eq. 3-4: raised-cosine depth envelope across the ladder, faded out over
	// the octave below 0.45*sr so stages above it never stack at the clamp
	let envL = (0.5 - 0.5 * cos(2.0 * pi * posL)) * clamp(log2(fTop / fRawL), 0.0, 1.0);
	let envR = (0.5 - 0.5 * cos(2.0 * pi * posR)) * clamp(log2(fTop / fRawR), 0.0, 1.0);
	let gL = dbtoa(lmax * envL);
	let gR = dbtoa(lmax * envR);

	// --- Left: TPT state-variable bandpass + notch ---
	let tL = tan(pi * fcL / sr);
	let a1L = 1.0 / (1.0 + tL * (tL + kq));
	let a2L = tL * a1L;
	let a3L = tL * a2L;
	let v3L = sigL - ic2L[m];
	let v1L = a1L * ic1L[m] + a2L * v3L;
	let v2L = ic2L[m] + a2L * ic1L[m] + a3L * v3L;
	ic1L[m] = 2.0 * v1L - ic1L[m];
	ic2L[m] = 2.0 * v2L - ic2L[m];
	sigL = sigL - (1.0 - gL) * kq * v1L;

	// --- Right ---
	let tR = tan(pi * fcR / sr);
	let a1R = 1.0 / (1.0 + tR * (tR + kq));
	let a2R = tR * a1R;
	let a3R = tR * a2R;
	let v3R = sigR - ic2R[m];
	let v1R = a1R * ic1R[m] + a2R * v3R;
	let v2R = ic2R[m] + a2R * ic1R[m] + a3R * v3R;
	ic1R[m] = 2.0 * v1R - ic1R[m];
	ic2R[m] = 2.0 * v2R - ic2R[m];
	sigR = sigR - (1.0 - gR) * kq * v1R;
}

let gain = dbtoa(clamp(out_gain_db, -24.0, 12.0));
out1 = sigL * gain;
out2 = sigR * gain;
