// ============================================================================
// flanger_midside_just.txt
// Mid/Side parallel flanger: two independent combs, one on the mid and one on
// the side, with the side comb tuned by a 5-limit ratio above the mid comb. The
// width pumps while the mono core stays put, so a sub-heavy bass can move
// without its fundamental wandering.
// From: Orchisama Das, "An Open Source Stereo Widening Plugin", DAFx-24
//       2024_DAFx24_paper_92 §3 (broadband decorrelation filters - width from
//       the difference channel, not from detuning the whole signal);
//       Esqueda, Välimäki & Parker, DAFx-15 §3 Fig. 6 (flanger as x + g*z^-D x);
//       Tong & Tojo, SMC-19 §2 (5-limit ratios: the mid and side comb spacings
//       are related by a ratio of small integers, never a cent value).
//
// Both combs derive from one delay, so the interval between them is exact at
// every point of the sweep: the side comb is always r times the mid comb,
// including at the deepest point of the LFO. That is the "just" part - an
// equal-tempered ratio would drift as the delays move.
// ============================================================================
// Inlets:
//   in1 / in2: stereo audio
//   in3: f0 in Hz. 0 (or out of range) = unknown: hold the last valid value.
//        Cold start is 55 Hz (A1). The mid comb spacing snaps to a whole number
//        of the note's harmonics, so mid_ms selects *which* multiple it lands on.
//   in4: 0..1 one-bar phasor - locks the sweep to the bar when moving,
//        free-running LFO at rate_hz when held at 0
// Outlets:
//   out1 / out2: stereo, wet only
// No dry/wet control, no limiter, no metering.
// ============================================================================

@param({ min: 0.5, max: 12 }) mid_ms = 3;                  // mid comb spacing
@param({ enum: ["1/1", "6/5", "5/4", "4/3", "3/2", "2/1"] }) side_ratio = 4; // side interval
@param({ min: 0.02, max: 2 }) rate_hz = 0.11;              // sweep rate
@param({ min: 0, max: 1 }) depth = 0.7;                    // sweep depth
@param({ min: -0.85, max: 0.85 }) feedback = 0.35;         // feedback
@param({ min: 0, max: 2 }) side_width = 1;                 // side width
@param({ min: -24, max: 12 }) out_gain_db = 0;             // output gain

@state bufM = new FixedSampleArray(2048);
@state bufS = new FixedSampleArray(2048);
@state wi: Int = 0;

@state dampM = 0.0;
@state dampS = 0.0;
@state dSm = 64.0;

@state f0Last = 55.0;
@state f0Sm = 55.0;

@state phF = 0.0;
@state phOff = 0.0;
@state prevBarIn = 0.0;

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = in2;
let f0In = in3;
let barIn = in4;

// --- in3: 0 = unknown, hold the last valid f0 (cold start 55 Hz) ------------
if (f0In > 20.0 && f0In < 1200.0) {
	f0Last = f0In;
}
let f0Slew = 1.0 - exp(0.0 - 1.0 / (0.02 * sr));
f0Sm = f0Sm + (f0Last - f0Sm) * f0Slew;
let f0 = clamp(f0Sm, 20.0, 1200.0);
let period = sr / f0;

// --- Sweep phase -----------------------------------------------------------
// The bar phase is carried as an offset against a free-running phase, so a
// moving phasor takes over exactly and a stopped one hands back with no jump.
let rate = clamp(rate_hz, 0.02, 2.0);
phF = wrap(phF + rate / sr, 0.0, 1.0);
if (abs(barIn - prevBarIn) > 0.0000001) {
	phOff = wrap(barIn - phF, 0.0, 1.0);
}
prevBarIn = barIn;
let ph = wrap(phF + phOff, 0.0, 1.0);

// --- Mid comb spacing: snap to a whole number of the note's harmonics -------
let want = clamp(mstosamps(clamp(mid_ms, 0.5, 12.0)), 8.0, 1900.0);
let mult = clamp(round(period / want), 1.0, 64.0);
let target = period / mult;
dSm = dSm + (target - dSm) * f0Slew;

// --- Sweep: both combs scale together, so the ratio stays exact ------------
let oct = clamp(depth, 0.0, 1.0) * 0.5 * sin(2.0 * pi * ph);
let dM = clamp(dSm * exp2(oct), 8.0, 1900.0);

let ratio = 1.0;
if (side_ratio == 1) { ratio = 1.2; }
else if (side_ratio == 2) { ratio = 1.25; }
else if (side_ratio == 3) { ratio = 1.3333333; }
else if (side_ratio == 4) { ratio = 1.5; }
else if (side_ratio == 5) { ratio = 2.0; }
let dS = clamp(dM / ratio, 8.0, 1900.0);

let fbAmt = clamp(feedback, -0.85, 0.85);
let dampC = 1.0 - exp(0.0 - 2.0 * pi * 6000.0 / sr);
let wid = clamp(side_width, 0.0, 2.0);

// --- Mid and side into their own delay lines -------------------------------
let mid = 0.5 * (xL + xR);
let side = 0.5 * (xL - xR);

bufM[wi] = mid + fbAmt * dampM;
bufS[wi] = side + fbAmt * dampS;

let rM = wi - dM;
while (rM < 0.0) { rM += 2048.0; }
let qM: Int = floor(rM);
let tapM = cubicinterp(rM - qM,
	bufM[(qM - 1 + 2048) & 2047],
	bufM[qM & 2047],
	bufM[(qM + 1) & 2047],
	bufM[(qM + 2) & 2047]);

let rS = wi - dS;
while (rS < 0.0) { rS += 2048.0; }
let qS: Int = floor(rS);
let tapS = cubicinterp(rS - qS,
	bufS[(qS - 1 + 2048) & 2047],
	bufS[qS & 2047],
	bufS[(qS + 1) & 2047],
	bufS[(qS + 2) & 2047]);

let midOut = 0.5 * (mid + tapM);
let sideOut = 0.5 * (side + tapS);

dampM = dampM + dampC * (midOut - dampM);
dampS = dampS + dampC * (sideOut - dampS);

wi = (wi + 1) & 2047;

let gain = dbtoa(clamp(out_gain_db, -24.0, 12.0));
out1 = (midOut + wid * sideOut) * gain;
out2 = (midOut - wid * sideOut) * gain;
