// ============================================================================
// filter_vplpc_vocal_effort_formant.txt
// Stereo Variable Pre-Emphasis Vocal Effort & Aspiration Formant Filter
// Implements Karl I. Nordstrom & Peter F. Driessen (DAFx-06, Paper p_157):
// "Variable Pre-Emphasis LPC for Modeling Vocal Effort in the Singing Voice"
// (Sections 1.1, 2.1, 2.2 & Fig. 1, 3, 4, 5).
// Models the physiological transition from lax/breathy voice to strained,
// high-effort vocal screaming:
//   1. Variable Pre-Emphasis / Spectral Tilt + 2.5 kHz Singer's Effort Hump
//      (Sec 2.1, Fig. 4a vs 4b): Low vocal effort exhibits a steep -18 dB/oct
//      glottal roll-off; high vocal effort flattens the glottal tilt via
//      impulsive nonlinear glottal closure compression and excites a prominent
//      2-3 kHz high-effort resonance hump!
//   2. Pitch-Synchronous Pulsed Aspiration Noise (Sec 2.2, Fig. 5): Blends
//      Gaussian noise pulsed at a 50% duty cycle locked to the tuba's fundamental
//      frequency f0 (in3, Hz) through matched highpass/lowpass crossover filters
//      before feeding the 4-formant vocal tract filter (1/A_F).
// ============================================================================
// Inlets:
//   in1: Left Audio Input (Tuba)
//   in2: Right Audio Input (if unpatched, mono in1 is used automatically)
//   in3: Tracked Tuba Pitch f0 in Hz (locks 50% duty-cycle aspiration noise & formants)
//   in4: Modulation Signal in Hz (modulates formant cutoff & vocal effort hump in Hz)
// Outlets:
//   out1: Left Variable Vocal Effort Formant Output
//   out2: Right Variable Vocal Effort Formant Output

@param({ min: 0.0, max: 1.0 }) vocal_effort = 0.70;
@param({ min: 0.0, max: 1.0 }) vowel_morph = 0.45;
@param({ min: 0.0, max: 1.0 }) pulsed_aspiration = 0.25;
@param({ min: 1800.0, max: 4200.0 }) effort_hump_hz = 2650.0;
@param({ min: 0.5, max: 1.4 }) pole_bwe_radius = 0.96;
@param({ min: 0.0, max: 1.5 }) pitch_track_amt = 0.80;
@param({ min: -2.0, max: 2.0 }) mod_hz_depth = 1.0;
@param({ min: 0.0, max: 1.0 }) dry_wet_mix = 0.90;

// Variable Pre-Emphasis / Glottal Source Tilting & 2.5 kHz Effort Hump filters (A_PN)
@state laxLpL = new onepole();
@state laxLpR = new onepole();
@state humpSvfL = new svf();
@state humpSvfR = new svf();

// Matched crossover filters for pitch-pulsed aspiration noise (DAFx-06 Sec 2.2, Fig. 5)
@state noiseHpL = new onepole();
@state noiseHpR = new onepole();
@state whiteL = new noise();
@state whiteR = new noise();
@state pulsePhasor = new phasor();

// 4-Formant Vocal Tract Filter Bank (1 / A_F)
@state tractF1L = new svf(); @state tractF2L = new svf();
@state tractF3L = new svf(); @state tractF4L = new svf();
@state tractF1R = new svf(); @state tractF2R = new svf();
@state tractF3R = new svf(); @state tractF4R = new svf();

// Envelope follower & fallback ZC pitch tracker
@state envRms = 0.0;
@state lpTrack = new onepole();
@state prevSig = 0.0;
@state zcCount = 0.0;
@state zcHz = 58.27;
@state smoothF0 = 58.27;
@state smoothEffort = 0.70;
@state smoothVowel = 0.45;

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = (abs(in2) > 0.0000001) ? in2 : xL;
let pitchIn = in3;
let modHz = in4;

// --- 1. Pitch Tracking (in3 in Hz) & Input Envelope ---
let mono = 0.5 * (xL + xR);
let envCoeff = 1.0 - exp(-1.0 / (0.008 * sr));
envRms += (mono * mono - envRms) * envCoeff;
let rms = sqrt(maximum(envRms, 0.0));

let monoLow = lpTrack.next(mono, 190.0);
zcCount += 1.0;
if (prevSig <= 0.0 && monoLow > 0.0) {
	if (zcCount > sr / 500.0 && zcCount < sr / 22.0) {
		zcHz = sr / zcCount;
	}
	zcCount = 0.0;
}
prevSig = monoLow;

let rawF0 = (pitchIn > 20.0 && pitchIn < 1200.0) ? pitchIn : zcHz;
let slew = 1.0 - exp(-1.0 / (0.005 * sr));
smoothF0 += (rawF0 - smoothF0) * slew;
let f0 = clamp(smoothF0, 24.0, 1000.0);

let mDepth = clamp(mod_hz_depth, -2.0, 2.0);
let modNorm = clamp((modHz * mDepth) / 700.0, -0.5, 0.5);
smoothEffort += (clamp(vocal_effort + 0.5 * modNorm, 0.0, 1.0) - smoothEffort) * slew;
smoothVowel += (clamp(vowel_morph + modNorm, 0.0, 1.0) - smoothVowel) * slew;

// --- 2. Glottal Source Effort Shaping & Variable Pre-Emphasis (DAFx-06 Sec 1.1 & 2.1, Fig. 4) ---
// Low effort = lax sinusoidal/warm slope; High effort = compressed vocal folds with reduced
// open quotient (sharp impulse-like edges) + 2-3 kHz high-effort spectral hump (Fig. 4b)
let eff = smoothEffort;
let foldDrive = 1.0 + 9.0 * pow(eff, 1.6);
let tenseL = tanh(xL * foldDrive);
let tenseR = tanh(xR * foldDrive);

let laxL = laxLpL.next(xL, 420.0);
let laxR = laxLpR.next(xR, 420.0);

let humpFreq = clamp(effort_hump_hz + mDepth * modHz * 0.5, 1200.0, 5000.0);
let humpL = humpSvfL.next(tenseL, humpFreq, 0.72)[2];
let humpR = humpSvfR.next(tenseR, humpFreq * 1.03, 0.72)[2];

let glottalL = linearinterp(eff, laxL, tenseL + 1.35 * eff * humpL);
let glottalR = linearinterp(eff, laxR, tenseR + 1.35 * eff * humpR);

// --- 3. Pitch-Synchronous 50% Duty-Cycle Pulsed Aspiration Noise (Sec 2.2, Fig. 5) ---
// "This noise was pulsed in sync with the frequency of the voice using a square
// envelope with a 50% duty cycle... Matched first-order Butterworth filters were
// used to low-pass the residual and high-pass the pulsed noise."
let ph = pulsePhasor.next(f0);
let sqGate = (ph < 0.5) ? 1.0 : 0.15;
let rawNL = whiteL.next() * sqGate * rms * 2.2;
let rawNR = whiteR.next() * sqGate * rms * 2.2;

let hpNoiseL = rawNL - noiseHpL.next(rawNL, 1400.0);
let hpNoiseR = rawNR - noiseHpR.next(rawNR, 1400.0);

let aspAmt = clamp(pulsed_aspiration, 0.0, 1.0);
let excL = glottalL + aspAmt * hpNoiseL;
let excR = glottalR + aspAmt * hpNoiseR;

// --- 4. Vocal Tract Formant Filter 1/A_F with Pole Bandwidth Expansion (Sec 2.1, Fig. 3) ---
// Morph across 5 singing vowels: OO -> OH -> AH -> EH -> EE
let vScaled = smoothVowel * 4.0;
let vIdx: Int = floor(clamp(vScaled, 0.0, 3.999));
let vFrac = vScaled - vIdx;

let f1A = 300.0; let f2A = 870.0;  let f3A = 2240.0;
let f1B = 450.0; let f2B = 1100.0; let f3B = 2450.0;
if (vIdx == 1) {
	f1A = 450.0; f2A = 1100.0; f3A = 2450.0;
	f1B = 730.0; f2B = 1090.0; f3B = 2440.0;
} else if (vIdx == 2) {
	f1A = 730.0; f2A = 1090.0; f3A = 2440.0;
	f1B = 530.0; f2B = 1840.0; f3B = 2480.0;
} else if (vIdx == 3) {
	f1A = 530.0; f2A = 1840.0; f3A = 2480.0;
	f1B = 270.0; f2B = 2290.0; f3B = 3010.0;
}

let pitchShiftHz = clamp(pitch_track_amt, 0.0, 1.5) * (f0 - 65.0) + mDepth * modHz;
let f1 = clamp(linearinterp(vFrac, f1A, f1B) + pitchShiftHz * 0.6, 150.0, 1300.0);
let f2 = clamp(linearinterp(vFrac, f2A, f2B) + pitchShiftHz * 1.2, 400.0, 3800.0);
let f3 = clamp(linearinterp(vFrac, f3A, f3B) + pitchShiftHz * 1.6, 1200.0, 5500.0);
let f4 = clamp(3400.0 + pitchShiftHz * 1.8, 2000.0, 7500.0);

// Bandwidth Expansion (BWE) pole scaling parameter (Sec 2.1): >1.0 pushes poles toward unit circle
let bwe = clamp(pole_bwe_radius, 0.5, 1.4);
let svfQ = clamp(0.45 + 0.38 * bwe, 0.50, 0.982);

let y1L = tractF1L.next(excL, f1, svfQ)[2];
let y2L = tractF2L.next(excL, f2, svfQ)[2];
let y3L = tractF3L.next(excL, f3, svfQ)[2];
let y4L = tractF4L.next(excL, f4, svfQ)[2];

let y1R = tractF1R.next(excR, f1 * 1.015, svfQ)[2];
let y2R = tractF2R.next(excR, f2 * 0.985, svfQ)[2];
let y3R = tractF3R.next(excR, f3 * 1.015, svfQ)[2];
let y4R = tractF4R.next(excR, f4 * 0.985, svfQ)[2];

// Upper formants gain increases with vocal_effort (Fig. 1: solid high-effort vs dashed breathy)
let upGain = 0.45 + 1.15 * eff;
let wetL = tanh((y1L + upGain * (0.9 * y2L + 0.75 * y3L + 0.55 * y4L)) * 0.95);
let wetR = tanh((y1R + upGain * (0.9 * y2R + 0.75 * y3R + 0.55 * y4R)) * 0.95);

let mix = clamp(dry_wet_mix, 0.0, 1.0);
out1 = linearinterp(mix, xL, wetL);
out2 = linearinterp(mix, xR, wetR);
