// ============================================================================
// snare_wire_collision.txt — triggered snare drum: six-mode batter head,
//                            snare wires driven by a one-sided collision
//                            force, wire texture from velvet noise
//
// From: A. Torin, B. Hamilton, "An Energy Conserving Finite Difference Scheme
//       for the Simulation of Collisions in Snare Drums", DAFx-14: §2.4
//       eq. 10-11 (mallet force kappa*[eta]+^alpha), §2.5 eq. 15-16 (snare
//       wires collide one-sidedly with the head, force kappa*[xi]+^beta);
//       R. Marogna, F. Avanzini, B. Bank, "Energy Based Synthesis of Tension
//       Modulation in Membranes", DAFx-10 P49: §2.2 (circular membrane modes
//       J_n(mu_nm r / R), mu_nm = Bessel zeros);
//       K. J. Werner, "Generalizations of Velvet Noise and their Use in 1-Bit
//       Music", DAFx-19 paper 53 (one signed impulse per period at a random
//       offset: sparse noise whose density is a timbre control);
//       S. Poirot, S. Bilbao, R. Kronland-Martinet, "A Coupled Resonant
//       Filter Bank for the Sound Synthesis of Nonlinear Sources", DAFx-23
//       paper 42: §5.2 (collisions as a nonlinear coupling between modes;
//       random processes add realism).
//
// Head: six complex one-pole resonators at the Bessel-zero ratios
// (0,1) 1.000, (1,1) 1.593, (2,1) 2.136, (0,2) 2.295, (3,1) 2.653,
// (1,2) 2.917, each fed by the stick's contact pulse times J_n(mu r) for
// strike radius r (centre = round, towards the rim = ringier). Higher modes
// decay faster; the head gets a small level lift below ~40 Hz so loudness
// stays even across tunings. Wires: the low-mode head displacement d gives a one-sided
// penetration [-d - gap]+, force = pen^1.5 (Torin eq. 16). The force
// envelope scales a velvet-noise stream per channel (separate L/R, so the
// wires are wide) that is high-passed and toned. The wires come in bursts
// at the head's own period, which is the snare buzz. A small gap
// (loose) keeps them colliding far down the decay.
// Every hit moves a random-walk strike radius (mode balance, contact time,
// detune, wire sensitivity). A retrigger adds into the ringing modes (the
// resonators are linear and never reset) with a shorter contact.
//
// Inlets:
//   in1: audio-rate trigger (single-sample click, rising edge above 0.1)
// Outlets:
//   out1 / out2: stereo (head centred, wires wide)
// No safety limiter, no metering.
// ============================================================================

// Tune (head fundamental)
@param({ min: 35, max: 800, exponent: 3 }) tune_hz = 190;
// Head decay (time to -40 dB)
@param({ min: 40, max: 800, exponent: 2 }) decay_ms = 180;
// Snare wires level
@param({ min: 0, max: 100 }) snares = 70;
// Wire gap: tight (0) -> loose (100)
@param({ min: 0, max: 100 }) wire_gap = 50;
// Wire tone (lowpass)
@param({ min: 1500, max: 12000, exponent: 2 }) tone_hz = 5000;
// Grain: crackly sparse (0) -> smooth dense (100)
@param({ min: 0, max: 100 }) grain = 70;
// Hit-to-hit variance
@param({ min: 0, max: 100 }) variance = 35;
// Output gain
@param({ min: -24, max: 12 }) out_gain_db = 0;

@state rng = new noise();
@state wireHpL = new onepole();
@state wireHpR = new onepole();
@state wireLp1L = new onepole();
@state wireLp2L = new onepole();
@state wireLp1R = new onepole();
@state wireLp2R = new onepole();
@state crackLp = new onepole();
@state dcLpL = new onepole();
@state dcLpR = new onepole();

// per mode: Bessel order, Bessel zero, strike weight, rotation, state
@state modeOrder = new FixedSampleArray(6);
@state modeMu = new FixedSampleArray(6);
@state modeW = new FixedSampleArray(6);
@state modeCos = new FixedSampleArray(6);
@state modeSin = new FixedSampleArray(6);
@state modeRe = new FixedSampleArray(6);
@state modeIm = new FixedSampleArray(6);

@state prevTrig = 0.0;
@state lastSr = 0.0;
@state lastTune = -1.0;
@state lastDecay = -1.0;
@state lastSnares = -1.0;
@state spot = 0.0;
@state detuneMul = 1.0;
@state headComp = 1.0;
@state gapMul = 1.0;
@state pulsePos = 1000000.0;
@state pulseLen = 24.0;
@state pulseGain = 0.0;
@state prevPulse = 0.0;
@state wireEnv = 0.0;
@state velPosL = 0.0;
@state velAtL = 0.0;
@state velSgnL = 1.0;
@state velPosR = 0.0;
@state velAtR = 0.0;
@state velSgnR = 1.0;

function init() {
	modeOrder[0] = 0.0;
	modeMu[0] = 2.404826;
	modeOrder[1] = 1.0;
	modeMu[1] = 3.831706;
	modeOrder[2] = 2.0;
	modeMu[2] = 5.135622;
	modeOrder[3] = 0.0;
	modeMu[3] = 5.520078;
	modeOrder[4] = 3.0;
	modeMu[4] = 6.380162;
	modeOrder[5] = 1.0;
	modeMu[5] = 7.015587;
}

// J_n(x) = (1/pi) * integral_0^pi cos(n t - x sin t) dt  (midpoint rule)
function besselJ(n, x) {
	let acc = 0.0;
	for (let k: Int = 0; k < 24; k++) {
		let t = pi * (k + 0.5) / 24.0;
		acc += cos(n * t - x * sin(t));
	}
	return acc / 24.0;
}

function setModes(sr) {
	let f0 = clamp(tune_hz, 35.0, 800.0) * detuneMul;
	let dec = clamp(decay_ms, 40.0, 800.0) * 0.001;
	// equal loudness across tunings: lift below ~40 Hz, exactly 1 at 190 Hz
	headComp = sqrt((1.0 + (40.0 / f0) * (40.0 / f0)) / (1.0 + (40.0 / 190.0) * (40.0 / 190.0)));
	// wires resting on the head drain it a little
	let damp = 1.0 - 0.25 * clamp(snares, 0.0, 100.0) * 0.01;
	for (let i: Int = 0; i < 6; i++) {
		let ratio = modeMu[i] / 2.404826;
		let f = minimum(f0 * ratio, sr * 0.45);
		let t60 = dec * damp * pow(ratio, -0.8);
		let rad = exp(-4.6 / (t60 * sr));
		let w = 2.0 * pi * f / sr;
		modeCos[i] = rad * cos(w);
		modeSin[i] = rad * sin(w);
	}
}

let sr = maximum(samplerate(), 1.0);
let v = clamp(variance, 0.0, 100.0) * 0.01;

if (sr != lastSr || tune_hz != lastTune || decay_ms != lastDecay || snares != lastSnares) {
	setModes(sr);
	lastSr = sr;
	lastTune = tune_hz;
	lastDecay = decay_ms;
	lastSnares = snares;
}

// --- trigger ---
let trigIn = abs(in1);
if (trigIn > 0.1 && prevTrig <= 0.1) {
	let u1 = rng.next();
	let u2 = rng.next();
	let u3 = rng.next();
	let ringing = clamp(abs(modeIm[0]) + abs(modeRe[0]), 0.0, 1.0);
	spot = clamp(0.55 * spot + 0.65 * u1, -1.0, 1.0);
	// strike radius: 0.3 of the radius, wandering 0.05 .. 0.6
	let r = clamp(0.3 + 0.3 * v * spot, 0.05, 0.6);
	for (let i: Int = 0; i < 6; i++) {
		modeW[i] = besselJ(modeOrder[i], modeMu[i] * r);
	}
	detuneMul = exp2(v * 0.15 * u2 / 12.0);
	gapMul = exp2(v * 0.5 * u3);
	setModes(sr);
	// stick contact 0.35 ms; shorter on a moving head
	let tc = 0.00035 * exp2(v * 0.5 * u2) * (1.0 - 0.3 * ringing);
	pulseLen = maximum(2.0, tc * sr);
	pulsePos = 0.0;
	// unit-area pulse: the low modes get the same push for any contact time
	pulseGain = 2.4 / pulseLen;
}
prevTrig = trigIn;

let pulse = 0.0;
if (pulsePos < pulseLen) {
	let p = sin(pi * pulsePos / pulseLen);
	pulse = p * sqrt(p) * pulseGain;
	pulsePos += 1.0;
}

// --- head: complex one-pole modal resonators, output = imaginary part ---
let head = 0.0;
let disp = 0.0;
for (let i: Int = 0; i < 6; i++) {
	let re = modeRe[i];
	let im = modeIm[i];
	let x = pulse * modeW[i];
	let nre = modeCos[i] * re - modeSin[i] * im + x;
	let nim = modeSin[i] * re + modeCos[i] * im;
	modeRe[i] = nre;
	modeIm[i] = nim;
	head += nim;
	if (i < 2) {
		disp += nim;
	}
}

// stick crack: velocity of the contact pulse, lowpassed
let crack = crackLp.next(pulse - prevPulse, 6000.0) * pulseLen * 0.6;
prevPulse = pulse;

// --- wires: one-sided collision with the resonant head ---
let loose = clamp(wire_gap, 0.0, 100.0) * 0.01;
let gap = (0.005 + 0.35 * pow(1.0 - loose, 1.5)) * gapMul;
let pen = maximum(0.0 - disp - gap, 0.0);
let force = pen * sqrt(pen);
let rel = exp(-1.0 / ((0.003 + 0.022 * loose) * sr));
if (force > wireEnv) {
	wireEnv += (force - wireEnv) * 0.3;
} else {
	wireEnv *= rel;
}

// velvet noise: one +-1 impulse per period Td, at a random offset
let dens = 300.0 * exp2(5.6 * clamp(grain, 0.0, 100.0) * 0.01);
let td = maximum(sr / dens, 1.0);
let velNorm = sqrt(td) * 0.35;
let velL = 0.0;
if (velPosL >= velAtL && velPosL < velAtL + 1.0) {
	velL = velSgnL;
}
velPosL += 1.0;
if (velPosL >= td) {
	velPosL -= td;
	velAtL = floor((0.5 + 0.5 * rng.next()) * (td - 1.0));
	velSgnL = (rng.next() < 0.0) ? -1.0 : 1.0;
}
let velR = 0.0;
if (velPosR >= velAtR && velPosR < velAtR + 1.0) {
	velR = velSgnR;
}
velPosR += 1.0;
if (velPosR >= td) {
	velPosR -= td;
	velAtR = floor((0.5 + 0.5 * rng.next()) * (td - 1.0));
	velSgnR = (rng.next() < 0.0) ? -1.0 : 1.0;
}

let tone = clamp(tone_hz, 1500.0, minimum(12000.0, sr * 0.45));
let wl = velL * velNorm * wireEnv;
let wr = velR * velNorm * wireEnv;
wl = wl - wireHpL.next(wl, 1800.0);
wr = wr - wireHpR.next(wr, 1800.0);
wl = wireLp2L.next(wireLp1L.next(wl, tone), tone);
wr = wireLp2R.next(wireLp1R.next(wr, tone), tone);
let wireAmt = clamp(snares, 0.0, 100.0) * 0.01 * 2.2;
let wireL = (0.7 * wl + 0.3 * wr) * wireAmt;
let wireR = (0.7 * wr + 0.3 * wl) * wireAmt;

let centre = 0.28 * headComp * head + crack;
let yl = centre + wireL;
let yr = centre + wireR;
yl = yl - dcLpL.next(yl, 20.0);
yr = yr - dcLpR.next(yr, 20.0);
let gain = dbtoa(clamp(out_gain_db, -24.0, 12.0));
out1 = yl * gain;
out2 = yr * gain;
