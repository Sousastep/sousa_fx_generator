// ============================================================================
// kick_808_tension.txt — triggered kick drum: sine body whose pitch falls with
//                        its energy (tension modulation), loopback-FM
//                        harmonics for small speakers, soft contact beater
//
// From: K. J. Werner, J. S. Abel, J. O. Smith III, "A Physically-Informed,
//       Circuit-Bendable, Digital Model of the Roland TR-808 Bass Drum
//       Circuit", DAFx-14: §4 (pulse shaper: the trigger is a short smooth
//       pulse, not a step), §8.1 (frequency shift on attack), §8.2 (pitch
//       sigh: the centre frequency rises while the swing is large);
//       R. Marogna, F. Avanzini, B. Bank, "Energy Based Synthesis of Tension
//       Modulation in Membranes", DAFx-10 P49: §2.3, §3.2 eq. 20 (quasistatic
//       tension is proportional to membrane energy, f ~ sqrt(tension));
//       J. Hsu, T. Smyth, "Percussion Synthesis using Loopback Frequency
//       Modulation Oscillators", SMC-19: §3.2 eq. 4-5, §3.4 eq. 8-9 (closed
//       form z0 = (b0 + e^jT)/(1 + b0 e^jT): b0 sets the harmonic series
//       independently of the integrated phase T);
//       F. Avanzini, D. Rocchesso, "Modeling Collision Sounds: Non-Linear
//       Contact Force", DAFx-01: §2.1 (contact force ~ compression^alpha,
//       contact time shortens with hardness).
//
// Body: phase-integrated oscillator at f = tune * sqrt(1 + k * tension),
// tension decaying with drop_ms and topped up by each hit (energy adds, so a
// retrigger into a ringing drum starts a little sharper). Its waveform is
// Re{z0} - b0 with b0 <= 0 driven by a fast envelope plus the amplitude, so
// the attack carries a harmonic series (audible on phones) that thins to a
// clean sine as the note dies (b0 is also capped so the series is -60 dB
// at Nyquist on high tunings). Gentle tanh for weight; a -3 dB tilt at
// 800 Hz keeps loudness even across the tuning range. Beater: half-sine^1.5
// contact pulse, lowpassed (no click) plus a band-passed "knock".
// Every hit moves a random-walk strike spot (detune, brightness, knock
// pitch, contact time); a retrigger ramps from the current level (no reset,
// phase continuous) with a shorter, quieter contact.
//
// Inlets:
//   in1: audio-rate trigger (single-sample click, rising edge above 0.1)
// Outlets:
//   out1 / out2: audio (mono kick on both)
// No safety limiter, no metering.
// ============================================================================

// Tune (settled pitch)
@param({ min: 35, max: 800, exponent: 3 }) tune_hz = 50;
// Decay (time to -40 dB)
@param({ min: 80, max: 1500, exponent: 2 }) decay_ms = 380;
// Pitch drop at the hit
@param({ min: 0, max: 36 }) drop_st = 14;
// Pitch drop time
@param({ min: 5, max: 200, exponent: 2 }) drop_ms = 35;
// Harmonics (loopback FM brightness + weight)
@param({ min: 0, max: 100 }) harmonics = 45;
// Beater
@param({ min: 0, max: 100 }) beater = 40;
// Hit-to-hit variance
@param({ min: 0, max: 100 }) variance = 35;
// Output gain
@param({ min: -24, max: 12 }) out_gain_db = 0;

@state rng = new noise();
@state thumpLp = new onepole();
@state dcLp = new onepole();

@state prevTrig = 0.0;
@state lastSr = 0.0;
@state lastDecay = -1.0;
@state lastDrop = -1.0;
@state kAmp = 0.0;
@state kTens = 0.0;
@state kBright = 0.0;

@state ph = 0.0;
@state ampEnv = 0.0;
@state ampStart = 0.0;
@state brightEnv = 0.0;
@state brightStart = 0.0;
@state atkPos = 1000000.0;
@state atkLen = 48.0;
@state tens = 0.0;

@state spot = 0.0;
@state detuneMul = 1.0;
@state harmMul = 1.0;

@state beatPos = 1000000.0;
@state beatLen = 48.0;
@state beatGain = 0.0;
@state kA1 = 0.0;
@state kA2 = 0.0;
@state kA3 = 0.0;
@state kK = 0.4;
@state ic1 = 0.0;
@state ic2 = 0.0;

let sr = maximum(samplerate(), 1.0);

if (sr != lastSr || decay_ms != lastDecay || drop_ms != lastDrop) {
	let dec = clamp(decay_ms, 80.0, 1500.0) * 0.001;
	let drp = clamp(drop_ms, 5.0, 200.0) * 0.001;
	kAmp = exp(-4.6 / (dec * sr));
	kTens = exp(-1.0 / (drp * sr));
	kBright = exp(-1.0 / ((0.02 + 1.2 * drp) * sr));
	lastSr = sr;
	lastDecay = decay_ms;
	lastDrop = drop_ms;
}

let v = clamp(variance, 0.0, 100.0) * 0.01;
let harm = clamp(harmonics, 0.0, 100.0) * 0.01;
let beat = clamp(beater, 0.0, 100.0) * 0.01;

// --- trigger: rising edge of the click ---
let trigIn = abs(in1);
if (trigIn > 0.1 && prevTrig <= 0.1) {
	let ringing = clamp(ampEnv, 0.0, 1.0);
	let u1 = rng.next();
	let u2 = rng.next();
	let u3 = rng.next();
	// strike spot drifts like a hand, it does not jump
	spot = clamp(0.55 * spot + 0.65 * u1, -1.0, 1.0);
	detuneMul = exp2(v * 0.2 * spot / 12.0);
	harmMul = 1.0 + v * 0.4 * abs(spot);
	if (ringing < 0.001) {
		ph = 0.0;
	}
	ampStart = ampEnv;
	brightStart = brightEnv;
	atkPos = 0.0;
	atkLen = maximum(1.0, 0.0008 * sr);
	// energy adds: a hit into a ringing head starts sharper
	tens = minimum(tens + 1.0 + v * 0.2 * u3, 1.35);
	// contact: harder beater = shorter contact; moving head = shorter, softer
	let tc = (0.0016 - 0.00115 * beat) * exp2(v * 0.7 * u2) * (1.0 - 0.3 * ringing);
	beatLen = maximum(2.0, tc * sr);
	beatPos = 0.0;
	beatGain = sqrt(beat) * (1.0 + v * 0.3 * u3) * (1.0 - 0.25 * ringing);
	let knockHz = clamp(2100.0 * exp2(v * 0.35 * spot), 200.0, sr * 0.4);
	let g = tan(pi * knockHz / sr);
	kK = 0.4;
	kA1 = 1.0 / (1.0 + g * (g + kK));
	kA2 = g * kA1;
	kA3 = g * kA2;
}
prevTrig = trigIn;

// --- envelopes: linear ramp from the current level, then exponential ---
if (atkPos < atkLen) {
	let a = atkPos / atkLen;
	ampEnv = ampStart + (1.0 - ampStart) * a;
	brightEnv = brightStart + (1.0 - brightStart) * a;
	atkPos += 1.0;
} else {
	ampEnv *= kAmp;
	brightEnv *= kBright;
}
tens *= kTens;

// --- body: tension-modulated pitch, closed-form loopback FM waveform ---
let kDrop = exp2(2.0 * clamp(drop_st, 0.0, 36.0) / 12.0) - 1.0;
let f0 = clamp(tune_hz, 35.0, 800.0) * detuneMul;
let freq = f0 * sqrt(1.0 + kDrop * tens) * (1.0 + 0.025 * ampEnv);
freq = clamp(freq, 1.0, sr * 0.45);
ph += freq / sr;
ph = ph - floor(ph);

// keep the harmonic series |b|^k under -60 dB at Nyquist for high tunings
let bLim = minimum(0.85, exp(-6.9 * freq / maximum(0.5 * sr - freq, 1.0)));
let b = 0.0 - clamp(harm * harmMul * (0.3 * ampEnv + 0.5 * brightEnv), 0.0, bLim);
let c = sin(2.0 * pi * ph);
let raw = (2.0 * b + (1.0 + b * b) * c) / (1.0 + 2.0 * b * c + b * b) - b;
let shaped = raw / (1.0 - b * b) * (1.0 + 0.5 * b);
let drive = 1.0 + 1.5 * harm;
// equal loudness across tunings: -3 dB at 800 Hz, exactly 1 at the 50 Hz default
let tuneComp = sqrt((1.0 + 0.00390625) / (1.0 + (f0 / 800.0) * (f0 / 800.0)));
let body = 0.8 * tuneComp * tanh(drive * shaped * ampEnv) / tanh(drive);

// --- beater: smooth contact pulse -> thump + knock ---
let pulse = 0.0;
if (beatPos < beatLen) {
	let p = sin(pi * beatPos / beatLen);
	pulse = p * sqrt(p) * beatGain;
	beatPos += 1.0;
}
let thump = thumpLp.next(pulse, 3500.0);
let v3 = pulse - ic2;
let v1 = kA1 * ic1 + kA2 * v3;
let v2 = ic2 + kA2 * ic1 + kA3 * v3;
ic1 = 2.0 * v1 - ic1;
ic2 = 2.0 * v2 - ic2;
let knock = kK * v1;

let mixed = body + 0.45 * thump + 3.0 * knock;
let y = mixed - dcLp.next(mixed, 15.0);
let gain = dbtoa(clamp(out_gain_db, -24.0, 12.0));
out1 = y * gain;
out2 = y * gain;
