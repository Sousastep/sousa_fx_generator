// ============================================================================
// yoi_cascade_formants.txt — five two-pole formant resonators in series
//                            (all-pole cascade, unity gain at DC) following
//                            U -> O -> I as in3 rises: a vowel with the
//                            natural formant tilt of a vocal tract
//
// From: P. R. Cook, "Toward the Perfect Audio Morph? Singing Voice Synthesis
//       and Processing", DAFx-98 COO62 (formants as the longitudinal modes
//       of the vocal tract tube below ~4 kHz; all-pole formant filter);
//       C. Traube, N. D'Alessandro, DAFx-05 P_104 §1.5 (vowels in the
//       (F1, F2) plane, Fig. 8 / Fig. 11).
//
// Signal (per channel):
//   v      = clamp(log2(f_in3 / 200) / 4, 0, 1)    200 Hz U, 800 O, 3200 I
//   F1..3  = log-interpolated  U (300, 870, 2240) -> O (570, 840, 2410)
//                              -> I (270, 2290, 3010) Hz; F4 3300, F5 3750
//            (all x formant shift)
//   y      = R5(R4(R3(R2(R1(x)))))     R = TPT SVF lowpass
//                                      w^2 / (s^2 + (w/Q) s + w^2):
//                                      unity at DC, peak ~Q at its formant
//   Q      = 2 * 15^in4 (2 .. 30)
//   out    = y * 0.7 * (Q / 2)^-0.4     (the tuba's low end passes at unity,
//                                       only the formant peaks grow)
//
// Inlets:
//   in1 / in2: stereo audio
//   in3: vowel position as a frequency in Hz (held when <= 1 Hz; 800 Hz = O
//        at start)
//   in4: resonance 0..1 (formant sharpness)
// Outlets:
//   out1 / out2: stereo (mono sum = plain stereo filter, see width)
// No safety limiter, no metering.
// ============================================================================

// Moves all five formants (semitones): - = bigger mouth, + = smaller
@param({ min: -12, max: 12 }) formant_shift_st = 0;
// Vowel position offset from in3 (semitones; 24 st = U to O)
@param({ min: -36, max: 36 }) freq_offset_st = 0;
// Width (%): formant-shifted side copy, cancels exactly in mono
@param({ min: 0, max: 100 }) width = 50;

// resonator states: slot = channel * 5 + formant (0 L, 1 R, 2 side copy)
@state ic1 = new FixedSampleArray(15);
@state ic2 = new FixedSampleArray(15);
@state fHold = 800.0;
@state fSm = 800.0;
@state resSm = 0.0;

// TPT SVF lowpass (unity DC, damping kd = 1/Q) on slot idx
function svfLp(idx: Int, x: number, g: number, kd: number): number {
	let a1 = 1.0 / (1.0 + g * (g + kd));
	let a2 = g * a1;
	let a3 = g * a2;
	let v3 = x - ic2[idx];
	let v1 = a1 * ic1[idx] + a2 * v3;
	let v2 = ic2[idx] + a2 * ic1[idx] + a3 * v3;
	ic1[idx] = 2.0 * v1 - ic1[idx];
	ic2[idx] = 2.0 * v2 - ic2[idx];
	return v2;
}

// Prewarped TPT gain for formant frequency 2^l2f Hz
function tptG(l2f: number, sr: number): number {
	return tan(pi * clamp(exp2(l2f), 20.0, sr * 0.45) / sr);
}

// Five formants in series, channel ch (log2 frequencies l1..l5)
function cascade(ch: Int, x: number, l1: number, l2: number, l3: number, l4: number, l5: number, kd: number, sr: number): number {
	let b: Int = ch * 5;
	let y = svfLp(b, x, tptG(l1, sr), kd);
	y = svfLp(b + 1, y, tptG(l2, sr), kd);
	y = svfLp(b + 2, y, tptG(l3, sr), kd);
	y = svfLp(b + 3, y, tptG(l4, sr), kd);
	return svfLp(b + 4, y, tptG(l5, sr), kd);
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = in2;
let aSm = 1.0 - exp(-200.0 / sr);

// --- Controls: in3 vowel position (held), in4 resonance, 5 ms smoothing ------
if (in3 > 1.0) {
	fHold = in3;
}
let fTarget = clamp(fHold * exp2(clamp(freq_offset_st, -36.0, 36.0) / 12.0), 20.0, 20000.0);
fSm = fSm + (fTarget - fSm) * aSm;
resSm = resSm + (clamp(in4, 0.0, 1.0) - resSm) * aSm;

// --- Vowel path U -> O -> I, interpolated in log2(Hz) ------------------------
let v = clamp(log2(fSm / 200.0) * 0.25, 0.0, 1.0);
let lo = (v < 0.5) ? 1.0 : 0.0;
let t = (v < 0.5) ? 2.0 * v : 2.0 * v - 1.0;
let shift = clamp(formant_shift_st, -12.0, 12.0) / 12.0;
let l1 = shift + lo * linearinterp(t, 8.2288, 9.1548) + (1.0 - lo) * linearinterp(t, 9.1548, 8.0768);
let l2 = shift + lo * linearinterp(t, 9.7649, 9.7142) + (1.0 - lo) * linearinterp(t, 9.7142, 11.1611);
let l3 = shift + lo * linearinterp(t, 11.1293, 11.2348) + (1.0 - lo) * linearinterp(t, 11.2348, 11.5555);
// F4 3300 Hz, F5 3750 Hz
let l4 = shift + 11.6883;
let l5 = shift + 11.8727;

let q = 2.0 * pow(15.0, resSm);
let kd = 1.0 / q;
let makeup = 0.7 * pow(q * 0.5, -0.4);
let wd = clamp(width, 0.0, 100.0) * 0.01;

// --- Filter: L, R, and a side copy on the mono sum shifted up to +1 octave --
let yL = cascade(0, xL, l1, l2, l3, l4, l5, kd, sr);
let yR = cascade(1, xR, l1, l2, l3, l4, l5, kd, sr);
let yD = cascade(2, 0.5 * (xL + xR), l1 + wd, l2 + wd, l3 + wd, l4 + wd, l5 + wd, kd, sr);
let side = wd * (yD - 0.5 * (yL + yR));

out1 = (yL + side) * makeup;
out2 = (yR - side) * makeup;
