// ============================================================================
// loopback_fm_growl.txt — closed-form loopback FM on the tuba's phase, with
//                         the feedback swept in bar steps for a growl
//
// From: J. Hsu, T. Smyth, "Percussion Synthesis using Loopback Frequency
//       Modulation Oscillators", SMC-19 §3.1 eqs. 2-3 (loopback FM, feedback
//       coefficient B), §3.2 eqs. 4-5 (closed form
//       z0 = (b0 + e^{jTheta}) / (1 + b0 e^{jTheta}), b0 = (sqrt(1-B^2)-1)/B),
//       §3.4 eq. 9 (time-varying b0 with an integrated phase Theta(n));
//       T. Smyth, "Representations of Self-Coupled Modal Oscillators with
//       Time-Varying Frequency", SMC-19 (self-coupled oscillator view).
//
// Theta is the tuba's own phase ramp (optionally an octave up/down), so pitch
// is fixed by the horn and only the timbre moves (the reason the paper gives
// for preferring z0 over z_c). |z0| = 1 for |b0| < 1, so the level stays put
// while the spectrum opens. At B = 0 the voice is exactly sin(2*pi*phase),
// the horn's fundamental, so "wet" blends in phase.
//
// Inlets:
//   in1 / in2: stereo tuba
//   in3: f0 in Hz. > 0 = tracked pitch (clamped 20..1000 Hz);
//        0 = disconnected -> fixed 55 Hz. No zero-crossing fallback.
//   in4: 0..1 one-bar phasor (step sweep of B). Held (no movement for
//        100 ms) -> internal 2 s bar (120 BPM 4/4).
//   in5: phase-locked tuba phasor 0..1 (Theta). Static -> internal ramp at
//        f0, continued from the last tracked phase.
//   in6: smoothed tuba envelope 0..1 (gate + dynamics to feedback).
//        Static -> internal peak follower on in1/in2.
// Outlets:
//   out1 / out2: stereo (dry keeps its image, voice is centred)
// No safety limiter, no metering.
// ============================================================================

// Feedback B
@param({ min: 0, max: 99 }) feedback = 60;
// Step sweep depth
@param({ min: 0, max: 100 }) sweep = 60;
// Sweep steps per bar
@param({ min: 1, max: 16 }) sweep_div = 4;
// Envelope to feedback
@param({ min: 0, max: 100 }) env_to_fb = 40;
// Octave
@param({ min: -1, max: 1 }) octave = 0;
// Tone (2-pole lowpass)
@param({ min: 200, max: 12000 }) tone_hz = 3000;
// Wet
@param({ min: 0, max: 100 }) wet = 100;
// Output gain
@param({ min: -24, max: 12 }) out_gain_db = 0;

@state f0Smooth = 55.0;

@state prevBarIn = 0.0;
@state barLast = 0.0;
@state barHold = 0.0;
@state freeBar = 0.0;
@state prevBarPh = 0.0;
@state barCount = 0.0;

@state prevPhIn = 0.0;
@state phHold = 0.0;
@state ownPh = 0.0;
@state prevOwnPh = 0.0;
@state flip = 0.0;

@state prevEnvIn = 0.0;
@state envHold = 0.0;
@state envInt = 0.0;

@state bSmooth = 0.0;
@state lp1 = new onepole();
@state lp2 = new onepole();

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = in2;
let pitchIn = in3;
let barIn = in4;
let phIn = in5;
let envIn = in6;

// --- f0: in3 > 0 is the pitch; 0 = disconnected -> fixed 55 Hz --------------
let f0Target = (pitchIn > 0.0) ? clamp(pitchIn, 20.0, 1000.0) : 55.0;
f0Smooth += (f0Target - f0Smooth) * (1.0 - exp(-1.0 / (0.004 * sr)));
let inc = clamp(f0Smooth, 20.0, 1000.0) / sr;

// --- Bar phase: hold-latched in4, else internal 2 s bar; 4-bar position -----
if (abs(barIn - prevBarIn) > 0.0000001) {
	barLast = barIn;
	barHold = 0.1 * sr;
} else if (barHold > 0.0) {
	barHold -= 1.0;
}
prevBarIn = barIn;
freeBar = wrap(freeBar + 0.5 / sr, 0.0, 1.0);
let barPh = (barHold > 0.0) ? wrap(barLast, 0.0, 1.0) : freeBar;
if (barPh < prevBarPh - 0.5) {
	barCount += 1.0;
	if (barCount >= 4.0) {
		barCount = 0.0;
	}
}
prevBarPh = barPh;
let pos4 = barCount + barPh;

// --- Tuba phase: in5 while it moves, else continue it at f0 -----------------
if (abs(phIn - prevPhIn) > 0.0000001) {
	phHold = 0.1 * sr;
} else if (phHold > 0.0) {
	phHold -= 1.0;
}
prevPhIn = phIn;
if (phHold > 0.0) {
	ownPh = wrap(phIn, 0.0, 1.0);
} else {
	ownPh = wrap(ownPh + inc, 0.0, 1.0);
}
// cycle-parity bit for the octave-down phase (half-rate ramp, still locked)
if (ownPh < prevOwnPh - 0.5) {
	flip = 1.0 - flip;
}
prevOwnPh = ownPh;

// --- Envelope: in6 while it moves, else internal peak follower --------------
let absMono = 0.5 * (abs(xL) + abs(xR));
let atk = 1.0 - exp(-1.0 / (0.003 * sr));
let rel = 1.0 - exp(-1.0 / (0.080 * sr));
envInt += (absMono - envInt) * ((absMono > envInt) ? atk : rel);
if (abs(envIn - prevEnvIn) > 0.0000001) {
	envHold = 0.1 * sr;
} else if (envHold > 0.0) {
	envHold -= 1.0;
}
prevEnvIn = envIn;
let envVal = (envHold > 0.0) ? clamp(envIn, 0.0, 1.0) : clamp(envInt, 0.0, 1.0);
let gate = clamp(sqrt(envVal) * 1.6, 0.0, 1.0);

// --- Theta: tuba phase, octave -1 / 0 / +1 ----------------------------------
let oct = round(clamp(octave, -1.0, 1.0));
let phO = ownPh;
if (oct < -0.5) {
	phO = 0.5 * (ownPh + flip);
} else if (oct > 0.5) {
	phO = wrap(2.0 * ownPh, 0.0, 1.0);
}
// quarter-cycle offset so that b0 = 0 gives sin(2*pi*phase)
let theta = 2.0 * pi * phO - 0.5 * pi;

// --- Feedback B: base, bar-step "yoi" contour, envelope push ---------------
let div = round(clamp(sweep_div, 1.0, 16.0));
let stepPh = wrap(pos4 * div, 0.0, 1.0);
let contour = 0.5 - 0.5 * cos(2.0 * pi * stepPh);
let fbBase = clamp(feedback, 0.0, 99.0) * 0.01;
let sw = clamp(sweep, 0.0, 100.0) * 0.01;
let bTarget = fbBase * (1.0 - sw + sw * contour);
bTarget += clamp(env_to_fb, 0.0, 100.0) * 0.01 * envVal * (0.99 - bTarget);
bTarget = clamp(bTarget, 0.0, 0.99);
bSmooth += (bTarget - bSmooth) * (1.0 - exp(-1.0 / (0.002 * sr)));

// --- Closed-form loopback FM (SMC-19 eqs. 4-5) ------------------------------
let bb = bSmooth;
let b0 = (bb > 0.0001) ? (sqrt(1.0 - bb * bb) - 1.0) / bb : (0.0 - 0.5 * bb);
let cTh = cos(theta);
let sTh = sin(theta);
let numR = b0 + cTh;
let numI = sTh;
let denR = 1.0 + b0 * cTh;
let denI = b0 * sTh;
let den2 = maximum(denR * denR + denI * denI, 0.000001);
let zRe = (numR * denR + numI * denI) / den2;

let tone = clamp(tone_hz, 200.0, minimum(12000.0, sr * 0.45));
let voice = lp2.next(lp1.next(zRe, tone), tone) * gate * 0.7;

// --- Wet: phase-locked crossfade --------------------------------------------
let w = clamp(wet, 0.0, 100.0) * 0.01;
let outGain = dbtoa(clamp(out_gain_db, -24.0, 12.0));
out1 = linearinterp(w, xL, voice) * outGain;
out2 = linearinterp(w, xR, voice) * outGain;
