// ============================================================================
// vdp_entrain_bass.txt — Van der Pol limit-cycle bass tuned to the tuba's f0
//                        and pulled into lock by the tuba's fundamental
//
// From: W. E. Weingartner, "Stable Limit Cycles as Tunable Signal Sources",
//       DAFx-25 paper 68: §2.1.1 eq. 4 (van der Pol: f(x) = -mu(1-x^2),
//       g(x) = x), §3.1 eqs. 12, 15 (generalised angular frequency
//       w = T_ang(mu)/T, amplitude normalisation by gamma_max), §3.2.1
//       (offline step counting -> lookup table, same solver as real time),
//       §3.2.2 eq. 16 (h = f T_ang / (S fs), substeps averaged for
//       antialiasing), §4.5 (mu as a timbre control);
//       M. Neukom, "Frequency Control of Nonlinear Oscillators", SMC-18
//       §2.2 (nonlinear extensions), §4.3 (control by external forces).
//
// The oscillator is solved at S = 8 substeps per sample with a semi-implicit
// Euler step (damping term implicit, so high mu stays stable). The step size
// comes from the table below, so the pitch stays at f0 whatever mu does.
// "lock" adds the tuba's phase-locked fundamental as an external force
// (Neukom §4.3): the limit cycle entrains to the horn and holds a steady phase
// relation instead of beating. mu breathes on each bar step: low mu = round
// sine, high mu = buzzy relaxation square.
//
// Table: T_ang(mu, h) and max|x| for mu = 0.1, 0.25, 0.5 ... 8 (33 rows,
// step 0.25) x h = 0.0005 * 2^j, j = 0..7 (8 columns), index mu_row*8 + j.
// Made offline with the same solver (80 time units settle, 3 periods
// counted on upward zero crossings). Simulated tuning error without lock:
// <= 5 cents up to 200 Hz for all mu, <= 20 cents at 500 Hz.
//
// Inlets:
//   in1 / in2: stereo tuba
//   in3: f0 in Hz. > 0 = tracked pitch (clamped 20..1000 Hz);
//        0 = disconnected -> fixed 55 Hz. No zero-crossing fallback.
//   in4: 0..1 one-bar phasor (mu breathing). Held (no movement for 100 ms)
//        -> internal 2 s bar (120 BPM 4/4).
//   in5: phase-locked fundamental sine -1..1, scaled by confidence (the
//        forcing). Static -> internal sine at f0.
//   in6: smoothed tuba envelope 0..1 (gate). Static -> internal peak follower.
// Outlets:
//   out1 / out2: stereo (dry keeps its image, voice is centred)
// No safety limiter, no metering.
// ============================================================================

@state f0Smooth = 55.0;

@state prevBarIn = 0.0;
@state barLast = 0.0;
@state barHold = 0.0;
@state freeBar = 0.0;
@state prevBarPh = 0.0;
@state barCount = 0.0;

@state ownPh = 0.0;
@state prevSinIn = 0.0;
@state sinHold = 0.0;

@state prevEnvIn = 0.0;
@state envHold = 0.0;
@state envInt = 0.0;

@state vx = 2.0;
@state vy = 0.0;
@state muSmooth = 2.5;
@state hPrev = 0.001;
@state lp1 = new onepole();
@state lp2 = new onepole();

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = in2;
let pitchIn = in3;
let barIn = in4;
let sinIn = in5;
let envIn = in6;

// --- f0: in3 > 0 is the pitch; 0 = disconnected -> fixed 55 Hz --------------
let f0Target = (pitchIn > 0.0) ? clamp(pitchIn, 20.0, 1000.0) : 55.0;
f0Smooth += (f0Target - f0Smooth) * (1.0 - exp(-1.0 / (0.004 * sr)));
let f0 = clamp(f0Smooth, 20.0, 1000.0);
ownPh = wrap(ownPh + f0 / sr, 0.0, 1.0);

// --- Bar phase: hold-latched in4, else internal 2 s bar; 4-bar position -----
if (abs(barIn - prevBarIn) > 0.0000001) {
	barLast = barIn;
	barHold = 0.1 * sr;
} else if (barHold > 0.0) {
	barHold -= 1.0;
}
prevBarIn = barIn;
freeBar = wrap(freeBar + 0.5 / sr, 0.0, 1.0);
let barPh = (barHold > 0.0) ? wrap(barLast, 0.0, 1.0) : freeBar;
if (barPh < prevBarPh - 0.5) {
	barCount += 1.0;
	if (barCount >= 4.0) {
		barCount = 0.0;
	}
}
prevBarPh = barPh;
let pos4 = barCount + barPh;

// --- Forcing: in5 locked sine while it moves, else internal sine at f0 ------
if (abs(sinIn - prevSinIn) > 0.0000001) {
	sinHold = 0.1 * sr;
} else if (sinHold > 0.0) {
	sinHold -= 1.0;
}
prevSinIn = sinIn;
let force = (sinHold > 0.0) ? clamp(sinIn, -1.0, 1.0) : sin(2.0 * pi * ownPh);

// --- Envelope: in6 while it moves, else internal peak follower --------------
let absMono = 0.5 * (abs(xL) + abs(xR));
let atk = 1.0 - exp(-1.0 / (0.003 * sr));
let rel = 1.0 - exp(-1.0 / (0.080 * sr));
envInt += (absMono - envInt) * ((absMono > envInt) ? atk : rel);
if (abs(envIn - prevEnvIn) > 0.0000001) {
	envHold = 0.1 * sr;
} else if (envHold > 0.0) {
	envHold -= 1.0;
}
prevEnvIn = envIn;
let envVal = (envHold > 0.0) ? clamp(envIn, 0.0, 1.0) : clamp(envInt, 0.0, 1.0);
let gate = clamp(sqrt(envVal) * 1.6, 0.0, 1.0);

// --- mu: base, breathing down towards a sine on each bar step ---------------
let div = round(clamp(sweep_div, 1.0, 16.0));
let stepPh = wrap(pos4 * div, 0.0, 1.0);
let contour = 0.5 - 0.5 * cos(2.0 * pi * stepPh);
let muTarget = clamp(mu * (1.0 - clamp(mu_sweep, 0.0, 100.0) * 0.01 * contour), 0.1, 8.0);
muSmooth += (muTarget - muSmooth) * (1.0 - exp(-1.0 / (0.003 * sr)));
let muNow = clamp(muSmooth, 0.1, 8.0);

// --- 2-D table lookup (mu x log2 h), bilinear; h by fixed-point iteration -
// h = f0 T_ang(mu, h) / (S fs) depends on the solver's own period at that h
// (§3.2.1: same solver offline and in real time), so the previous sample's
// h is used to read the table; it converges within a few samples.
let fi = (muNow < 0.25) ? (muNow - 0.1) / 0.15 : (muNow - 0.25) / 0.25 + 1.0;
fi = clamp(fi, 0.0, 31.999);
let i0: Int = trunc(fi);
let fr = fi - i0;
let hj = clamp(log2(maximum(hPrev, 0.0000001) / 0.0005), 0.0, 6.999);
let j0: Int = trunc(hj);
let fh = hj - j0;
let a00: Int = i0 * 8 + j0;
let a10: Int = a00 + 8;
let tA = tangTab[a00] + fh * (tangTab[a00 + 1] - tangTab[a00]);
let tB = tangTab[a10] + fh * (tangTab[a10 + 1] - tangTab[a10]);
let tAng = tA + fr * (tB - tA);
let xA = xmaxTab[a00] + fh * (xmaxTab[a00 + 1] - xmaxTab[a00]);
let xB = xmaxTab[a10] + fh * (xmaxTab[a10 + 1] - xmaxTab[a10]);
let xMax = xA + fr * (xB - xA);

// --- Solve: S substeps, semi-implicit Euler, averaged (DAFx-25 eq. 16) ------
let subS = 8.0;
let h = f0 * tAng / (subS * sr);
hPrev = h;
let fAmp = clamp(lock, 0.0, 100.0) * 0.015 * force;
let acc = 0.0;
for (let k: Int = 0; k < 8; k++) {
	let den = maximum(1.0 - h * muNow * (1.0 - vx * vx), 0.1);
	vy = (vy + h * (fAmp - vx)) / den;
	vx = vx + h * vy;
	acc += vx;
}
// recover from any numerical blow-up (NaN fails the comparison too)
if (!(abs(vx) < 10.0) || !(abs(vy) < 200.0)) {
	vx = 2.0;
	vy = 0.0;
	acc = 0.0;
}
let osc = acc / (subS * xMax);

let tone = clamp(tone_hz, 200.0, minimum(12000.0, sr * 0.45));
let voice = lp2.next(lp1.next(osc, tone), tone) * gate * 0.8;

// --- Mix: crossfade with an oscillator that is entrained to the horn --------
let w = clamp(mix, 0.0, 100.0) * 0.01;
let outGain = dbtoa(clamp(out_gain_db, -24.0, 12.0));
out1 = linearinterp(w, xL, voice) * outGain;
out2 = linearinterp(w, xR, voice) * outGain;
