// ============================================================================
// harmonizer_minishift_doubler.txt
// Doubles the input with two mini-detuned voices for stereo width and
// thickness.
// From: M. Hoffman & P. Cook, "Real-Time Dissonancizers: Two Dissonance-
//       Augmenting Audio Effects", DAFx-08, Paper 30, Sec. 3-4 (detuned
//       spectral duplicates beat against the source and thicken it), and
//       O. Das, "An Open Source Stereo Widening Plugin", DAFx-24, Paper 92,
//       Sec. 1 (decorrelation between the channels is what creates width).
//       The pitch engine is the PSOLA grain re-spacing of DAFx-11 Paper 52
//       Sec. 4.1-4.2: two voices at +/- detune_cents, one per side.
// in1/in2: stereo audio (in2 silent for 250 ms -> in1 is used for both)
// in3: tuba f0 in Hz, used only for the grain length (no pitch tracking);
//      0 (or anything outside 20..1200 Hz) holds the last valid value;
//      fallback_hz is used live until the first valid value arrives
// in4: one-bar phasor 0..1 -> detune breathing (exception to the Hz
//      contract); 100 ms after in4 stops moving an internal 0.13 Hz
//      oscillator drives the breathing (motion = 0 for a static minishift)
// Latency of the doubled pair: pre_delay_ms + 0..1 f0 period (grains start
// at the newest analysis mark)
// out1/out2: stereo out
// ============================================================================

// Mini-detune per side, in cents
@param({ min: 0.0, max: 40.0 }) detune_cents = 12.0;
// Pre-delay of the doubled pair
@param({ min: 0.0, max: 30.0 }) pre_delay_ms = 12.0;
// Voice A left <-> voice B right
@param({ min: 0.0, max: 100.0 }) width = 70.0;
// Dry <-> doubled pair balance
@param({ min: 0.0, max: 100.0 }) mix = 50.0;
// Bar-rate breathing of the detune, in %
@param({ min: 0.0, max: 100.0 }) motion = 20.0;
// f0 used until in3 reports a pitch
@param({ min: 40.0, max: 300.0 }) fallback_hz = 82.0;

@state bufL = new FixedSampleArray(16384);
@state bufR = new FixedSampleArray(16384);
@state winTab = new FixedSampleArray(1025);
@state aStart = new FixedIntArray(12);
@state aAge = new FixedIntArray(12);
@state aLen = new FixedIntArray(12);
@state bStart = new FixedIntArray(12);
@state bAge = new FixedIntArray(12);
@state bLen = new FixedIntArray(12);
@state wi: Int = 0;
@state markIdx: Int = 0;
@state gLenI: Int = 256;
@state slotA: Int = 0;
@state slotB: Int = 0;
@state f0Hold = 0.0;
@state haveF0: Int = 0;
@state markAcc = 0.0;
@state synA = 0.0;
@state synB = 0.0;
@state ratioA = 1.0;
@state ratioB = 1.0;
@state prevBarIn = 0.0;
@state barHold = 0.0;
@state in2Hold = 0.0;
@state freeLfo = new phasor();
@state srCached = 0.0;
@state ratioK = 0.01;

function init() {
	for (let i = 0; i < 1025; i++) {
		winTab[i] = 0.5 - 0.5 * cos(2.0 * pi * i / 1024.0);
	}
	for (let g = 0; g < 12; g++) {
		aStart[g] = 0;
		aAge[g] = 100000;
		aLen[g] = 0;
		bStart[g] = 0;
		bAge[g] = 100000;
		bLen[g] = 0;
	}
}

let sr = maximum(samplerate(), 1.0);
if (srCached != sr) {
	srCached = sr;
	ratioK = 1.0 - exp(-1.0 / (0.03 * sr));
}

let xL = in1;
// in2 counts as patched for 250 ms after any non-zero sample (a per-sample test
// would swap in xL for single samples at in2's zero crossings)
if (abs(in2) > 0.0000001) {
	in2Hold = 0.25 * sr;
} else if (in2Hold > 0.0) {
	in2Hold -= 1.0;
}
let xR = (in2Hold > 0.0) ? in2 : xL;
let pIn = in3;
let barIn = in4;

// --- 1. f0 conditioning: hold the last valid value, never divide by in3 ---
if (pIn > 20.0 && pIn < 1200.0) {
	f0Hold = pIn;
	haveF0 = 1;
}
let f0 = haveF0 ? f0Hold : clamp(fallback_hz, 40.0, 300.0);
let period = sr / clamp(f0, 40.0, 1200.0);

// --- 2. opposite detunes; motion breathes the spread between them ---
// Bar phasor counts as active for 100 ms after it last moved (a constant in4 must not freeze the breathing)
if (abs(barIn - prevBarIn) > 0.000001) {
	barHold = 0.1 * sr;
} else if (barHold > 0.0) {
	barHold -= 1.0;
}
prevBarIn = barIn;
let freePh = freeLfo.next(0.13);
let driftPhase = (barHold > 0.0) ? wrap(barIn, 0.0, 1.0) : freePh;
let det = clamp(detune_cents, 0.0, 40.0);
let wob = 0.01 * clamp(motion, 0.0, 100.0) * det * sin(2.0 * pi * driftPhase);
let spread = det + wob;
let rA = exp2(spread / 1200.0);
let rB = exp2((0.0 - spread) / 1200.0);
ratioA += (rA - ratioA) * ratioK;
ratioB += (rB - ratioB) * ratioK;

let hopA = maximum(period / maximum(ratioA, 0.25), 8.0);
let hopB = maximum(period / maximum(ratioB, 0.25), 8.0);
let lf = 2.0 * maximum(period, maximum(hopA, hopB)) + 0.05 * period;
lf = clamp(lf, 64.0, 10000.0);
let lInt: Int = floor(lf);
gLenI = lInt;

let preSamps: Int = floor(clamp(pre_delay_ms, 0.0, 30.0) * 0.001 * sr);

// --- 3. analysis marks every tuba period; two synthesis grain streams ---
// A grain reads forward at unity speed, so it may start at the newest mark
// (start <= write index is all that is required); each grain keeps the
// length it was born with so a note change never truncates a live window.
bufL[wi] = xL;
bufR[wi] = xR;

markAcc += 1.0;
if (markAcc >= period) {
	markAcc -= period;
	markIdx = wi;
}

synA += 1.0;
if (synA >= hopA) {
	synA -= hopA;
	aStart[slotA] = (markIdx - preSamps + 16384) & 16383;
	aAge[slotA] = 0;
	aLen[slotA] = gLenI;
	slotA = (slotA + 1) % 12;
}

synB += 1.0;
if (synB >= hopB) {
	synB -= hopB;
	bStart[slotB] = (markIdx - preSamps + 16384) & 16383;
	bAge[slotB] = 0;
	bLen[slotB] = gLenI;
	slotB = (slotB + 1) % 12;
}

// --- 4. voice A: Hann overlap-add normalised by its window sum ---
let accA = 0.0;
let accB = 0.0;
let wsumA = 0.0;
let wsumB = 0.0;
for (let g = 0; g < 12; g++) {
	let ag = aAge[g];
	let gl = aLen[g];
	if (ag < gl) {
		let xi: Int = floor(ag * 1024.0 / gl);
		if (xi > 1024) { xi = 1024; }
		let wv = winTab[xi];
		let idx = (aStart[g] + ag) & 16383;
		accA += wv * bufL[idx];
		wsumA += wv;
		aAge[g] = ag + 1;
	}
}

// --- 5. voice B ---
for (let g = 0; g < 12; g++) {
	let ag = bAge[g];
	let gl = bLen[g];
	if (ag < gl) {
		let xi: Int = floor(ag * 1024.0 / gl);
		if (xi > 1024) { xi = 1024; }
		let wv = winTab[xi];
		let idx = (bStart[g] + ag) & 16383;
		accB += wv * bufR[idx];
		wsumB += wv;
		bAge[g] = ag + 1;
	}
}
wi = (wi + 1) & 16383;

// --- 6. voice A left / voice B right, through a mono-safe mid/side ---
let vA = 0.0;
let vB = 0.0;
if (wsumA > 0.000001) { vA = accA / wsumA; }
if (wsumB > 0.000001) { vB = accB / wsumB; }
let mid = 0.5 * (vA + vB);
let side = 0.5 * (vA - vB);
let wide = clamp(width / 100.0, 0.0, 1.0);
let wetL = mid + wide * side;
let wetR = mid - wide * side;

let w = clamp(mix / 100.0, 0.0, 1.0);
out1 = linearinterp(w, xL, wetL);
out2 = linearinterp(w, xR, wetR);
