// ============================================================================
// harmonizer_psola_interval.txt
// One pitch-synchronous harmony voice at a fixed musical interval.
// From: A. von dem Knesebeck, S. Kraft & U. Zoelzer, "Realtime System for
//       Backing Vocal Harmonization", DAFx-11, Paper 52, Sec. 4.1-4.2
//       (PSOLA pitch shifters feed each backing voice), and D. Arfib &
//       V. Verfaille, "Driving Pitch-Shifting and Time-Scaling Algorithms with
//       Adaptive and Gestural Techniques", DAFx-03, Paper 20, Sec. 2
//       (PSOLA = analysis grains on pitch marks, re-spaced on synthesis).
//
// The tuba's f0 (in3) gives the analysis period: a pitch mark is stored every
// period, and each synthesis grain is copied forward from the last mark at
// rate 1.0 (content untouched, spectral envelope/formants preserved) and
// re-spaced every P/ratio samples, so the output F0 is the tuba's f0 times the
// interval ratio. Hann grains of two periods (longer when a downward shift
// needs more overlap) are overlap-added and normalised by the summed window.
// Each grain keeps the length it was born with (no truncation on note changes).
// The left and right voices are separate grain streams drifting in opposite
// directions (+/- drift_cents), so the voice has width even on a mono tuba.
// Latency of the voice: 0..1 f0 period (grains start at the newest mark).
// in1/in2: stereo audio (in2 silent for 250 ms -> in1 is used for both)
// in3: tuba f0 in Hz; 0 (or anything outside 20..1200 Hz) holds the last valid
//      value; fallback_hz is used live until the first valid value arrives
// in4: one-bar phasor 0..1 (exception to the Hz contract); 100 ms after it
//      stops moving the drift runs from an internal 0.13 Hz oscillator
// out1/out2: stereo out (dry crossfaded with the harmony voice)
// ============================================================================

// Interval above (+) or below (-) the tuba
@param({ min: -12.0, max: 12.0 }) interval = 7.0;
// Dry <-> harmony balance
@param({ min: 0.0, max: 100.0 }) mix = 40.0;
// Extra grain length, % of a period (more = smoother splices)
@param({ min: 5.0, max: 60.0 }) grain_smooth = 5.0;
// Stereo width of the harmony voice
@param({ min: 0.0, max: 100.0 }) width = 70.0;
// Bar-rate drift depth, in cents
@param({ min: 0.0, max: 25.0 }) drift_cents = 4.0;
// f0 used until in3 reports a pitch
@param({ min: 40.0, max: 300.0 }) fallback_hz = 82.0;

@state bufL = new FixedSampleArray(16384);
@state bufR = new FixedSampleArray(16384);
@state winTab = new FixedSampleArray(1025);
@state lStart = new FixedIntArray(12);
@state lAge = new FixedIntArray(12);
@state lLen = new FixedIntArray(12);
@state rStart = new FixedIntArray(12);
@state rAge = new FixedIntArray(12);
@state rLen = new FixedIntArray(12);
@state wi: Int = 0;
@state markIdx: Int = 0;
@state gLenL: Int = 256;
@state gLenR: Int = 256;
@state slotL: Int = 0;
@state slotR: Int = 0;
@state f0Hold = 0.0;
@state haveF0: Int = 0;
@state markAcc = 0.0;
@state synL = 0.0;
@state synR = 0.0;
@state ratioL = 1.0;
@state ratioR = 1.0;
@state prevBarIn = 0.0;
@state barHold = 0.0;
@state in2Hold = 0.0;
@state freeLfo = new phasor();
@state srCached = 0.0;
@state ratioK = 0.01;

function init() {
	for (let i = 0; i < 1025; i++) {
		winTab[i] = 0.5 - 0.5 * cos(2.0 * pi * i / 1024.0);
	}
	for (let g = 0; g < 12; g++) {
		lStart[g] = 0;
		lAge[g] = 100000;
		lLen[g] = 0;
		rStart[g] = 0;
		rAge[g] = 100000;
		rLen[g] = 0;
	}
}

let sr = maximum(samplerate(), 1.0);
if (srCached != sr) {
	srCached = sr;
	ratioK = 1.0 - exp(-1.0 / (0.03 * sr));
}

let xL = in1;
// in2 counts as patched for 250 ms after any non-zero sample (a per-sample test
// would swap in xL for single samples at in2's zero crossings)
if (abs(in2) > 0.0000001) {
	in2Hold = 0.25 * sr;
} else if (in2Hold > 0.0) {
	in2Hold -= 1.0;
}
let xR = (in2Hold > 0.0) ? in2 : xL;
let pIn = in3;
let barIn = in4;

// --- 1. f0 conditioning: hold the last valid value, never divide by in3 ---
if (pIn > 20.0 && pIn < 1200.0) {
	f0Hold = pIn;
	haveF0 = 1;
}
let f0 = haveF0 ? f0Hold : clamp(fallback_hz, 40.0, 300.0);
let period = sr / clamp(f0, 40.0, 1200.0);

// --- 2. interval +/- bar-rate drift -> L/R ratios (30 ms glide) ---
let semis = clamp(interval, -12.0, 12.0);
// Bar phasor counts as active for 100 ms after it last moved (a constant in4 must not freeze the drift)
if (abs(barIn - prevBarIn) > 0.000001) {
	barHold = 0.1 * sr;
} else if (barHold > 0.0) {
	barHold -= 1.0;
}
prevBarIn = barIn;
let freePh = freeLfo.next(0.13);
let driftPhase = (barHold > 0.0) ? wrap(barIn, 0.0, 1.0) : freePh;
let driftSemis = 0.01 * clamp(drift_cents, 0.0, 25.0) * sin(2.0 * pi * driftPhase);
let rL = exp2((semis + driftSemis) / 12.0);
let rR = exp2((semis - driftSemis) / 12.0);
ratioL += (rL - ratioL) * ratioK;
ratioR += (rR - ratioR) * ratioK;

// --- 3. analysis marks every tuba period, synthesis grains every P/ratio ---
let smoothP = 0.01 * clamp(grain_smooth, 5.0, 60.0) * period;
let hopL = maximum(period / maximum(ratioL, 0.25), 8.0);
let hopR = maximum(period / maximum(ratioR, 0.25), 8.0);
let lfL = clamp(2.0 * maximum(period, hopL) + smoothP, 64.0, 10000.0);
let lfR = clamp(2.0 * maximum(period, hopR) + smoothP, 64.0, 10000.0);
let lIntL: Int = floor(lfL);
let lIntR: Int = floor(lfR);
gLenL = lIntL;
gLenR = lIntR;

bufL[wi] = xL;
bufR[wi] = xR;

markAcc += 1.0;
if (markAcc >= period) {
	markAcc -= period;
	markIdx = wi;
}

// A grain reads forward at unity speed, so it may start at the newest mark
// (start <= write index is all that is required).
synL += 1.0;
if (synL >= hopL) {
	synL -= hopL;
	lStart[slotL] = markIdx;
	lAge[slotL] = 0;
	lLen[slotL] = gLenL;
	slotL = (slotL + 1) % 12;
}

synR += 1.0;
if (synR >= hopR) {
	synR -= hopR;
	rStart[slotR] = markIdx;
	rAge[slotR] = 0;
	rLen[slotR] = gLenR;
	slotR = (slotR + 1) % 12;
}

// --- 4. Hann overlap-add of the active grains, normalised by the window sum ---
let accL = 0.0;
let accR = 0.0;
let wsumL = 0.0;
let wsumR = 0.0;
for (let g = 0; g < 12; g++) {
	let ag = lAge[g];
	let gl = lLen[g];
	if (ag < gl) {
		let xi: Int = floor(ag * 1024.0 / gl);
		if (xi > 1024) { xi = 1024; }
		let wv = winTab[xi];
		let idx = (lStart[g] + ag) & 16383;
		accL += wv * bufL[idx];
		wsumL += wv;
		lAge[g] = ag + 1;
	}
}
for (let g = 0; g < 12; g++) {
	let ag = rAge[g];
	let gl = rLen[g];
	if (ag < gl) {
		let xi: Int = floor(ag * 1024.0 / gl);
		if (xi > 1024) { xi = 1024; }
		let wv = winTab[xi];
		let idx = (rStart[g] + ag) & 16383;
		accR += wv * bufR[idx];
		wsumR += wv;
		rAge[g] = ag + 1;
	}
}
wi = (wi + 1) & 16383;

// --- 5. mono-safe width (0 = both streams centred, 100 = L stream left /
//        R stream right), then dry/wet ---
let gL = 0.0;
let gR = 0.0;
if (wsumL > 0.000001) { gL = accL / wsumL; }
if (wsumR > 0.000001) { gR = accR / wsumR; }
let gm = 0.5 * (gL + gR);
let gs = 0.5 * (gL - gR);
let sideGain = clamp(width / 100.0, 0.0, 1.0);
let wetL = gm + sideGain * gs;
let wetR = gm - sideGain * gs;

// The voice fades out below half a semitone, where it would only beat with dry
let voiceGate = clamp(abs(semis) / 0.5, 0.0, 1.0);
let w = clamp(mix / 100.0, 0.0, 1.0) * voiceGate;
out1 = linearinterp(w, xL, wetL);
out2 = linearinterp(w, xR, wetR);
