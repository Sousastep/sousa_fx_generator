// ============================================================================
// 06_asymmetric_delay_adfm.txt
// Unix-Style Effect 6: Delay-Line Asymmetric Adaptive FM (ASDFM1 / ASDFM2)
// Implements Lazzarini, Timoney & Lysaght (DAFx-08):
// "Asymmetric-Spectra Methods for Adaptive FM Synthesis" (Sec 2.1 & 2.2, Eq. 1-4).
// Applies adaptive phase modulation to the full stereo tuba signal via a cubic-
// interpolated variable delay line, ring-modulated by an exponentially mapped
// cosine with Palamin symmetry parameter r and modified-Bessel I0 normalization.
// ============================================================================
// Inlets:
//   in1: Left Audio Input (Full Tuba Carrier)
//   in2: Right Audio Input (Full Tuba Carrier)
//   in3: Tracked Fundamental Pitch in Hz (from fluid.pitch~; fallback ZC if <=0)
// Outlets:
//   out1: Left Asymmetric AdFM Output
//   out2: Right Asymmetric AdFM Output

@param({ min: 0.25, max: 4.0 }) cm_ratio = 1.0;
@param({ min: 0.0, max: 6.0 }) mod_index = 2.0;
@param({ min: -1.0, max: 1.0 }) symmetry_r = 0.35;
@param({ min: 0.0, max: 1.0 }) bessel_mode = 1.0;

// Delay lines (4096 samples per channel, power-of-2 wrap)
@state delayBufL = new FixedSampleArray(4096);
@state delayBufR = new FixedSampleArray(4096);
@state writeIdx: Int = 0;

// Modulator phasor
@state modPhasor = new phasor();

// Fallback ZC pitch tracker
@state lpTrack = new onepole();
@state prevSig = 0.0;
@state zcCount = 0.0;
@state zcHz = 55.0;
@state smoothF0 = 55.0;

// Rational polynomial approximation of ln(I0(x)) for Modified Bessel normalization
// Prevents exponential overload at large k * |r +/- 1/r| (Paper Sec 2.2 & 4.2)
function lnBesselI0(x: number): number {
	let ax = abs(x);
	let res = 0.0;
	if (ax < 3.75) {
		let t = ax / 3.75;
		let t2 = t * t;
		let i0 = 1.0 + t2 * (3.5156229 + t2 * (3.0899424 + t2 * (1.2067492 + t2 * (0.2659732 + t2 * (0.0360768 + t2 * 0.0045813)))));
		res = log(i0);
	} else {
		// Asymptotic expansion: ln(I0(x)) ~ |x| - 0.5*ln(2*pi*|x|)
		res = ax - 0.5 * log(2.0 * pi * ax) + 0.125 / ax;
	}
	return res;
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = in2;
let pitchIn = in3;

// --- 1. Pitch Conditioning ---
let monoLow = lpTrack.next(0.5 * (xL + xR), 180.0);
zcCount += 1.0;
if (prevSig <= 0.0 && monoLow > 0.0) {
	if (zcCount > sr / 500.0 && zcCount < sr / 22.0) {
		zcHz = sr / zcCount;
	}
	zcCount = 0.0;
}
prevSig = monoLow;

let rawF0 = (pitchIn > 20.0 && pitchIn < 1000.0) ? pitchIn : zcHz;
smoothF0 += (rawF0 - smoothF0) * (1.0 - exp(-1.0 / (0.008 * sr)));
let f0 = clamp(smoothF0, 25.0, 1000.0);

// --- 2. Write Incoming Tuba Carrier to Circular Delay Buffer ---
delayBufL[writeIdx] = xL;
delayBufR[writeIdx] = xR;

// --- 3. Compute Asymmetric AdFM Parameters (Paper Eq. 3 & 4) ---
let k = pow(clamp(mod_index / 6.0, 0.0, 1.0), 1.3) * 6.0;
// symmetry_r in [-1, 1] maps exponentially to r in [0.25, 4.0] (0 -> r=1 symmetric)
let r = pow(4.0, clamp(symmetry_r, -1.0, 1.0));
let rPlus = 0.5 * (r + 1.0 / r);
let rMinus = 0.5 * (r - 1.0 / r);

// Blend between ASDFM1 (Eq. 3: standard Bessel Jn) and ASDFM2 (Eq. 4: Modified Bessel In moving formants)
let mMode = clamp(bessel_mode, 0.0, 1.0);
let kDelay = k * linearinterp(mMode, rPlus, abs(rMinus));
let kAmp = k * linearinterp(mMode, rMinus, rPlus);
// Keep kAmp bounded for numerical stability
kAmp = clamp(kAmp, -8.0, 8.0);

// Modulator frequency fm = f0 / cm_ratio
let fm = clamp(f0 / clamp(cm_ratio, 0.25, 4.0), 8.0, 2000.0);
let phL = modPhasor.next(fm);
let phR = wrap(phL + 0.06, 0.0, 1.0); // Slight stereo modulator phase spread

// Raised, scaled, inverted sine for delay modulation (Paper Sec 2.1: pi/2 offset from cosine)
let modSinL = 0.5 * (1.0 - sin(2.0 * pi * phL));
let modSinR = 0.5 * (1.0 - sin(2.0 * pi * phR));

// Delay modulation depth in samples: adt = (kDelay / (pi * f0)) * sr * modSin + 2 samples offset
let maxDelaySamps = clamp((kDelay / (pi * f0)) * sr, 0.0, 2000.0);
let delaySampsL = 2.0 + maxDelaySamps * modSinL;
let delaySampsR = 2.0 + maxDelaySamps * modSinR;

// --- 4. Cubic Interpolated Delay Read ---
let readPosL = writeIdx - delaySampsL;
while (readPosL < 0.0) { readPosL += 4096.0; }
let idxL0: Int = floor(readPosL);
let fracL = readPosL - idxL0;
let iLm1: Int = (idxL0 - 1 + 4096) % 4096;
let iL0: Int = idxL0 % 4096;
let iL1: Int = (idxL0 + 1) % 4096;
let iL2: Int = (idxL0 + 2) % 4096;
let delL = cubicinterp(fracL, delayBufL[iLm1], delayBufL[iL0], delayBufL[iL1], delayBufL[iL2]);

let readPosR = writeIdx - delaySampsR;
while (readPosR < 0.0) { readPosR += 4096.0; }
let idxR0: Int = floor(readPosR);
let fracR = readPosR - idxR0;
let iRm1: Int = (idxR0 - 1 + 4096) % 4096;
let iR0: Int = idxR0 % 4096;
let iR1: Int = (idxR0 + 1) % 4096;
let iR2: Int = (idxR0 + 2) % 4096;
let delR = cubicinterp(fracR, delayBufR[iRm1], delayBufR[iR0], delayBufR[iR1], delayBufR[iR2]);

writeIdx = (writeIdx + 1) % 4096;

// --- 5. Exponential Cosine Ring Modulation with I0 Normalization ---
let modCosL = cos(2.0 * pi * phL);
let modCosR = cos(2.0 * pi * phR);
let lnI0 = lnBesselI0(kAmp);

let envModL = exp(kAmp * modCosL - lnI0);
let envModR = exp(kAmp * modCosR - lnI0);

out1 = delL * envModL;
out2 = delR * envModR;
