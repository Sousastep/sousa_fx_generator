// ============================================================================
// chorus_leslie_doppler_sdf.txt
// Standalone Stereo Chorus Effect: Dual-Rotor Leslie Rotating Speaker Chorus
// Synthesizes 3 DAFx Rotary Speaker Papers into one stereo effect:
//   1. Julius O. Smith, Stefania Serafin, Jonathan Abel & David Berners (DAFx-02):
//      "Doppler Simulation and the Leslie" (Sections 3.2, 3.6 & 4.1, Eq. 10-20, Fig. 2-3)
//      + Sascha Disch & Udo Zoelzer (DAFx-99, Sec 4, Eq. 12-13, Fig. 8-9):
//      Models the 180-degree opposite dual rotating horns with exact circular Doppler
//      delay modulation (~0.5 ms depth), synchronous directional radiation AM, and
//      stereo 0.7 cross-reflection cabinet mixing (DAFx-99 Fig. 9).
//   2. Jussi Pekonen, Tapani Pihlajamaki & Vesa Valimaki (DAFx-11, Paper 49):
//      "Computationally Efficient Hammond Organ Synthesis" (Sections 3.1 & 3.2,
//      Eq. 1-5, Fig. 2, Table 1):
//      Splits the signal at an 800 Hz crossover into Treble Horn and Bass Drum pathways,
//      applying time-varying Spectral Delay Filters (SDFs: 4-stage Treble with Mb = -0.75,
//      Ms = 0.20; 3-stage Bass with Mb = -0.92, Ms = 0.04) and independent motor inertia
//      with 0.1 Hz treble/bass motor mistuning!
// ============================================================================
// Inlets:
//   in1: Left Audio Input (Tuba)
//   in2: Right Audio Input (if unpatched, mono in1 is used automatically)
//   in3: Optional Speed Throw Switch or Envelope [0..1] (added to speed_chorale_tremolo).
//        Exception to the in3-pitch contract: values above 20 (a pitch in Hz) are ignored.
//   in4: unused
// Outlets:
//   out1: Left Stereo Leslie Cabinet Chorus Output
//   out2: Right Stereo Leslie Cabinet Chorus Output

@param({ min: 0.0, max: 1.0 }) speed_chorale_tremolo = 0.25;   // rotor speed
@param({ min: 0.0, max: 1.0 }) doppler_horn_depth = 0.65;       // horn Doppler depth
@param({ min: 0.0, max: 1.0 }) sdf_dispersion_depth = 0.70;     // SDF phase depth
@param({ min: 0.0, max: 1.0 }) directional_am_depth = 0.55;     // rotor AM depth
@param({ min: 0.0, max: 18.0 }) tube_amp_growl_db = 4.5;        // amp drive
@param({ min: 0.0, max: 1.0 }) cabinet_stereo_width = 0.80;     // stereo width
@param({ min: 0.0, max: 1.0 }) dry_wet_mix = 1.0;               // dry/wet

// 800 Hz Crossover Filters (DAFx-11 Sec 3.2: 2-stage cascaded one-poles per channel)
@state xLp1L = new onepole(); @state xLp2L = new onepole();
@state xLp1R = new onepole(); @state xLp2R = new onepole();

// Dual-Horn Interpolated Doppler Delay Buffers (512 samples per horn, DAFx-02 Sec 4.1)
@state hornBufL = new FixedSampleArray(512);
@state hornBufR = new FixedSampleArray(512);
@state writeIdx: Int = 0;

// Spectral Delay Filter (SDF) Allpass Chains (DAFx-11 Table 1: Treble N=4, Bass N=3)
@state sdfTrebXL = new FixedSampleArray(4); @state sdfTrebYL = new FixedSampleArray(4);
@state sdfTrebXR = new FixedSampleArray(4); @state sdfTrebYR = new FixedSampleArray(4);
@state sdfBassXL = new FixedSampleArray(3); @state sdfBassYL = new FixedSampleArray(3);
@state sdfBassXR = new FixedSampleArray(3); @state sdfBassYR = new FixedSampleArray(3);

// Independent Treble Horn & Bass Drum Motor Phasors + Inertia Smoothers (DAFx-11 Sec 3.2)
@state hornPhasor = new phasor();
@state drumPhasor = new phasor();
@state smoothHornHz = 1.8;
@state smoothDrumHz = 1.7;
@state in2Hold = 0.0;

let sr = maximum(samplerate(), 1.0);
let xL = in1;
// in2 counts as patched for 250 ms after any non-zero sample (a per-sample test
// would swap in xL for single samples at in2's zero crossings)
if (abs(in2) > 0.0000001) {
	in2Hold = 0.25 * sr;
} else if (in2Hold > 0.0) {
	in2Hold -= 1.0;
}
let xR = (in2Hold > 0.0) ? in2 : xL;
let throwIn = (in3 > 20.0) ? 0.0 : clamp(in3, 0.0, 1.0);

// --- 1. Leslie Tube Amplifier Soft Overdrive ("Growl", DAFx-02 Sec 1) ---
// Level-matched at a -12 dBFS (0.25) reference: more drive saturates peaks
// without making normal playing levels louder
let growlGain = dbtoa(clamp(tube_amp_growl_db, 0.0, 18.0));
let normGrowl = 0.25 / tanh(growlGain * 0.7 * 0.25);
let ampL = tanh(xL * growlGain * 0.7) * normGrowl;
let ampR = tanh(xR * growlGain * 0.7) * normGrowl;

// --- 2. 800 Hz Cabinet Crossover into Bass Drum & Treble Horn (DAFx-11 Sec 3.2) ---
let bassInL = xLp2L.next(xLp1L.next(ampL, 800.0), 800.0);
let bassInR = xLp2R.next(xLp1R.next(ampR, 800.0), 800.0);
let trebInL = ampL - bassInL;
let trebInR = ampR - bassInR;

// --- 3. Dual Motor Speeds with Physical Acceleration Inertia (DAFx-11 Sec 3.2) ---
// Slow (Chorale) 0.85 Hz, Fast (Tremolo) 6.2 Hz, bass drum 0.1 Hz below the horn
let speedCtrl = clamp(speed_chorale_tremolo + throwIn, 0.0, 1.0);
let targetHornHz = 0.85 + 5.35 * speedCtrl;
let targetDrumHz = targetHornHz - 0.1; // 0.1 Hz motor mistuning (Henricksen / Pekonen Sec 3.2)

// Treble horn accelerates in ~0.8s; heavy wooden bass drum takes ~2.5s!
let hornSlew = 1.0 - exp(-1.0 / (0.18 * sr));
let drumSlew = 1.0 - exp(-1.0 / (0.55 * sr));
smoothHornHz += (targetHornHz - smoothHornHz) * hornSlew;
smoothDrumHz += (targetDrumHz - smoothDrumHz) * drumSlew;

let phHorn = hornPhasor.next(smoothHornHz);
let phDrum = drumPhasor.next(smoothDrumHz);

// Quadrature & 180-deg opposite horn modulators (DAFx-99 Fig. 8-9, DAFx-02 Eq. 15-20)
let sinHornA = sin(2.0 * pi * phHorn);
let cosHornA = cos(2.0 * pi * phHorn);
let sinHornB = 0.0 - sinHornA; // Opposite horn B (180 degrees out of phase)
let cosHornB = 0.0 - cosHornA;

let sinDrumL = sin(2.0 * pi * phDrum);
let sinDrumR = cos(2.0 * pi * phDrum);

// --- 4. Treble Pathway: Dual-Horn Interpolated Doppler + 4-Stage SDF (DAFx-02 & DAFx-11) ---
hornBufL[writeIdx] = trebInL;
hornBufR[writeIdx] = trebInR;

// DAFx-99 Sec 4: Horn Doppler modulation depth is ~0.5 ms around 1.2 ms center delay
let dopDepthSamps = mstosamps(0.55 * clamp(doppler_horn_depth, 0.0, 1.0));
let centerSamps = mstosamps(1.2);

let delHornA = centerSamps + dopDepthSamps * sinHornA;
let rpHornA = writeIdx - clamp(delHornA, 2.0, 506.0);
while (rpHornA < 0.0) { rpHornA += 512.0; }
let iHornA: Int = floor(rpHornA);
let hornAL = cubicinterp(rpHornA - iHornA, hornBufL[(iHornA - 1 + 512) & 511], hornBufL[iHornA & 511], hornBufL[(iHornA + 1) & 511], hornBufL[(iHornA + 2) & 511]);

let delHornB = centerSamps + dopDepthSamps * sinHornB;
let rpHornB = writeIdx - clamp(delHornB, 2.0, 506.0);
while (rpHornB < 0.0) { rpHornB += 512.0; }
let iHornB: Int = floor(rpHornB);
let hornBR = cubicinterp(rpHornB - iHornB, hornBufR[(iHornB - 1 + 512) & 511], hornBufR[iHornB & 511], hornBufR[(iHornB + 1) & 511], hornBufR[(iHornB + 2) & 511]);
writeIdx = (writeIdx + 1) & 511;

// DAFx-11 Table 1 Treble SDF: N = 4 stages, Mb = -0.75, Ms = 0.20 * sdf_dispersion_depth
let sdfDepth = clamp(sdf_dispersion_depth, 0.0, 1.0);
let mTrebL = clamp(-0.75 + 0.20 * sdfDepth * sinHornA, -0.96, 0.96);
let mTrebR = clamp(-0.75 + 0.20 * sdfDepth * sinHornB, -0.96, 0.96);

let tSigL = hornAL;
let tSigR = hornBR;
for (let k: Int = 0; k < 4; k++) {
	let yL = mTrebL * tSigL + sdfTrebXL[k] - mTrebL * sdfTrebYL[k];
	sdfTrebXL[k] = tSigL; sdfTrebYL[k] = yL; tSigL = yL;

	let yR = mTrebR * tSigR + sdfTrebXR[k] - mTrebR * sdfTrebYR[k];
	sdfTrebXR[k] = tSigR; sdfTrebYR[k] = yR; tSigR = yR;
}

// --- 5. Bass Pathway: 3-Stage Time-Varying SDF (DAFx-11 Table 1: N = 3, Mb = -0.92, Ms = 0.04) ---
let mBassL = clamp(-0.92 + 0.04 * sdfDepth * sinDrumL, -0.97, -0.50);
let mBassR = clamp(-0.92 + 0.04 * sdfDepth * sinDrumR, -0.97, -0.50);

let bSigL = bassInL;
let bSigR = bassInR;
for (let k: Int = 0; k < 3; k++) {
	let yL = mBassL * bSigL + sdfBassXL[k] - mBassL * sdfBassYL[k];
	sdfBassXL[k] = bSigL; sdfBassYL[k] = yL; bSigL = yL;

	let yR = mBassR * bSigR + sdfBassXR[k] - mBassR * sdfBassYR[k];
	sdfBassXR[k] = bSigR; sdfBassYR[k] = yR; bSigR = yR;
}

// --- 6. Synchronous Directional Horn/Drum AM & Cabinet Cross-Mixing (DAFx-99 Fig. 8-9) ---
// Loudest when the horn mouth points at the mic, which is also its closest (minimum-delay,
// zero-Doppler) point: delay goes as +sin, so the AM goes as -sin
let amAmt = clamp(directional_am_depth, 0.0, 1.0) * 0.45;
let hornModL = tSigL * (1.0 - amAmt * sinHornA);
let hornModR = tSigR * (1.0 - amAmt * sinHornB);
let drumModL = bSigL * (1.0 + amAmt * 0.7 * sinDrumL);
let drumModR = bSigR * (1.0 + amAmt * 0.7 * sinDrumR);

// DAFx-99 Fig. 9: cross-mixing of opposite horns inside the wooden cabinet
// (cross gain 1.0 = mono at width 0, 0.35 at width 1; Fig. 9's 0.7 is width ~0.46).
// cabGain keeps the level of coherent (bass) content constant across the width range.
let crossMix = 1.0 - 0.65 * clamp(cabinet_stereo_width, 0.0, 1.0);
let cabGain = 1.11 / (1.0 + crossMix);
let cabL = (hornModL + crossMix * hornModR) * cabGain + (drumModL + crossMix * drumModR) * cabGain;
let cabR = (hornModR + crossMix * hornModL) * cabGain + (drumModR + crossMix * drumModL) * cabGain;

let mix = clamp(dry_wet_mix, 0.0, 1.0);
out1 = linearinterp(mix, xL, cabL);
out2 = linearinterp(mix, xR, cabR);
