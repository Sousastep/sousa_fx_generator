// ============================================================================
// chorus_erb_allpass_reson_deco.txt
// Standalone Stereo Chorus Effect: Perceptually Informed ERB Prime-Delay Allpass
// & Decaying-Resonator Spatial Chorus (Zero Magnitude Coloration!)
// Synthesizes 2 DAFx Perceptual Spatial Decorrelation Papers:
//   1. Elliot Kermit-Canfield & Jonathan Abel (DAFx-16, Paper 30):
//      "Signal Decorrelation using Perceptually Informed Allpass Filters"
//      (Sections 3.1, 3.2 & 3.3, Eq. 1-6, Fig. 1-3):
//      Cascades conjugate-symmetric biquad allpass sections with ERB-spaced pole
//      frequencies, pole radii bounded by the 30 ms perceptual group-delay limit,
//      and mutually prime multi-sample delays z^(-P_k) so phase wraps rapidly
//      around the unit circle with strictly flat 0 dB magnitude response!
//   2. Jon Fagerstrom, Nils Meyer-Kahlen, Sebastian J. Schlecht & Vesa Valimaki
//      (DAFx-25, Paper 80): "Perceptual Decorrelator Based on Resonators"
//      (Sections 3 & 4, Table 1, Eq. 4-14, Fig. 2-4):
//      Borrows the idea of the listening-test-derived ERB profile (Table 1: most group
//      delay in the low/mid bands, much less at high frequencies so transients stay tight).
//      Measured on this cascade at 48 kHz, defaults: mean group delay ~9 ms at 60-200 Hz,
//      ~3 ms at 200-800 Hz, ~1-1.5 ms above. The z^-P structure replicates every stage's
//      poles across the whole band, so the HF taper is done with a crossover that routes
//      the top band around the cascade (hf_transient_tightness).
// ============================================================================
// Inlets:
//   in1: Left Audio Input (Tuba)
//   in2: Right Audio Input (if unpatched, mono in1 is used automatically)
//   in3: Optional Envelope Modulator [0..1] (modulates allpass pole angles on attacks).
//        Exception to the in3-pitch contract: values above 20 (a pitch in Hz) are ignored.
//   in4: Optional 1-Bar Phasor [0..1] (exception to the in4-in-Hz contract): while it
//        moves, drift LFO 1 runs at 1 cycle per bar
// Outlets:
//   out1: Left ERB Perceptual Allpass Chorus Output
//   out2: Right ERB Perceptual Allpass Chorus Output

@param({ min: 0.1, max: 1.0 }) erb_group_delay_scale = 0.70;    // prime delay scale
@param({ min: 0.0, max: 1.0 }) pole_radius_resonance = 0.72;    // pole radius
@param({ min: 0.0, max: 1.0 }) pole_drift_depth = 0.45;         // drift depth
@param({ min: 0.08, max: 3.5 }) drift_rate_hz = 0.42;           // drift rate
@param({ min: 0.0, max: 1.0 }) hf_transient_tightness = 0.75;   // keep highs undelayed
@param({ min: 0.0, max: 1.0 }) dry_wet_chorus_mix = 0.50;       // dry/wet

// 6-Stage Prime-Delay Biquad Allpass Cascade Buffers per Channel (DAFx-16 Sec 3.3)
// Prime delay lengths (in samples at 48 kHz, erb_group_delay_scale = 1):
//   Left:  P_L = [19, 13, 11, 7, 5, 3]
//   Right: P_R = [23, 17, 7, 11, 3, 5]
// Each stage uses a 128-sample circular buffer to store x[n - P], x[n - 2P], y[n - P], y[n - 2P]
@state apXBufL = new FixedSampleArray(768); @state apYBufL = new FixedSampleArray(768);
@state apXBufR = new FixedSampleArray(768); @state apYBufR = new FixedSampleArray(768);
@state wIdx: Int = 0;

@state lfoPh1 = new phasor();
@state lfoPh2 = new phasor();
@state prevBarIn = 0.0;
@state barHold = 0.0;
@state in2Hold = 0.0;

// HF bypass crossover (complementary: lo + hi = input)
@state xoLp1L = new onepole(); @state xoLp2L = new onepole();
@state xoLp1R = new onepole(); @state xoLp2R = new onepole();

// Nearest prime in [2, 61], so scaled delays stay mutually prime (DAFx-16 Sec 3.3)
function snapPrime(n: number): number {
	if (n < 2.5) return 2.0;
	if (n < 4.0) return 3.0;
	if (n < 6.0) return 5.0;
	if (n < 9.0) return 7.0;
	if (n < 12.0) return 11.0;
	if (n < 15.0) return 13.0;
	if (n < 18.0) return 17.0;
	if (n < 21.0) return 19.0;
	if (n < 26.0) return 23.0;
	if (n < 30.0) return 29.0;
	if (n < 34.0) return 31.0;
	if (n < 39.0) return 37.0;
	if (n < 42.0) return 41.0;
	if (n < 45.0) return 43.0;
	if (n < 50.0) return 47.0;
	if (n < 56.0) return 53.0;
	if (n < 60.0) return 59.0;
	return 61.0;
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
// in2 counts as patched for 250 ms after any non-zero sample (a per-sample test
// would swap in xL for single samples at in2's zero crossings)
if (abs(in2) > 0.0000001) {
	in2Hold = 0.25 * sr;
} else if (in2Hold > 0.0) {
	in2Hold -= 1.0;
}
let xR = (in2Hold > 0.0) ? in2 : xL;
let envMod = (in3 > 20.0) ? 0.0 : clamp(in3, 0.0, 1.0);
let barIn = in4;

let rate = clamp(drift_rate_hz, 0.08, 3.5);
// Bar phasor counts as active for 100 ms after it last moved (a constant in4 must not freeze the LFO)
if (abs(barIn - prevBarIn) > 0.000001) {
	barHold = 0.1 * sr;
} else if (barHold > 0.0) {
	barHold -= 1.0;
}
prevBarIn = barIn;

let freePh1 = lfoPh1.next(rate);
let ph1 = (barHold > 0.0) ? wrap(barIn, 0.0, 1.0) : freePh1;
let ph2 = lfoPh2.next(rate * 1.23);

let driftAmt = clamp(pole_drift_depth, 0.0, 1.0) * 0.28;
let gdScale = clamp(erb_group_delay_scale, 0.1, 1.0);
let baseRad = 0.52 + 0.40 * clamp(pole_radius_resonance, 0.0, 1.0);
let hfTight = clamp(hf_transient_tightness, 0.0, 1.0);

// hf_transient_tightness: crossover from 12 kHz (tightness 0) down to 1.5 kHz (tightness 1);
// the band above it is blended around the cascade by the same amount, so at 0 the whole
// signal is allpassed and at 1 everything above ~1.5 kHz stays undelayed
let xoHz = clamp(1500.0 * exp2(3.0 * (1.0 - hfTight)), 200.0, sr * 0.45);
let hiL = xL - xoLp2L.next(xoLp1L.next(xL, xoHz), xoHz);
let hiR = xR - xoLp2R.next(xoLp1R.next(xR, xoHz), xoHz);
let sigL = xL - hfTight * hiL;
let sigR = xR - hfTight * hiR;
let srScale = sr / 48000.0;

// --- Evaluate 6 ERB-Spaced Prime-Delay Biquad Allpass Stages (DAFx-16 Eq. 2-4 & DAFx-25 Table 1) ---
// Pole-angle frequencies (before the z^-P substitution, which maps each to (f + k*sr) / P):
// Stage 0: 120 Hz  -> prime delays 19 / 23
// Stage 1: 260 Hz  -> prime delays 13 / 17
// Stage 2: 520 Hz  -> prime delays 11 / 7
// Stage 3: 1050 Hz -> prime delays 7 / 11
// Stage 4: 2200 Hz -> prime delays 5 / 3
// Stage 5: 4800 Hz -> prime delays 3 / 5 (stages 3-5 radii also reduced by hf_transient_tightness)
for (let s: Int = 0; s < 6; s++) {
	let baseFreq = 120.0;
	let pBaseL: Int = 19;
	let pBaseR: Int = 23;
	let rWeight = 1.0;

	if (s == 1) {
		baseFreq = 260.0;  pBaseL = 13; pBaseR = 17; rWeight = 1.0;
	} else if (s == 2) {
		baseFreq = 520.0;  pBaseL = 11; pBaseR = 7;  rWeight = 0.98;
	} else if (s == 3) {
		baseFreq = 1050.0; pBaseL = 7;  pBaseR = 11; rWeight = 0.94 - 0.10 * hfTight;
	} else if (s == 4) {
		baseFreq = 2200.0; pBaseL = 5;  pBaseR = 3;  rWeight = 0.88 - 0.22 * hfTight;
	} else if (s == 5) {
		baseFreq = 4800.0; pBaseL = 3;  pBaseR = 5;  rWeight = 0.80 - 0.35 * hfTight;
	}

	// Scale prime delays by erb_group_delay_scale and sample rate, then snap back to a prime
	// in [2 .. 61] samples so 2*P <= 122 < 128
	let pL: Int = snapPrime(pBaseL * gdScale * srScale);
	let pR: Int = snapPrime(pBaseR * gdScale * srScale);

	// Modulate pole angles in counter-quadrature between Left and Right
	let lfoL = sin(2.0 * pi * ph1 + s * 1.0472) * driftAmt + 0.2 * envMod;
	let lfoR = cos(2.0 * pi * ph2 + s * 1.0472) * driftAmt - 0.2 * envMod;

	let fL = clamp(baseFreq * pow(2.0, lfoL), 40.0, sr * 0.44);
	let fR = clamp(baseFreq * pow(2.0, -lfoR * 1.07), 40.0, sr * 0.44);

	let rL = clamp(baseRad * rWeight, 0.35, 0.94);
	let rR = clamp(baseRad * rWeight * 0.98, 0.35, 0.94);

	// DAFx-16 Eq. 2 & 4: Conjugate-symmetric biquad allpass difference equation with z^(-P):
	// y[n] = r^2 * x[n] - 2*r*cos(w) * x[n - P] + x[n - 2P] + 2*r*cos(w) * y[n - P] - r^2 * y[n - 2P]
	let c1L = -2.0 * rL * cos(2.0 * pi * fL / sr);
	let c2L = rL * rL;
	let c1R = -2.0 * rR * cos(2.0 * pi * fR / sr);
	let c2R = rR * rR;

	let baseOffset: Int = s * 128;
	let idx0: Int   = baseOffset + (wIdx & 127);
	let idx1L: Int  = baseOffset + ((wIdx - pL + 128) & 127);
	let idx2L: Int  = baseOffset + ((wIdx - 2 * pL + 128) & 127);
	let idx1R: Int  = baseOffset + ((wIdx - pR + 128) & 127);
	let idx2R: Int  = baseOffset + ((wIdx - 2 * pR + 128) & 127);

	let yL = c2L * sigL + c1L * apXBufL[idx1L] + apXBufL[idx2L] - c1L * apYBufL[idx1L] - c2L * apYBufL[idx2L];
	apXBufL[idx0] = sigL;
	apYBufL[idx0] = yL;
	sigL = yL;

	let yR = c2R * sigR + c1R * apXBufR[idx1R] + apXBufR[idx2R] - c1R * apYBufR[idx1R] - c2R * apYBufR[idx2R];
	apXBufR[idx0] = sigR;
	apYBufR[idx0] = yR;
	sigR = yR;
}

wIdx = (wIdx + 1) & 127;

// Add back the bypassed top band
sigL = sigL + hfTight * hiL;
sigR = sigR + hfTight * hiR;

// Mix dry + decorrelated allpass outputs:
// At dry_wet_chorus_mix = 1.0 and hf_transient_tightness = 0, the wet path is a pure allpass
// (flat magnitude); with tightness > 0 the crossover region picks up some coloration.
// At dry_wet_chorus_mix = 0.5, sums dry + time-varying prime-delay allpass for dense multi-notch chorus!
let mix = clamp(dry_wet_chorus_mix, 0.0, 1.0);
out1 = cos(mix * 0.5 * pi) * xL + sin(mix * 0.5 * pi) * sigL;
out2 = cos(mix * 0.5 * pi) * xR + sin(mix * 0.5 * pi) * sigR;
