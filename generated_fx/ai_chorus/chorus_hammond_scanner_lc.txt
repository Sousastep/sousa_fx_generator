// ============================================================================
// chorus_hammond_scanner_lc.txt
// Standalone Stereo Chorus Effect: Hammond B-3 18-Stage LC Ladder & 9-Tap
// "There-and-Back-Again" Capacitive Scanner Vibrato/Chorus (V1-V3 / C1-C3)
// Implements Kurt James Werner, W. Ross Dunkel & Francois G. Germain (DAFx-16, Paper 54):
// "A Computational Model of the Hammond Organ Vibrato/Chorus using Wave Digital Filters"
// (Sections 2.1, 2.2, 3 & Table 1-2, Fig. 1-3).
// Models the complete Hammond Organ Vibrato/Chorus line box and electromechanical scanner:
//   1. 18 Dispersive Lowpass-Allpass Ladder Stages (v_1 .. v_19) with the progressive
//      6-stage input voltage-divider attenuation (R1+..R6+, R1-..R6-, Table 1) and
//      inductor-capacitor phase dispersion + passband ripple.
//   2. Exact Table 2 Tap Wiring Matrix for Depth Settings V1/C1, V2/C2, and V3/C3
//      connecting ladder nodes v_1..v_19 to the 9 scanner terminals t_1..t_9:
//        - V1/C1: [v1, v2, v3, v4, v5, v6, v7, v8, v9]
//        - V2/C2: [v1, v2, v3, v5, v7, v9, v11, v12, v13]
//        - V3/C3: [v1, v2, v4, v7, v10, v13, v16, v18, v19]
//   3. 16-Stack Capacitive Rotor Scanner (Section 3, Fig. 3): Scans linearly "there-and-
//      back-again" from t_1 -> t_9 -> t_1 using overlapping trapezoidal plate capacitances
//      v_out = eta * v_alpha + (1 - eta) * v_beta, with stereo rotor phase offset!
// ============================================================================
// Inlets:
//   in1: Left Audio Input (Tuba)
//   in2: Right Audio Input (if unpatched, mono in1 is used automatically)
//   in3: Optional Envelope Modulator [0..1] (modulates scanner depth V1->V3 dynamically).
//        Exception to the in3-pitch contract: values above 20 (a pitch in Hz) are ignored.
//   in4: Optional 1-Bar Phasor [0..1] (exception to the in4-in-Hz contract): while it
//        moves, the scanner makes 4 there-and-back sweeps per bar
// Outlets:
//   out1: Left Hammond Scanner Chorus/Vibrato Output
//   out2: Right Hammond Scanner Chorus/Vibrato Output

@param({ min: 1.0, max: 3.0 }) scanner_depth_c1_c3 = 2.5;      // depth V1/C1 .. V3/C3
@param({ min: 0.5, max: 12.0 }) scanner_rate_hz = 6.87;         // scanner rate
@param({ min: 1800.0, max: 12000.0 }) lc_cutoff_hz = 4800.0;    // line box cutoff
@param({ min: 0.0, max: 0.75 }) lc_passband_ripple = 0.35;      // ladder ripple
@param({ min: 0.0, max: 1.0 }) stereo_rotor_spread = 0.50;      // L/R rotor offset
@param({ min: 0.0, max: 1.0 }) chorus_to_vibrato = 0.0;         // chorus -> vibrato

// 18-Stage Dispersive LC Ladder State Arrays (Left & Right channels, nodes v_1..v_19)
@state ladApXL = new FixedSampleArray(18); @state ladApYL = new FixedSampleArray(18);
@state ladLpL  = new FixedSampleArray(18);
@state nodeVL  = new FixedSampleArray(19);

@state ladApXR = new FixedSampleArray(18); @state ladApYR = new FixedSampleArray(18);
@state ladLpR  = new FixedSampleArray(18);
@state nodeVR  = new FixedSampleArray(19);

// Scanner synchronous motor phasor (commonly cited B-3 scanner vibrato rate ~6.87 Hz)
@state motorPhasor = new phasor();
@state prevBarIn = 0.0;
@state barHold = 0.0;
@state in2Hold = 0.0;

// Evaluate Table 2 Tap Selection & Fig. 3 "There-and-Back-Again" Capacitive Scanner

let sr = maximum(samplerate(), 1.0);
let xL = in1;
// in2 counts as patched for 250 ms after any non-zero sample (a per-sample test
// would swap in xL for single samples at in2's zero crossings)
if (abs(in2) > 0.0000001) {
	in2Hold = 0.25 * sr;
} else if (in2Hold > 0.0) {
	in2Hold -= 1.0;
}
let xR = (in2Hold > 0.0) ? in2 : xL;
let envMod = (in3 > 20.0) ? 0.0 : clamp(in3, 0.0, 1.0);
let barIn = in4;

// --- 1. Propagate Signal Through 18-Stage Dispersive LC Line Box (Sec 2.1, Table 1) ---
let fc = clamp(lc_cutoff_hz, 1800.0, 12000.0);
let ripple = clamp(lc_passband_ripple, 0.0, 0.75);

// Dispersive allpass coefficient + lowpass cutoff per LC section (Sec 5.1: the real ladder
// delays ~0.85 ms in total with its passband edge at ~7075 Hz; both scale with lc_cutoff_hz).
// First-order allpass (g + z^-1)/(1 + g z^-1) has DC group delay (1 - g)/(1 + g) samples.
let stageDelay = 0.00085 * (7075.0 / fc) * sr / 18.0;
let apG = clamp((1.0 - stageDelay) / (1.0 + stageDelay), -0.95, 0.95);
// Per-stage lowpass at 5.1 x fc puts the 18-stage cascade's -3 dB point near fc,
// so every tap stays flat in the passband and deeper taps only roll off faster above it
let lpAlpha = clamp(1.0 - exp(-2.0 * pi * fc * 5.1 / sr), 0.05, 0.999);

nodeVL[0] = xL;
nodeVR[0] = xR;
let curL = xL;
let curR = xR;

for (let k: Int = 0; k < 18; k++) {
	// Backward-wave reflection coupling from next node (non-unidirectional LC ladder ripple, Sec 2)
	let nextNodeL = (k < 17) ? nodeVL[k + 1] : 0.0;
	let nextNodeR = (k < 17) ? nodeVR[k + 1] : 0.0;
	let inStageL = curL + ripple * 0.06 * (nextNodeL - curL);
	let inStageR = curR + ripple * 0.06 * (nextNodeR - curR);

	// Dispersive phase delay stage
	let apOutL = apG * inStageL + ladApXL[k] - apG * ladApYL[k];
	ladApXL[k] = inStageL; ladApYL[k] = apOutL;

	let apOutR = apG * inStageR + ladApXR[k] - apG * ladApYR[k];
	ladApXR[k] = inStageR; ladApYR[k] = apOutR;

	// Inductor-capacitor lowpass roll-off per stage
	ladLpL[k] += (apOutL - ladLpL[k]) * lpAlpha;
	ladLpR[k] += (apOutR - ladLpR[k]) * lpAlpha;

	// First 6 stages (x in [1..6]) include subtle Rk+/Rk- voltage divider attenuation (Table 1)
	let atten = (k < 6) ? 0.988 : 0.996;
	curL = ladLpL[k] * atten;
	curR = ladLpR[k] * atten;

	nodeVL[k + 1] = curL;
	nodeVR[k + 1] = curR;
}

// --- 2. Electromechanical Capacitive Rotor Scanner (Sec 2.2 & 3, Fig. 3, Table 2) ---
let rate = clamp(scanner_rate_hz, 0.5, 12.0);
// Bar phasor counts as active for 100 ms after it last moved (a constant in4 must not freeze the scanner)
if (abs(barIn - prevBarIn) > 0.000001) {
	barHold = 0.1 * sr;
} else if (barHold > 0.0) {
	barHold -= 1.0;
}
prevBarIn = barIn;

let freePh = motorPhasor.next(rate);
let phL = (barHold > 0.0) ? wrap(barIn * 4.0, 0.0, 1.0) : freePh;
let spread = clamp(stereo_rotor_spread, 0.0, 1.0) * 0.5;
let phR = wrap(phL + spread, 0.0, 1.0);

let effDepth = clamp(scanner_depth_c1_c3 + envMod * 1.2, 1.0, 3.0);
// 16-segment "there-and-back-again" scan across terminals t_1 (idx 0) .. t_9 (idx 8) on nodeVL
let triScanL = triangle(wrap(phL, 0.0, 1.0), 0.5) * 8.0; // in [0.0 .. 8.0]
let termIdxL: Int = clamp(floor(triScanL), 0, 7);
let etaL = triScanL - termIdxL; // Capacitive voltage divider overlap C_alpha / (C_alpha + C_beta)
let nextTL: Int = termIdxL + 1;
let v1Tap0L: Int = termIdxL;
let v1Tap1L: Int = termIdxL + 1;
let v2Tap0L: Int = (termIdxL <= 2) ? termIdxL : ((termIdxL <= 6) ? (2 + (termIdxL - 2) * 2) : (termIdxL + 4));
let v2Tap1L: Int = (nextTL <= 2) ? nextTL : ((nextTL <= 6) ? (2 + (nextTL - 2) * 2) : (nextTL + 4));
let v3Tap0L: Int = (termIdxL <= 1) ? termIdxL : ((termIdxL == 2) ? 3 : ((termIdxL <= 6) ? (3 + (termIdxL - 2) * 3) : (termIdxL + 10)));
let v3Tap1L: Int = (nextTL <= 1) ? nextTL : ((nextTL == 2) ? 3 : ((nextTL <= 6) ? (3 + (nextTL - 2) * 3) : (nextTL + 10)));
let outV1L = (1.0 - etaL) * nodeVL[v1Tap0L] + etaL * nodeVL[v1Tap1L];
let outV2L = (1.0 - etaL) * nodeVL[v2Tap0L] + etaL * nodeVL[v2Tap1L];
let outV3L = (1.0 - etaL) * nodeVL[v3Tap0L] + etaL * nodeVL[v3Tap1L];
let dScanL = clamp(effDepth, 1.0, 3.0);
let scanL = (dScanL <= 2.0) ? linearinterp(dScanL - 1.0, outV1L, outV2L) : linearinterp(dScanL - 2.0, outV2L, outV3L);

let triScanR = triangle(wrap(phR, 0.0, 1.0), 0.5) * 8.0;
let termIdxR: Int = clamp(floor(triScanR), 0, 7);
let etaR = triScanR - termIdxR;
let nextTR: Int = termIdxR + 1;
let v1Tap0R: Int = termIdxR;
let v1Tap1R: Int = termIdxR + 1;
let v2Tap0R: Int = (termIdxR <= 2) ? termIdxR : ((termIdxR <= 6) ? (2 + (termIdxR - 2) * 2) : (termIdxR + 4));
let v2Tap1R: Int = (nextTR <= 2) ? nextTR : ((nextTR <= 6) ? (2 + (nextTR - 2) * 2) : (nextTR + 4));
let v3Tap0R: Int = (termIdxR <= 1) ? termIdxR : ((termIdxR == 2) ? 3 : ((termIdxR <= 6) ? (3 + (termIdxR - 2) * 3) : (termIdxR + 10)));
let v3Tap1R: Int = (nextTR <= 1) ? nextTR : ((nextTR == 2) ? 3 : ((nextTR <= 6) ? (3 + (nextTR - 2) * 3) : (nextTR + 10)));
let outV1R = (1.0 - etaR) * nodeVR[v1Tap0R] + etaR * nodeVR[v1Tap1R];
let outV2R = (1.0 - etaR) * nodeVR[v2Tap0R] + etaR * nodeVR[v2Tap1R];
let outV3R = (1.0 - etaR) * nodeVR[v3Tap0R] + etaR * nodeVR[v3Tap1R];
let dScanR = clamp(effDepth, 1.0, 3.0);
let scanR = (dScanR <= 2.0) ? linearinterp(dScanR - 1.0, outV1R, outV2R) : linearinterp(dScanR - 2.0, outV2R, outV3R);

// --- 3. Chorus (C1/C2/C3) vs. Vibrato (V1/V2/V3) Switch (Fig. 1, Sec 2.1) ---
// In Chorus mode (switch open), dry signal passes through Rc = 22 kOhm and mixes with scanner
let vibBlend = clamp(chorus_to_vibrato, 0.0, 1.0);
let chorusL = 0.58 * xL + 0.58 * scanL;
let chorusR = 0.58 * xR + 0.58 * scanR;

out1 = linearinterp(vibBlend, chorusL, scanL);
out2 = linearinterp(vibBlend, chorusR, scanR);
