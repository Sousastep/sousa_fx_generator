// ============================================================================
// comp_crest_flux_autoballistic.txt
// Stereo Crest-Factor & Spectral-Flux Adaptive Auto-Ballistic FET/VCA Compressor
// Based on Maddams, Finn & Reiss (DAFx-2012, "An Autonomous Method for
// Multi-Track Dynamic Range Compression", Sec. 4.2, Fig. 2A/2B) and
// Hilsamer & Herzog (DAFx-2014, Sec. 2-3, Fig. 1)
// Inlets:
//   in1 / in2: stereo audio (if in2 stays silent for 250 ms, in1 feeds both sides)
//   in3 / in4: not used
// Outlets:
//   out1 / out2: stereo, dry/wet by mix
// ============================================================================

// Threshold
@param({ min: -40.0, max: 0.0 }) thresh_db = -20.0;
// Ratio
@param({ min: 1.5, max: 30.0 }) ratio = 8.0;
// Variable knee mode
@param({ min: 0.0, max: 1.0 }) var_knee_mode = 0.65;
// Base attack
@param({ min: 0.1, max: 50.0 }) base_attack_ms = 8.0;
// Base release
@param({ min: 20.0, max: 800.0 }) base_release_ms = 180.0;
// Crest / flux adaptation
@param({ min: 0.0, max: 1.0 }) crest_flux_adapt = 0.75;
// FET saturation
@param({ min: 0.0, max: 1.0 }) fet_slam_sat = 0.40;
// Dry/wet
@param({ min: 0.0, max: 1.0 }) mix = 1.0;

// Decoupled Peak and RMS detectors for instantaneous Crest Factor calculation
@state peak_env = 0.0001;
@state rms_sq_env = 0.0001;

// 2-band split for real-time Spectral Flux onset detection (Maddams Sec. 4.2)
@state flux_lp = 0.0;
@state lo_fast = 0.0;
@state lo_slow = 0.0;
@state hi_fast = 0.0;
@state hi_slow = 0.0;
@state spec_flux_env = 0.0;

// Smoothed gain-reduction state in dB
@state gr_smooth_db = 0.0;

@state in2Hold = 0.0;

// Sample-rate-only coefficients, recomputed when the rate changes
@state last_sr = 0.0;
@state a_pk_att = 0.0;
@state a_pk_rel = 0.0;
@state a_rms = 0.0;
@state g_flx = 0.0;
@state a_on_fast = 0.0;
@state a_on_slow = 0.0;
@state a_flx_att = 0.0;
@state a_flx_rel = 0.0;

// Helper: Soft-knee / Variable-knee static gain computer (Maddams Sec. 4.2.1 Fig. 2A)
// In "Threshold Mode" (var_knee_mode -> 1), knee width equals |thresh_db| and ratio -> 40:1
function adaptive_knee_gr_db(
    x_db: number,
    T: number,
    inv_R: number,
    vk: number
): number {
    // Fixed 3 dB knee in Ratio mode vs |T| knee width in Threshold mode (DAFx-2012 Sec. 4.2.1)
    let knee_w = (1.0 - vk) * 3.0 + vk * max(abs(T), 3.0);
    let diff = x_db - T;
    let slope = inv_R - 1.0;
    if (2.0 * diff < -knee_w) {
        return 0.0;
    } else if (2.0 * abs(diff) <= knee_w) {
        let u = diff + 0.5 * knee_w;
        return slope * (u * u) / max(2.0 * knee_w, 0.001);
    } else {
        return slope * diff;
    }
}

let sr_safe = max(samplerate(), 8000.0);
if (sr_safe != last_sr) {
    last_sr = sr_safe;
    // Fast peak tracker (1 ms attack, 25 ms decay) & RMS tracker (45 ms window)
    a_pk_att = 1.0 - exp(-1.0 / (0.001 * sr_safe));
    a_pk_rel = 1.0 - exp(-1.0 / (0.025 * sr_safe));
    a_rms = 1.0 - exp(-1.0 / (0.045 * sr_safe));
    g_flx = 1.0 - exp(-2.0 * PI * 1400.0 / sr_safe);
    // Onset detector: 4 ms band envelope against a 60 ms band envelope
    a_on_fast = 1.0 - exp(-1.0 / (0.004 * sr_safe));
    a_on_slow = 1.0 - exp(-1.0 / (0.060 * sr_safe));
    a_flx_att = 1.0 - exp(-1.0 / (0.002 * sr_safe));
    a_flx_rel = 1.0 - exp(-1.0 / (0.050 * sr_safe));
}

let dryL = in1;
// in2 counts as patched for 250 ms after any non-zero sample (a per-sample test
// would swap in dryL for single samples at in2's zero crossings)
if (abs(in2) > 0.0000001) {
    in2Hold = 0.25 * sr_safe;
} else if (in2Hold > 0.0) {
    in2Hold -= 1.0;
}
let dryR = (in2Hold > 0.0) ? in2 : dryL;

// Stereo-linked instantaneous magnitude and squared energy
let x_abs = max(abs(dryL), abs(dryR));
let x_sq = 0.5 * (dryL * dryL + dryR * dryR);

let a_pk = (x_abs > peak_env) ? a_pk_att : a_pk_rel;
peak_env = peak_env + a_pk * (x_abs - peak_env);

rms_sq_env = a_rms * x_sq + (1.0 - a_rms) * rms_sq_env;
let rms_val = sqrt(max(rms_sq_env, 1e-12));

// Crest Factor in dB = 20*log10(peak / rms); a steady sine reads ~3 dB
let crest_db = clamp(20.0 * log10(max(peak_env, 1e-6) / max(rms_val, 1e-6)), 0.0, 24.0);
let crest_norm = clamp((crest_db - 3.0) / 15.0, 0.0, 1.0);

// Real-time 2-band Spectral Flux detector: positive rise of each band's short
// envelope over its long envelope. Relative, so it is level independent and
// ~0 on a sustained note (a raw sample-to-sample |x| difference would read
// every rising half cycle of a steady tone as an onset).
let mono = 0.5 * (dryL + dryR);
flux_lp = flux_lp + g_flx * (mono - flux_lp);
let hi_sig = mono - flux_lp;

let lo_e = abs(flux_lp);
let hi_e = abs(hi_sig);
lo_fast = lo_fast + a_on_fast * (lo_e - lo_fast);
lo_slow = lo_slow + a_on_slow * (lo_e - lo_slow);
hi_fast = hi_fast + a_on_fast * (hi_e - hi_fast);
hi_slow = hi_slow + a_on_slow * (hi_e - hi_slow);
// 1e-4 (-80 dB) floor keeps silence and noise from reading as onsets
let d_flux = max(0.0, lo_fast / (lo_slow + 0.0001) - 1.0) + 1.8 * max(0.0, hi_fast / (hi_slow + 0.0001) - 1.0);

let a_flx = (d_flux > spec_flux_env) ? a_flx_att : a_flx_rel;
spec_flux_env = spec_flux_env + a_flx * (d_flux - spec_flux_env);
let flux_norm = clamp(spec_flux_env * 0.5, 0.0, 1.0);

// Combined transient activity index [0..1]
let adapt = clamp(crest_flux_adapt, 0.0, 1.0);
let activity = adapt * clamp(0.55 * crest_norm + 0.45 * flux_norm, 0.0, 1.0);

// Modulate attack and release time constants inversely with Crest Factor & Spectral Flux
// High crest / high spectral flux -> attack down to 12% of base (40 us floor) & fast recovery
let t_att_ms = clamp(base_attack_ms, 0.1, 50.0) * (1.0 - 0.88 * activity);
let t_rel_ms = clamp(base_release_ms, 20.0, 800.0) * (1.0 - 0.78 * activity);

let alpha_att = 1.0 - exp(-1.0 / (max(t_att_ms * 0.001, 0.00004) * sr_safe));
let alpha_rel = 1.0 - exp(-1.0 / (max(t_rel_ms * 0.001, 0.008) * sr_safe));

let T = clamp(thresh_db, -40.0, 0.0);
let R = clamp(ratio, 1.5, 30.0);
let vk = clamp(var_knee_mode, 0.0, 1.0);
// Morph the slope (1/R), not R: a linear R morph toward 40:1 made the ratio
// knob nearly inert for any var_knee_mode above ~0.3
let inv_R = (1.0 - vk) / R + vk / 40.0;
let x_level_db = atodb(max(peak_env, 1e-6));

let raw_gr_db = adaptive_knee_gr_db(x_level_db, T, inv_R, vk);
let target_atten_db = 0.0 - raw_gr_db;

if (target_atten_db > gr_smooth_db) {
    gr_smooth_db = gr_smooth_db + alpha_att * (target_atten_db - gr_smooth_db);
} else {
    gr_smooth_db = gr_smooth_db + alpha_rel * (target_atten_db - gr_smooth_db);
}

// Automatic makeup: about half the reduction a 0 dBFS peak would get at the
// effective (morphed) ratio
let makeup_db = -0.52 * T * (1.0 - inv_R);
let lin_gain = dbtoa(-gr_smooth_db + makeup_db);

let uL = dryL * lin_gain;
let uR = dryR * lin_gain;

// 1176-style FET "All-Buttons-In" slam: tanh(u*d)/d has unity small-signal gain
// (no level jump as fet_slam_sat leaves 0), and d rises with gain reduction so
// the clip point drops from 0 dBFS (no GR) to about -12 dBFS at 20 dB GR
let sat = clamp(fet_slam_sat, 0.0, 1.0);
let drv_fet = 1.0 + 0.15 * gr_smooth_db;
let wetL = uL + sat * (tanh(uL * drv_fet) / drv_fet - uL);
let wetR = uR + sat * (tanh(uR * drv_fet) / drv_fet - uR);

let mx = clamp(mix, 0.0, 1.0);
out1 = (1.0 - mx) * dryL + mx * wetL;
out2 = (1.0 - mx) * dryR + mx * wetR;
