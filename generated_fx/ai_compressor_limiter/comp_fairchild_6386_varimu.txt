// ============================================================================
// comp_fairchild_6386_varimu.txt
// Stereo Fairchild 670 / 6386 Remote-Cutoff Triode Variable-Mu Feedback Limiter
// Based on Raffensperger (DAFx-2012, "Toward a Wave Digital Filter Model of
// the Fairchild 670 Limiter", Eqs. 1, 10-14, Fig. 8, Table 3)
// Inlets:
//   in1 / in2: stereo audio (if in2 stays silent for 250 ms, in1 feeds both sides)
//   in3 / in4: not used
// Outlets:
//   out1 / out2: stereo, dry/wet by mix
// ============================================================================

@param({ min: -24.0, max: 36.0 }) input_gain_db = 0.0;
@param({ min: 0.0, max: 24.0 }) drive_db = 8.0;
@param({ min: 0.0, max: 1.0 }) ac_thresh = 0.55;
@param({ min: 0.0, max: 1.0 }) dc_knee = 0.25;
@param({ min: 1.0, max: 6.0 }) tc_mode = 5.0;
@param({ min: 0.0, max: 1.0 }) tube_harmonics = 0.5;
@param({ min: 0.0, max: 1.0 }) lat_vert_mode = 0.35;
@param({ min: 0.0, max: 1.0 }) mix = 1.0;
@param({ min: -24.0, max: 36.0 }) output_gain_db = 0.0;

// Two complete limiter pairs: Left/Right and Lateral/Vertical (Mid/Side).
// lat_vert_mode crossfades their outputs. (Blending the L/R and M/S matrices
// themselves cannot work with independent per-channel gain reduction: at the
// old default 0.35 mono input came out several dB quieter on the right, and
// the blended matrix is singular at 0.5.)

// Feedback output state (unit delay breaking feedback loop, Eq. 13)
@state y_prev_L = 0.0;
@state y_prev_R = 0.0;
@state y_prev_M = 0.0;
@state y_prev_S = 0.0;

// 3-Capacitor sidechain network states (CT fast, CU medium, CV slow memory, Fig. 8)
// Left
@state v_ct_L = 0.0;
@state v_cu_L = 0.0;
@state v_cv_L = 0.0;
// Right
@state v_ct_R = 0.0;
@state v_cu_R = 0.0;
@state v_cv_R = 0.0;
// Mid / Lateral
@state v_ct_M = 0.0;
@state v_cu_M = 0.0;
@state v_cv_M = 0.0;
// Side / Vertical
@state v_ct_S = 0.0;
@state v_cu_S = 0.0;
@state v_cv_S = 0.0;

// Transformer DC-blocking & high-frequency inductive roll-off states
@state dc_x_L = 0.0;
@state dc_y_L = 0.0;
@state dc_x_R = 0.0;
@state dc_y_R = 0.0;
@state hf_lp_L = 0.0;
@state hf_lp_R = 0.0;

@state in2Hold = 0.0;

// Helper: Raffensperger Eq. 10 Class-B sidechain crossover nonlinearity
function fairchild_stage1(vpot: number, phi_dc_prime: number): number {
	let a = clamp(vpot - phi_dc_prime, -40.0, 40.0);
	let b = clamp(-vpot - phi_dc_prime, -40.0, 40.0);
	let num = log(1.0 + exp(a));
	let den = log(1.0 + exp(b));
	return num - den;
}

// Helper: Stage 1-3 sidechain voltage and germanium diode-bridge current (Raffensperger Eqs. 10-14)
function sidechain_current(fb: number, sc_scale: number, phi_dc_prime: number, v_ct: number): number {
	let v_st1 = fairchild_stage1(fb * sc_scale, phi_dc_prime);
	let v_st3 = clamp(8.4 * v_st1, -100.0, 100.0);
	let v_diff = abs(v_st3) - v_ct;
	let i_nom = 0.008 * log(1.0 + exp(clamp(4.0 * v_diff - 1.2, -20.0, 30.0)));
	return clamp(i_nom, 0.0, 0.5);
}

// Helper: Push-Pull 6386 remote-cutoff variable-mu triode amplifier (Raffensperger Sec. 2 & 6)
// Models remote-cutoff variable transconductance and balanced odd-order harmonic saturation
// that increases with gain reduction (Fig. 2, Fig. 12, Fig. 13)
function varimu_6386_amp(x: number, v_sc: number, harm_amt: number): number {
	// Remote-cutoff 6386 transconductance curve: smooth variable-mu attenuation as grid bias goes negative
	let gr_gain = pow(1.0 + 0.14 * v_sc + 0.008 * v_sc * v_sc, -1.35);
	// Raffensperger Sec. 6 Fig. 13: balanced push-pull cancels even harmonics, while 3rd-harmonic
	// distortion rises from -21 dB at light GR to -14 dB under heavy GR
	let gr_depth = clamp(1.0 - gr_gain, 0.0, 1.0);
	let u = x * gr_gain;
	let drive_sat = 1.0 + harm_amt * (0.45 + 1.85 * gr_depth);
	let u_driven = u * drive_sat;
	// Balanced push-pull soft-saturation (pure odd harmonics) normalized for unity small-signal gain
	let sat_wave = tanh(u_driven) / drive_sat;
	// Add subtle 3rd-order triode curvature before transformer ceiling
	let k3 = harm_amt * (0.08 + 0.22 * gr_depth);
	return tanh(sat_wave + k3 * sat_wave * sat_wave * sat_wave);
}

let sr_safe = maximum(samplerate(), 8000.0);
let dt = 1.0 / sr_safe;

// Input trim (plain gain stage, ahead of the transformer/6386 drive)
let g_in = dbtoa(clamp(input_gain_db, -24.0, 36.0));
let dryL = in1 * g_in;
// in2 counts as patched for 250 ms after any non-zero sample (a per-sample test
// would swap in in1 for single samples at in2's zero crossings)
if (abs(in2) > 0.0000001) {
	in2Hold = 0.25 * sr_safe;
} else if (in2Hold > 0.0) {
	in2Hold -= 1.0;
}
let dryR = ((in2Hold > 0.0) ? in2 : in1) * g_in;

// Input transformer step gain
let in_gain = dbtoa(clamp(drive_db, 0.0, 24.0));
let xL = dryL * in_gain;
let xR = dryR * in_gain;

// Crossfade between Left/Right (0) and Lateral/Vertical Mid-Side (1).
// A pair whose weight is 0 is not computed (lv = 0 or 1 costs one pair; in
// between costs two). Its state is held, and it re-enters at a tiny weight.
let lv = clamp(lat_vert_mode, 0.0, 1.0);
let inv_sqrt2 = 0.7071067811865475;

// Sidechain AC and DC threshold controls (Raffensperger Eq. 10, Fig. 9 & 12)
// phi_dc sets Class-B crossover bias (soft vari-mu knee at 0 -> hard limiting knee at 1)
// phi_ac scales the sidechain input potentiometer V_pot
let phi_ac = clamp(ac_thresh, 0.0, 1.0);
let phi_dc = clamp(dc_knee, 0.0, 1.0);
let phi_dc_prime = 1.2 + 9.5 * phi_dc;
let sc_scale = (1.5 + 22.0 * pow(phi_ac, 1.25)) * (1.0 + 0.55 * phi_dc);

// Fairchild 670 6-position Time Constant network (Fig. 8 & Table 3)
// Exact RC values from Table 3:
//   Pos 1: CT=2uF, RT=51.9k  (tau_att=0.2ms, tau_rel=104ms)
//   Pos 2: CT=2uF, RT=149.9k (tau_att=0.2ms, tau_rel=300ms)
//   Pos 3: CT=4uF, RT=220k   (tau_att=0.4ms, tau_rel=880ms)
//   Pos 4: CT=8uF, RT=220k   (tau_att=0.8ms, tau_rel=1760ms)
//   Pos 5: CT=4uF, RT=220k, CU=8uF, RU=100k (dual program-dependent release)
//   Pos 6: CT=2uF, RT=220k, CU=8uF, RU=100k, CV=20uF, RV=100k (triple program-dependent release)
let sel = clamp(tc_mode, 1.0, 6.0);
let c_t = 2.0;
let tau_rel_fast = 0.104;
if (sel < 1.5) {
	c_t = 2.0;
	tau_rel_fast = 0.104;
} else if (sel < 2.5) {
	c_t = 2.0;
	tau_rel_fast = 0.300;
} else if (sel < 3.5) {
	c_t = 4.0;
	tau_rel_fast = 0.880;
} else if (sel < 4.5) {
	c_t = 8.0;
	tau_rel_fast = 1.760;
} else if (sel < 5.5) {
	c_t = 4.0;
	tau_rel_fast = 0.450;
} else {
	c_t = 2.0;
	tau_rel_fast = 0.250;
}

// Fast attack charging rate (scaled by 1 / CT)
let charge_scale = 2800.0 / c_t;
let g_cu = (sel >= 4.5) ? (1.0 / (0.10 * c_t)) : 0.0;
let g_cv = (sel >= 5.5) ? (1.0 / (0.10 * c_t)) : 0.0;

let harm = clamp(tube_harmonics, 0.0, 1.0);

let wetL = 0.0;
let wetR = 0.0;

// --- Left/Right pair --------------------------------------------------------
if (lv < 1.0) {
	// Update 3-capacitor network for Left (Fig. 8)
	let i_sc_L = sidechain_current(y_prev_L, sc_scale, phi_dc_prime, v_ct_L);
	let d_ct_L = i_sc_L * charge_scale - (v_ct_L / tau_rel_fast) - g_cu * (v_ct_L - v_cu_L) - g_cv * (v_ct_L - v_cv_L);
	v_ct_L = clamp(v_ct_L + dt * d_ct_L, 0.0, 60.0);
	let d_cu_L = (sel >= 4.5) ? ((v_ct_L - v_cu_L) / 0.80) : (0.0 - v_cu_L / 1.50);
	v_cu_L = clamp(v_cu_L + dt * d_cu_L, 0.0, 60.0);
	let d_cv_L = (sel >= 5.5) ? ((v_ct_L - v_cv_L) / 2.00) : (0.0 - v_cv_L / 3.00);
	v_cv_L = clamp(v_cv_L + dt * d_cv_L, 0.0, 60.0);

	// Update 3-capacitor network for Right (Fig. 8)
	let i_sc_R = sidechain_current(y_prev_R, sc_scale, phi_dc_prime, v_ct_R);
	let d_ct_R = i_sc_R * charge_scale - (v_ct_R / tau_rel_fast) - g_cu * (v_ct_R - v_cu_R) - g_cv * (v_ct_R - v_cv_R);
	v_ct_R = clamp(v_ct_R + dt * d_ct_R, 0.0, 60.0);
	let d_cu_R = (sel >= 4.5) ? ((v_ct_R - v_cu_R) / 0.80) : (0.0 - v_cu_R / 1.50);
	v_cu_R = clamp(v_cu_R + dt * d_cu_R, 0.0, 60.0);
	let d_cv_R = (sel >= 5.5) ? ((v_ct_R - v_cv_R) / 2.00) : (0.0 - v_cv_R / 3.00);
	v_cv_R = clamp(v_cv_R + dt * d_cv_R, 0.0, 60.0);

	// Apply 6386 Push-Pull Variable-Mu Amplifier
	let y_L = varimu_6386_amp(xL, v_ct_L, harm);
	let y_R = varimu_6386_amp(xR, v_ct_R, harm);
	// Store feedback output state
	y_prev_L = y_L;
	y_prev_R = y_R;
	wetL = (1.0 - lv) * y_L;
	wetR = (1.0 - lv) * y_R;
}

// --- Lateral/Vertical (Mid/Side) pair ---------------------------------------
if (lv > 0.0) {
	let mid_in = (xL + xR) * inv_sqrt2;
	let side_in = (xL - xR) * inv_sqrt2;

	// Update 3-capacitor network for Mid (Fig. 8)
	let i_sc_M = sidechain_current(y_prev_M, sc_scale, phi_dc_prime, v_ct_M);
	let d_ct_M = i_sc_M * charge_scale - (v_ct_M / tau_rel_fast) - g_cu * (v_ct_M - v_cu_M) - g_cv * (v_ct_M - v_cv_M);
	v_ct_M = clamp(v_ct_M + dt * d_ct_M, 0.0, 60.0);
	let d_cu_M = (sel >= 4.5) ? ((v_ct_M - v_cu_M) / 0.80) : (0.0 - v_cu_M / 1.50);
	v_cu_M = clamp(v_cu_M + dt * d_cu_M, 0.0, 60.0);
	let d_cv_M = (sel >= 5.5) ? ((v_ct_M - v_cv_M) / 2.00) : (0.0 - v_cv_M / 3.00);
	v_cv_M = clamp(v_cv_M + dt * d_cv_M, 0.0, 60.0);

	// Update 3-capacitor network for Side (Fig. 8)
	let i_sc_S = sidechain_current(y_prev_S, sc_scale, phi_dc_prime, v_ct_S);
	let d_ct_S = i_sc_S * charge_scale - (v_ct_S / tau_rel_fast) - g_cu * (v_ct_S - v_cu_S) - g_cv * (v_ct_S - v_cv_S);
	v_ct_S = clamp(v_ct_S + dt * d_ct_S, 0.0, 60.0);
	let d_cu_S = (sel >= 4.5) ? ((v_ct_S - v_cu_S) / 0.80) : (0.0 - v_cu_S / 1.50);
	v_cu_S = clamp(v_cu_S + dt * d_cu_S, 0.0, 60.0);
	let d_cv_S = (sel >= 5.5) ? ((v_ct_S - v_cv_S) / 2.00) : (0.0 - v_cv_S / 3.00);
	v_cv_S = clamp(v_cv_S + dt * d_cv_S, 0.0, 60.0);

	let y_M = varimu_6386_amp(mid_in, v_ct_M, harm);
	let y_S = varimu_6386_amp(side_in, v_ct_S, harm);
	y_prev_M = y_M;
	y_prev_S = y_S;
	// Decode back from Lateral/Vertical (M/S)
	wetL = wetL + lv * (y_M + y_S) * inv_sqrt2;
	wetR = wetR + lv * (y_M - y_S) * inv_sqrt2;
}

// Output transformer gentle high-frequency silk & DC block (using lowercase pi!)
let g_hf = 1.0 - exp(-2.0 * pi * 16500.0 / sr_safe);
hf_lp_L = hf_lp_L + g_hf * (wetL - hf_lp_L);
hf_lp_R = hf_lp_R + g_hf * (wetR - hf_lp_R);

// Automatic makeup / output level trim so output stays controlled across drive & threshold
let out_comp = (1.0 + 0.55 * phi_ac) / pow(in_gain, 0.45 + 0.35 * (1.0 - phi_ac));
let procL = hf_lp_L * out_comp;
let procR = hf_lp_R * out_comp;

dc_y_L = procL - dc_x_L + 0.9992 * dc_y_L;
dc_x_L = procL;
dc_y_R = procR - dc_x_R + 0.9992 * dc_y_R;
dc_x_R = procR;

let m = clamp(mix, 0.0, 1.0);

// Output trim (plain gain stage, after the sidechain feedback tap)
let g_out = dbtoa(clamp(output_gain_db, -24.0, 36.0));
out1 = ((1.0 - m) * dryL + m * dc_y_L) * g_out;
out2 = ((1.0 - m) * dryR + m * dc_y_R) * g_out;
