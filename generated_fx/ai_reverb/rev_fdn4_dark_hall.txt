// ============================================================================
// rev_fdn4_dark_hall.txt
// A four-line FDN hall whose tail is dark and sub-free.
// From: Beltrán et al., DAFx-99, §5.1 — Jot's reverberator: a bank of delay
//       lines with a unitary matrix A (A·A^T = I) in the feedback loop, plus a
//       global absorptive filter; the 4x4 Hadamard matrix used here is the
//       smallest unitary mixing matrix. Also Dahl & Jot, "A Reverberator Based
//       on Absorbent All-Pass Filters", DAFx-00, §2.1 (unitary-feedback delay
//       networks), for the losses-in-the-loop form. The input diffusers are
//       the series allpass chain of Dattorro, "Effect Design Part 1", JAES
//       1997 (lengths 142/107/379/277 at 29.761 kHz), so the hall is dense
//       from its first reflections instead of a handful of discrete echoes.
// 100% wet. No dry path, no limiter, no metering.
// Inlets:
//   in1 / in2: stereo audio
//   in3: bound only (no pitch-relative parameter)
//   in4: 0..1 one-bar phasor, bound only (hall time is in seconds, not bars)
// Outlets:
//   out1 / out2: stereo hall, wet only
// ============================================================================

@param({ min: 0, max: 250, exponent: 2 }) predelay_ms = 30;      // predelay
@param({ min: 15, max: 120 }) size_ms = 45;         // room size
@param({ min: 1, max: 20, exponent: 2 }) decay_s = 6.0;          // decay
@param({ min: 250, max: 8000, exponent: 3 }) damp_hz = 1500;     // damping
@param({ min: 20, max: 500, exponent: 2 }) lowcut_hz = 80;       // low cut
@param({ min: 0, max: 1 }) mod_depth = 0.3;         // modulation
@param({ min: 0, max: 1 }) width = 1.0;             // width
@param({ min: -24, max: 12 }) out_gain_db = 0;      // output gain

// 32768 samples: 250 ms predelay and the 2.13 x 120 ms line fit up to 96 kHz
@state preL = new FixedSampleArray(32768);
@state preR = new FixedSampleArray(32768);
@state preIdx: Int = 0;

@state line0 = new FixedSampleArray(32768);
@state line1 = new FixedSampleArray(32768);
@state line2 = new FixedSampleArray(32768);
@state line3 = new FixedSampleArray(32768);
@state wIdx: Int = 0;

// Input diffusion: four series allpasses per channel (shared write index)
@state dfL0 = new FixedSampleArray(2048);
@state dfL1 = new FixedSampleArray(2048);
@state dfL2 = new FixedSampleArray(2048);
@state dfL3 = new FixedSampleArray(2048);
@state dfR0 = new FixedSampleArray(2048);
@state dfR1 = new FixedSampleArray(2048);
@state dfR2 = new FixedSampleArray(2048);
@state dfR3 = new FixedSampleArray(2048);
@state dfIdx: Int = 0;

@state lp0 = 0.0;
@state lp1 = 0.0;
@state lp2 = 0.0;
@state lp3 = 0.0;
@state hx0 = 0.0;
@state hx1 = 0.0;
@state hx2 = 0.0;
@state hx3 = 0.0;
@state hy0 = 0.0;
@state hy1 = 0.0;
@state hy2 = 0.0;
@state hy3 = 0.0;
@state modPh = 0.0;
// Input low cut: two cascaded one-pole highpasses per channel
@state icxL0 = 0.0;
@state icyL0 = 0.0;
@state icxL1 = 0.0;
@state icyL1 = 0.0;
@state icxR0 = 0.0;
@state icyR0 = 0.0;
@state icxR1 = 0.0;
@state icyR1 = 0.0;

// One-pole lowpass coefficient b whose gain at the damping frequency is
// m = exp(-kLoss * len): solves (1-b)^2 / (1 - 2b cos w + b^2) = m^2.
// Loss is per second rather than per pass, so long and short lines (and
// small and large rooms) darken at the same rate - Jot's absorptive filter
function dampCoef(len, cosW, kLoss) {
	let m2 = exp(0.0 - 2.0 * kLoss * len);
	let den = maximum(1.0 - m2, 0.000000000001);
	let a = (1.0 - m2 * cosW) / den;
	return a - sqrt(maximum(a * a - 1.0, 0.0));
}

// Same for the scaled DC blocker (1+R)/2 * (1 - z^-1) / (1 - R z^-1): pole R
// whose gain at the low-cut frequency is m = exp(-kLoss * len)
function cutCoef(len, cosW, kLoss) {
	let m2 = exp(0.0 - 2.0 * kLoss * len);
	let q = 0.5 * (1.0 - cosW);
	let den = maximum(m2 - q, 0.000000000001);
	let bq = (m2 * cosW + q) / den;
	return bq - sqrt(maximum(bq * bq - 1.0, 0.0));
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = in2;
let unused_in3 = in3; // keeps the in3 port present
let barIn = in4;      // accepted for the batch contract; hall time is seconds

// --- Predelay (ms) ---------------------------------------------------------
preL[preIdx] = xL;
preR[preIdx] = xR;

let preSamps = clamp(mstosamps(predelay_ms), 1.0, 32700.0);

let rpL = preIdx - preSamps;
while (rpL < 0.0) { rpL += 32768.0; }
let piL: Int = floor(rpL);
let pL = cubicinterp(rpL - piL,
	preL[(piL - 1 + 32768) & 32767],
	preL[piL & 32767],
	preL[(piL + 1) & 32767],
	preL[(piL + 2) & 32767]);

let rpR = preIdx - preSamps;
while (rpR < 0.0) { rpR += 32768.0; }
let piR: Int = floor(rpR);
let pR = cubicinterp(rpR - piR,
	preR[(piR - 1 + 32768) & 32767],
	preR[piR & 32767],
	preR[(piR + 1) & 32767],
	preR[(piR + 2) & 32767]);

preIdx = (preIdx + 1) & 32767;

// --- Input diffusion: y = -g*x + d, d' = x + g*y (unity-gain allpass) ------
// Right channel lengths are ~7% longer so the two sides diffuse differently
let dsr = sr / 29761.0;
let dl0: Int = floor(clamp(142.0 * dsr, 1.0, 2040.0));
let dl1: Int = floor(clamp(107.0 * dsr, 1.0, 2040.0));
let dl2: Int = floor(clamp(379.0 * dsr, 1.0, 2040.0));
let dl3: Int = floor(clamp(277.0 * dsr, 1.0, 2040.0));
let dr0: Int = floor(clamp(152.0 * dsr, 1.0, 2040.0));
let dr1: Int = floor(clamp(115.0 * dsr, 1.0, 2040.0));
let dr2: Int = floor(clamp(406.0 * dsr, 1.0, 2040.0));
let dr3: Int = floor(clamp(297.0 * dsr, 1.0, 2040.0));

let zL0 = dfL0[(dfIdx - dl0 + 2048) & 2047];
let aL0 = zL0 - 0.75 * pL;
dfL0[dfIdx] = pL + 0.75 * aL0;
let zL1 = dfL1[(dfIdx - dl1 + 2048) & 2047];
let aL1 = zL1 - 0.75 * aL0;
dfL1[dfIdx] = aL0 + 0.75 * aL1;
let zL2 = dfL2[(dfIdx - dl2 + 2048) & 2047];
let aL2 = zL2 - 0.625 * aL1;
dfL2[dfIdx] = aL1 + 0.625 * aL2;
let zL3 = dfL3[(dfIdx - dl3 + 2048) & 2047];
let aL3 = zL3 - 0.625 * aL2;
dfL3[dfIdx] = aL2 + 0.625 * aL3;

let zR0 = dfR0[(dfIdx - dr0 + 2048) & 2047];
let aR0 = zR0 - 0.75 * pR;
dfR0[dfIdx] = pR + 0.75 * aR0;
let zR1 = dfR1[(dfIdx - dr1 + 2048) & 2047];
let aR1 = zR1 - 0.75 * aR0;
dfR1[dfIdx] = aR0 + 0.75 * aR1;
let zR2 = dfR2[(dfIdx - dr2 + 2048) & 2047];
let aR2 = zR2 - 0.625 * aR1;
dfR2[dfIdx] = aR1 + 0.625 * aR2;
let zR3 = dfR3[(dfIdx - dr3 + 2048) & 2047];
let aR3 = zR3 - 0.625 * aR2;
dfR3[dfIdx] = aR2 + 0.625 * aR3;
dfIdx = (dfIdx + 1) & 2047;

// --- Input low cut (12 dB/oct at lowcut_hz): no sub ever enters the hall --
let inCutF = clamp(lowcut_hz, 10.0, sr * 0.45);
let inCutC = exp(0.0 - 2.0 * pi * inCutF / sr);
let hpL0 = inCutC * (icyL0 + aL3 - icxL0);
icxL0 = aL3; icyL0 = hpL0;
let hpL1 = inCutC * (icyL1 + hpL0 - icxL1);
icxL1 = hpL0; icyL1 = hpL1;
let hpR0 = inCutC * (icyR0 + aR3 - icxR0);
icxR0 = aR3; icyR0 = hpR0;
let hpR1 = inCutC * (icyR1 + hpR0 - icxR1);
icxR1 = hpR0; icyR1 = hpR1;

let dL = hpL1;
let dR = hpR1;

// --- Line lengths: mutually prime-ish ratios of the room size -------------
let baseSamps = clamp(mstosamps(size_ms), 8.0, 15000.0);
let modPhInc = 0.37 / sr;
modPh = wrap(modPh + modPhInc, 0.0, 1.0);
let modAmt = clamp(mod_depth, 0.0, 1.0) * 6.0;

let len0 = clamp(baseSamps * 1.00 + modAmt * sin(2.0 * pi * modPh), 8.0, 32700.0);
let len1 = clamp(baseSamps * 1.31 + modAmt * sin(2.0 * pi * modPh + 1.5707963), 8.0, 32700.0);
let len2 = clamp(baseSamps * 1.71 + modAmt * sin(2.0 * pi * modPh + 3.1415927), 8.0, 32700.0);
let len3 = clamp(baseSamps * 2.13 + modAmt * sin(2.0 * pi * modPh + 4.7123890), 8.0, 32700.0);

// --- Read the four lines --------------------------------------------------
let rd0 = wIdx - len0;
while (rd0 < 0.0) { rd0 += 32768.0; }
let i0: Int = floor(rd0);
let y0 = cubicinterp(rd0 - i0,
	line0[(i0 - 1 + 32768) & 32767],
	line0[i0 & 32767],
	line0[(i0 + 1) & 32767],
	line0[(i0 + 2) & 32767]);

let rd1 = wIdx - len1;
while (rd1 < 0.0) { rd1 += 32768.0; }
let i1: Int = floor(rd1);
let y1 = cubicinterp(rd1 - i1,
	line1[(i1 - 1 + 32768) & 32767],
	line1[i1 & 32767],
	line1[(i1 + 1) & 32767],
	line1[(i1 + 2) & 32767]);

let rd2 = wIdx - len2;
while (rd2 < 0.0) { rd2 += 32768.0; }
let i2: Int = floor(rd2);
let y2 = cubicinterp(rd2 - i2,
	line2[(i2 - 1 + 32768) & 32767],
	line2[i2 & 32767],
	line2[(i2 + 1) & 32767],
	line2[(i2 + 2) & 32767]);

let rd3 = wIdx - len3;
while (rd3 < 0.0) { rd3 += 32768.0; }
let i3: Int = floor(rd3);
let y3 = cubicinterp(rd3 - i3,
	line3[(i3 - 1 + 32768) & 32767],
	line3[i3 & 32767],
	line3[(i3 + 1) & 32767],
	line3[(i3 + 2) & 32767]);

// --- Absorptive losses in the loop: one-pole lowpass + DC blocker ----------
// At damp_hz and at lowcut_hz the tail loses an extra 45 dB per second
// (5.181 Np/s) on top of the decay_s rate, whatever the room size; beyond
// them it dies faster still. The DC blocker is scaled by (1 + R) / 2 so its
// gain is <= 1 at every frequency (unscaled it reaches 2 / (1 + R))
let dampF = clamp(damp_hz, 50.0, sr * 0.45);
let dampCos = cos(2.0 * pi * dampF / sr);
let kLoss = 5.181 / sr;
let b0 = dampCoef(len0, dampCos, kLoss);
let b1 = dampCoef(len1, dampCos, kLoss);
let b2 = dampCoef(len2, dampCos, kLoss);
let b3 = dampCoef(len3, dampCos, kLoss);
let cutF = clamp(lowcut_hz, 10.0, sr * 0.45);
let cutCos = cos(2.0 * pi * cutF / sr);
let r0 = cutCoef(len0, cutCos, kLoss);
let r1 = cutCoef(len1, cutCos, kLoss);
let r2 = cutCoef(len2, cutCos, kLoss);
let r3 = cutCoef(len3, cutCos, kLoss);

lp0 = lp0 + (1.0 - b0) * (y0 - lp0);
lp1 = lp1 + (1.0 - b1) * (y1 - lp1);
lp2 = lp2 + (1.0 - b2) * (y2 - lp2);
lp3 = lp3 + (1.0 - b3) * (y3 - lp3);

let h0 = 0.5 * (1.0 + r0) * (lp0 - hx0) + r0 * hy0;
let h1 = 0.5 * (1.0 + r1) * (lp1 - hx1) + r1 * hy1;
let h2 = 0.5 * (1.0 + r2) * (lp2 - hx2) + r2 * hy2;
let h3 = 0.5 * (1.0 + r3) * (lp3 - hx3) + r3 * hy3;
hx0 = lp0; hy0 = h0;
hx1 = lp1; hy1 = h1;
hx2 = lp2; hy2 = h2;
hx3 = lp3; hy3 = h3;

// --- Per-line decay gain (each line is its own loop length) ---------------
// Every other gain in the loop is <= 1 and the matrix is unitary, so the
// clamp is only a guard; it no longer caps short lines at long decays
let rt60 = clamp(decay_s, 1.0, 20.0);
let g0 = clamp(exp(0.0 - 6.907755 * (len0 / sr) / rt60), 0.0, 0.9995);
let g1 = clamp(exp(0.0 - 6.907755 * (len1 / sr) / rt60), 0.0, 0.9995);
let g2 = clamp(exp(0.0 - 6.907755 * (len2 / sr) / rt60), 0.0, 0.9995);
let g3 = clamp(exp(0.0 - 6.907755 * (len3 / sr) / rt60), 0.0, 0.9995);

let f0 = h0 * g0;
let f1 = h1 * g1;
let f2 = h2 * g2;
let f3 = h3 * g3;

// --- Unitary mixing: 4x4 Hadamard / 2 -------------------------------------
let m0 = 0.5 * (f0 + f1 + f2 + f3);
let m1 = 0.5 * (f0 - f1 + f2 - f3);
let m2 = 0.5 * (f0 + f1 - f2 - f3);
let m3 = 0.5 * (f0 - f1 - f2 + f3);

// --- Inject the (pre-delayed) input and advance ---------------------------
line0[wIdx] = m0 + 0.5 * dL;
line1[wIdx] = m1 + 0.5 * dR;
line2[wIdx] = m2 + 0.5 * dL;
line3[wIdx] = m3 + 0.5 * dR;
wIdx = (wIdx + 1) & 32767;

// --- Output: two disjoint (orthogonal) tap pairs -------------------------
// Taken after the loop filters, so even the first pass through each line is
// damped and low-cut. Lines 0+3 / 1+2 were picked by measurement: on a mono
// tuba the two sides come out balanced and nearly uncorrelated, so the hall
// survives a mono sum
let wetL = 0.7071068 * (h0 + h3);
let wetR = 0.7071068 * (h1 + h2);

// Energy normalisation from the mean loop gain, plus a measured correction:
// the absorptive filters cap how long the tuba band rings, so without it a
// 20 s / 120 ms hall came out ~9 dB under a 1 s / 15 ms one
let gMean = 0.25 * (g0 + g1 + g2 + g3);
let normTrim = pow(rt60 / 6.0, 0.25) * pow(baseSamps / mstosamps(45.0), 0.12);
let norm = 2.35 * sqrt(maximum(1.0 - gMean * gMean, 0.0001)) * normTrim;
let sL = wetL * norm;
let sR = wetR * norm;

// --- Width (mid/side) -----------------------------------------------------
let mid = 0.5 * (sL + sR);
let side = 0.5 * (sL - sR) * clamp(width, 0.0, 1.0);
let oL = mid + side;
let oR = mid - side;

let g = dbtoa(clamp(out_gain_db, -24.0, 12.0));
out1 = oL * g;
out2 = oR * g;
