// ============================================================================
// rev_plate_dattorro.txt
// A Dattorro plate: dense bright metal with a slow wobble.
// From: Fernando A. Beltrán, José R. Beltrán, Nicolas Holzem & Adrian Gogu,
//       "Matlab Implementation of Reverberation Algorithms", DAFx-99, §3 and
//       Table 1 — the Dattorro (JAES 1997) network: predelay -> bandwidth
//       lowpass -> four input allpasses -> two cross-coupled tank lines with
//       decay diffusion allpasses, damping and 12 output taps.
// 100% wet. No dry path, no limiter, no metering.
// Inlets:
//   in1 / in2: stereo audio
//   in3: bound only (no pitch-relative parameter in a plate)
//   in4: 0..1 one-bar phasor, bound only (a plate has no tempo-locked time)
// Outlets:
//   out1 / out2: stereo plate, wet only
// ============================================================================

@param({ min: 0, max: 200 }) predelay_ms = 20;      // predelay
@param({ min: 0.3, max: 12 }) decay_s = 2.0;        // decay
@param({ min: 500, max: 16000 }) damp_hz = 6000;    // damping
@param({ min: 0, max: 1 }) diffuse = 0.65;          // diffusion
@param({ min: 0, max: 3 }) wobble_ms = 0.8;         // wobble depth
@param({ min: 0.05, max: 2 }) wobble_hz = 0.35;     // wobble rate
@param({ min: 0, max: 1 }) width = 1.0;             // width
@param({ min: -24, max: 12 }) out_gain_db = 0;      // output gain

@state preL = new FixedSampleArray(16384);
@state preR = new FixedSampleArray(16384);
@state preIdx: Int = 0;

// Four input diffusion allpasses per channel: row 0..3 = left, 4..7 = right
@state inAp = new FixedSampleArray(8, 2048);
@state iaIdx: Int = 0;

// Two decay-diffusion allpasses per tank half: row 0=L1, 1=L2, 2=R1, 3=R2
@state tankAp = new FixedSampleArray(4, 4096);
@state taIdx: Int = 0;

// Tank delay lines: 4453 / 1800 samples (left), 4217 / 2656 (right) @ 29.76 kHz
@state t1L = new FixedSampleArray(16384);
@state t1R = new FixedSampleArray(16384);
@state t1Idx: Int = 0;
@state t2L = new FixedSampleArray(16384);
@state t2R = new FixedSampleArray(16384);
@state t2Idx: Int = 0;

@state dmpL = 0.0;
@state dmpR = 0.0;
@state wobPh = 0.0;
@state xfbL = 0.0;
@state xfbR = 0.0;
@state gInAp = 0.6;

// One Schroeder allpass stage on a tank row (fractional, for the wobble)
function tankStep(row: Int, xin: number, len: number): number {
	let rp = taIdx - len;
	while (rp < 0.0) { rp += 4096.0; }
	let i0: Int = floor(rp);
	let xd = cubicinterp(rp - i0,
		tankAp[row][(i0 - 1 + 4096) & 4095],
		tankAp[row][i0 & 4095],
		tankAp[row][(i0 + 1) & 4095],
		tankAp[row][(i0 + 2) & 4095]);
	let v = xin + 0.5 * xd;
	tankAp[row][taIdx] = v;
	return xd - 0.5 * v;
}

// One input diffusion allpass stage
function inputStep(row: Int, xin: number, len: number): number {
	let rp = iaIdx - len;
	while (rp < 0.0) { rp += 2048.0; }
	let i0: Int = floor(rp);
	let xd = cubicinterp(rp - i0,
		inAp[row][(i0 - 1 + 2048) & 2047],
		inAp[row][i0 & 2047],
		inAp[row][(i0 + 1) & 2047],
		inAp[row][(i0 + 2) & 2047]);
	let v = xin + gInAp * xd;
	inAp[row][iaIdx] = v;
	return xd - gInAp * v;
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = in2;
let unused_in3 = in3; // keeps the in3 port present
let barIn = in4;      // accepted for the batch contract; a plate has no bar time
let srScale = sr / 29761.0;

// --- Predelay (ms) ---------------------------------------------------------
preL[preIdx] = xL;
preR[preIdx] = xR;

let preSamps = clamp(mstosamps(predelay_ms), 1.0, 16300.0);

let rpL = preIdx - preSamps;
while (rpL < 0.0) { rpL += 16384.0; }
let piL: Int = floor(rpL);
let pL = cubicinterp(rpL - piL,
	preL[(piL - 1 + 16384) & 16383],
	preL[piL & 16383],
	preL[(piL + 1) & 16383],
	preL[(piL + 2) & 16383]);

let rpR = preIdx - preSamps;
while (rpR < 0.0) { rpR += 16384.0; }
let piR: Int = floor(rpR);
let pR = cubicinterp(rpR - piR,
	preR[(piR - 1 + 16384) & 16383],
	preR[piR & 16383],
	preR[(piR + 1) & 16383],
	preR[(piR + 2) & 16383]);

preIdx = (preIdx + 1) & 16383;

// --- Input diffusion: four allpasses per channel (DAFx-99 §3.1) ------------
gInAp = 0.40 + 0.35 * clamp(diffuse, 0.0, 1.0);
let l0 = clamp(142.0 * srScale, 4.0, 2000.0);
let l1 = clamp(107.0 * srScale, 4.0, 2000.0);
let l2 = clamp(379.0 * srScale, 4.0, 2000.0);
let l3 = clamp(277.0 * srScale, 4.0, 2000.0);

let diffL = inputStep(0, pL, l0);
diffL = inputStep(1, diffL, l1);
diffL = inputStep(2, diffL, l2);
diffL = inputStep(3, diffL, l3);

let diffR = inputStep(4, pR, l0);
diffR = inputStep(5, diffR, l1);
diffR = inputStep(6, diffR, l2);
diffR = inputStep(7, diffR, l3);

iaIdx = (iaIdx + 1) & 2047;

// --- Tank timing, decay and damping ---------------------------------------
let t1LenL = clamp(4453.0 * srScale, 16.0, 16300.0);
let t1LenR = clamp(4217.0 * srScale, 16.0, 16300.0);
let t2LenL = clamp(1800.0 * srScale, 16.0, 16300.0);
let t2LenR = clamp(2656.0 * srScale, 16.0, 16300.0);

// One half of the tank is one loop pass; the decay gain is applied once per
// half, so the full L->R->L cycle is what sets RT60.
let halfSec = 0.5 * (t1LenL + t2LenL) / sr;
let decay = clamp(exp(0.0 - 6.907755 * halfSec / clamp(decay_s, 0.3, 12.0)), 0.0, 0.92);

let dampF = clamp(damp_hz, 200.0, sr * 0.45);
let dampC = 1.0 - exp(0.0 - 2.0 * pi * dampF / sr);

// Slow wobble on the decay-diffusion allpasses (kills the metallic ring)
wobPh = wrap(wobPh + clamp(wobble_hz, 0.01, 5.0) / sr, 0.0, 1.0);
let wobSamps = sin(2.0 * pi * wobPh) * clamp(wobble_ms, 0.0, 3.0) * 0.001 * sr;

let ap1Len = clamp(672.0 * srScale + wobSamps, 8.0, 4000.0);
let ap2Len = clamp(908.0 * srScale - wobSamps, 8.0, 4000.0);

// --- Left tank half -------------------------------------------------------
let aL = diffL + xfbL;
let yA_L = tankStep(0, aL, ap1Len);

t1L[t1Idx] = yA_L;
let dL = t1L[(t1Idx - t1LenL) & 16383];
dmpL = dmpL + dampC * (dL - dmpL);

t2L[t2Idx] = dmpL * decay;
let eL = t2L[(t2Idx - t2LenL) & 16383];
let yB_L = tankStep(1, eL, ap2Len);

// --- Right tank half ------------------------------------------------------
let aR = diffR + xfbR;
let yA_R = tankStep(2, aR, ap1Len);

t1R[t1Idx] = yA_R;
let dR = t1R[(t1Idx - t1LenR) & 16383];
dmpR = dmpR + dampC * (dR - dmpR);

t2R[t2Idx] = dmpR * decay;
let eR = t2R[(t2Idx - t2LenR) & 16383];
let yB_R = tankStep(3, eR, ap2Len);

t1Idx = (t1Idx + 1) & 16383;
t2Idx = (t2Idx + 1) & 16383;
taIdx = (taIdx + 1) & 4095;

// Cross-coupling: each half drives the other on the next sample
xfbL = yB_R;
xfbR = yB_L;

// --- Output taps (DAFx-99 Table 1, reduced to four alternating taps) ------
let rawL = 0.50 * dL + 0.45 * eR + 0.30 * yA_L - 0.25 * yB_R;
let rawR = 0.50 * dR + 0.45 * eL + 0.30 * yA_R - 0.25 * yB_L;

let norm = 0.85 * sqrt(maximum(1.0 - decay * decay, 0.0001));
let wetL = rawL * norm;
let wetR = rawR * norm;

// --- Width (mid/side) -----------------------------------------------------
let mid = 0.5 * (wetL + wetR);
let side = 0.5 * (wetL - wetR) * clamp(width, 0.0, 1.0);
let oL = mid + side;
let oR = mid - side;

let g = dbtoa(clamp(out_gain_db, -24.0, 12.0));
out1 = oL * g;
out2 = oR * g;
