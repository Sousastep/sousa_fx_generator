// ============================================================================
// rev_modal_tuned.txt
// Eight 5-limit room modes locked to the tuba's f0.
// From: Jonathan S. Abel & Kurt James Werner, "Distortion and Pitch Processing
//       Using a Modal Reverberator Architecture", DAFx-15 Paper 72, §2 (the
//       modal reverberator: parallel resonant filters, h_m(t) = e^{(j*w_m -
//       a_m)t}, decay set per mode) and §3.3 (mode frequencies scaled by a
//       ratio -> pitch). The equal-tempered 2^(sigma/12) of that section is
//       replaced here by 5-limit ratios.
// 100% wet. No dry path, no limiter, no metering.
// Inlets:
//   in1 / in2: stereo audio (summed to excite the mode bank)
//   in3: f0 in Hz — 0 means unknown, the last value is held (55 Hz at startup)
//   in4: 0..1 one-bar phasor, bound only (modes follow f0, not the bar)
// Outlets:
//   out1 / out2: stereo modal tail, wet only
// ============================================================================

@param({ min: 0, max: 100 }) predelay_ms = 15;      // predelay
@param({ min: 0.5, max: 20 }) decay_s = 5;          // decay
@param({ min: 0, max: 1 }) tilt = 0.45;             // brightness
@param({ min: 0, max: 25 }) detune_cents = 8;       // detune
@param({ min: 0, max: 1 }) sub_weight = 0.35;       // sub octave
@param({ min: 0, max: 1 }) width = 0.8;             // width
@param({ min: -24, max: 12 }) out_gain_db = 0;      // output gain

@state preL = new FixedSampleArray(8192);
@state preR = new FixedSampleArray(8192);
@state preIdx: Int = 0;

// 5-limit mode ratios: 1/2, 1/1, 6/5, 5/4, 3/2, 5/3, 2/1, 5/2
@state modeRatio = new FixedSampleArray([0.5, 1.0, 1.2, 1.25, 1.5, 1.6666667, 2.0, 2.5]);

@state a1L = new FixedSampleArray(8);
@state a2L = new FixedSampleArray(8);
@state gL = new FixedSampleArray(8);
@state y1L = new FixedSampleArray(8);
@state y2L = new FixedSampleArray(8);

@state a1R = new FixedSampleArray(8);
@state a2R = new FixedSampleArray(8);
@state gR = new FixedSampleArray(8);
@state y1R = new FixedSampleArray(8);
@state y2R = new FixedSampleArray(8);

@state f0Hold = 55.0;
@state ctlCnt: Int = 0;

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = in2;
let barIn = in4; // accepted for the batch contract; modes follow f0, not the bar

// --- f0: read, hold when unknown (0), never fall back to a derivative ------
let fin = in3;
if (fin > 20.0 && fin < 1200.0) { f0Hold = fin; }
let f0 = clamp(f0Hold, 20.0, 1200.0);

// --- Resonator coefficients, updated every 32 samples (control rate) ------
if (ctlCnt == 0) {
	let rt60 = clamp(decay_s, 0.5, 20.0);
	let tiltBase = 0.45 + 0.60 * clamp(tilt, 0.0, 1.0);
	let det = clamp(detune_cents, 0.0, 25.0) * 0.000577623;
	let subW = clamp(sub_weight, 0.0, 1.0);
	let tiltG = 1.0;

	for (let m: Int = 0; m < 8; m++) {
		// higher modes die a little sooner, as in a real room
		let modeDecay = rt60 / (1.0 + 0.5 * m / 7.0);
		let rAmt = exp(0.0 - 6.907755 / (modeDecay * sr));
		let fm = clamp(f0 * modeRatio[m], 10.0, sr * 0.45);

		let wL = 2.0 * pi * clamp(fm * (1.0 - det), 10.0, sr * 0.45) / sr;
		let wR = 2.0 * pi * clamp(fm * (1.0 + det), 10.0, sr * 0.45) / sr;

		let amp = tiltG;
		if (m == 0) { amp = tiltG * subW; }

		a1L[m] = 2.0 * rAmt * cos(wL);
		a2L[m] = rAmt * rAmt;
		gL[m] = amp * 2.0 * sin(wL) * (1.0 - rAmt);

		a1R[m] = 2.0 * rAmt * cos(wR);
		a2R[m] = rAmt * rAmt;
		gR[m] = amp * 2.0 * sin(wR) * (1.0 - rAmt);

		tiltG = tiltG * tiltBase;
	}
}
ctlCnt = (ctlCnt + 1) & 31;

// --- Predelay (ms) ---------------------------------------------------------
preL[preIdx] = xL;
preR[preIdx] = xR;

let preSamps = clamp(mstosamps(predelay_ms), 1.0, 8180.0);

let rpL = preIdx - preSamps;
while (rpL < 0.0) { rpL += 8192.0; }
let piL: Int = floor(rpL);
let pL = cubicinterp(rpL - piL,
	preL[(piL - 1 + 8192) & 8191],
	preL[piL & 8191],
	preL[(piL + 1) & 8191],
	preL[(piL + 2) & 8191]);

let rpR = preIdx - preSamps;
while (rpR < 0.0) { rpR += 8192.0; }
let piR: Int = floor(rpR);
let pR = cubicinterp(rpR - piR,
	preR[(piR - 1 + 8192) & 8191],
	preR[piR & 8191],
	preR[(piR + 1) & 8191],
	preR[(piR + 2) & 8191]);

preIdx = (preIdx + 1) & 8191;

// --- Eight resonators per channel -----------------------------------------
let xin = 0.5 * (pL + pR);
let accL = 0.0;
let accR = 0.0;

for (let m: Int = 0; m < 8; m++) {
	let vL = gL[m] * xin + a1L[m] * y1L[m] - a2L[m] * y2L[m];
	y2L[m] = y1L[m];
	y1L[m] = vL;
	accL = accL + vL;

	let vR = gR[m] * xin + a1R[m] * y1R[m] - a2R[m] * y2R[m];
	y2R[m] = y1R[m];
	y1R[m] = vR;
	accR = accR + vR;
}

let wetL = accL * 1.0;
let wetR = accR * 1.0;

// --- Width (mid/side) -----------------------------------------------------
let mid = 0.5 * (wetL + wetR);
let side = 0.5 * (wetL - wetR) * clamp(width, 0.0, 1.0);
let oL = mid + side;
let oR = mid - side;

let g = dbtoa(clamp(out_gain_db, -24.0, 12.0));
out1 = oL * g;
out2 = oR * g;
