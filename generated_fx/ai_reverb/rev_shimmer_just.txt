// ============================================================================
// rev_shimmer_just.txt
// Shimmer: every feedback pass climbs one 5-limit interval.
// From: Jonathan S. Abel & Kurt James Werner, "Distortion and Pitch Processing
//       Using a Modal Reverberator Architecture", DAFx-15 Paper 72, §1 — a
//       pitch shifter inside a reverberator's feedback (the shimmer of
//       ValhallaShimmer, their ref. [26]); their 2^(sigma/12) shift is
//       replaced here by 5-limit ratios. The shifter itself is the classic
//       two-head, Hann-crossfaded delay-line (doppler) shifter, and the loop
//       is diffused by series Schroeder allpasses, before and inside the
//       loop, as in Beltrán et al., DAFx-99, §2.5. An unshifted body loop
//       runs in parallel and shift_mix blends the two outputs.
// 100% wet. No dry path, no limiter, no metering.
// Inlets:
//   in1 / in2: stereo audio
//   in3: bound only (this effect shifts by a fixed 5-limit interval, not by f0)
//   in4: 0..1 one-bar phasor, bound only (shimmer time is in seconds)
// Outlets:
//   out1 / out2: stereo shimmer, wet only
// ============================================================================

@param({ min: 0, max: 200, exponent: 2 }) predelay_ms = 25;      // predelay
@param({ min: 1, max: 30, exponent: 2 }) decay_s = 8;            // decay
@param({ enum: ["5/4", "4/3", "3/2", "5/3", "2/1", "5/2", "3/1"] }) interval = 4; // interval
@param({ min: 0, max: 1 }) shift_mix = 0.55;        // shimmer amount
@param({ min: 500, max: 12000, exponent: 2 }) tone_hz = 4000;    // damping
@param({ min: 40, max: 200 }) window_ms = 90;       // grain window
@param({ min: 0, max: 1 }) width = 0.9;             // width
@param({ min: -24, max: 12 }) out_gain_db = 0;      // output gain

@state preL = new FixedSampleArray(16384);
@state preR = new FixedSampleArray(16384);
@state preIdx: Int = 0;

// Shimmer loop line: written with input + feedback, read by the two grain
// heads (65536: a 200 ms window's 2.5-window reach fits up to 96 kHz)
@state shL = new FixedSampleArray(65536);
@state shR = new FixedSampleArray(65536);
@state wIdx: Int = 0;

// Body loop line: the same input in a parallel loop that is never shifted
@state bdL = new FixedSampleArray(16384);
@state bdR = new FixedSampleArray(16384);
@state bIdx: Int = 0;

// Diffusion allpasses: rows 0,1 = left input, 2,3 = right input,
// rows 4,5 = left shimmer loop, 6,7 = right shimmer loop,
// rows 8,9 = left body loop, 10,11 = right body loop
@state apBuf = new FixedSampleArray(12, 4096);
@state apIdx: Int = 0;

@state phL = 0.0;
@state phR = 0.37;
@state dmpL = 0.0;
@state dmpR = 0.0;
@state bdDmpL = 0.0;
@state bdDmpR = 0.0;

function apStep(row: Int, xin: number, len: number, gAp: number): number {
	let rp = apIdx - len;
	while (rp < 0.0) { rp += 4096.0; }
	let i0: Int = floor(rp);
	let xd = apBuf[row][i0 & 4095];
	let v = xin + gAp * xd;
	apBuf[row][apIdx] = v;
	return xd - gAp * v;
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = in2;
let unused_in3 = in3; // keeps the in3 port present
let barIn = in4;      // accepted for the batch contract; shimmer time is seconds

// --- 5-limit interval ------------------------------------------------------
let ratio = 1.25;
if (interval == 1) { ratio = 1.3333333; }
else if (interval == 2) { ratio = 1.5; }
else if (interval == 3) { ratio = 1.6666667; }
else if (interval == 4) { ratio = 2.0; }
else if (interval == 5) { ratio = 2.5; }
else if (interval == 6) { ratio = 3.0; }

// --- Predelay (ms) ---------------------------------------------------------
preL[preIdx] = xL;
preR[preIdx] = xR;

let preSamps = clamp(mstosamps(predelay_ms), 1.0, 16300.0);

let rpL = preIdx - preSamps;
while (rpL < 0.0) { rpL += 16384.0; }
let piL: Int = floor(rpL);
let pL = cubicinterp(rpL - piL,
	preL[(piL - 1 + 16384) & 16383],
	preL[piL & 16383],
	preL[(piL + 1) & 16383],
	preL[(piL + 2) & 16383]);

let rpR = preIdx - preSamps;
while (rpR < 0.0) { rpR += 16384.0; }
let piR: Int = floor(rpR);
let pR = cubicinterp(rpR - piR,
	preR[(piR - 1 + 16384) & 16383],
	preR[piR & 16383],
	preR[(piR + 1) & 16383],
	preR[(piR + 2) & 16383]);

preIdx = (preIdx + 1) & 16383;

// --- Diffusion before the loop --------------------------------------------
let diffL = apStep(0, pL, clamp(mstosamps(7.1), 4.0, 4000.0), 0.65);
diffL = apStep(1, diffL, clamp(mstosamps(3.3), 4.0, 4000.0), 0.65);
let diffR = apStep(2, pR, clamp(mstosamps(8.3), 4.0, 4000.0), 0.65);
diffR = apStep(3, diffR, clamp(mstosamps(4.1), 4.0, 4000.0), 0.65);

// --- Two parallel loops ---------------------------------------------------
// The body loop is never shifted and the shimmer loop always is; shift_mix
// blends their outputs (equal power, outside both loops). Blending static and
// shifted reads inside one loop lost up to 3 dB per pass, which capped the
// real decay near 3.5 s whatever decay_s said
let smix = clamp(shift_mix, 0.0, 1.0);
let rt60 = clamp(decay_s, 1.0, 30.0);

let wSamps = clamp(mstosamps(window_ms), 64.0, 20000.0);
let baseSamps = 1.5 * wSamps;

// Loop times: grain heads average 2 windows, the body taps are fixed
// (67.3 / 79.9 ms); each loop adds its two allpasses, counted at 1.5x their
// length (their recirculation; calibrated so the body's RT60 = decay_s)
let shLoopL = (2.0 * wSamps + mstosamps(1.5 * 54.8)) / sr;
let shLoopR = (2.0 * wSamps + mstosamps(1.5 * 62.2)) / sr;
let bdLoopL = mstosamps(67.3 + 1.5 * 49.0) / sr;
let bdLoopR = mstosamps(79.9 + 1.5 * 56.0) / sr;
let decayL = clamp(exp(0.0 - 6.907755 * shLoopL / rt60), 0.0, 0.99);
let decayR = clamp(exp(0.0 - 6.907755 * shLoopR / rt60), 0.0, 0.99);
let bdDecayL = clamp(exp(0.0 - 6.907755 * bdLoopL / rt60), 0.0, 0.99);
let bdDecayR = clamp(exp(0.0 - 6.907755 * bdLoopR / rt60), 0.0, 0.99);

let toneF = clamp(tone_hz, 200.0, sr * 0.45);
let toneC = 1.0 - exp(0.0 - 2.0 * pi * toneF / sr);

// One grain step per sample. The delay shrinks by (ratio-1) samples per sample,
// so the read runs ratio times faster than the write -> pitch up by `ratio`.
let phInc = (ratio - 1.0) / wSamps;
phL = wrap(phL - phInc, 0.0, 1.0);
phR = wrap(phR - phInc, 0.0, 1.0);

// --- Shimmer loop, left: two heads, Hann crossfade --------------------------
let d1L = baseSamps + phL * wSamps;
let ph2L = wrap(phL + 0.5, 0.0, 1.0);
let d2L = baseSamps + ph2L * wSamps;

let r1L = wIdx - d1L;
while (r1L < 0.0) { r1L += 65536.0; }
let q1L: Int = floor(r1L);
let h1L = linearinterp(r1L - q1L, shL[q1L & 65535], shL[(q1L + 1) & 65535]);

let r2L = wIdx - d2L;
while (r2L < 0.0) { r2L += 65536.0; }
let q2L: Int = floor(r2L);
let h2L = linearinterp(r2L - q2L, shL[q2L & 65535], shL[(q2L + 1) & 65535]);

let xfL = 0.5 - 0.5 * cos(2.0 * pi * phL);
let shiftL = xfL * h1L + (1.0 - xfL) * h2L;
dmpL = dmpL + toneC * (shiftL - dmpL);

// --- Shimmer loop, right ----------------------------------------------------
let d1R = baseSamps + phR * wSamps;
let ph2R = wrap(phR + 0.5, 0.0, 1.0);
let d2R = baseSamps + ph2R * wSamps;

let r1R = wIdx - d1R;
while (r1R < 0.0) { r1R += 65536.0; }
let q1R: Int = floor(r1R);
let h1R = linearinterp(r1R - q1R, shR[q1R & 65535], shR[(q1R + 1) & 65535]);

let r2R = wIdx - d2R;
while (r2R < 0.0) { r2R += 65536.0; }
let q2R: Int = floor(r2R);
let h2R = linearinterp(r2R - q2R, shR[q2R & 65535], shR[(q2R + 1) & 65535]);

let xfR = 0.5 - 0.5 * cos(2.0 * pi * phR);
let shiftR = xfR * h1R + (1.0 - xfR) * h2R;
dmpR = dmpR + toneC * (shiftR - dmpR);

// --- Body loop: fixed taps, same tone filter --------------------------------
let bsL = bIdx - clamp(mstosamps(67.3), 4.0, 16300.0);
while (bsL < 0.0) { bsL += 16384.0; }
let qbL: Int = floor(bsL);
let statL = linearinterp(bsL - qbL, bdL[qbL & 16383], bdL[(qbL + 1) & 16383]);
bdDmpL = bdDmpL + toneC * (statL - bdDmpL);

let bsR = bIdx - clamp(mstosamps(79.9), 4.0, 16300.0);
while (bsR < 0.0) { bsR += 16384.0; }
let qbR: Int = floor(bsR);
let statR = linearinterp(bsR - qbR, bdR[qbR & 16383], bdR[(qbR + 1) & 16383]);
bdDmpR = bdDmpR + toneC * (statR - bdDmpR);

// --- Diffuse every pass, then close both loops -------------------------------
// Unity-gain allpasses: they smear each pass into a wash (without them the
// tail was a string of discrete echoes 1.5..2.5 windows apart)
let lpL = apStep(4, dmpL, clamp(mstosamps(23.1), 4.0, 4000.0), 0.6);
lpL = apStep(5, lpL, clamp(mstosamps(31.7), 4.0, 4000.0), 0.6);
let lpR = apStep(6, dmpR, clamp(mstosamps(26.3), 4.0, 4000.0), 0.6);
lpR = apStep(7, lpR, clamp(mstosamps(35.9), 4.0, 4000.0), 0.6);
let bpL = apStep(8, bdDmpL, clamp(mstosamps(19.7), 4.0, 4000.0), 0.6);
bpL = apStep(9, bpL, clamp(mstosamps(29.3), 4.0, 4000.0), 0.6);
let bpR = apStep(10, bdDmpR, clamp(mstosamps(22.9), 4.0, 4000.0), 0.6);
bpR = apStep(11, bpR, clamp(mstosamps(33.1), 4.0, 4000.0), 0.6);
apIdx = (apIdx + 1) & 4095;

shL[wIdx] = diffL + lpL * decayL;
shR[wIdx] = diffR + lpR * decayR;
wIdx = (wIdx + 1) & 65535;
bdL[bIdx] = diffL + bpL * bdDecayL;
bdR[bIdx] = diffR + bpR * bdDecayR;
bIdx = (bIdx + 1) & 16383;

// --- Output: equal-power blend of body and shimmer ---------------------------
// Each loop is energy-normalised from its own per-pass gain; the shimmer gets
// a measured lift for long decays (its energy leaks upward through the tone
// filter), and 0.9 sets the overall level
let normBd = sqrt(maximum(1.0 - bdDecayL * bdDecayL, 0.0001));
let normSh = sqrt(maximum(1.0 - decayL * decayL, 0.0001)) * pow(maximum(rt60, 8.0) / 8.0, 0.4);
let gBd = cos(0.5 * pi * smix) * normBd * 0.9;
let gSh = sin(0.5 * pi * smix) * normSh * 0.9;
let wetL = bpL * gBd + lpL * gSh;
let wetR = bpR * gBd + lpR * gSh;

// --- Width (mid/side) -----------------------------------------------------
let mid = 0.5 * (wetL + wetR);
let side = 0.5 * (wetL - wetR) * clamp(width, 0.0, 1.0);
let oL = mid + side;
let oR = mid - side;

let g = dbtoa(clamp(out_gain_db, -24.0, 12.0));
out1 = oL * g;
out2 = oR * g;
