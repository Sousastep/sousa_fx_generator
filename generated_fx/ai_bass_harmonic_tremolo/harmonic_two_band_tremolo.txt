// ============================================================================
// harmonic_two_band_tremolo.txt
// Two-band harmonic tremolo: the input is split into a low and a high band at
// a 5-limit multiple of the tracked note, and the two bands are gain-modulated
// in antiphase by one LFO, so the note pumps low <-> high instead of simply
// ducking. "depth" is a phase-coherent dry/wet: the split is complementary
// (high = input - low), so at depth 0 both band gains are 1 and the output is
// the input sample for sample - there is no comb notch to cross-fade.
// From: Disch & Zoelzer, "Modulation and Delay Line Based Digital Audio
//       Effects", DAFx-99, §2 Eqs. 1-2 (the AM and ring-modulation laws that
//       the two band gains interpolate between);
//       Pekonen & Valimaki, "Computationally Efficient Hammond Organ
//       Synthesis", DAFx-11, §1-2 (5 Hz tremolo, and the original organ
//       tremolo is deliberately stronger on the high partials than on the low
//       ones - the antiphase band gains do the same thing);
//       Huovilainen, "Enhanced Digital Models for Analog Modulation Effects",
//       DAFx-05, §2 (the phase-split/notch family whose pumping this band
//       tremolo reproduces without a delay line).
//
// Inlets:
//   in1 / in2: stereo audio
//   in3: tracked f0 in Hz (20..1200). 0 or out of range = hold the last valid
//        value; cold start 55 Hz (A1).
//   in4: 0..1 one-bar phasor. bar_sync pulls the LFO rate to the nearest
//        whole number of cycles per bar and phase-locks it to the downbeat;
//        held at 0 the LFO runs free at rate_hz.
// Outlets:
//   out1 / out2: stereo audio (exactly dry at depth = 0)
//   out3: transient click, 0..1, single-sample detection signal
// No safety limiter, no metering, no presets.
// ============================================================================

@param({ min: 0.05, max: 12.0 }) rate_hz = 3.2;        // tremolo rate
@param({ min: 0.0, max: 1.0 }) depth = 0.85;           // dry/wet
@param({ min: 0.0, max: 1.0 }) spread = 0.5;           // L/R LFO offset
@param({ min: 0.0, max: 1.0 }) shape = 0.25;           // sine <-> square
@param({ enum: ["2/1", "3/1", "5/2", "4/1", "6/1", "8/1"] }) split_ratio = 1; // split = f0 x
@param({ min: 0.0, max: 1.0 }) bar_sync = 0.7;         // downbeat pull
@param({ min: -24.0, max: 12.0 }) out_gain_db = 0.0;   // output gain

// Two cascaded one-poles per channel make the low band; the high band is the
// exact complement, so low + high = input.
@state loL1 = new onepole(); @state loL2 = new onepole();
@state loR1 = new onepole(); @state loR2 = new onepole();

@state f0Last = 55.0;
@state f0Sm = 55.0;
@state xoSm = 165.0;

@state lfoPh = 0.0;
@state prevBar = 0.0;
@state barLast = 0.0;
@state barHold = 0.0;
@state barHz = 0.5;
@state barHzSm = 0.5;

@state tFast = 0.0;
@state tSlow = 0.0;

let sr = maximum(samplerate(), 1.0);

// --- in3: hold the last valid f0, then slew it ----------------------------
let f0In = in3;
if (f0In > 20.0 && f0In < 1200.0) {
	f0Last = f0In;
}
let f0Slew = 1.0 - exp(0.0 - 1.0 / (0.02 * sr));
f0Sm = f0Sm + (f0Last - f0Sm) * f0Slew;
let f0 = clamp(f0Sm, 20.0, 1200.0);

// --- 5-limit split multiple of the played note ----------------------------
let mult = 3.0;
if (split_ratio == 0) { mult = 2.0; }
else if (split_ratio == 2) { mult = 2.5; }
else if (split_ratio == 3) { mult = 4.0; }
else if (split_ratio == 4) { mult = 6.0; }
else if (split_ratio == 5) { mult = 8.0; }

let xoTarget = clamp(f0 * mult, 30.0, sr * 0.45);
let xoSlew = 1.0 - exp(0.0 - 1.0 / (0.03 * sr));
xoSm = xoSm + (xoTarget - xoSm) * xoSlew;
let xo = clamp(xoSm, 30.0, sr * 0.45);

// --- free-running LFO, pulled onto the bar ---------------------------------
// The phasor's own slope gives the bar rate; bar_sync blends the LFO rate
// toward the nearest whole number of cycles per bar and then locks the phase
// to the downbeat. Because the rate is corrected first, the phase error is
// constant once settled, so the pull converges once and is inert after that.
let barIn = in4;
let barStep = barIn - prevBar;
if (barStep > 0.0000001) {
	barLast = barIn;
	barHz = clamp(barStep * sr, 0.05, 20.0);
	barHold = 0.1 * sr;
}
else if (barStep < -0.5) {
	barLast = barIn;
	barHold = 0.1 * sr;          // bar line; no usable rate estimate
}
else if (barHold > 0.0) {
	barHold -= 1.0;
}
prevBar = barIn;

let hzSlew = 1.0 - exp(0.0 - 1.0 / (0.05 * sr));
barHzSm = barHzSm + (barHz - barHzSm) * hzSlew;

let bs = clamp(bar_sync, 0.0, 1.0);
let rate = clamp(rate_hz, 0.05, 12.0);
let barOK = (barHold > 0.0) ? 1.0 : 0.0;
let nDiv = maximum(floor(rate / barHzSm + 0.5), 1.0);
let rateEff = rate;
if (barOK > 0.5) {
	rateEff = linearinterp(bs, rate, nDiv * barHzSm);
}
lfoPh = wrap(lfoPh + clamp(rateEff, 0.05, 20.0) / sr, 0.0, 1.0);

// 200 ms tau: the initial settle is a glide in frequency, never a jump in gain.
if (barOK > 0.5 && bs > 0.01) {
	let target = wrap(barLast * nDiv, 0.0, 1.0);
	let err = wrap(target - lfoPh, -0.5, 0.5);
	let kPull = 1.0 - exp(0.0 - 1.0 / (0.2 * sr));
	lfoPh = wrap(lfoPh + bs * err * kPull, 0.0, 1.0);
}

let phL = lfoPh;
let phR = wrap(lfoPh + 0.5 * clamp(spread, 0.0, 1.0), 0.0, 1.0);

// sine -> cube: the cube lingers at the extremes and crosses zero faster,
// i.e. a squarer tremolo from the same phase.
let shapeAmt = clamp(shape, 0.0, 1.0);
let sL = sin(2.0 * pi * phL);
let sR = sin(2.0 * pi * phR);
let mL = linearinterp(shapeAmt, sL, sL * sL * sL);
let mR = linearinterp(shapeAmt, sR, sR * sR * sR);

// --- complementary split and antiphase band gains -------------------------
let lowL = loL2.next(loL1.next(in1, xo), xo);
let lowR = loR2.next(loR1.next(in2, xo), xo);
let highL = in1 - lowL;
let highR = in2 - lowR;

let d = clamp(depth, 0.0, 1.0);
let gLoL = maximum(1.0 - d * mL, 0.0);
let gHiL = maximum(1.0 + d * mL, 0.0);
let gLoR = maximum(1.0 - d * mR, 0.0);
let gHiR = maximum(1.0 + d * mR, 0.0);

// --- transient click: fast-vs-slow rectified envelope, 0..1 ---------------
let rect = abs(0.5 * (in1 + in2));
let aFast = 1.0 - exp(0.0 - 1.0 / (0.003 * sr));
let aSlow = 1.0 - exp(0.0 - 1.0 / (0.060 * sr));
tFast = tFast + (rect - tFast) * aFast;
tSlow = tSlow + (rect - tSlow) * aSlow;
out3 = clamp(maximum(tFast - 1.4 * tSlow, 0.0) * 12.0, 0.0, 1.0);

let og = dbtoa(clamp(out_gain_db, -24.0, 12.0));
out1 = (lowL * gLoL + highL * gHiL) * og;
out2 = (lowR * gLoR + highR * gHiR) * og;
