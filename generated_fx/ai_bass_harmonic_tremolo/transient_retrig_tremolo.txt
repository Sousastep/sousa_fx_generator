// ============================================================================
// transient_retrig_tremolo.txt
// Attack-keyed tremolo: the LFO restarts on every tuba tongue, so a phrase
// always opens the same way instead of the wobble landing wherever it likes.
// A fast-vs-slow rectified envelope detects the attack; on detection the LFO
// phase is pulled back to its open peak and the tremolo ducks out for ~40 ms so
// the consonant punches through. The keyed phase offset unwinds over 60 ms, so
// the wobble glides back into the groove instead of stepping. sync chooses what
// sets the rate: a free rate, pulses per bar from in4, or a 5-limit division of
// the tracked pitch.
// "depth" is a phase-coherent dry/wet: at 0 the gain is exactly 1 and the
// output is the input sample for sample.
// From: Duxbury, Sandler & Davies, "A Hybrid Approach to Musical Note Onset
//       Detection", DAFx-02, §1-2 (energy-based detection function on a
//       subband / fast-vs-slow envelope - the detector used here);
//       McGovern et al., "Rhythm-Synchronized Effects Control with Modulation
//       Keying and Bezier Splines", DAFx-09 paper 10, §2.3 (modulation keying:
//       an LFO whose state is keyed by events rather than running free);
//       Disch & Zoelzer, "Modulation and Delay Line Based Digital Audio
//       Effects", DAFx-99, §2 Eq. 1 (the amplitude-modulation law).
//
// Inlets:
//   in1 / in2: stereo audio
//   in3: tracked f0 in Hz (20..1200) - rate source in "pitch" sync. 0 or out of
//        range = hold the last valid value; cold start 55 Hz (A1).
//   in4: 0..1 one-bar phasor - rate source in "bar" sync. Held at 0 the tremolo
//        falls back to rate_hz, so it never freezes.
// Outlets:
//   out1 / out2: stereo audio (exactly dry at depth = 0)
//   out3: transient click, 0..1, single-sample detection signal
// No safety limiter, no metering, no presets.
// ============================================================================

@param({ enum: ["free", "bar", "pitch"] }) sync = 1;   // rate source
@param({ min: 0.1, max: 12.0 }) rate_hz = 4.0;         // free rate
@param({ min: 1, max: 16, steps: 16 }) pulses = 4;     // pulses per bar
@param({ min: 0.0, max: 1.0 }) depth = 0.8;            // dry/wet
@param({ min: 0.0, max: 1.0 }) retrigger = 0.7;        // attack keying amount
@param({ min: 0.0, max: 1.0 }) shape = 0.3;            // open <-> gated
@param({ min: 0.0, max: 1.0 }) sens = 0.4;             // detector sensitivity
@param({ min: -24.0, max: 12.0 }) out_gain_db = 0.0;   // output gain

@state f0Last = 55.0;
@state f0Sm = 55.0;

@state ph = 0.0;
@state prevBar = 0.0;
@state barLast = 0.0;
@state barHold = 0.0;
@state barHz = 0.5;
@state barHzSm = 0.5;
@state phOff = 0.0;
@state keyOff = 0.0;
@state wasBar = 0.0;
@state duck = 0.0;
@state refr = 0.0;

@state tFast = 0.0;
@state tSlow = 0.0;

let sr = maximum(samplerate(), 1.0);

// --- in3: hold the last valid f0, then slew it ----------------------------
let f0In = in3;
if (f0In > 20.0 && f0In < 1200.0) {
	f0Last = f0In;
}
let f0Slew = 1.0 - exp(0.0 - 1.0 / (0.02 * sr));
f0Sm = f0Sm + (f0Last - f0Sm) * f0Slew;
let f0 = clamp(f0Sm, 20.0, 1200.0);

// --- in4: measure the bar rate, and hold it through a dropout -------------
let barIn = in4;
let barStep = barIn - prevBar;
if (barStep > 0.0000001) {
	barLast = barIn;
	barHz = clamp(barStep * sr, 0.05, 20.0);
	barHold = 0.1 * sr;
}
else if (barStep < -0.5) {
	barLast = barIn;
	barHold = 0.1 * sr;          // wrap: a bar line, but no rate estimate
}
else if (barHold > 0.0) {
	barHold -= 1.0;
}
prevBar = barIn;

let hzSlew = 1.0 - exp(0.0 - 1.0 / (0.05 * sr));
barHzSm = barHzSm + (barHz - barHzSm) * hzSlew;

// --- rate source ----------------------------------------------------------
let nPulses = clamp(floor(pulses + 0.5), 1.0, 16.0);
let baseRate = clamp(rate_hz, 0.1, 12.0);
if (sync == 1 && barHold > 0.0) {
	baseRate = clamp(nPulses * barHzSm, 0.05, 12.0);
}
else if (sync == 2) {
	baseRate = clamp(f0 / 16.0, 0.05, 12.0);
}

// --- LFO phase ------------------------------------------------------------
// Bar sync reads the phase straight off the phasor (sample-exact). The offset
// absorbed on entry decays over 50 ms, so arriving in bar sync is a glide from
// the free phase, not a jump; in steady state the offset is exactly 0.
let barOK = (sync == 1 && barHold > 0.0);
if (barOK > 0.5) {
	let target = wrap(barLast * nPulses, 0.0, 1.0);
	if (wasBar < 0.5) {
		phOff = wrap(ph - target, -0.5, 0.5);
	}
	let aOff = exp(0.0 - 1.0 / (0.05 * sr));
	phOff = phOff * aOff;
	ph = wrap(target + phOff, 0.0, 1.0);
}
else {
	ph = wrap(ph + baseRate / sr, 0.0, 1.0);
}
wasBar = barOK;

// --- attack detection -----------------------------------------------------
let rect = abs(0.5 * (in1 + in2));
let aFast = 1.0 - exp(0.0 - 1.0 / (0.003 * sr));
let aSlow = 1.0 - exp(0.0 - 1.0 / (0.060 * sr));
tFast = tFast + (rect - tFast) * aFast;
tSlow = tSlow + (rect - tSlow) * aSlow;
let click = clamp(maximum(tFast - 1.4 * tSlow, 0.0) * 12.0, 0.0, 1.0);
out3 = click;

// --- keying: pull the phase to the open peak and duck the tremolo ---------
// The key offset is expressed as a phase offset rather than a state reset, so
// it works the same in bar sync (where the clock is read from in4) and in free
// sync, and it unwinds over 60 ms instead of stepping.
let r = clamp(retrigger, 0.0, 1.0);
let thr = 0.45 - 0.40 * clamp(sens, 0.0, 1.0);
if (refr > 0.0) {
	refr -= 1.0;
}
if (click > thr && r > 0.02 && refr <= 0.0) {
	keyOff = 0.0 - ph * r;
	duck = r;
	refr = 0.05 * sr;
}
let aKey = exp(0.0 - 1.0 / (0.06 * sr));
keyOff = keyOff * aKey;
let aDuck = exp(0.0 - 1.0 / (0.04 * sr));
duck = duck * aDuck;

// --- wave: open at phase 0, closed at phase 0.5 ---------------------------
let phOut = wrap(ph + keyOff, 0.0, 1.0);
let w = 0.5 + 0.5 * cos(2.0 * pi * phOut);
let kShape = 1.0 + 3.0 * clamp(shape, 0.0, 1.0);
let wk = pow(w, kShape);

// --- gain: depth 0 is exactly dry; duck lifts it back to 1 on an attack ---
let d = clamp(depth, 0.0, 1.0);
let g = 1.0 - d * (1.0 - wk) * (1.0 - duck);

let og = dbtoa(clamp(out_gain_db, -24.0, 12.0));
out1 = in1 * g * og;
out2 = in2 * g * og;
