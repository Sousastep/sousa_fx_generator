// ============================================================================
// adaa_fold_liquid.txt — liquid-bass folder: an antiderivative-antialiased
//                        sine wavefolder on the band above a complementary
//                        crossover, key-tracked drive, bar swell; the lows
//                        pass untouched
//
// From: F. Esqueda, H. Pöntynen, V. Välimäki, J. D. Parker, "Virtual Analog
//       Buchla 259 Wavefolder", DAFx-17 paper 82: §2 (folding: the output
//       turns back each time the input crosses a fold threshold, so the
//       spectrum brightens with level), §3 (first-order antiderivative
//       antialiasing: y = (F(x[n]) - F(x[n-1])) / (x[n] - x[n-1]), with the
//       midpoint fallback when the step is tiny);
//       L. Gabrielli, S. Squartini et al., "Simplifying Antiderivative
//       Antialiasing with Lookup Table Integration", DAFx-25 paper 30 §1-2
//       (ADAA1 formulation and its ill-conditioned case).
// Parent: ai_freq_mod_phase_dist/12_adaa_chebyshev_wavefolder (5 stars),
//         buchla259_parallel_wavefolder (4 stars).
//
// Folder: f(u) = sin(u), F(u) = -cos(u), u = (pi/2)(g*high + bias), so a
// fold starts where g*|high| + bias passes 1. Makeup 1/sqrt(g).
// Split: low = one-pole lowpass, high = x - low (sums back exactly).
// Dry/wet acts on the high band only and the low band is the untouched
// horn, so the sub never moves in phase, at any blend.
//
// Inlets:
//   in1 / in2: stereo tuba
//   in3: f0 in Hz. > 0 = tracked pitch (clamped 20..1000 Hz);
//        0 = disconnected -> fixed 55 Hz. No zero-crossing fallback.
//   in4: 0..1 one-bar phasor (drive swell). Held (no movement for 100 ms) ->
//        internal 2 s bar (120 BPM 4/4).
//   in5: smoothed tuba volume envelope 0..1 (louder = more fold). Static ->
//        internal envelope follower.
// Outlets:
//   out1 / out2: stereo
// No safety limiter, no metering.
// ============================================================================

// Fold drive (dB)
@param({ min: 0, max: 30 }) drive_db = 12;
// Fold asymmetry (opposite on L/R)
@param({ min: -60, max: 60 }) bias = 15;
// Drive key tracking (less drive on high notes)
@param({ min: 0, max: 100 }) keytrack = 50;
// Bar swell of the drive
@param({ min: 0, max: 100 }) bar_swell = 40;
// Swells per bar
@param({ min: 1, max: 16 }) swell_div = 4;
// Crossover: untouched below (Hz)
@param({ min: 40, max: 300 }) xover_hz = 110;
// Dry/wet of the high band
@param({ min: 0, max: 100 }) blend = 100;
// Output gain
@param({ min: -24, max: 12 }) out_gain_db = 0;

@state f0Smooth = 55.0;

@state prevBarIn = 0.0;
@state barLast = 0.0;
@state barHold = 0.0;
@state freeBar = 0.0;

@state prevEnvIn = 0.0;
@state envHold = 0.0;
@state envInt = 0.0;

@state gSm = 4.0;
@state prevU = new FixedSampleArray(2);

@state lpL = new onepole();
@state lpR = new onepole();
@state dcBlkL = new onepole();
@state dcBlkR = new onepole();

// ADAA1 sine folder for channel ch (F = -cos).
function foldAdaa(ch: Int, u: number): number {
	let u1 = prevU[ch];
	prevU[ch] = u;
	let du = u - u1;
	let y = 0.0;
	if (abs(du) < 0.00001) {
		y = sin(0.5 * (u + u1));
	} else {
		y = (cos(u1) - cos(u)) / du;
	}
	return y;
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = in2;
let pitchIn = in3;
let barIn = in4;
let envIn = in5;

// --- f0: in3 > 0 is the pitch; 0 = disconnected -> fixed 55 Hz --------------
let f0Target = (pitchIn > 0.0) ? clamp(pitchIn, 20.0, 1000.0) : 55.0;
f0Smooth += (f0Target - f0Smooth) * (1.0 - exp(-1.0 / (0.004 * sr)));
let f0 = clamp(f0Smooth, 20.0, 1000.0);

// --- Bar phase: in4 while it moves, else internal 2 s bar -------------------
if (abs(barIn - prevBarIn) > 0.0000001) {
	barLast = barIn;
	barHold = 0.1 * sr;
} else if (barHold > 0.0) {
	barHold -= 1.0;
}
prevBarIn = barIn;
freeBar = wrap(freeBar + 0.5 / sr, 0.0, 1.0);
let barPh = (barHold > 0.0) ? wrap(barLast, 0.0, 1.0) : freeBar;

// --- Envelope: in5 while it moves, else internal follower ------------------
let absMono = 0.5 * (abs(xL) + abs(xR));
let atk = 1.0 - exp(-1.0 / (0.005 * sr));
let rel = 1.0 - exp(-1.0 / (0.120 * sr));
envInt += (absMono - envInt) * ((absMono > envInt) ? atk : rel);
if (abs(envIn - prevEnvIn) > 0.0000001) {
	envHold = 0.1 * sr;
} else if (envHold > 0.0) {
	envHold -= 1.0;
}
prevEnvIn = envIn;
let env = (envHold > 0.0) ? clamp(envIn, 0.0, 1.0) : clamp(sqrt(maximum(envInt, 0.0)) * 1.6, 0.0, 1.0);

// --- Drive: base x key tracking x bar swell x envelope ----------------------
let div = round(clamp(swell_div, 1.0, 16.0));
let swellShape = 0.5 - 0.5 * cos(2.0 * pi * wrap(barPh * div, 0.0, 1.0));
let sw = clamp(bar_swell, 0.0, 100.0) * 0.01;
let kt = clamp(keytrack, 0.0, 100.0) * 0.01;
let gT = dbtoa(clamp(drive_db, 0.0, 30.0)) * pow(55.0 / f0, 0.5 * kt) * (1.0 - sw + sw * swellShape) * (0.5 + 0.5 * env);
gSm += (gT - gSm) * (1.0 - exp(-1.0 / (0.005 * sr)));
let g = maximum(gSm, 0.05);
let makeup = 1.0 / sqrt(maximum(g, 1.0));

// --- Complementary split -------------------------------------------------------
let xo = clamp(xover_hz, 40.0, 300.0);
let lowL = lpL.next(xL, xo);
let lowR = lpR.next(xR, xo);
let hiL = xL - lowL;
let hiR = xR - lowR;

// --- Fold the high band (opposite bias on L/R) -------------------------------
let bs = clamp(bias, -60.0, 60.0) * 0.01;
// subtract the folder's resting value first, otherwise the swelling makeup
// would turn the bias offset into a sub-audio pulse
let rest = sin(0.5 * pi * bs);
let fL = (foldAdaa(0, 0.5 * pi * (g * hiL + bs)) - rest) * makeup;
let fR = (foldAdaa(1, 0.5 * pi * (g * hiR - bs)) + rest) * makeup;
// asymmetric folding still makes signal-dependent DC: block it
let wetL = fL - dcBlkL.next(fL, 10.0);
let wetR = fR - dcBlkR.next(fR, 10.0);

// --- Dry/wet on the high band; lows always untouched --------------------------
let b = clamp(blend, 0.0, 100.0) * 0.01;
let oL = lowL + linearinterp(b, hiL, wetL);
let oR = lowR + linearinterp(b, hiR, wetR);

// --- Output gain (last stage) -----------------------------------------------
let outGain = dbtoa(clamp(out_gain_db, -24.0, 12.0));
out1 = oL * outGain;
out2 = oR * outGain;
