// ============================================================================
// sideband_sub_growl.txt — lower-sideband sub-harmonic growl: adaptive
//                          split-sideband FM of the horn, the bar morphing
//                          between lower (sub) and upper (screech) sidebands
//
// From: V. Lazzarini, J. Timoney, T. Lysaght, "Asymmetric-Spectra Methods
//       for Adaptive FM Synthesis", DAFx-08 paper 42: §3.1 eq. 5-14 (split
//       sidebands: cos(I sin wt) carries the even sidebands, sin(I sin wt)
//       the odd ones; single-sideband products with the analytic carrier
//       separate upper from lower), §3.2 eq. 15-20 (adaptive SpSB: the
//       carrier is the input signal), §4.3 (sub-harmonic multiphonics from
//       lower sidebands, screech from upper sidebands).
// Parent: ai_freq_mod_phase_dist/15_inst_split_sideband_morph_synth and
//         07_adaptive_split_sideband (5 and 4 stars).
//
// Implementation: e^(jI sin t) = sum_n J_n(I) e^(jnt) (J_-n = (-1)^n J_n).
// Each sideband n gets an upper/lower weight and an odd/even weight; the
// output is Re{z * m(t)}, z the analytic horn. With every weight at 1 this is
// exactly PM of the horn. J_0..J_10 come from Miller's backward recurrence,
// updated every 32 samples.
// Dry/wet: the dry is the real branch of the same Hilbert pair, so the blend
// does not comb.
//
// Inlets:
//   in1 / in2: stereo tuba
//   in3: f0 in Hz. > 0 = tracked pitch (clamped 20..1000 Hz);
//        0 = disconnected -> fixed 55 Hz. No zero-crossing fallback.
//   in4: 0..1 one-bar phasor (sideband morph). Held (no movement for
//        100 ms) -> internal 2 s bar (120 BPM 4/4).
//   in5: smoothed tuba volume envelope 0..1 (opens the index). Static ->
//        internal envelope follower.
// Outlets:
//   out1 / out2: stereo
// No safety limiter, no metering.
// ============================================================================

// Carrier:modulator ratio (fm = f0 / ratio)
@param({ min: 0.5, max: 4 }) cm_ratio = 2;
// FM index
@param({ min: 0, max: 6 }) index = 2;
// Lower (-) / upper (+) sidebands
@param({ min: -100, max: 100 }) low_high = -60;
// Odd (-) / even (+) sidebands
@param({ min: -100, max: 100 }) odd_even = 0;
// Bar morph of low/high
@param({ min: 0, max: 100 }) bar_morph = 50;
// Morphs per bar
@param({ min: 1, max: 16 }) morph_div = 2;
// Dry/wet (phase-aligned dry)
@param({ min: 0, max: 100 }) blend = 100;
// Output gain
@param({ min: -24, max: 12 }) out_gain_db = 0;

@state f0Smooth = 55.0;

@state prevBarIn = 0.0;
@state barLast = 0.0;
@state barHold = 0.0;
@state freeBar = 0.0;

@state prevEnvIn = 0.0;
@state envHold = 0.0;
@state envInt = 0.0;

@state modPh = 0.0;
@state lhSm = -0.6;
@state oeSm = 0.0;
@state ctlCount: Int = 0;
@state jn = new FixedSampleArray(11);

@state reX = new FixedSampleArray(2, 4);
@state reY = new FixedSampleArray(2, 4);
@state imX = new FixedSampleArray(2, 4);
@state imY = new FixedSampleArray(2, 4);
@state reDel = new FixedSampleArray(2);

@state dcBlkL = new onepole();
@state dcBlkR = new onepole();

// Real branch of the 4-stage quadrature pair, plus one sample of delay to
// line it up with the imaginary branch. y = c*(x + y1) - x1 per stage.
function hilbRe(ch: Int, x: number): number {
	let s = x;
	let c = 0.0;
	for (let i: Int = 0; i < 4; i++) {
		c = (i == 0) ? 0.4794008656 : ((i == 1) ? 0.8762184935 : ((i == 2) ? 0.9765975895 : 0.9974992559));
		let y = c * (s + reY[ch][i]) - reX[ch][i];
		reX[ch][i] = s;
		reY[ch][i] = y;
		s = y;
	}
	let out = reDel[ch];
	reDel[ch] = s;
	return out;
}

// Imaginary branch; this coefficient order gives the negative Hilbert, so negate.
function hilbIm(ch: Int, x: number): number {
	let s = x;
	let c = 0.0;
	for (let i: Int = 0; i < 4; i++) {
		c = (i == 0) ? 0.1617584983 : ((i == 1) ? 0.7330289323 : ((i == 2) ? 0.9453497003 : 0.9905991567));
		let y = c * (s + imY[ch][i]) - imX[ch][i];
		imX[ch][i] = s;
		imY[ch][i] = y;
		s = y;
	}
	return 0.0 - s;
}

// J_0..J_10 of x into jn[] (Miller backward recurrence, normalised by
// J_0 + 2 sum J_2k = 1).
function besselAll(x: number) {
	if (x < 0.000001) {
		for (let n: Int = 0; n < 11; n++) {
			jn[n] = (n == 0) ? 1.0 : 0.0;
		}
	} else {
		let jp1 = 0.0;
		let jc = 1.0e-30;
		let sum = 0.0;
		for (let k: Int = 40; k > 0; k--) {
			let jm1 = (2.0 * k / x) * jc - jp1;
			jp1 = jc;
			jc = jm1;
			if (abs(jc) > 1.0e10) {
				jc *= 1.0e-10;
				jp1 *= 1.0e-10;
				sum *= 1.0e-10;
				for (let r: Int = 0; r < 11; r++) {
					jn[r] = jn[r] * 1.0e-10;
				}
			}
			let n: Int = k - 1;
			if (n < 11) {
				jn[n] = jc;
			}
			if (n > 0 && (n % 2) == 0) {
				sum += 2.0 * jc;
			}
		}
		sum += jc;
		for (let n: Int = 0; n < 11; n++) {
			jn[n] = jn[n] / sum;
		}
	}
}

// Re{(re + j im) * m(t)} with the weighted split-sideband modulator m.
function splitPM(re: number, im: number, th: number, wU: number, wL: number, wO: number, wE: number): number {
	let mRe = jn[0];
	let mIm = 0.0;
	let c1 = cos(th);
	let s1 = sin(th);
	let pc = 1.0;
	let ps = 0.0;
	for (let n: Int = 1; n < 11; n++) {
		let t = pc * c1 - ps * s1;
		ps = pc * s1 + ps * c1;
		pc = t;
		let odd = (n % 2) == 1;
		let wp = odd ? wO : wE;
		let u = wU * wp * jn[n];
		let l = wL * wp * jn[n] * (odd ? -1.0 : 1.0);
		mRe += (u + l) * pc;
		mIm += (u - l) * ps;
	}
	return re * mRe - im * mIm;
}

let sr = maximum(samplerate(), 1.0);
let xL = in1;
let xR = in2;
let pitchIn = in3;
let barIn = in4;
let envIn = in5;

// --- f0: in3 > 0 is the pitch; 0 = disconnected -> fixed 55 Hz --------------
let f0Target = (pitchIn > 0.0) ? clamp(pitchIn, 20.0, 1000.0) : 55.0;
f0Smooth += (f0Target - f0Smooth) * (1.0 - exp(-1.0 / (0.004 * sr)));
let f0 = clamp(f0Smooth, 20.0, 1000.0);

// --- Bar phase: in4 while it moves, else internal 2 s bar -------------------
if (abs(barIn - prevBarIn) > 0.0000001) {
	barLast = barIn;
	barHold = 0.1 * sr;
} else if (barHold > 0.0) {
	barHold -= 1.0;
}
prevBarIn = barIn;
freeBar = wrap(freeBar + 0.5 / sr, 0.0, 1.0);
let barPh = (barHold > 0.0) ? wrap(barLast, 0.0, 1.0) : freeBar;

// --- Envelope: in5 while it moves, else internal follower ------------------
let absMono = 0.5 * (abs(xL) + abs(xR));
let atk = 1.0 - exp(-1.0 / (0.005 * sr));
let rel = 1.0 - exp(-1.0 / (0.120 * sr));
envInt += (absMono - envInt) * ((absMono > envInt) ? atk : rel);
if (abs(envIn - prevEnvIn) > 0.0000001) {
	envHold = 0.1 * sr;
} else if (envHold > 0.0) {
	envHold -= 1.0;
}
prevEnvIn = envIn;
let env = (envHold > 0.0) ? clamp(envIn, 0.0, 1.0) : clamp(sqrt(maximum(envInt, 0.0)) * 1.6, 0.0, 1.0);

// --- Control rate: index -> Bessel table every 32 samples --------------------
if (ctlCount <= 0) {
	besselAll(clamp(index, 0.0, 6.0) * (0.4 + 0.6 * env));
	ctlCount = 32;
}
ctlCount -= 1;

// --- Sideband weights: low/high + bar morph, odd/even ------------------------
let div = round(clamp(morph_div, 1.0, 16.0));
let lhT = clamp(clamp(low_high, -100.0, 100.0) * 0.01 + clamp(bar_morph, 0.0, 100.0) * 0.01 * sin(2.0 * pi * wrap(barPh * div, 0.0, 1.0)), -1.0, 1.0);
let ctl = 1.0 - exp(-1.0 / (0.01 * sr));
lhSm += (lhT - lhSm) * ctl;
oeSm += (clamp(odd_even, -100.0, 100.0) * 0.01 - oeSm) * ctl;
let wU = minimum(1.0, 1.0 + lhSm);
let wL = minimum(1.0, 1.0 - lhSm);
let wE = minimum(1.0, 1.0 + oeSm);
let wO = minimum(1.0, 1.0 - oeSm);

// --- Modulator -------------------------------------------------------------
let fm = clamp(f0 / clamp(cm_ratio, 0.5, 4.0), 5.0, sr * 0.1);
modPh = wrap(modPh + fm / sr, 0.0, 1.0);
let thL = 2.0 * pi * modPh;
let thR = 2.0 * pi * wrap(modPh + 0.06, 0.0, 1.0);

// --- Analytic horn x split-sideband modulator ---------------------------------
let reL = hilbRe(0, xL);
let imL = hilbIm(0, xL);
let reR = hilbRe(1, xR);
let imR = hilbIm(1, xR);
let spL = splitPM(reL, imL, thL, wU, wL, wO, wE);
let spR = splitPM(reR, imR, thR, wU, wL, wO, wE);
// makeup: sum of J_n^2 over all n is 1, so the kept power is known exactly
let kept = jn[0] * jn[0];
for (let n: Int = 1; n < 11; n++) {
	let wp = ((n % 2) == 1) ? wO : wE;
	kept += (wU * wU + wL * wL) * wp * wp * jn[n] * jn[n];
}
let makeup = 1.0 / sqrt(maximum(kept, 0.1));
// lower sidebands of the fundamental can reach 0 Hz: block DC
let wetL = (spL - dcBlkL.next(spL, 10.0)) * makeup;
let wetR = (spR - dcBlkR.next(spR, 10.0)) * makeup;

// --- Dry/wet (dry = Hilbert real branch) ------------------------------------
let b = clamp(blend, 0.0, 100.0) * 0.01;
let oL = linearinterp(b, reL, wetL);
let oR = linearinterp(b, reR, wetR);

// --- Output gain (last stage) -----------------------------------------------
let outGain = dbtoa(clamp(out_gain_db, -24.0, 12.0));
out1 = oL * outGain;
out2 = oR * outGain;
